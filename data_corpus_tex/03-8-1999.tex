Ниже $L_q$ --- комплексное пространство с нормой
$\|f\|_{L_q}=\Big(\int\limits_{-1}^1|f(x)|^q\omega(x)dx\Big)^{1/q}$,
$1\le q<\infty$, $\omega(x)=(1-x^2)^{\gamma-1/2}$ --- вес $(0<\gamma<1)$,
$L_\infty=L_\infty[-1,1]$ --- вещественное пространство измеримых,
ограниченных в существенном функций,
$D=D[-1,1]$ --- пространство абсолютно непрерывных на $[-1,1]$ функций.

Рассмотрим действующий в $L_2$ оператор $T$
$$
Ty=-(1-x^2)y''+(2\gamma+1)xy'.
$$
Его собственным числам $\lambda_n=n(n+2\gamma)$
соответствуют ортонормированные в $L_2$ собственные функции
$v_n(x)=h_n^{-1/2}C_n^\gamma(x)$, где
$h_n=\frac{\sqrt\pi (2\gamma)_n\Gamma(\gamma+1/2)}
{(n+\gamma)n!\Gamma(\gamma)}$,
$(\alpha)_n=\frac{\Gamma(\alpha+n)}{\Gamma(\alpha)}$,
$C_n^\gamma(x)$ --- многочлены Гегенбауэра ([1], с.\,176),
$n=\overline{1,\infty}$. Использованы обозначения справочника [1].
Так как $v_n$ $(n=\overline{1,\infty})$ полны
в $L_2$ ([1], с.\,160), то можно показать, что $T$
--- самосопряженный оператор.

Пусть $P:L_2\to L_2$ --- оператор умножения на функцию $p$ $(p\in L_\infty)$.
Обозначим собственные значения оператора
$T+P$ через $\mu_n$, занумерованные в порядке возрастания
действительных частей с учетом алгебраической кратности,
а через $u_n$ --- соответствующие им ортонормированные собственные функции.

Поскольку резольвенты
$R_\lambda(T)$ и $R_\lambda({T+P})$ операторов $T$ и $T+P$
являются интегральными операторами ([2], с.\,219),
то, обозначая их ядра через $K_T(x,y,\lambda)$ и
$K_{T+P}(x,y,\lambda)$ соответственно,
можем доказать утверждение.

\begin{thm} Положим
\begin{gather}
\begin{align}
\alpha_j(n,x,y)&=\frac{(-1)^j}{2\pi i}
\int_{T_n}([K_T\circ P]^j\circ K_T)(x,y,\lambda)d\lambda\quad
(j=\overline{1,t}), \\
%
\varphi_t(n,x,y)&=\frac{(-1)^{t+1}}{2\pi i}
\int_{T_n}([K_T\circ P]^{t+1}\circ K_{T+P})(x,y,\lambda)d\lambda
\quad (t\ge0),\notag
\end{align}
\\
(K\circ P\circ Q)(x,y,\lambda)=
\int_{-1}^1 K(x,z,\lambda)p(z)Q(z,y,\lambda)\omega(z)dz,\notag
\end{gather}
где $T_n$ --- замкнутый регулярный контур, охватывающий только собственные
значения $\lambda_1,\dotsc,\lambda_n$ и отстоящий от них на расстоянии не
менее чем $n$.

Если $p\in L_\infty$, то
\begin{gather}
\sum_{j=1}^n u_j(x)\overline{u_j(y)}=
\sum_{j=1}^n v_j(x)\overline{v_j(y)} +
\sum_{j=1}^t\alpha_j(n,x,y)+\varphi_t(n,x,y), \\
%
\bigg\|
\sum_{j=1}^n u_j(x)\overline{u_j(y)}-\sum_{j=1}^n v_j(x)
\overline{v_j(y)}-\alpha_1(n,x,y)
\bigg\|_{L_q}=O\bigg(\frac{\ln n}{n^{1/2}}\bigg), \quad 1\le q\le2.
\end{gather}
\end{thm}

Для доказательства потребуются три леммы.

\begin{lem} Если $\lambda\in\gamma_n=\left\{\lambda\mid\re\lambda=
\frac{\lambda_{n+1}+\lambda_n}2\right\}$,
то $\sum\limits_{i=1}^\infty\frac1{|\lambda_i-\lambda|}=
O\big(\frac{\ln n}n\big)$.
\end{lem}

{\bf Доказательство.}
\begin{multline*}
\sum_{i=1}^\infty \frac1{|\lambda_i-\lambda|}\le
\sum_{i=1}^\infty\frac1{|\lambda_i-\re\lambda|}=
\sum_{i=1}^n\frac1{\re\lambda-\lambda_i}+\frac1{n+\frac{2\gamma+1}2} +
\sum_{i=n+2}^\infty\frac1{\lambda_i-\re\lambda}\le \\
%
\le\int_0^n\frac{dx}{\re\lambda-x^2-2\gamma x}+
\frac1{n+\frac{2\gamma+1}2} +
\int_{n+1}^\infty\frac{dx}{x^2+2\gamma x-\re\lambda}=
O\bigg(\frac{\ln n}n\bigg).\quad\square
\end{multline*}

Пусть $\Pi_n=\big\{\lambda\mid\lambda=
\frac{\lambda_{n+1}+\lambda_n}2+
\big(\frac{\lambda_{n+1}+\lambda_n}2+n\big)e^{i\varphi},
\ \frac\pi2\le\varphi\le\frac32\pi\big\}$.

\begin{lem} Если $\lambda\in\Pi_n$,
то $\sum\limits_{i=1}^\infty\frac1{|\lambda_i-\lambda|}
= O\big(\frac1{n^{1/2}}\big)$.
\end{lem}

{\bf Доказательство.}
$$
\sum_{i=1}^\infty \frac1{|\lambda_i-\lambda|}\le
\sum_{i=1}^n\frac1{n+\lambda_i} +
\frac1{\frac{\lambda_{n+1}+\lambda_n}2+n} +
\int_{n+1}^\infty\frac{dx}{x^2+2\gamma x-\re\lambda} =
O\bigg(\frac1{n^{1/2}}\bigg).
$$

\begin{lem} Имеют место асимптотические равенства
\begin{gather}
v_n(\cos\alpha)=\frac{O(1)\cos[(n+\gamma)\alpha-\gamma\pi/2]}
{(\sin\alpha)^\gamma},\quad 0<\alpha<\pi; \\
%
\|v_n\|_{L_q}=O(1), \quad 1\le q\le 2, \notag
\end{gather}
где $O(1)$ --- константа, не зависящая от $n$.
\end{lem}

{\bf Доказательство.}
Для многочленов Гегенбауэра известна ([1], с.\,198) равномерная по
$n$ и $\alpha$ асимптотика
$$
C_n^\gamma(\cos\alpha)=\frac{\Gamma(2\gamma+n)}{(\gamma)_{n+1}}\,
\frac{O(1)\cos[(n+\gamma)\alpha-\gamma\pi/2]}{(\sin\alpha)^\gamma},
\quad 0<\alpha<\pi.
$$
Тогда (4) следует из того, что
$$
h_n^{-1/2}\frac{\Gamma(2\gamma+n)}{(\gamma)_{n+1}}O(1)=
\bigg[\frac{(n+\gamma)n!\Gamma(2\gamma+n)}{[\Gamma(\gamma+n+1)]^2}
\bigg]^{1/2}O(1)=\bigg[
\frac{n}{n+\gamma}\prod_{k=1}^{n-1}
\frac{1+{2\gamma}/k}{(1+{2\gamma}/k)^2}
\bigg]^{1/2}O(1)=O(1).
$$
Отсюда
$$
\|v_n\|_{L_q}=\bigg(
\int_0^\pi|v_n(\cos\alpha)|^q(\sin\alpha)^{2\gamma}d\alpha
\bigg)^{1/q} = O(1), \quad 1\le q\le 2.
$$

{\bf Доказательство теоремы 1.} Выберем в качестве  $T_n$ контур,
состоящий из отрезка
$\gamma(n)=\big\{\lambda\mid\lambda=\frac{\lambda_{n+1}+\lambda_n}2+
ri,\ -\frac{\lambda_{n+1}+\lambda_n}2-n\le r\le
\frac{\lambda_{n+1}+\lambda_n}2+n\big\}$
и полуокружности $\Pi_n$. Такой контур удовлетворяет условию теоремы.
A поскольку $\lambda\notin\sigma(T)\cup\sigma(T+P)$, то, используя тождество
Гильберта
$R_\lambda(T+P)-R_\lambda(T)=-R_\lambda(T)PR_\lambda(T+P)$, получим
равенство
\begin{equation}
R_\lambda(T+P)=\sum_{l=0}^t(-1)^l[R_\lambda(T)P]^lR_\lambda(T)+
(-1)^{t+1}[R_\lambda(T)P]^{t+1}R_\lambda(T+P).
\end{equation}
Далее, если не указано противное, интегрирование ведется от $-1$ до 1. Имеем
\begin{multline*}
[R_\lambda(T)P]^lR_\lambda(T)v(x)=[R_\lambda(T)P]^{(l-1)}R_\lambda(T)
\int p(x)K_T(x,y,\lambda)v(y)\omega(y)dy=\\
%
=[R_\lambda(T)P]^{(l-1)}\int\bigg(\int K_T(x,y_1,\lambda)p(y_1)
K_T(y_1,y,\lambda)\omega(y_1)dy_1\bigg)v(y)\omega(y)dy=\dotsb= \\
%
=\int\bigg(\idotsint K_T(x,y_1,\lambda)p(y_1)K_T(y_1,y_2,\lambda)\dotsc
p(y_l)\cdot \\
\cdot K_T(y_l,y,\lambda)\omega(y_1)\dotsc\omega(y_l) dy_1\dotsc
dy_l\bigg)v(y)\omega(y)dy= \\
%
\int\bigg(\idotsint K_T(x,y_1,\lambda)\dotsc p(y_{l-1})(K_T\circ
P\circ K_T)(y_{l-1},y,\lambda)\omega(y_1)\dotsc\omega(y_{l-1})
dy_1\dotsc dy_{l-1}\bigg)v(y)\omega(y)dy= \\
%
=\dotsb=\int([K_T\circ P]^l\circ K_T)(x,y,\lambda)v(y)\omega(y)dy.
\end{multline*}
Используя полученное выражение для $[R_\lambda(T)P]^lR_\lambda(T)v(x)$
при переходе от операторов в равенстве (5) к их ядрам, будем иметь
\begin{multline}
K_{T+P}(x,y,\lambda)=K_T(x,y,\lambda)+
\sum_{l=1}^t(-1)^l([K_T\circ P]^l\circ K_T)(x,y,\lambda) + \\
+ (-1)^{t+1}([K_T\circ P]^{t+1}\circ K_{T+P})(x,y,\lambda).
\end{multline}
Можно показать, что в любой компактной части комплексной плоскости,
не содержащей точек
спектра операторов $T$ и $T+P$, имеют место разложения
$$
K_T(x,y,\lambda)=\sum_{k=1}^\infty\frac{v_k(x)\overline{v_k(y)}}
{\lambda_k-\lambda},
\quad K_{T+P}(x,y,\lambda)=\sum_{j=1}^\infty
\frac{u_j(x)\overline{u_j(y)}}{\mu_j-\lambda}.
$$
Подставляя их в левую часть равенства (6) и интегрируя по контуру $T_n$,
получим (2).

Докажем (3). Используя разложение для преобразования правой части
равенства (6), имеем
\begin{multline}
([K_T\circ P]^l\circ K_T)(x,y,\lambda)=
\idotsint K_T(x,y_1,\lambda)p(y_1)\dotsc
p(y_l)K_T(y_l,y,\lambda)\omega(y_1)\dotsc\omega(y_l) dy_1\dotsc dy_l= \\
%
=\idotsint \sum_{k=1}^\infty\frac{v_k(x)\overline{v_k(y_1)}}
{\lambda_k-\lambda}p(y_1)K_T(y_1,y_2,\lambda)\dotsc
K_T(y_{l-1},y_l,\lambda)p(y_l)\cdot \\
\cdot\sum_{j=1}^\infty
\frac{v_j(y_l)\overline{v_j(y)}}{\lambda_j-\lambda}
\omega(y_1)\dotsc\omega(y_l) dy_1\dots dy_l= \\
%
=\dotsb=\sum_{k,j}\frac{v_k(x){\displaystyle \int}
\overline{v_k(y_1)}p(y_1)[R_\lambda(T)P]^{l-1}v_j(y_1)\omega(y_1)dy_1
\overline{v_j(y)}}
{(\lambda_k-\lambda)(\lambda_j-\lambda)}= \\
%
=\sum_{k,j}\frac{v_k(x)(P[R_\lambda(T)P]^{l-1}v_j,v_k)\overline{v_j(y)}}
{(\lambda_k-\lambda)(\lambda_j-\lambda)}.
\end{multline}
Подставим найденное выражение вместо
$([K_T\circ P]^l\circ K_T)(x,y,\lambda)$ в (6). Получим
\begin{multline*}
\sum_{j=1}^\infty\frac{u_j(x)\overline{u_j(y)}}{\mu_j-\lambda}-
\sum_{k=1}^\infty\frac{v_k(x)\overline{v_k(y)}}{\lambda_k-\lambda}=
\sum_{l=1}^t(-1)^l\sum_{k,j}
\frac{v_k(x)(P[R_\lambda(T)P]^{l-1}v_j,v_k)\overline{v_j(y)}}
{(\lambda_k-\lambda)(\lambda_j-\lambda)}+ \\
%
(-1)^{t+1}\sum_{k,j}\frac{v_k(x)(P[R_\lambda(T)P]^tu_j,v_k)
\overline{u_j(y)}}{(\lambda_k-\lambda)(\mu_j-\lambda)}.
\end{multline*}
Далее, используя обобщенное неравенство Минковского ([3], с.\,178),
из (1) и (7) найдем
\begin{multline*}
\|\alpha_l(n,x,y)\|_{L_q}=\bigg\|\frac{(-1)^l}{2\pi i}
\int_{T_n}\sum_{k,j}
\frac{v_k(x)(P[R_\lambda(T)P]^{l-1}v_j,v_k)\overline{v_j(y)}}
{(\lambda-\lambda_k)(\lambda-\lambda_j)}d\lambda\bigg\|_{L_q}= \\
%
=\frac1{2\pi}\bigg[\int\int\bigg|
\int_{T_n}\sum_{k,j}
\frac{v_k(x)(P[R_\lambda(T)P]^{l-1}v_j,v_k)\overline{v_j(y)}}
{(\lambda-\lambda_k)(\lambda-\lambda_j)}d\lambda
\bigg|^q \omega(x)\omega(y)dx\,dy\bigg]^{1/q}\le \\
%
\le\frac1{2\pi}\sum_{k,j}\bigg[\int\int\bigg(
\int_{T_n}\bigg|\frac{v_k(x)(P[R_\lambda(T)P]^{l-1}v_j,v_k)\overline{v_j(y)}}
{(\lambda-\lambda_k)(\lambda-\lambda_j)}\bigg|d\lambda
\bigg)^q \omega(x)dx\,\omega(y)dy\bigg]^{1/q} = \\
%
=\frac1{2\pi}\sum_{k,j}\|v_k\|_{L_q}\|v_j\|_{L_q}
\bigg(\int_{T_n}\frac{|(P[R_\lambda(T)P]^{l-1}v_j,v_k)|}
{|\lambda-\lambda_k|\,|\lambda-\lambda_j|}d\lambda
\bigg)^{q\frac1q} = \\
%
= O(1)\int_{T_n}\sum_{k,j}\frac{|(P[R_\lambda(T)P]^{l-1}v_j,v_k)|}
{|\lambda-\lambda_k|\,|\lambda-\lambda_j|}d\lambda =
O(1)\int_{T_n}\sum_{k,j}
\frac{\|P\|^l\|R_\lambda(T)\|^{l-1}\|v_j\|_{L_2}\|v_k\|_{L_2}}
{|\lambda-\lambda_k|\,|\lambda-\lambda_j|}d\lambda = \\
%
=O(1)\|P\|^l\int_{T_n}\|R_\lambda(T)\|^{l-1}\sum_{k,j=1}^\infty
\frac1{|\lambda{-}\lambda_k|\,|\lambda{-}\lambda_j|}d\lambda{=}
O(1)\|P\|^l\bigg(\int_{\Pi_n}{+}\int_{\gamma(n)}\bigg)
\|R_\lambda(T)\|^{l-1}O\bigg(\frac{\ln n}{n^{3/2}}\bigg)d\lambda.
\end{multline*}
Здесь мы перешли от интеграла по контуру к криволинейному интегралу 1-го
рода по кривой $T_n$.
Продолжим преобразование, учитывая, что $P$ --- ограниченный оператор,
$\|R_\lambda(T)\|\le 1/{\rho(\lambda,\sigma(T))}$
где $\rho(\lambda,\sigma(T))$ ---
расстояние от $\lambda$ до спектра оператора $T$ ([4], с.\,367),
\begin{multline*}
\|P\|^lO\bigg(\frac{\ln n}{n^{3/2}}\bigg)\bigg
(\int_{\Pi_n}+\int_{\gamma(n)}\bigg)\|R_\lambda(T)\|^{l-1}d\lambda=
O\bigg(\frac{\ln n}{n^{3/2}}\bigg)\bigg(\int_{\Pi_n}+\int_{\gamma(n)}
\bigg)\frac{d\lambda}{n^{l-1}}= \\
%
= O\bigg(\frac{\ln n}{n^{3/2}}\bigg)
\frac{(\pi+2)(n^2+2n+2n\gamma+\gamma+1/2)}{n^{l-1}}=
O\bigg(\frac{\ln n}{n^{l-3/2}}\bigg).
\end{multline*}
Отсюда видно, что если $l\ge2$, то
$\|\alpha_l(n,x,y)\|_{L_q}\to 0$ при $n\to\infty$. Далее
$$
\|\varphi_t(n,x,y)\|_{L_q}=
\bigg\|\sum_{l=t+1}^\infty \alpha_l(n,x,y)\bigg\|_{L_q} \le
\sum_{l=t+1}^\infty \|\alpha_l(n,x,y)\|_{L_q} =
O(\ln n)\sum_{l=t+1}^\infty \frac1{n^{l-3/2}} =
O\bigg(\frac1{n^{t-1/2}}\bigg).
$$
Таким образом, если $t\ge1$ (при этом $l\ge2$), то
$\|\varphi_t(n,x,y)\|_{L_q}\to 0$ при $n\to\infty$. Теперь из (2)
имеем
$$
\bigg\|\sum_{j=1}^n u_j(x)\overline{u_j(y)}-
\sum_{j=1}^n v_j(x)\overline{v_j(y)}-
\alpha_1(n,x,y)\bigg\|_{L_q}\le
\|\alpha_2(n,x,y)\|_{L_q}+\|\varphi_2(n,x,y)\|_{L_q}=
O\bigg(\frac{\ln n}{n^{1/2}}\bigg).\quad\square
$$

Аналогичная теорема была получена в работе [5]
(с дискретным положительным псевдодифференциальным порядка $m$ $(m>1)$
оператором $T$ и требованием гладкости функции $p$).
\end{notenonum}

Накладывая на $p$ более жесткие условия, усилим вторую часть доказанного
утверждения.

\begin{thm} Если $p\in D$ и $p(-1)=p(1)=0$, то
$$
\bigg\|\sum_{j=1}^n u_j(x)\overline{u_j(y)}-
\sum_{j=1}^n v_j(x)\overline{v_j(y)}\bigg\|_{L_q}=
O\bigg(\frac{\ln n}{n^{1/2}}\bigg),\quad 1\le q\le 2.
$$
\end{thm}

{\bf Доказательство.} Имеем
\begin{multline*}
\bigg[\int\int
|v_k(x)(Pv_j,v_k)\overline{v_j(y)}|^q\omega(x)\omega(y)dxdy
\bigg]^{1/q}\le
|(Pv_j,v_k)|\,\|v_k\|_{L_q}\|v_j\|_{L_q}=O(1)|(Pv_j,v_k)|= \\
%
=O(1)\bigg|
\bigg(\int_0^\varepsilon+\int_{\pi-\varepsilon}^\pi\bigg)
p(\cos\alpha)v_j(\cos\alpha)v_k(\cos\alpha)(\sin\alpha)^{2\gamma}d\alpha
\bigg|+ \\
%
+O(1)\bigg|
\int_\varepsilon^{\pi-\varepsilon}
p(\cos\alpha)v_j(\cos\alpha)v_k(\cos\alpha)(\sin\alpha)^{2\gamma}d\alpha
\bigg|=I_1+I_2.
\end{multline*}
Из оценки
$(\sin\alpha)^\gamma|C_n^\gamma(\cos\alpha)|<\big(\frac n2\big)^{\gamma-1}
[\Gamma(\gamma)]^{-1}$, $0\le\alpha\le\pi$ ([1], с.\,205)
аналогично лемме 3 получим
$|v_n(\cos\alpha)|=\frac{O(n^\gamma)}{(\sin\alpha)^\gamma}$.
Отсюда
$$
I_1=\bigg|O(k^\gamma j^\gamma)
\bigg(\int_0^\varepsilon+\int_{\pi-\varepsilon}^\pi\bigg)
p(\cos\alpha) d\alpha\bigg| =
O(k^\gamma j^\gamma\varepsilon^2)=O((k+j)^{-2}).
$$
Здесь $p(\cos\alpha)$ вблизи $\alpha=0$, $\alpha=\pi$
ведет себя как $\sin\alpha$
и взято $\varepsilon=(kj)^{-\gamma/2}(k+j)^{-1}$.

Далее, используя асимптотику (4), найдем
\begin{multline*}
I_2=
O(1)\bigg|
\int_\varepsilon^{\pi-\varepsilon}
p(\cos\alpha)\cos[(j+\gamma)\alpha-\gamma\pi/2]
\cos[(k+\gamma)\alpha-\gamma\pi/2] d\alpha\bigg|= \\
%
=O(1)\bigg|
\int_\varepsilon^{\pi-\varepsilon}
p(\cos\alpha)\cos[(k+j+2\gamma)\alpha-\gamma\pi]d\alpha+
\int_\varepsilon^{\pi-\varepsilon}
p(\cos\alpha)\cos[(k-j)\alpha]d\alpha\bigg| = \\
%
=\frac{O(1)}{k+j+2\gamma}+\frac{O(1)}{k+j+2\gamma}\bigg|
\int_\varepsilon^{\pi-\varepsilon}
p'(\cos\alpha)\sin[(k+j+2\gamma)\alpha-\gamma\pi]d\alpha\bigg|+ \\
%
+\frac{O(1)}{|k-j|}+\frac{O(1)}{|k-j|}\bigg|
\int_\varepsilon^{\pi-\varepsilon}
p'(\cos\alpha)\sin[(k-j)\alpha]d\alpha
\bigg|=\frac{O(1)}{|k-j|}.
\end{multline*}

Так как условие теоремы 1 выполняется, то достаточно оценить
$\|\alpha_1(n,x,y)\|_{L_q}$. Из (1) и (7), обозначая
$v_k(x)(Pv_j,v_k)\overline{v_j(y)}$ через $q_{kj}$,
имеем
\begin{multline*}
\|\alpha_1(n,x,y)\|_{L_q}=\bigg[
\int\int\bigg|\frac{-1}{2\pi i}\int_{T_n}\sum_{k,j}
\frac{q_{kj}}
{(\lambda-\lambda_k)(\lambda-\lambda_j)}d\lambda
\bigg|^q\omega(x)\omega(y)dx\,dy\bigg]^{1/q}= \\
%
=\bigg[\int\int\bigg|\bigg[
\sum_{k=1}^n\sum_{j=1}^n+\sum_{k=n+1}^\infty\sum_{j=1}^n+
\sum_{k=1}^n\sum_{j=n+1}^\infty+\sum_{k=n+1}^\infty\sum_{j=n+1}^\infty
\bigg]\cdot \\
\cdot\bigg[\frac{q_{kj}}{2\pi i}\int_{T_n}\frac{d\lambda}
{(\lambda-\lambda_k)(\lambda-\lambda_j)}\bigg]
\bigg|^q\omega(x)\omega(y)dx\,dy\bigg]^{1/q}= \\
%
=\bigg[\int\int\bigg|\bigg[
\sum_{k=n+1}^\infty\sum_{j=1}^n\frac{q_{kj}}{\lambda_j-\lambda_k}+
\sum_{k=1}^n\sum_{j=n+1}^\infty\frac{q_{kj}}{\lambda_k-\lambda_j}
\bigg]\bigg|^q\omega(x)\omega(y)dx\,dy\bigg]^{1/q}\le \\
%
\le\sum_{k=n+1}^\infty\sum_{j=1}^n
\frac{[\int\int|q_{kj}|^q\omega(x)\omega(y)dx\,dy]^{1/q}+
[\int\int|q_{jk}|^q\omega(x)\omega(y)dx\,dy]^{1/q}}
{\lambda_k-\lambda_j}=
\sum_{k=n+1}^\infty\sum_{j=1}^n\frac{O(1)}{(k-j)(\lambda_k-\lambda_j)}= \\
%
=O(1)\sum_{k=n+1}^\infty\int_0^n\frac{dx}{(k-x)^2(k+x)} =
O(1)\int_n^\infty\frac{\ln x dx}{x^2}+\frac{O(1)n}{(n+1)^2}+
O(1)n\int_{n+1}^\infty\frac{dx}{x^2(x-n)}=
O\bigg(\frac{\ln n}{n}\bigg).
\end{multline*}
Таким образом,
$$
\bigg\|\sum_{j=1}^n u_j(x)\overline{u_j(y)}-\sum_{j=1}^n v_j(x)
\overline{v_j(y)}\bigg\|_{L_q}=
O\bigg(\frac{\ln n}n\bigg)+O\bigg(\frac{\ln n}{n^{1/2}}\bigg)=
O\bigg(\frac{\ln n}{n^{1/2}}\bigg).\quad\square
$$

\begin{cornonum} Если $p\in D$ и $p(-1)=p(1)=0$, то
$$
\bigg|\sum_{j=1}^n [\mu_j-\lambda_j-(Pv_j,v_j)]\bigg|=
O\bigg(\frac{\ln n}{n^{1/2}}\bigg).
$$
\end{cornonum}

{\bf Доказательство.} Поскольку по теореме 1 из [6] имеет место неравенство
$$
\sum_{j=1}^n(Pu_j,u_j)\le\sum_{j=1}^n \mu_j-\lambda_j\le\sum_{j=1}^n(Pv_j,v_j),
$$
то
\begin{multline*}
\bigg|\sum_{j=1}^n [\mu_j-\lambda_j-(Pv_j,v_j)]\bigg|\le
\bigg|\sum_{j=1}^n[(Pu_j,u_j)-(Pv_j,v_j)]\bigg|= \\
%
=\bigg|\sum_{j=1}^n\bigg[\int p(x)u_j(x)\overline{u_j(x)}\omega(x)dx-
\int p(x)v_j(x)\overline{v_j(x)}\omega(x)dx\bigg]\bigg|\le \\
%
\le \max_{x\in[-1,1]}|p(x)|
\bigg\|\sum_{j=1}^n u_j(x)\overline{u_j(x)}-
\sum_{j=1}^n v_j(x)\overline{v_j(x)}\bigg\|_{L_1}
=O\bigg(\frac{\ln n}{n^{1/2}}\bigg).
\end{multline*}
