\Section{Определения и обозначения}
Обобщением функции типа Миттаг"--~Леффлера на случай двух переменных мы называем~[\citen{assimptE:bib:Vestnik2010apr}] целую функцию, определяемую степенным рядом
\begin{equation}
\label{assimptE__1 }
E_{\alpha, \beta }  (x, y; \mu)=\sum_{k, l =0}^{\infty }\frac{x^k  y^l  }{\Gamma \left(k\alpha +l\beta +\mu \right)},
\end{equation}
где $\mu$ "--- произвольный, вообще говоря, комплексный параметр.

Обозначим через $\gamma(\varepsilon; \theta)$ $(\varepsilon >0$, $0<\theta \le \pi )$ контур, пробегаемый в направлении неубывания $\arg \zeta$ и состоящий из следующих частей:
\begin{enumerate}
\item  луч $\arg \zeta =-\theta $, $\left|\zeta \right|\ge \varepsilon $;
\item  дуга $-\theta \le \arg \zeta \le \theta $ окружности $\left|\zeta \right|=\varepsilon $;
\item  луч $\arg \zeta =\theta $, $\left|\zeta \right|\ge \varepsilon $.
\end{enumerate}

\Section{Интегральные представления} 
Докажем несколько лемм об интегральных представлениях обобщения функции типа Миттаг"--~Леффлера~(\ref{assimptE__1 }) в различных областях
изменения её аргументов.
\begin{lemma}[1]
Пусть $\alpha,\,\beta \in (0, 2),$ $\alpha \beta <2,$ $\mu$ "--- любое комплексное число$,$ а $\theta$ удовлетворяет условию
\begin{equation}
\label{assimptE__4_}
\frac{\pi \alpha \beta }{2} <\theta \le \min \{\pi ,\pi \alpha \beta \}.
\end{equation}
Если $x\in G^{(-)} (\varepsilon _\alpha ; \theta_a ),$
$y\in G^{(-)} (\varepsilon _\beta  ;\theta _\beta ),$
где $\varepsilon _{\alpha } =\varepsilon ^{1/\beta} ,$
$\varepsilon _{\beta } =\varepsilon ^{1/ \alpha} $
и $\theta _{\alpha } =\theta /{\beta } ,$ $\theta _{\beta } =\theta /{\alpha } ,$
то имеет место следующее интегральное представление{\rm:}
\begin{equation}
\label{assimptE__5_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (\varepsilon; \theta)}
\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} }{\left(\zeta ^{1/ \alpha}  -y\right)\left(\zeta ^{1/\beta} -x\right)} d \zeta.
\end{equation}
\end{lemma}

\begin{proof}
Положим сначала $|x|<\varepsilon _{\alpha } $. С учётом того,
что $\varepsilon _{\alpha } =\varepsilon ^{1/\beta} =\bigl(\varepsilon _{\beta }^{\alpha } \bigr)^{1/\beta} =\varepsilon_\beta^{\alpha/\beta}$, очевидно, что 
\begin{equation*}
\mathop{\sup }\limits_{\zeta \in \gamma (\varepsilon_\beta; \theta_\beta)} \bigl|x\zeta ^{-\alpha/ \beta  } \bigr|<1.
\end{equation*}
Запишем функцию $E_{\alpha, \beta }  (x, y; \mu)$ в соответствии с определением (\ref{assimptE__1 }) и преобразуем, сворачивая один из рядов по определению функции типа Миттаг"--~Леффлера одного аргумента~[\citen{assimptE:bib:Djabrashyan}]:
\begin{multline*}
E_{\alpha, \beta }  (x, y; \mu)=\sum_{k=0}^{\infty }\sum _{l=0}^{\infty }\frac{x^k  y^l  }{\Gamma (\alpha k+\beta l+\mu )}=
\\
=\sum_{k=0}^{\infty }x^k  \sum _{l=0}^{\infty }\frac{y^l  }{\Gamma \left[\beta l+(\alpha k+\mu )\right]}=\sum _{k=0}^{\infty }x^k  E_{\beta } (y; \alpha k+l)  .
\end{multline*} 
В силу условий леммы можно воспользоваться известным интегральным представлением
для~$E_\beta (y; \alpha k+l) $ (см. \cite[формула (2.2)]{assimptE:bib:Djabrashyan}), причём в качестве параметров, определяющих контур $\gamma$, возьмём
введённые выше $\varepsilon_\beta$ и $\theta_\beta$, что допустимо в соответствии
с~(\ref{assimptE__4_}) и с учётом равенства $\theta_\beta =\theta /\alpha$.
Тогда, если $y\in G^{(-)} (\varepsilon _\beta  ;\theta _\beta )$, то
\begin{multline} 
\label{assimptE__7_}
E_{\alpha, \beta }  (x, y; \mu)=\sum _{k=0}^{\infty }x^k  E_{\beta } (y; \alpha k+l) =
\sum _{k=0}^{\infty }x^k   \frac{1}{2\pi i} \frac{1}{\beta } \int\limits_{\gamma (\varepsilon_\beta; \theta_\beta ) }
\frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1-\alpha k-\mu }{\beta } } }{\zeta -y} d \zeta=
\\
=\frac{1}{2\pi i} \frac{1}{\beta } \int\limits_{\gamma (\varepsilon_\beta; \theta_\beta ) }\frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1-\mu }{\beta } } }{\zeta -y} \left\{\sum _{k=0}^{\infty }\left(x\zeta^{-\alpha/\beta}  \right)^{k}  \right\}d \zeta=
\\
=\frac{1}{2\pi i} \frac{1}{\beta } \int\limits_{\gamma (\varepsilon_\beta; \theta_\beta ) }\frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1+\alpha -\mu }{\beta } } }{\left(\zeta -y\right)\left(\zeta^{\alpha /\beta}  -x\right)} d \zeta.
\end{multline}
Преобразуем интегральное представление~(\ref{assimptE__7_}), приведя его к
интегралу по контуру $\gamma (\varepsilon; \theta)$:
\begin{multline*}
\frac{1}{2\pi i} \frac{1}{\beta } \!\!\! \int\limits_{\gamma (\varepsilon_\beta; \theta_\beta ) } \!\!\! \frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1+\alpha -\mu }{\beta } } }{\left(\zeta -y\right)\left(\zeta^{\alpha /\beta}  -x\right)} d\zeta  = 
\frac{1}{2\pi i} \frac{1}{\beta } \!\!\! \int\limits_{\gamma (\varepsilon; \theta)} \!\!\! \frac{e^{\left(\xi ^{1/ \alpha}  \right)^{1/\beta} } \left(\xi ^{1/ \alpha}  \right)^{\frac{1+\alpha -\mu }{\beta } } }{\left(\xi ^{1/ \alpha}  -y\right)\left(\xi ^{1/\beta} -x\right)}  \frac{1}{\alpha } \xi ^{\frac{1-\alpha }{\alpha } } d\xi =
\\
=\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (\varepsilon; \theta)}\frac{e^{\xi^{1/(\alpha \beta)}  } \xi ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} }{\left(\xi ^{1/ \alpha}  -y\right)\left(\xi ^{1/\beta} -x\right)}  d\xi .
\end{multline*}
%
%\opium

Полученный интеграл абсолютно сходится и является аналитической функцией от $x$ и $y$ при $x\in G^{(-)} (\varepsilon_\alpha; \theta_a)$, $y\in G^{(-)} (\varepsilon _\beta  ;\theta _\beta )$.

Очевидно, что круг $|x|<\varepsilon _{\alpha }$ содержится в области $G^{(-)} (\varepsilon_\alpha; \theta_\alpha) $
для всех значений $\theta_\alpha$ из промежутка $\left(\pi \alpha/ 2 , \min \{\pi, \pi \alpha \}\right)$.
Следовательно, по принципу аналитического продолжения представление \eqref{assimptE__5_} имеет
место всюду в области $G^{(-)} (\varepsilon_\alpha; \theta_\alpha) $.
\end{proof}

\begin{lemma}[2]
Пусть $\alpha,\,\beta \in (0, 2),$ $\alpha \beta <2,$ $\mu$ "---
любое комплексное число$,$ а $\theta$ удовлетворяет условию~\eqref{assimptE__4_}$.$
Если $x\in G^{(-)} (\varepsilon_\alpha; \theta_a)$,
$y\in G^{(+)} (\varepsilon _\beta  ;\theta _\beta ),$
где $\varepsilon _{\alpha } =\varepsilon ^{1/\beta} ,$
$\varepsilon _{\beta } =\varepsilon ^{1/ \alpha} $
и $\theta _{\alpha } =\theta/ \beta ,$
$\theta _{\beta } =\theta/ \alpha ,$ то имеет место следующее интегральное
представление:
\begin{equation}
\label{assimptE__9_}
E_{\alpha ,\beta }  (x, y; \mu)=\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (\varepsilon; \theta)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} }{\left(\zeta ^{1/ \alpha}  -y\right)\left(\zeta ^{1/\beta} -x\right)} d \zeta.
\end{equation}
\end{lemma}


\begin{proof}
По условию леммы точка $y$ лежит справа от контура $\gamma (\varepsilon _\beta  ;\theta _\beta )$, т.\,е. $y\in G^{(+)} (\varepsilon _\beta  ;\theta _\beta )$.
Тогда для любого $\varepsilon _{\beta 1} >|y|$ очевидно  $y\in G^{(-)} (\varepsilon_{\beta 1}; \theta _\beta) $,
а $x\in G^{(-)} (\varepsilon_{\alpha 1}; \theta_a) $,
потому по формуле \eqref{assimptE__7_} имеем представление 
\begin{equation}
\label{assimptE__11_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{2\pi i} \frac{1}{\beta } \int\limits_{\gamma \left(\varepsilon _{\beta 1};\,\theta _{\beta } \right)}\frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1+\alpha -\mu }{\beta } } }{\left(\zeta -y\right)\left(\zeta^{\alpha /\beta}  -x\right)} d \zeta .
\end{equation}

С другой стороны, если
$\varepsilon _{\beta } <|y|<\varepsilon _{\beta 1}$,
$\left|\arg y\right|<\theta _{\beta } $, то по теореме Коши справедлива формула
\begin{equation}
\label{assimptE__12_}
\frac{1}{2\pi i} \frac{1}{\beta } \int\limits_{\gamma (\varepsilon_{\beta 1}; \theta _\beta) -\gamma (\varepsilon _\beta  ;\theta _\beta )}\frac{e^{\zeta ^{1/\beta}} \zeta ^{\frac{1+\alpha -\mu }{\beta } } }{\left(\zeta -y\right)\left(\zeta^{\alpha /\beta}  -x\right)} d \zeta =\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} .
\end{equation}

Из \eqref{assimptE__11_} и \eqref{assimptE__12_} немедленно получается представление \eqref{assimptE__9_}.
\end{proof}

\begin{lemma}[3]
Пусть $\alpha,\,\beta \in (0, 2),$ $\alpha \beta <2,$ $\mu$ "---
любое комплексное число$,$ а $\theta$ удовлетворяет условию~\eqref{assimptE__4_}$.$
Если $x\in G^{(+)} (\varepsilon_\alpha; \theta_a),$
$y\in G^{(+)} (\varepsilon _\beta  ;\theta _\beta ),$
где $\varepsilon _{\alpha } =\varepsilon ^{1/\beta} ,$
$\varepsilon _{\beta } =\varepsilon ^{1/ \alpha} $ и
$\theta _{\alpha } =\theta/ \beta ,$ $\theta _{\beta } =\theta/ \alpha ,$
то имеет место следующее интегральное представление{\rm:}
\begin{multline}
\label{assimptE__15_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y} +\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} + \\ + \frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (\varepsilon; \theta)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} }{\left(\zeta ^{1/ \alpha}  -y\right)\left(\zeta ^{1/\beta} -x\right)} d \zeta .
\end{multline}
\end{lemma}


\begin{proof}
По условию леммы точки \textit{x} и \textit{y} лежат по правую сторону от контуров
$\gamma \left(\varepsilon _{\alpha } ;\theta _{\beta } \right)$ и
$\gamma (\varepsilon _\beta  ;\theta _\beta )$ соответственно, т.\,е.
$x\in G^{(+)} (\varepsilon_\alpha; \theta_a)$,
$y\in G^{(+)} (\varepsilon _\beta  ;\theta _\beta )$.
Параметрам $\varepsilon _{\alpha } $ и $\varepsilon _{\beta } $ соответствует $\varepsilon $.
Выберем такое $\varepsilon _{1}$ ($\varepsilon _{1} >\varepsilon $), чтобы одна
из координат оказалась справа от контура, а другая слева (такое всегда возможно,
если $x^{\beta } \ne y^{\alpha }$). Для определённости пусть
$x\in G^{(-)} (\varepsilon_{\alpha 1}; \theta_a) $,
$y\in G^{(+)} (\varepsilon_{\beta 1}; \theta _\beta) $
(т.\,е. $x<y$). Тогда по формуле \eqref{assimptE__9_} из леммы 2 имеет место интегральное представление
\begin{equation}
\label{assimptE__16_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma \left(\varepsilon _{1} ;\theta \right)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} }{\left(\zeta ^{1/ \alpha}  -y\right)\left(\zeta ^{1/\beta} -x\right)} d \zeta .
\end{equation}
Запишем интеграл в \eqref{assimptE__16_} в виде
\begin{equation*}
\frac{1}{2\pi i} \frac{1}{\alpha } \int\limits_{\gamma \left(\varepsilon _{\alpha 1} ;\theta _{\alpha } \right)}\frac{e^{\zeta ^{1/ \alpha}  } \zeta ^{\frac{1+\beta -\mu }{\alpha } } }{\left(\zeta^{\beta/ \alpha}  -y\right)\left(\zeta -x\right)} d \zeta .
\end{equation*}
Если $\varepsilon _{\alpha } <|x|<\varepsilon _{\alpha 1} $,
$|\arg x|<\theta _{\alpha }$, то по теореме Коши справедлива формула
\begin{equation}
\label{assimptE__17_}
\frac{1}{2\pi i} \frac{1}{\alpha } \int\limits_{\gamma \left(\varepsilon _{\alpha 1} ;\theta _{\alpha } \right)-\gamma (\varepsilon_\alpha; \theta_\alpha) }\frac{e^{\zeta ^{1/ \alpha}  } \zeta ^{\frac{1+\beta -\mu }{\alpha } } }{\left(\zeta^{\beta/ \alpha}  -y\right)\left(\zeta -x\right)} d \zeta =\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y} .
\end{equation}
Из \eqref{assimptE__16_} и \eqref{assimptE__17_} получается представление \eqref{assimptE__15_}.
\end{proof}

\Section{Асимптотические свойства} 
Особый интерес представляют асимптотические свойства функции
$E_{\alpha, \beta }  (x, y; \mu)$ при больших по модулю значениях
аргументов $|x|$ и $|y|$.

\smallskip

\begin{theorem}
Пусть $\alpha,\,\beta \in (0, 2),$ $\alpha \beta <2,$ $\mu$ "--- любое комплексное число$,$ $a$ "--- любое вещественное число$,$ удовлетворяющих условию
\begin{equation*}
\frac{\pi \alpha \beta }{2} <a\le \min \left\{\pi ,\pi \alpha \beta \right\}.
\end{equation*}
Тогда для любого целого $p\ge 1$ при $|x|\to \infty $ и~$|y|\to \infty $ справедливы следующие асимптотические формулы{\rm:}


\begin{itemize}
\item[\rm1)] при $|\arg x|\le a/\beta  $ и
$\left|\arg y\right|\le a/\alpha {:}$
\begin{multline}
\label{assimptE__19_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y} +\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +
\\
+\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{x^{-k} y^{-l} }{\Gamma (\mu -k\alpha -l\beta)} + o\bigl(|xy|^{-1}|y|^{-p_\alpha}\bigr) + o\bigl(|xy|^{-1}|x|^{-p_\beta}\bigr);
\end{multline}


\item[\rm2)] при $|\arg x|\le a/\beta $ и $a/\alpha  <\left|\arg y\right|\le \pi {:}$
\begin{multline}
\label{assimptE__20_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y}+
\\
+\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{x^{-k} y^{-l} }{\Gamma (\mu -k\alpha -l\beta)} + o\left(|xy|^{-1}|y|^{-p_\alpha}\right) + o\left(|xy|^{-1}|x|^{-p_\beta}\right);
\end{multline}
%
%\opium

\item[\rm3)] при $a/\beta  <|\arg x|\le \pi$ и $\left|\arg y\right|\le a/\alpha {:}$
\begin{multline}
\label{assimptE__21_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +
\\
+\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{x^{-k} y^{-l} }{\Gamma (\mu -k\alpha -l\beta)} + o\left(|xy|^{-1}|y|^{-p_\alpha}\right) + o\left(|xy|^{-1}|x|^{-p_\beta}\right);
\end{multline}



\item[\rm 4)] при $a/\beta  <|\arg x|\le \pi $ и $a/\alpha  <\left|\arg y\right|\le \pi {:}$
\begin{multline}
\label{assimptE__22_}
E_{\alpha, \beta }  (x, y; \mu)=\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{x^{-k} y^{-l} }{\Gamma (\mu -k\alpha -l\beta)}  + \\+ o\left(|xy|^{-1}|y|^{-p_\alpha}\right) + o\left(|xy|^{-1}|x|^{-p_\beta}\right).
\end{multline}

\end{itemize}

\end{theorem}
\begin{proof}
При ограничениях $|\arg x|\le a/\beta  $ и $\left|\arg y\right|\le a/\alpha $ число $b$ выбираем  так, чтобы
\begin{equation}
\label{assimptE__23_}
\frac{\pi \alpha \beta }{2} <a<b\le \min \left\{\pi ,\pi \alpha \beta \right\}.
\end{equation}

Нетрудно показать, что имеет место разложение
\begin{equation}
\label{assimptE__24_}
\frac{1}{\left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} = \sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{\zeta ^{\frac{l-1}{\alpha } +\frac{k-1}{\beta } } }{y^l  x^k  }   +\frac{x^{p_{\beta } } \zeta ^{\frac{p_{\alpha } }{\alpha } } +y^{p_{\alpha } } \zeta ^{\frac{p_{\beta } }{\beta } } -\zeta ^{\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} .
\end{equation}

Воспользуемся формулой \eqref{assimptE__15_} из леммы 3. В \eqref{assimptE__24_} положим
$\varepsilon =1$, тогда справа от контура $\gamma (1; b)$ (т.е.
в области $G^{(+)} (1; b)$) для функции
$E_{\alpha, \beta }  (x, y; \mu)$ с учётом  \eqref{assimptE__24_} получаем
представление
\begin{multline}
\label{assimptE__25_}
E_{\alpha ,\beta }  (x, y; \mu)=\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y} +\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +
\\
+\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{1}{2\pi i} \frac{1}{\alpha \beta } \biggl(\int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{l-1}{\alpha } +\frac{k-1}{\beta } } d\zeta  \biggr)x^{-k} y^{-l}+
\\
+\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} \frac{x^{p_{\beta } } \zeta ^{\frac{p_{\alpha } }{\alpha } } +y^{p_{\alpha } } \zeta ^{\frac{p_{\beta } }{\beta } } -\zeta ^{\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d \zeta .
\end{multline}
Согласно формуле Ханкеля для гамма-функции~[\citen{assimptE:bib:Markushevich}]
\begin{equation*}
\frac{1}{\Gamma \left(s\right)} =\frac{1}{2\pi i} \int _{\gamma (\varepsilon; \tau) }e^{u} u^{-s} du ,
\end{equation*}
имеем 
\begin{multline*}
\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{l-1}{\alpha } +\frac{k-1}{\beta } } d\zeta  = \\ = \frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1-\mu }{\alpha \beta } -1+\frac{l}{\alpha } +\frac{k}{\beta } } d\zeta  =
\\
=\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{-\frac{1}{\alpha \beta } \left(\mu -l\beta -\alpha k\right)+\frac{1}{\alpha \beta } -1} d\zeta  =\frac{1}{\Gamma (\mu -k\alpha -l\beta)} .
\end{multline*}
Отсюда \eqref{assimptE__25_} в силу \eqref{assimptE__23_} переписываем в виде
\begin{multline}
\label{assimptE__26_}
E_{\alpha, \beta }  (x, y; \mu)=\frac{1}{\alpha } \frac{e^{x^{1/ \alpha}  } x^{\frac{1+\beta -\mu }{\alpha } } }{x^{\beta /\alpha} -y} +\frac{1}{\beta } \frac{e^{y^{1/\beta} } y^{\frac{1+\alpha -\mu }{\beta } } }{y^{\alpha/\beta}  -x} +\sum _{k=1}^{p_{\beta } }\sum _{l=1}^{p_{\alpha } }\frac{x^{-k} y^{-l} }{\Gamma (\mu -k\alpha -l\beta)} +
\\
+\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int _{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} \frac{x^{p_{\beta } } \zeta ^{\frac{p_{\alpha } }{\alpha } } +y^{p_{\alpha } } \zeta ^{\frac{p_{\beta } }{\beta } } -\zeta ^{\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d \zeta.
\end{multline}

Рассмотрим последнее слагаемое в формуле \eqref{assimptE__26_}:
\begin{multline}
\label{assimptE__27_}
\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1} \frac{x^{p_{\beta } } \zeta ^{\frac{p_{\alpha } }{\alpha } } +y^{p_{\alpha } } \zeta ^{\frac{p_{\beta } }{\beta } } -\zeta ^{\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d \zeta =
\\
=\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{p_{\alpha } }{\alpha } } }{y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d\zeta +
\\
+\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int _{\gamma (1; b)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d\zeta  -
\\
-\frac{1}{2\pi i} \frac{1}{\alpha \beta } \int\limits_{\gamma (1; b)}\frac{e^{\zeta ^{1/(\alpha \beta)} }   \zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } }{x^{p_{\beta } } y^{p_{\alpha } } \left(\zeta ^{1/\beta} -x\right)\left(\zeta ^{1/ \alpha}  -y\right)} d \zeta =I_{1} +I_{2} +I_{3} .
\end{multline}
Оценим каждый интеграл из \eqref{assimptE__27_} при больших $|x|$ и $|y|$,
предполагая, что $|\arg x|\le a/\beta $ и
$\left|\arg y\right|\le a/\alpha $.

Заметим,  что когда $|\arg x|\le a/\beta $ и $|x|$ достаточно велико, то 
\begin{equation*}
\min\limits_{\zeta \in \gamma (1; b)} \left|\zeta ^{1/\beta} -x\right|=|x|\sin \left(b/\beta -a/\beta  \right)=|x|\sin \Bigl(\frac{b-a}{\beta } \Bigr).
\end{equation*}
Аналогично для $|\arg y|\le a/\alpha  $ и достаточно большого $|y|$ имеем
\begin{equation*}
\min\limits_{\zeta \in \gamma (1; b)} \left|\zeta ^{1/ \alpha}  -y\right|=|y|\sin \left(b/\alpha -a/\alpha  \right)=|y|\sin \Bigl(\frac{b-a}{\alpha } \Bigr).
\end{equation*} 
Поэтому для больших $|x|$ и $|y|$ при $|\arg x|\le a/\beta $ и $\left|\arg y\right|\le a/\alpha  $ получаем оценку
\begin{equation*}
\left|I_{1} \right|\le \frac{|x|^{-1} |y|^{-p_{\alpha } -1} }{2\pi \alpha \beta \sin \left(\frac{b-a}{\beta } \right)\sin \left(\frac{b-a}{\alpha } \right)} \int\limits_{\gamma (1; b)}\left|e^{\zeta ^{1/(\alpha \beta)} }   \right|\left|\zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{p_{\alpha } }{\alpha } } \right|\left|d\zeta \right|,
\end{equation*}
причём интеграл справа сходится, так как на лучах
$\arg \zeta =\pm b$ ($|\zeta |\ge 1$), входящих в~состав контура $\gamma (1; b)$, выполняется равенство
\begin{equation*}
\Bigl|e^{\zeta ^{1/(\alpha \beta)} }   \Bigr|=\exp \left\{\left|\zeta \right|^{\frac{1}{\alpha \beta } } \cos \frac{b}{\alpha \beta } \right\},
\end{equation*}
а  согласно условию \eqref{assimptE__23_}, $\cos \frac{b}{\alpha \beta } <0$.
Итак, $I_{1} =o\left(|xy|^{-1}|y|^{-p_\alpha}\right)$. 
Аналогично $I_{2} =o\left(|xy|^{-1}|x|^{-p_\beta}\right)$.

Обратимся к третьему слагаемому из \eqref{assimptE__27_}:
\begin{equation*}
\left|I_{3} \right|\le \frac{|x|^{-p_{\beta } -1} |y|^{-p_{\alpha } -1} }{2\pi \alpha \beta \sin \left(\frac{b-a}{\beta } \right)\sin \left(\frac{b-a}{\alpha } \right)} \int _{\gamma (1; b)}\left|e^{\zeta ^{1/(\alpha \beta)} }   \right|\left|\zeta ^{\frac{1+\alpha +\beta -\mu }{\alpha \beta } -1+\frac{p_{\alpha } }{\alpha } +\frac{p_{\beta } }{\beta } } \right|\left|d\zeta \right|.
\end{equation*}
Очевидно, что $I_{3} =o\left(|xy|^{-1}|y|^{-p_\alpha}|x|^{-p_\beta}\right)$.

Итак, $I=o\left(|xy|^{-1}|y|^{-p_\alpha}\right) + o\left(|xy|^{-1}|x|^{-p_\beta}\right)$.

Доказательства пунктов 2), 3) и 4) проводятся по той же схеме, что приведена выше.
\end{proof}
