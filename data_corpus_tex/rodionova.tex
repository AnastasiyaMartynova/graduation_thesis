\title{Задача с интегральным условием для одного пространственного уравнения гиперболического типа, вырождающегося на координатных плоскостях}
{Задача с интегральным условием для одного пространственного уравнения гиперболического \dots}

\author{И.\,Н.~Родионова}{Р\,о\,д\,и\,о\,н\,о\,в\,а~И.\,Н.}

\institute{\samgu.}


\email{E-mail}{mvdolg@ssu.samara.ru}

\otherinf{\fullauthor{Ирина Николаевна Родионова}\regalia{к.ф.-м.н., доцент} \position{доцент} \dept{каф. математики и бизнес"=информатики; научно"=исследовательская лаборатория математической физики} }

\engtitle{The problem with integral condition for one space analog of hyperbolic type equation degenerated on~coordinate planes}

\engauthor{I.\,N.~Rodionova}{R\,o\,d\,i\,o\,n\,o\,v\,a~I.\,N.}

\otherenginfm{\fullauthor{Irina N.~Rodionova} \regalia{Ph.\,D. (Phys. \& Math.)} \position{Associate Professor} 
 \dept{Dept. of Mathematics \& Business Informatics; Research Lab. of Mathematical Physics}
}

\engabstract{For the full equation of the third order in a three-dimensional Euclidean space the boundary value problem with interface on non-characteristic plane in the area limited by planes of a singularity of the equation factors is solved.}



\date{01/X/2010}
\revisiondate{21/II/2011}

\maketitle


Рассмотрим на множестве $H=H_1\cup H_2$ уравнение
\begin{equation}
U_{xyz}+\frac{\gamma}{z}U_{xy}+\frac{\beta}{y}U_{xz}+\frac{\alpha}{x}U_{yz}+
\frac{\beta\gamma}{yz}U_x+\frac{\alpha\gamma}{xz}U_y+\frac{\alpha\beta}{xy}U_z+ \frac{\alpha\beta\gamma}{xyz}U=0,
\label{rod1}
\end{equation}
где %$\alpha$, $\beta$, $\gamma$ "--- $\text{const}$; 
$ \alpha$,
$\beta$, $\gamma \in (0, 1)$; 
$H_1=\{(x,\,y,\,z) : 0<y<x<h,$ $0<z<+\infty\}$,
$H_2=\{(x,\,y,\,z) : 0<y<x<h,$ $ 0<z<+\infty\}$, %$h$ "--- $\text{const}$, 
$h>0$.

В соответствии с классификацией
%%(см., например, монографию
\cite{gegalov}
%%и ссылки в ней)
уравнений со старшими частными производными ${\partial^N}/({\partial x_1 \dots \partial x_N})$ уравнение (\ref{rod1}) относится к уравнениям гиперболического типа.

\begin{task}
Найти на множестве $H$ непрерывное решение уравнения~\eqref{rod1} с~усло\-виями
\begin{gather}
\int _0^yU(t,\,y,\,z)dt=\psi(y,\,z), \quad (y,\,z)\in{\cal D}_1=\{0<y<h,\,0<z<+\infty\}, \label{rod2}\\
\bigl(z^\gamma U(x,\,y,\,z)\bigr)_{z=0}=f_1(x,\,y), \quad (x,\,y)\in{\cal D}_2=\{0<x<y<h\}, \label{rod3}\\
\bigl(z^\gamma U(x,\,y,\,z)\bigr)_{z=0}=f_2(x,\,y), \quad (x,\,y)\in{\cal D}_3=\{0<y<x<h\}, \label{rod4}\\
\bigl(y^\beta U(x,\,y,\,z)\bigr)_{y=0}=\varphi(x,z), \quad (x,\,z)\in{\cal D}_4=\{0<x<h,\, 0<z<+\infty\} \label{rod5}
\end{gather}
и условиями сопряжения на плоскости $x=y${\rm:}
\begin{gather}
\lim_{y\to x+0} U(x,\,y,\,z)=\lim_{y\to x-0} U(x,\,y,\,z); \label{rod6}\\
\lim_{y \to x+0} x^{2\alpha}(U_x-U_y)=\lim_{y\to x-0} [x^{2\beta}(U_x-U_y)]+ (x^{2\beta}U(x,x,z))'_x. \label{rod7}
\end{gather}
На заданные функции налагаются следующие требования: $f_1(x,\,y),$ $f_2(x,\,y),$
 $\varphi(x,\,z),$ $\psi(y,\,z)$ непрерывны в~своих областях определения вместе со своими смешанными частными
производными второго порядка{\rm;}
\begin{gather}
f_1(x,\,x)=f_2(x,\,x)=0, \quad 
\frac{\partial f_1}{\partial x}\Big|_{y=x} =
\frac{\partial f_2}{\partial x}\Big|_{y=x}, \quad
\frac{\partial f_1}{\partial y}\Big|_{y=x} =
\frac{\partial f_2}{\partial y}\Big|_{y=x} ; \label{rod9}\\
\notag \int _0^y f_1(t,\,y)dt=0, \quad 0\leq y\leq h.
\end{gather}
\end{task}

\begin{remark}
Простейшим примером функции $f_1(x,\,y)$, удовлетворяющей
указанным условиям, является
$$
f_1(x,\,y)=f_{11}(y)\bigl[x^\gamma(y-x)^\delta-x^\delta(y-x)^\gamma\bigr],
\quad \gamma>1, \quad \delta>1, \quad f_{11}(y)\in
C^{(1)}_{[0,h]}.
$$
\end{remark}

Полагая для определённости $2\alpha-\beta>0$, потребуем
представления
\begin{equation}
\psi(y,\,z)=y\psi^*(y,z), \quad \varphi(x,\,z)=x\varphi^*(x,\,z),
\label{rod10}
\end{equation}
где 
$\psi^*$, $\varphi^*$ удовлетворяют условиям непрерывности в своих областях определения вместе со своими смешанными частными производными второго порядка, причём
$\varphi(y,\,0)=\psi(x,\,0)=0$. Для решения поставленной задачи
возьмём за основу полученное методом Римана в~работе~\cite{rodlit1} решение задачи Коши"--~Гурса для
уравнения~(\ref{rod1}) в области $H_1$ при выполнении условий
\begin{gather*}
U(x,\,x,\,z)=\tau_1(x,\,z),\quad
(U_x-U_y)_{x=y}=\nu_1(x,\,z), \quad (x,\,z)\in {\cal D}_4
\end{gather*}
и~(\ref{rod3}). Оно имеет вид~\cite{rodlit1}:
\begin{multline}
U(x,\,y,\,z)=z^{-\gamma}f_1(x,\,y)+\frac{1}{2}\Bigl[\Bigl(\frac{y}{x}\Bigr)^\alpha\tau_1(y,\,z)+
\Bigl(\frac{x}{y}\Bigr)^\beta\tau_1(x,\,z)\Bigr]- \label{rod11}
\\
-x^{-\alpha}y^{-\beta}\int _x^yt^{\alpha+\beta}\nu_1(t,\,z)dt+(\beta-\alpha)
x^{-\alpha}y^{-\beta}\int _x^yt^{\alpha+\beta-1}\tau_1(t,\,z)dt.
\end{multline}

С помощью интегрального представления функции $\tau_1$ упростим формулу~(\ref{rod11}), положив
\begin{equation}
\tau_1(x,\,z)=\int _0^xT_1(s,\,z)\left(\frac{s}{x}\right)^{2\beta}
ds, \label{rod12}
\end{equation}
где $T_1(x, z)$ "--- новая функция, непрерывная в ${\cal D}_4$ вместе со
своей производной ${\partial T_1}/{\partial z}$;
$x^{2\beta}T_1(x,\,z)$ интегрируема по $x$ в $[0,\,h]$ при любом
$z\in [0,\,+\infty)$,
\begin{equation}
T_1(x,\,0)=0. \label{rod13}
\end{equation}
 Подставляя в функцию~(\ref{rod11}) вместо $\tau_1$ её интегральное представление~(\ref{rod12}), после некоторых преобразований получаем
\begin{gather}
U(x,y,z){=}z^{-\gamma}f_1(x,y){+}\frac{1}{x^\beta
y^\beta}\int _0^xT_1(s,z)s^{2\beta}ds{+}\frac{1}{x^\alpha
y^\beta}\int _x^yN_1(s,z)s^{\alpha{+}\beta}ds, \label{rod14}\\
N_1(s,\,z)=\frac{1}{2}[T_1(s,\,z)-\nu_1(s,\,z)]. \label{rod15}
\end{gather}

Аналогичными рассуждениями получаем в области $H_2$ решение задачи
Коши"--~Гурса с условиями
\begin{gather*}
U(x,\,x,\,z)=\tau_2(x,\,z),\quad
(U_x-U_y)_{x=y}=\nu_2(x,\,z), \quad (x,\,z)\in {\cal D}_4
\end{gather*}
и~(\ref{rod4}), которое, при  представлении
\begin{equation}
\tau_2(x,\,z)=\int _0^xT_2(s,\,z)\left(\frac{s}{x}\right)^{2\alpha}
ds, \label{rod16}
\end{equation}
имеет вид
\begin{multline}
U(x,y,z)=\frac{1}{x^\alpha
y^\alpha}\int _0^xT_2(s,z)s^{2\alpha}ds+  \frac{1}{x^\alpha
y^\beta}\int _y^xN_2(s,z)s^{\alpha{+}\beta}ds+ \\ + z^{-\gamma}f_2(x,y),
\label{rod17}
\end{multline}
\begin{gather}
N_2(s,\,z)=\frac{1}{2}[T_2(s,\,z)+\nu_2(s,\,z)], \label{rod18}
\end{gather}
где $T_2(x, z)$ вводится по  аналогии с $T_1 (x, z)$ и удовлетворяет условию $T_2(x, 0)= 0$.

Функции~(\ref{rod14}), (\ref{rod17}) подчиним условиям~(\ref{rod2}), (\ref{rod5}) соответственно, из которых находим
\begin{gather}
N_2(x,\,z)x^{\alpha+\beta}=-\bigl[x^\alpha\varphi(x,z)\bigr]^p_x,
%\label{rod19}
\notag
\\
\int _0^yT_1(s,\,z)s^{2\beta}ds+\frac{N_1(y,\,z)y^{1+2\beta}}{1-\alpha}=
y^\beta \bigl[y^\beta\psi(y,\,z)\bigr]'_y . \label{rod20}
\end{gather}

Условия сопряжения~(\ref{rod7}), соотношения~(\ref{rod15}),
(\ref{rod18}) приводят к интегральному уравнению
$$
x^{2\beta}T_1(x,\,z)+\frac{2(1-\alpha)}{x}\int _0^xT_1(s,\,z)s^{2\beta}ds=
\frac{{\cal G}(x,\,z)}{x},
$$
где
\begin{equation*}
{\cal
G}(x,\,z)=\frac{x^{\alpha-\beta+1}[x^\alpha\varphi(x,\,z)]'_x}{1-\alpha}
+ x^\beta[x^\beta\psi(x,\,z)]'_x, %\label{rod21}
\end{equation*}
единственное решение которого получено методом последовательных приближений:
\begin{equation}
T_1(x,\,z)x^{2\beta}=\frac{{\cal
G}(x,\,z)}{x}-\frac{2(1-\alpha)}{x}\int _0^x\frac{{\cal
G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt,
\label{rod22}
\end{equation}
при этом \eqref{rod22}
удовлетворяет условиям~(\ref{rod13}).
Из условий сопряжения~(\ref{rod6}), представлений~(\ref{rod12}), (\ref{rod16}), а также формулы~(\ref{rod22}) находим
\begin{equation}
\int _0^xT_2(s,\,z)s^{2\alpha}ds=x^{2\alpha-2\beta}\int _0^x\frac{{\cal
G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt.
\label{rod23}
\end{equation}
Из~(\ref{rod20}) имеем
\begin{equation}
N_1(x,\,z)x^{\alpha+\beta}=(1-\alpha)x^{\alpha-\beta-1}\Bigl[x^\beta(x^\beta\psi(x,\,z))'_x-
\frac{{\cal G}(t,\,z)}{t}
\Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt\Bigr]. \label{rod24}
\end{equation}

Подставляя найденные значения $N_i$, $T_i$ в
формулы~(\ref{rod11}), (\ref{rod14}) получаем в явном виде решение
поставленной задачи:
\begin{multline}
U(x,\,y,\,z)=z^{-\gamma}f_1(x,\,y)+\frac{1}{x^\beta y^\beta}
\int _0^x\frac{{\cal G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt+\label{rod25}\\
+ \frac{1-\alpha}{x^\alpha y^\beta}\int _x^ys^{\alpha-1}\bigl[s^\beta\psi(s,\,z)\bigr]'_sds- \\
-
\frac{1-\alpha}{(3\alpha-\beta-2)x^\alpha y^\beta}
\biggl[y^{\alpha-\beta}\int _0^y\frac{{\cal G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt-\\
- \int _x^y{\cal
G}(t,\,z)t^{\alpha-\beta-1}dt-x^{\alpha-\beta}\int _0^x\frac{{\cal
G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt\biggr]
\end{multline}
в области $H_1$ и
\begin{multline}
U(x,\,y,\,z)=
\frac{y^{\alpha-2\beta}}{x^\alpha}\int _0^y\frac{{\cal
G}(t,\,z)}{t} \Bigl(\frac{t}{x}\Bigr)^{2(1-\alpha)}dt+
z^{-\gamma}f_2(x,\,y)+\\
+ \frac{1}{x^\alpha
y^\beta}\bigl[x^\alpha\varphi(x,\,z)-y^\alpha\varphi(y,\,z)\bigr]
\label{rod26}
\end{multline}
 в области $H_2$.


Проверкой показано, что при  выполнении условий непрерывности, налагаемых на заданные функции, и условий~(\ref{rod9}),
(\ref{rod10}) функция, определяемая формулами~(\ref{rod25}),
(\ref{rod26}), является решением поставленной задачи.
Единственность следует из единственности решения задачи
Коши"--~Гурса, полученного методом Римана, и единственности решения
интегрального уравнения, к которому свелась задача.

При ограничениях, налагаемых на коэффициенты уравнения и данные
задачи, решение ее непрерывно на множестве
$H\cup\{(x,\,y,\,z):0<x=y<h,$ $0<z<+\infty\}$, а на координатных
плоскостях, являющихся частью границы множества $H$, обращается в
бесконечность порядка меньше единицы.
