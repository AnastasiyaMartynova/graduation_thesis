\ruabstract{Для уравнения с частной дробной производной Римана"--~Лиувилля исследована однозначная разрешимость задачи с~обобщенным оператором дробного интегро"=дифференцирования в~краевом условии. 
Теорема единственности решения поставленной задачи доказана на основании принципа экстремума для нелокального параболического уравнения и принципа экстремума для операторов дробного дифференцирования в~смысле Римана"--~Лиувилля. 
Доказательство существования решения эквивалентно сводится к~вопросу разрешимости дифференциального уравнения дробного порядка. 
Решение рассматриваемой задачи получено в~явном виде.}


\enabstract{The unique solvability is investigated for the problem of equation with partial fractional derivative of Riemann--Liouville and boundary condition that contains the generalized operator of fractional integro-differentiation. The uniqueness theorem for the solution of the problem is proved on the basis of the principle of optimality for a nonlocal parabolic equation and the principle of extremum for the operators of fractional differentiation in the sense of Riemann--Liouville. The proof of the existence of solutions is equivalent to the problem of solvability of differential equations of fractional order. The solution is obtained in explicit form.}

\Section{Постановка задачи}
Рассмотрим уравнение
\begin{equation}
\label{taras:eq1}
\left\{\begin{array} {ll}
u_{xx}-D_{0+,y}^{\alpha} u=0, & y>0, \\
(-y)^{m} u_{xx}-u_{yy}+a (-y)^{\frac m {2}-1}u_{x}=0, & y<0,
\end{array} \right.
\end{equation}
где  $D^{ \alpha}_{0+,y} $ "--- частная дробная производная Римана"--~Лиувилля от функции $u(x, y)$ порядка $ \alpha$ $({0< \alpha <1})$  \cite[с.~42, 44]{taras:1}:
\begin{equation}
\label{taras:eq2}
\left(D^{ \alpha}_{0+,y} u \right)(x,y) = 
\Bigl(\frac{ \partial}{ \partial y} \Bigr) \frac{1}{\Gamma (1- \alpha)}  \int  ^{y}_{0} \frac{u(x,t) \,dt} {(y-t)^{ \alpha}},  \quad 0< \alpha <1, \quad  y>0,
\end{equation}
$a$ "--- вещественная постоянная, 
$m > 2$, 
в~конечной области $\Omega$, ограниченной отрезками $AA_{0}$, $BB_{0}$, $A_{0}B_{0}$ прямых $x=0$, $x=1$, $y=1$ соответственно, лежащих в полуплоскости $y>0$, и характеристиками
\[
AC : x-\frac{2}{m+2}  (-y)^{\frac {m+2}{2}}=0, \quad  BC: x+\frac{2}{m+2}  (-y)^{\frac{m+2}{2}}=1
\]
уравнения \eqref{taras:eq1} в полуплоскости $y<0$.


Пусть $\Omega_{1}= \Omega \cap (y>0)$, $\Omega_{2}= \Omega \cap (y<0)$, $I \equiv AB$ "--- единичный интервал $0<x<1$ прямой $y=0$.

\smallskip

\begin{newthm}{Задача}Найти решение $u(x,y)$ уравнения \eqref{taras:eq1} в области $\Omega$ при $y \ne 0,$ удовлетворяющее краевым условиям
\begin{gather}
\label{taras:eq3}
u(0, y) = \varphi_{1}(y), \quad  u(1, y) = \varphi_{2}(y), \quad  0 \leqslant y \leqslant 1,
\\
\label{taras:eq4}
A_{1} \bigl( I^{ -\beta^{*}, 0, \beta^{*}+\beta-1}_{0+} u[ \Theta_{0}(t)] \bigr) (x)+A_{2} u(x, 0)= A_{3} g(x),
\end{gather}
а также условиям сопряжения
\begin{gather}
\label{taras:eq5}
\lim_{y \to 0+} y^{1-\alpha} u(x,y) = \lim_{y \to 0-} u(x,y), \quad  \forall \, x \in \overline{I},
\\
\label{taras:eq6}
\lim_{y \to 0+} y^{1-\alpha} \left( y^{1-\alpha}  u(x,y) \right)_{y} = \lim_{y \to 0-} u_{y}(x,y), \quad  \forall \, x \in  I,
\end{gather}
где $A_{i}$ $(i=1, 2, 3)$ "--- действительные константы такие$,$ что
\begin{equation}
\label{taras:eq7}
-\frac{\Gamma(\beta) A_{2}}{\Gamma(\beta^{*}+\beta)}< A_{1}<0 \quad \text{либо} \quad 0< A_{1}< -\frac{\Gamma(\beta) A_{2}}{\Gamma(\beta^{*}+\beta)},
\end{equation}
$\varphi_{i}(y)$ $(i=1, 2),$ $g(x)$ "--- заданные функ\-ции такие$,$ что 
\begin{gather}
\label{taras:eq8}
g(x) \in C^{1}(\overline{I}) \cap C^{3}( I),
\\
\label{taras:eq9}
\varphi_{1}(0)=  \varphi_{2}(0) =0,
\\
\label{taras:eq10}
y^{1- \alpha}\varphi_{1}(y), \, \ \  y^{1- \alpha}\varphi_{2}(y) \in C ( [0, 1]),
\end{gather}
$ \Theta_{0}(x)$ "--- точка пересечения характеристик уравнения \eqref{taras:eq1}$,$ выходящих
из точки $(x, 0)\in I,$ с характеристикой $AC;$
\[
\beta^{*}= \frac{m-2a}{2(m+2)}, \quad \beta =\frac{m+2a}{2(m+2)}, \quad
|a|< \frac{m}{2}, \quad 0< \beta^{*} < \frac{1}{2}, \quad 0< \beta < \frac{1}{2}.
\]
\end{newthm}

\smallskip

Будем искать решение $u(x,y)$ поставленной задачи в классе дважды дифференцируемых функций в области $\Omega$ таких, что
\begin{equation}
\label{taras:eq11}  
\begin{array}{l}
y^{1-\alpha} u(x,y) \in C( \overline{\Omega}_{1}), \quad u(x,y) \in C( \overline{\Omega}_{2}), \\
y^{1-\alpha} \left( y^{1-\alpha}  u \right)_{y} \in C( \Omega_{1} \cup \{ (x,y): 0<x<1, y=0 \}), \\
u_{xx} \in C(\Omega_{1} \cup \Omega_{2}), \quad u_{yy} \in C( \overline{\Omega}_{2}).
\end{array}
\end{equation}
$  (I^{ \alpha, \beta, \eta}_{0+}f)(x)$ "--- оператор обобщенного дробного интегро"=дифференцирования с~гипергеометрической функцией Гаусса $F(a, b;c;z)$, введенный М.~Сайго \cite{taras:2} (см. также \cite[с.~326--327]{taras:1}) и~имеющий при действительных $ \alpha$, $ \beta$, $ \eta$ и $x>0$ вид
\begin{equation}
\label{taras:eq3.1}
\bigl(I^{ \alpha, \beta, \eta}_{0+} f \bigr)(x)= \left\{
\begin{array}{l}
\displaystyle
\frac{x^{- \alpha - \beta}}{ \Gamma (\alpha)} \!\! \int  ^{x}_{0} \!\! (x-t)^{ \alpha -1} F \Bigl( \alpha + \beta, - \eta; \alpha; 1- \frac{t}{x}\Bigr) f(t) dt, \, \alpha >0, \\[5mm]
\displaystyle
\Bigl( \frac{d}{dx} \Bigr)^{n} \bigl(I^{ \alpha+n, \beta-n, \eta-n}_{0+} f\bigr)(x),  \ \alpha \le 0, \ n=[- \alpha]+1,
\end{array} \!\!\!\!
\right.
\end{equation}

Заметим, что если $ \alpha >0$, то справедливы формулы
\begin{equation}
\label{taras:eq4.1} {\begin{array}{l}
\bigl(I^{ \alpha, - \alpha, \eta}_{0+} f \bigr)(x)= \left(I^{ \alpha}_{0+} f \right)(x), \quad 
\bigl(I^{- \alpha, \alpha, \eta}_{0+} f \bigr)(x)= \left(D^{ \alpha}_{0+} f \right)(x),
\end{array}}
\end{equation}
в частности
\begin{equation}
\label{taras:eq5.1}
\big(I^{ 0, 0, \eta}_{0+} f \bigr)(x)= f(x), \quad \bigl(I^{ 0, 0, \eta}_{1-} f \bigr)(x)= f(x),
\end{equation}
где $\left(I^{ \alpha}_{0+} f \right)(x)$ и $\left(D^{ \alpha}_{0+} f \right)(x)$ "--- операторы дробного интегрирования и дифференцирования Римана"--~Лиувилля порядка $ \alpha>0$ \cite[с.~42, 44]{taras:1}:
\begin{equation}
\label{taras:eq6.1}  
\begin{array}{l}
\displaystyle
\bigl(I^{ \alpha}_{0+} f \bigr)(x)= \frac{1}{\Gamma (\alpha)} \int  ^{x}_{0} (x-t)^{ \alpha -1} f(t) dt, \quad \alpha >0, \ x>0, \\[3mm]
\displaystyle
\Bigl(D^{ \alpha}_{0+} f \Bigr)(x)= \Bigl(\frac{d}{dx} \Bigr)^{n} \frac{1}{\Gamma (n- \alpha)}  \int  ^{x}_{0} (x-t)^{ n-\alpha -1} f(t) dt, \quad \alpha > 0, \ n=[ \alpha]+1.
\end{array}
\end{equation}

Истоком настоящей задачи послужили публикации \cite{taras:3, taras:4}.

\Section{Единственность решения задачи}
Пусть существует решение исследуемой задачи. Введем обозначения
\begin{gather}
       \label{taras:eq15}
\lim_{y \to 0+} y^{1-\alpha} u(x,y) = \tau_{1}(x), \quad \lim_{y \to 0-} u(x,y)= \tau_{2}(x),
\\
       \label{taras:eq16}
\lim_{y \to 0+} y^{1-\alpha} \left( y^{1-\alpha}  u(x,y) \right)_{y} = \nu_{1}(x), \quad \lim_{y \to 0-} u_{y}(x,y) =\nu_{2}(x).
\end{gather}

Известно (см., например, \cite{taras:5, taras:6}), что решение уравнения \eqref{taras:eq1} в области $\Omega_{1}$, удовлетворяющее условию \eqref{taras:eq3} и условию
\begin{equation}
       \label{taras:eq17}
\lim_{y \to 0+} y^{1-\alpha} u(x,y) = \tau_{1}(x), \quad x \in \overline{I},
\end{equation}
дается формулой
\begin{multline}
\label{taras:eq18}
 u(x,y) = \int _{0}^{y} \varphi_{1} (\eta) G_{\xi} (x, y, 0, \eta)  d \eta  - 
 \\
- \int _{0}^{y} \varphi_{2} (\eta) G_{\xi} (x, y, 1, \eta)  d \eta +
\Gamma (\alpha) \int _{0}^{1} \tau_{1}( \xi) G(x, y, \xi, 0)  d \xi ,
\end{multline}
где
\begin{multline}
\label{taras:eq19}
G(x,y, \xi, \eta) = \frac{(y- \eta)^{\beta-1}}{2} 
\sum_{n=- \infty}^{ \infty} 
\left\{ e_{1, \beta}^{1, \beta} 
\Bigl( - \frac{|x- \xi +2n|}{(y- \eta)^{\beta}} \Bigr) - \right.
\\
\left.
- e_{1, \beta}^{1, \beta} \Bigl( - \frac{|x+ \xi +2n|}{(y- \eta)^{\beta}} \Bigr) \right\}, 
\quad \beta = \frac{\alpha}{2},
\end{multline}
\[
e_{b, c}^{p, q}(z)= \sum_{k=0}^{ \infty} \frac{z^{k}}{ \Gamma(bk+p) \Gamma(q-ck)}, \quad b>c, \quad b>0, \quad z \in \mathbb C.
\]

Также известно (см., например, \cite{taras:5}), что функциональное соотношение между $\tau_{1}(x)$ и $\nu_{1}(x)$, принесенное из параболической части $\Omega_{1}$ на линию $y=0$, имеет вид
\begin{equation}
\label{taras:eq20}
\nu_{1}(x) = \frac{1}{ \Gamma(1+ \alpha)} \tau_{1}''(x).
\end{equation}


Найдем соотношение между $\tau_{2}(x)$ и $\nu_{2}(x)$, принесенное на линию $y=0$ из гиперболической части $\Omega_{2}$ области $\Omega$. 
Для этого воспользуемся решением задачи Коши, которое в области $\Omega_{2}$ имеет вид \cite{taras:7}
\begin{multline}
\label{taras:eq21} 
u(x,y) =  \frac{ \Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta^{*}) \Gamma( \beta)}  \times \\
\times 
\int _{0}^{1} \tau_{2} 
\Bigl( x
+ \frac{2}{m+2} (-y)^{ \frac{m+2}{2}} (2t-1) \Bigr) t^{ \beta -1} (1-t)^{ \beta^{*}-1} \, dt + \\
+ \frac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*}) \Gamma(1- \beta)}  (-y) \times \hspace{4cm}\phantom{1}
\\ 
\times \int_{0}^{1} \nu_{2} \Bigl( x
+ \frac{2}{m+2} (-y)^{ \frac{m+2}{2}} (2t-1) \Bigr) t^{- \beta^{*}} (1-t)^{- \beta}  dt.
\end{multline}

Используя формулу \eqref{taras:eq21} и соотношения \eqref{taras:eq3.1}, получим
\begin{multline}
u[ \Theta_{0}(x)] = \frac{ \Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta)} 
\bigl( I_{0+}^{ \beta^{*}, 0, \beta-1} \tau _{2} (t)\bigr)(x) +
\\
+\frac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} 
\Bigl( \frac{m+2}{4} \Bigr) ^{ \frac{2}{m+2}} 
\bigl(I_{0+}^{1- \beta, \beta^{*}+ \beta -1, \beta-1} \nu_{2}(t) \bigr)(x).
\end{multline}

Подставляя $u[ \Theta_{0}(x)]$ в краевое условие \eqref{taras:eq4}, учитывая \eqref{taras:eq4.1} и~\eqref{taras:eq5.1}, получим
\begin{multline}
\label{taras:eq22} 
\Bigl(  A_{1} \frac{\Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta)} + A_{2} \Bigr) \tau_{2}(x)+ 
\\
+
A_{1} \frac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} \Bigl( \frac{m+2}{4} \Bigr)^{\frac{2}{m+2}} 
\bigl( I_{0+}^{1- \beta-\beta^{*}} \nu_{2} (t)\bigr)  (x)=A_{3} g(x).

Применив к обеим частям полученного равенства оператор $D_{0+}^{ 1- \beta-\beta^{*}}$, с~учетом
\[
\left( D_{0+}^{\alpha} (I_{0+}^{\alpha} f)(t) \right)(x)= f(x), \quad \alpha>0
\]

Рассмотрим соответствующую однородную задачу: $g(x)=0$.

Выразим из последнего выражения $\nu_{2}(x)$, в результате получим
\begin{equation}
\label{taras:eq23}
\nu_{2}(x) = \lambda_{1} \bigl( D_{0+}^{ 1- \beta-\beta^{*}} \tau_{2}(t) \bigr)(x),
\end{equation}
где
\[
\lambda_{1}=
-\frac{ A_{1}\dfrac{\Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta)} +A_{2}}{A_{1} \dfrac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} \Bigl( \dfrac{m+2}{4} \Bigr)^{\frac{2}{m+2} }}.
\]

Докажем две леммы.

\hypertarget{taras:l1}{}

\smallskip

\begin{lemma}[1]Если функция $\tau_{1}(x)$ достигает положительного максимума $($отрицательного минимума$)$ на отрезке $[0,  1]$ в точке  $x=x_{0}$ $(0<x_{0}<1),$ то $\nu_{1}(x_{0}) \le0$ $(\nu_{1}(x_{0}) \ge0).$
\end{lemma}

\smallskip

\hypertarget{taras:l2}{}

\begin{lemma}[2]Если функция $\tau_{2}(x)$  достигает положительного максимума $($отрицательного минимума$)$ на отрезке $[0, 1]$ в точке  $x=x_{0}$ $(0<x_{0}<1)$ и~выполняются условия $g(x)=0$ и \eqref{taras:eq7}$,$ то $\nu_{2}(x_{0}) >0$ $(\nu_{2}(x_{0}) <0).$
\end{lemma}

\smallskip
Справедливость леммы~\hyperlink{taras:l1}{1} непосредственно следует из соотношения \eqref{taras:eq20}.

\smallskip

Справедливость леммы~\hyperlink{taras:l2}{2} вытекает из соотношения \eqref{taras:eq23}, условий \eqref{taras:eq7} \linebreak и~$g(x)=0$ и~свойства строгой положительности (строгой отрицательности) производной дробного порядка 
$\bigl( D_{0+}^{ \gamma} \tau_{2} (t) \bigr) (x)$ в точке положительного максимума (отрицательного минимума) 
\cite[с.~105]{taras:8}.



Справедлива следующая теорема.

\smallskip

\begin{theorem}[1]Пусть выполняются неравенства \eqref{taras:eq7} и условия сопряжения \eqref{taras:eq5}$,$ \eqref{taras:eq6}$.$ Тогда$,$ если существует решение исследуемой задачи$,$ то оно единственно$.$
\end{theorem}

\smallskip

\begin{proof}
Используем ранее введенные обозначения точек $A_{0}(0, 1)$ и  $B_{0}(1, 1)$. 
Пусть $u(x,y)$ "--- решение однородной задачи, но $u(x,y) \ne 0$ 
в~области $\widetilde{ \Omega} = \Omega_{1} \cup \overline{AB} \cup \overline{AA_{0}} \cup \overline{A_{0}B_{0}} \cup \overline{BB_{0}}$. 
Пусть точка $Q$ "--- точка положительного максимума и точка $Q \in \widetilde{ \Omega}$.

В силу однородных условий точка $Q$ не может принадлежать $\overline{AA_{0}} \cup \overline{BB_{0}}$. 
На основании принципа экстремума для нелокального параболического уравнения \cite[с.~47]{taras:9} точка $Q$ не может принадлежать $\Omega_{1} \cup \overline{A_{0}B_{0}}$ и~поэтому  точка~$Q \in I = AB$, точка $Q(x_{0}, 0)$.

По лемме \hyperlink{taras:l1}{1} $$\nu_{1}(x_{0}) \le0,\quad  x_{0} \in I.$$ 
Так как $\tau_{1}(x)= \tau_{2}(x)= \tau(x)$, $$ \underset {\overline{I}}{\max }  \tau(x)= \tau(x_{0}).$$

В силу условий \eqref{taras:eq3}, \eqref{taras:eq16} по лемме~\hyperlink{taras:l2}{2} имеем $$\nu_{2}(x_{0}) >0,$$ что противоречит условию \eqref{taras:eq6}.

Следовательно, $u(x,y) \equiv 0$ в $\overline {\Omega_{1}}$ и, в частности, $\tau_{1}(x_{0})=0$ на $I$. 
Но тогда из равенства $\tau_{1}(x)= \tau_{2}(x)$ следует, что $\nu_{2}(x)=0$.

Отсюда по формуле \eqref{taras:eq21} получаем, что $u(x,y) \equiv 0$ и в области $\overline{ \Omega_{2}}$.
\end{proof}

\Section{Существование решения задачи}
Продифференцируем обе части \eqref{taras:eq22} по $x$ и учтем
\[
\left( D_{0+}^{ \alpha} \varphi \right) (x) = \bigl( I_{0+}^{ -\alpha, \alpha, \eta} \varphi \bigr) (x) = 
\Bigl( \frac{d}{dx} \Bigr)^{n} \left(I_{0+}^{n- \alpha} \varphi \right) (x),
\]
в результате получим
\begin{multline}
\label{taras:eq24} 
\Bigl(  A_{1} \frac{\Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta)} + A_{2} \Bigr) \tau''_{2}(x)+ \\
+ A_{1} \frac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} 
\Bigl( \frac{m+2}{4} \Bigr)^{\frac{2}{m+2}} \bigl( D_{0+}^{1+ \beta+\beta^{*}} \nu_{2} (t)\bigr)  (x)=A_{3} g''(x).
\end{multline}

Полагая $\tau_{1}(x)=\tau_{2}(x)=\tau(x)$, $\nu_{1}(x)=\nu_{2}(x)=\nu(x)$ и учитывая \eqref{taras:eq20}, получим дифференциальное уравнение дробного порядка $ 1+ \beta+\beta^{*}$:
\begin{equation}
\label{taras:eq25}
\bigl( D_{0+}^{1+ \beta+\beta^{*}} \nu(t) \bigr)(x)= \lambda_{2} \nu(x)+\lambda_{3} g''(x),
\end{equation}
где
\[
\lambda_{2}=
-\frac{ \Bigl( A_{1}\dfrac{\Gamma( \beta^{*}+ \beta)}{ \Gamma( \beta)} +A_{2} \Bigr) \Gamma(1+\alpha)}{A_{1} \dfrac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} \Bigl( \dfrac{m+2}{4} \Bigr)^{\frac{2}{m+2} }}, \quad
\lambda_{3}=-\frac{ A_{3}}{A_{1} \dfrac{ \Gamma(2- \beta^{*}- \beta)}{ \Gamma(1- \beta^{*})} \Bigl( \dfrac{m+2}{4} \Bigr)^{\frac{2}{m+2} }}.
\]

Известно \cite{taras:1}, что общее решение дифференциального
уравнения дробного порядка $\alpha >0$
\begin{equation}
      \label{taras:eq26}
\left( D_{0+}^{ \alpha} y(t)\right)(x)- \lambda y(x)= h(x), \quad \alpha>0, \quad n= - [- \alpha]
\end{equation}
дается формулой
\begin{equation}
      \label{taras:eq27}
y(x) = \sum_{k=1}^{n} c_{k} x^{ \alpha-k} E_{ \alpha, 1+\alpha-k} ( \lambda x^{ \alpha}) + 
\int _{0}^{x} (x-t)^{ \alpha-1} E_{ \alpha, \alpha} \bigl( \lambda (x-t)^{ \alpha} \bigr) h(t)  dt.
\end{equation}

\noindent Здесь $c_{1}, c_2, \dots, c_{n}$ "--- произвольные постоянные, а функции $ E_{ \alpha, 1+\alpha-k} ( \lambda x^{ \alpha})$ и \linebreak ${E_{ \alpha, \alpha} \bigl( \lambda (x-t)^{ \alpha}\bigr)}$ "--- специальные случаи функции Миттаг"--~Леффлера $ E_{ \alpha, \beta}(z)$, определяемой равенством
\begin{equation}
       \label{taras:eq28}
E_{ \alpha, \beta}(z)= \sum_{m=0}^{ \infty} \frac{z^{m}}{ \Gamma( \alpha m + \beta)}, \quad  \alpha>0, \quad \beta >0; \quad E_{ \alpha}(z) \equiv E_{ \alpha, 1}(z)
\end{equation}
и являющейся целой функцией от $z$ \cite{taras:1}.

Уравнение \eqref{taras:eq25} "--- уравнение вида \eqref{taras:eq26} с 
$y(x)= \nu(x)$, $ \alpha = 1+\beta+\beta^{*}$, $\lambda = \lambda_{2}$ и~$h(x) = \lambda_{3} g''(x)$. 
Так как $1< 1+\beta+\beta^{*} < 2$, общее решение вида \eqref{taras:eq27} для уравнения \eqref{taras:eq25} дается формулой
\begin{multline}
\label{taras:eq29}
\nu(x) = c_{1} x^{\beta+\beta^{*}} E_{  1+\beta+\beta^{*}, 1+\beta+\beta^{*}} 
\bigl( \lambda_{1} x^{ 1+\beta+\beta^{*}} \bigr) + 
\\
+ c_{2} x^{ \beta+\beta^{*}-1} E_{ 1+\beta+\beta^{*}, \beta+\beta^{*}} 
\bigl( \lambda_{1} x^{ 1+\beta+\beta^{*}} \bigr) + 
\\
+ \lambda_{2} \int _{0}^{x} (x-t)^{ \beta+\beta^{*} } E_{ 1+\beta+\beta^{*}, 1+\beta+\beta^{*}} 
\bigl( \lambda_{1} (x-t)^{ 1+\beta+\beta^{*}} \bigr) g''(t) \, dt,
\end{multline}
где $c_{1}$, $c_{2}$ "--- произвольные постоянные.

Подставляя \eqref{taras:eq29} в~\eqref{taras:eq22} (с $ \tau_{2} = \tau$, $\nu_{2}=\nu$), получаем выражение для $ \tau(x)$:
\begin{multline}
\label{taras:eq30}
\tau (x) = c_{1}^{*} 
\bigl( I_{0+}^{1-\beta-\beta^{*}}    t^{  \beta+\beta^{*}} E_{ 1+\beta+\beta^{*}, 1+\beta+\beta^{*}} \bigl( \lambda_{1} t^{ 1+\beta+\beta^{*}} \bigr) \bigr) (x)+ \\
+c_{2}^{*} \bigl( I_{0+}^{1-\beta-\beta^{*}}  t^{  \beta+\beta^{*}-1} E_{ 1+\beta+\beta^{*},\beta+\beta^{*}} \bigl( \lambda_{1} t^{ 1+\beta+\beta^{*}} \bigr) \bigr) (x)  +\\
+ \lambda_{4} 
\biggl( I_{0+}^{1-\beta-\beta^{*}}  \int _{0}^{t} (t-s)^{ \beta+\beta^{*}} E_{ 1+\beta+\beta^{*}, 1+\beta+\beta^{*}} \bigl( \lambda_{1} (t-s)^{ 1+\beta+\beta^{*}} \bigr)  \lambda_{2} g''(s)  ds \biggr) (x)+
\\
+\frac{A_{3} \Gamma(\beta) }{A_{1}\Gamma(\beta+\beta^{*})+A_{2}\Gamma(\beta)} g(x),
\end{multline}
где
\[
\lambda_{4} = - \frac{\Gamma(2-\beta-\beta^{*})}{\Gamma(1-\beta^{*})} \, \frac{A_{1} \Gamma(\beta) }{A_{1}\Gamma(\beta+\beta^{*})+A_{2}\Gamma(\beta)}  \Bigl( \frac{m+2}{4} \Bigr)^{\frac{2}{m+2}},
\]
\[
c_{1}^{*} = \lambda_{4} c_{1}, \quad c_{2}^{*} = \lambda_{4} c_{2}.
\]

Для приведения выражения \eqref{taras:eq30} к более простому виду воспользуемся двумя леммами, доказанными в~работе~\cite{taras:3}.

\begin{lemma}[3]Если $0< \beta < {1}/{2},$ $\lambda \in \mathbb C,$ то
\[
\bigl( I_{0+}^{1-2 \beta}    t^{2 \beta} E_{  2 \beta +1,  2 \beta+1} 
\bigl( \lambda t^{ 2 \beta +1} \bigr) \bigr) (x) = x E_{2 \beta +1,  2} \bigl( \lambda x^{2 \beta +1} \bigr)
\]
и
\[
\bigl( I_{0+}^{1- 2\beta }    t^{ 2 \beta - 1} E_{  2 \beta +1, 2 \beta} \bigl( \lambda t^{ 2 \beta +1} \bigr) \bigr) (x) = 
E_{  2 \beta +1, 1} \bigl( \lambda x^{ 2 \beta +1} \bigr)  \equiv E_{  2 \beta +1} \bigl( \lambda x^{ 2 \beta +1} \bigr).
\]
\end{lemma}

\begin{lemma}[4]Если $0< \beta < {1}/{2},$ $\lambda \in \mathbb C,$ то
\begin{multline}
\biggl( I_{0+}^{1- 2 \beta}  \int _{0}^{t} (t-s)^{2 \beta } E_{ 2 \beta +1, 2 \beta+1} \bigl( \lambda (t-s)^{ 2 \beta +1} \bigr) h(s) \, ds \biggr) (x) =
\\
 = \int _{0}^{x} (x-s) E_{ 2 \beta +1, 2} \bigl( \lambda (x-s)^{ 2 \beta +1} \bigr) h(s) \, ds.
\end{multline}
\end{lemma}

Полагая $ 2 \beta = \beta+\beta^{*}$, $ \lambda = \lambda_{1}$, $h(s)=\lambda_{2}g''(s)$, из равенства \eqref{taras:eq30} находим формулу для $ \tau(x)$:
\begin{multline}
\label{taras:eq31}
\tau (x) = c_{1}^{*} x E_{  1+\beta+\beta^{*},2} 
\bigl( \lambda_{1} x^{ 1+\beta+\beta^{*}} \bigr) + c_{2}^{*} E_{ 1+\beta+\beta^{*}} 
\bigl( \lambda_{1} x^{ 1+\beta+\beta^{*}} \bigr) + 
\\
+ \lambda_{4} \int _{0}^{x} (x-t) E_{ 1+\beta+\beta^{*}, 2} 
\bigl( \lambda_{1} (x-t)^{ 1+\beta+\beta^{*}} \bigr)  \lambda_{2} g''(s)ds  +
\\
+ \frac{A_{3} \Gamma(\beta) }{A_{1}\Gamma(\beta+\beta^{*})+A_{2}\Gamma(\beta)} g(x).
\end{multline}

Для нахождения постоянных  $c_{1}^{*}$, $c_{2}^{*}$ мы можем использовать соотношение $\tau(0)=\tau(1)=0$, вытекающее из условия \eqref{taras:eq9}. 
Подставляя $x=0$ в \eqref{taras:eq31} и~учитывая вытекающее из \eqref{taras:eq26} равенство $E_{  1+\beta+\beta^{*}}(0)=1$, получим 
\begin{equation}
\label{taras:eq32}
c_{2}^{*}=\lambda_{5}\,g(0), \quad \lambda_{5}=
- \frac{A_{3}\Gamma(\beta)}{A_{1}\Gamma(\beta^{*}+\beta)+A_{2} \Gamma(\beta)} .
\end{equation}

Подставляя $x=1$ в формулу \eqref{taras:eq31} и~учитывая выражение \eqref{taras:eq32}, находим
\begin{multline}
\label{taras:eq33}
c_{1}^{*}= \frac{1}{ E_{  1+ \beta+\beta^{*},  2} \left( \lambda_{1} \right)} 
\biggl( \lambda_{5} g (1)- \lambda_{5} g (0) E_{  1+ \beta+\beta^{*}} \left( \lambda_{1} \right) -   
\\
- \lambda_{2} \lambda_{4} \int _{0}^{1} (1-s) E_{ 1+\beta+\beta^{*},2} \bigl( \lambda_{1} (1-s)^{1+\beta+\beta^{*}} \bigr)   g(s)ds \biggr).
\end{multline}

Таким образом, подставляя \eqref{taras:eq31} в формулу \eqref{taras:eq18} (с~$\tau_{1}=\tau$), получим явное решение $u(x,y)$ исследуемой задачи. 
Это завершает доказательство существования решения исходной задачи.
