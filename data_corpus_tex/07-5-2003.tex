Рассматривается интегральный оператор $B$, $s(t)=B x$, где
\begin{equation}
s(t) = \frac{1}{2 \pi} \int_0^{2\pi} x(\tau) \ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau,
\end{equation}
$x(t)$ --- непрерывная $2\pi$ периодическая функция,
$c>\pi/2$. Следует
отметить, что результаты работы сохраняются и при $c>1$, но вычисления
будут более громоздкими.

\begin{thm}
Для функции $(1)$ справедливы оценки
\begin{align}
\omega(s,h) &\leq \frac{8}{\pi} \int_0^h \frac{\omega(x,\tau)}{\tau}
\ln^{-m}\frac{c}{\tau} d\tau +
\pi h \int_h^{\pi/2} \frac{\omega(x,\tau)}{\tau^2}
\ln^{-m}\frac{c}{\tau} d\tau+ \notag \\
%
&\quad +\pi hm \int_h^{\pi/2} \frac{\omega(x,\tau)}{\tau^2}
\ln^{-1-m} \frac{c}{\tau} d\tau + \frac{\omega(x,h)}{2} \ln^{-m}\frac{c}{h},
\quad 0<h \leq \frac{\pi}{2}, \\
%
\|s\| &\leq \frac{8}{\pi} \int_0^{\pi/2}
\frac{\omega(x,\tau)}{\tau} \ln^{-m}\frac{c}{\tau} d\tau,
\quad 0<h \leq \frac{\pi}{2}.
\end{align}
\end{thm}

{\bf Доказательство.} Учитывая, что
$$
\frac{1}{2 \pi} \int_0^{2\pi} x(\tau) \ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau=
\frac{1}{2 \pi} \int_{- \pi}^{\pi} x(\tau) \ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau,
$$
а также равенство
$$
\frac{1}{2 \pi} \int_0^{2\pi}\ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau =0,
$$
при любом $0<h< \pi$ находим
\begin{gather*}
s(t+h)-s(t) =
\frac{1}{2 \pi} \int_{- \pi}^{\pi} [x(t+\tau)-x(t+h)] \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} d\tau - \\
%
- \frac{1}{2 \pi} \int_{- \pi}^{\pi} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau = \\
%
=\frac{1}{2 \pi} \int_{- 2h}^{2h} [x(t+\tau)-x(t+h)] \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} d\tau - \\
%
\displaybreak[3]
- \frac{1}{2 \pi} \int_{- 2h}^{2h} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau + \\
%
+\frac{1}{2 \pi} \int_{- \pi}^{-2h} [x(t+\tau)-x(t+h)] \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} d\tau + \\
%
+ \frac{1}{2 \pi} \int_{2h}^{\pi} [x(t+\tau)-x(t+h)] \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} d\tau - \\
%
- \frac{1}{2 \pi} \int_{- \pi}^{-2h} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau - \\
%
- \frac{1}{2 \pi} \int_{2h}^{\pi} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau =\sum_{k=1}^4 J_k,
\end{gather*}
где
\begin{gather*}
J_1=
\frac{1}{2 \pi} \int_{- 2h}^{2h} [x(t+\tau)-x(t+h)] \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} d\tau, \\
%
J_2=
\frac{1}{2 \pi} \int_{- 2h}^{2h} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau , \\
%
J_3=\frac{1}{\pi} \left( \int_{-\pi}^{-2h}+
\int_{2h}^{\pi} \right) [x(t+\tau)-x(t)] \times \\
%
\times\left[\ctg\frac{\tau-h}{2}\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} -
\ctg \frac{\tau}{2} \ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} \right] d\tau
\equiv J_3' + J_3'', \\
%
J_4 = \frac{1}{2\pi} \int_{-\pi}^{-2h} [x(t) -x(t+h)]
\ctg \frac{\tau-h}{2} \ln^{-m}\frac{c}{|\sin(\tau-h)/2|} d\tau+ \\
%
+\frac{1}{2\pi} \int_{2h}^{\pi} [x(t) -x(t+h)]
\ctg \frac{\tau-h}{2} \ln^{-m}\frac{c}{|\sin(\tau-h)/2|} d\tau= \\
%
=\frac{[x(t+h)-x(t)]}{2\pi}
\int_{2h}^{\pi} \left[- \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} + \ctg \frac{\tau+h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau+h}{2}|} \right] d\tau.
\end{gather*}
Оценим интеграл $J_1$:
\begin{multline}
|J_1| \leq
\frac{1}{2 \pi} \left| \int_{- 2h}^{0} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau \right| + \\
%
+\frac{1}{2 \pi} \left| \int_{0}^{2h} [x(t+\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau \right| \leq \\
%
\leq
\frac{2}{\pi} \left| \int_{0}^{h} [x(t+2\tau)-x(t)] \ctg \frac{\tau}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau \right| \leq
\frac{4}{\pi} \int_0^h \frac{\omega(x,\tau)}{\tau}
\ln^{-m}\frac{c}{\tau} d\tau.
\end{multline}
Aналогично
\begin{equation}
|J_2| \leq \frac{4}{\pi} \int_0^h
\frac{\omega(x,\tau)}{\tau} \ln^{-m}\frac{c}{\tau} d\tau. \tag{4$'$}
\end{equation}
Оценим теперь $|J_3|$, $J_3 = J_3'+J_3''$,
\begin{multline}
|J_3'| = \frac{1}{2\pi} \bigg| \int_{-\pi}^{-2h}[x(t+\tau) -x(t)]
\bigg[ \tg \frac{\tau}{2} \ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} -
\tg \frac{\tau-h}{2} \ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|}\bigg]\times\\
%
\times \ctg \frac{\tau-h}{2}
\ctg \frac{\tau}{2} \ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau \bigg| \leq \\
%
\leq \frac{1}{2\pi} \int_{-\pi}^{-2h}|x(t+\tau) -x(t)|
\bigg| \bigg[ \tg \frac{\tau}{2} - \tg \frac{\tau-h}{2} \bigg]
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|}+ \\
%
+ \tg \frac{\tau-h}{2} \bigg[
\ln^{m}\frac{c}{|\sin \frac{\tau}{2}|} -
\ln^{-m}\frac {c}{|\sin \frac{\tau-h}{2}|} \bigg]\bigg| \times \\
%
\times\left|\ctg \frac{\tau-h}{2} \ctg
\frac{\tau}{2} \ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|} d\tau \right| \leq \\
%
\leq \frac{1}{2\pi} \int_{2h}^{\pi} \omega(x,\tau)
\left| \frac{\sin(h/2)}{\sin((\tau+h)/2) \sin(\tau/2)}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} \right| d\tau + \\
%
+ \frac{1}{2\pi} \int_{2h}^{\pi} \omega(x,\tau)
\left| \frac{mh \cos(\tau /2+ \theta h/2)}{\sin(\tau /2+\theta h/2)} \right|
\times \\
%
\times \left| \ln^{-m}\frac{c}{|\sin \frac{\tau+h}{2}|}
\ln^{-m}\frac{c}{|\sin \frac{\tau}{2}|}
\ln^{m-1} \frac{c}{\sin((\tau+\theta h)/2)} d\tau \right| \leq \\
%
\leq \frac{ \pi h}{2} \int_h^{\pi/2} \frac{\omega(x,\tau)}{\tau^2}
\ln^{-m}\frac{c}{\tau} d\tau + \frac{\pi hm}{2} \times
\int_h^{\pi/2} \frac{\omega(x,\tau)}{\tau^2}
\ln^{m+1} \frac{c}{\tau} d\tau.
\end{multline}
Интеграл $J_3''$ оценивается аналогично.

Оценим $J_4$:
\begin{gather}
|J_4|=\frac{|x(t+h)-x(t)|}{2\pi} \left|\int_{2h}^{\pi}
\left[ \ctg \frac{\tau-h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-h}{2}|} - \ctg \frac{\tau+h}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau+h}{2}|} \right] d\tau \right|
\leq \notag \\
%
\leq \frac{\omega(x,h)}{2\pi(1-m)}
\left|\ln^{1-m} \frac{c}{|\sin ((\tau-h)/2)|} \left.\right|_{2h}^{\pi}-
\ln^{1-m} \frac{c}{|\sin ((\tau+h)/2)|}\left.\right|_{2h}^{\pi}
\right| \leq \notag \\
%
\leq \frac{\omega(x,h)}{2} \ln^{-m}\frac{c}{h}.
\end{gather}

Так как
\begin{equation}
|s(t+h)-s(t)| \leq \sum_{k=1}^4 |J_k|,
\end{equation}
то из оценок (4)--(6), учитывая, что функция
$$
f(x) = \int_0^x \frac{\omega(x,\tau)}{\tau} \ln^{-m}\frac{c}{\tau} d\tau +
x\int_x^{\pi/2} \frac{\omega(x,\tau)}{\tau^2} \ln^{-m}\frac{c}{\tau} d\tau +
x \int_x^{\pi/2} \frac{\omega(x,\tau)}{\tau^2} \ln^{-m-1}
\frac{c}{\tau} d\tau
$$
является возрастающей, переходя в (7) к точным верхним граням,
получаем требуемое неравенство (2).

Неравенство (3) доказывается аналогично неравенству (6):
\begin{multline*}
|s(t)| = \left|\frac{1}{2 \pi} \int_0^{2\pi} x(\tau) \ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau \right| = \\
%
= \left|\frac{1}{2 \pi} \int_0^{2\pi} [x(t)-x(\tau)] \ctg \frac{\tau-t}{2}
\ln^{-m}\frac{c}{|\sin \frac{\tau-t}{2}|} d\tau \right| \leq \\
%
\leq \frac{8}{\pi} \int_0^{\pi/2}
\frac{\omega(x,\tau)}{\tau} \ln^{-m}\frac{c}{\tau} d\tau. \quad\square
\end{multline*}

Введем множество $\Phi$ положительных возрастающих функций $\varphi$,
удовлетворяющих условиям
$$
\begin{array}{l}
\text{a)} \ \lim\limits_{\delta \rightarrow 0} \varphi(\delta) =0, 
\hspace{2.8cm}\text{в)}\ \sup\limits_{\sigma >0}
\dfrac{1}{\varphi(\sigma)} \displaystyle\int_0^\sigma
\dfrac{\varphi(t)}{t} dt < \infty , \\[3mm]
\text{б)} \ \varphi(\delta) \text{ почти возрастает}, 
\hspace{1cm}\text{г)} \ \sup\limits_{\sigma >0} \dfrac{\sigma}{\varphi(\sigma)}
\displaystyle\int_{\sigma}^{\pi/2}
\dfrac{\varphi(t)}{t^2} dt < \infty.
\end{array}
$$

На $[0,\pi/2]$ введем функцию
\begin{equation}
f(\varphi,\delta) = \int_0^\delta \frac{\varphi(\tau)}{\tau}
\ln^{-m}\frac{c}{\tau} d\tau + \delta \int_\delta^{\pi/2}
\frac{\varphi(\tau)}{\tau^2}
\ln^{-m}\frac{c}{\tau} d\tau +
\delta \int_\delta^{\pi/2} \frac{\varphi(\tau)}{\tau^2}
\ln^{-m-1} \frac{c}{\tau} d\tau.
\end{equation}

\begin{lemnonum}
Пусть функция $\varphi \in \Phi$ такова, что
$$
\int_0^{\pi/2} \frac{\varphi(\tau)}{\tau}
\ln^{-m}\frac{c}{\tau} d\tau < \infty.
$$
Тогда 

$1$. $f(\varphi,\delta)$ возрастает$;$

$2$. $\lim\limits_{\delta \rightarrow 0} f(\varphi, \delta) =0;$

$3$. $\frac{f(\varphi,\delta)}{\delta}$ убывает$;$

$4$. если $\varphi_1 = o(\varphi_2)$, то
$f(\varphi_1,\delta) = o(f(\varphi_2,\delta));$

$5$. если $\varphi_1 \sim \varphi_2$, то
$f(\varphi_1,\delta) \sim f(\varphi_2,\delta);$

$6$. если $\varphi_1 = O({\varphi_2})$, то
$f(\varphi_1,\delta) = O({f(\varphi_2,\delta)})$.
\end{lemnonum}

Доказательство леммы проводится аналогично [1]. Из теоремы 1 вытекает

\begin{thm}
Пусть $\varphi \in \Phi $ удовлетворяет условию
\begin{equation}
\int_0^{\pi/2} \frac{\varphi(\tau)}{\tau}
\ln^{-m}\frac{c}{\tau} d\tau < \infty.
\end{equation}
Тогда оператор $B$ ограничен и переводит $H_\varphi$ в 
$H_{f(\varphi,\delta)}$,
где $f(\varphi,\delta)$ определяется равенством $(8)$.
\end{thm}

Обозначим через $\Phi H$ множество всех $\varphi(\delta) \in \Phi$,
для которых выполняются условия (9) и
$$
f(\varphi,\delta) = O\bigg({\varphi(\delta)
\ln^{-m}\frac{c}{\delta}}\bigg), \quad \delta \in (0,\pi/2).
$$
Тогда из теоремы 2 непосредственно вытекает

\begin{thm}
Если $\varphi \in \Phi H$ и
$$
\psi(\delta) = \varphi(\delta) \ln^{-m}\frac{c}{\delta}, \quad 0<\delta<1,
$$
то оператор $B$ действует из $H_\varphi$ в $H_\psi$ и ограничен.
\end{thm}

Из этой теоремы следует, что
оператор $B$ действует в обобщенных г\"ельдеровых пространствах и
вполне непрерывен в этих пространствах.
