\Section{Постановка задачи}
Рассматривается уравнение
\begin{equation}\label{main_equation}
    \left\{
    \begin{array}{ll}
      u_{xx}-D_{0+,\:y}^{\alpha}u=0\;&(y>0,\;0<\alpha<1),\\
      xu_{xx}+yu_{yy}+pu_x+qu_y=0\;&(y<0,\;\frac{1}{2}<p,\;q<1,\;q\geqslant p),\\
    \end{array}
    \right.
\end{equation}
где $D_{0+,\:y}^{\alpha}$ "--- частная дробная производная Римана"--~Лиувилля порядка $\alpha$ \linebreak ($0\hm<\alpha<1$) от функции $u(x,\,y)$ по второй
переменной \cite[c.~341]{sayg:SKM}:
\begin{equation*}%\label{partial_fractional_derivative}
 \left(D_{0+,\:y}^{\alpha}u\right)(x,\,y)=\frac{\partial}{\partial y}\frac{1}{\Gamma(1-\alpha)}\int _0^y\frac{u(x,\,t)}{(y-t)^{\alpha}}\,dt\quad (0<\alpha<1,\;y>0).
\end{equation*}
Результаты настоящей работы являются продолжением исследований, проведённых для уравнения~(\ref{main_equation}) в~\cite{sayg:KR1}.
Уравнение~(\ref{main_equation}) рассматривается в~области $D$, которая представляет собой объединение квадрата $D^+=\{(x,y):\;0\leqslant x$, $ y\leqslant1\}$, единичного интервала $I=(0, 1)$ прямой $y=0$ и области $D^-$, лежащей в нижней полуплоскости $(y<0)$ и~ограниченной характеристиками $AC$: $x+y=0$, $BC$: $\sqrt{x}+\sqrt{-y}=1$ уравнения~(\ref{main_equation}), отрезком $[0, 1]$ прямой $y=0$, $A(0, 0)$, $B(1, 0)$.

В настоящей работе используется $\left(I^{\alpha,\:\beta,\:\eta}_{0+}f\right)(x)$ "--- оператор обобщённого дробного интегро"=дифференцирования с~гипергеометрической функцией Гаусса $F(a,\,b;\,c;\,z)$, введённый в~\cite{sayg:S} (см. также \cite[c.~326--327]{sayg:SKM}) и имеющий при
действительных $\alpha$, $\beta$, $\eta$ и $x>0$ вид
\begin{equation*}%\label{gen_operator}
    \left(I^{\alpha,\:\beta,\:\eta}_{0+}f\right)(x)=\left\{
    \begin{array}{r}
    \displaystyle
      \dfrac{x^{-\alpha-\beta}}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}F\left(\alpha+\beta,\,-\eta;\,\alpha;\,1-\dfrac{t}{x}\right)f(t)\,dt\\
      (\alpha>0),\\
      \left(\dfrac{d}{dx}\right)^{n}\left(I^{\alpha+n,\beta-n,\eta-n}_{0+}f\right)(x)\quad(\alpha\leqslant0,n=[-\alpha]+1),\\
    \end{array}
    \right.
\end{equation*}
в частности
%\begin{equation*}%\label{operator's_property}
    $\left(I^{0,\:0,\:\eta}_{0+}f\right)(x)=f(x).$
%\end{equation*}
Отметим, что если $\alpha>0$, то справедливы следующие формулы:
\begin{equation}\label{operator's_property_a=-b}
  \left(I^{\alpha,\:-\alpha,\:\eta}_{0+}f\right)(x)=\left(I^{\alpha}_{0+}f\right)(x),\qquad
  \left(I^{-\alpha,\:\alpha,\:\eta}_{0+}f\right)(x)=\left(D^{\alpha}_{0+}f\right)(x),
\end{equation}
где $\left(I^{\alpha}_{0+}f\right)(x)$ и $\left(D^{\alpha}_{0+}f\right)(x)$ "--- операторы дробного
соответственно интегрирования и дифференцирования Римана"--~Лиувилля порядка $\alpha>0$
\cite[c.~42, 44]{sayg:SKM}:
\begin{gather}\label{riemann-liouville_derivative}
  \displaystyle
  \left(I^{\alpha}_{0+}f\right)(x)=\frac{1}{\Gamma(\alpha)}\int_0^x\frac{f(t)}{(x-t)^{1-\alpha}}\,dt\quad
  (\alpha>0,\;x>0),
\\
\label{riemann-liouville_integral}
  \displaystyle
  \left(D^{\alpha}_{0+}f\right)(x)=\left(\frac{d}{dx}\right)^{n}\frac{1}{\Gamma(n-\alpha)}\int_0^x\frac{f(t)}{(x-t)^{1-n+\alpha}}\,dt\quad
  (\alpha>0,\;n=[\alpha]+1),
\end{gather}
где $[\alpha]$ "--- целая часть $\alpha$.

Для уравнения~(\ref{main_equation}) ищется решение в~области $D$, удовлетворяющее краевым условиям
\begin{gather}\label{boundary_condition_1}
  u(0,\,y)=\varphi_0(y), \quad u(1,\,y)=\varphi_1(y), \quad 0<y<1,
\\
\label{boundary_condition_2}
    \begin{array}{c}
    \displaystyle
      \left(I_{0+}^{\frac{1}{2}-q,\:\frac{1+q-p}{2},\frac{p+q-1}{2}}t^{q-1}u\left[\theta_0(t)\right]\right)(x)=Ax^{\frac{p+q-3}{2}}u(x,\,0)+Bx^{\frac{p+q-1}{2}+\varepsilon}\\
    \left(0<x<1,\;\frac{1}{2}<p,\;q<1,\;q\geqslant p,\;\varepsilon>0\right),\\
    \end{array}
\end{gather}
а также условиям сопряжения
\begin{gather}\label{connection_1}
  \lim\limits_{y\rightarrow0+}y^{1-\alpha}u(x,\,y)=\lim\limits_{y\rightarrow0-}u(x,\,y)\quad (0\leqslant x\leqslant1),
\\
\label{connection_2}
  \lim\limits_{y\rightarrow0+}y^{1-\alpha}\left(y^{1-\alpha}u(x,\,y)\right)_y=\lim\limits_{y\rightarrow0-}(-y)^{q}u_y(x,\,y)\quad (0<x<1).
\end{gather}
Здесь $\theta_0(x)={x}/{4}-i{x}/{4}$ "--- точка пересечения характеристик уравнения~(\ref{main_equation}), выходящих из точек $(x,\,0)$ с характеристикой $AC$; $\varphi_0(y)$, $\varphi_1(y)$ "--- заданные функции, такие, что 
%\begin{equation}\label{functions}
  $y^{1-\alpha}\varphi_0(y)$, $ y^{1-\alpha}\varphi_1(y)\in C(0,\,1)$;
%\end{equation}
$A$ и $B$ "--- действительные константы, на которые ниже накладываются дополнительные условия.

Решение $u(x,\,y)$ поставленной задачи ищется в~классе дважды дифференцируемых в области $D$ функций таких, что
\begin{equation}\label{u''}
\begin{array}{c}
  y^{1-\alpha}u(x,\,y)\in C(\overline{D}^+),\qquad u(x,\,y)\in C(\overline{D}^-), \qquad 
  u_{xx}\in C\left(D^+\cup D^-\right),\\ u_{yy}\in C(\overline{D}^-), \quad 
  y^{1-\alpha}\left(y^{1-\alpha}u\right)_y \in C\left(D^+\cup \left\{(x,\,y)\textrm{:}\;0<x<1,\;y=0\right\}\right).
\end{array}
\end{equation}

\Section{Единственность решения задачи}
Пусть решение исследуемой задачи существует. Введём  обозначения:
\begin{gather*}
  \lim\limits_{y\rightarrow0+}y^{1-\alpha}u(x,\,y)=\tau_1(x),\quad \lim\limits_{y\rightarrow0-}u(x,\,y)=\tau_2(x),
\\
  \lim\limits_{y\rightarrow0+}y^{1-\alpha}\left(y^{1-\alpha}u(x,\,y)_y\right)=\nu_1(x),\quad \lim\limits_{y\rightarrow0-}(-y)^{q}u_y(x,\,y)=\nu_2(x).
\end{gather*}

Известно \cite[c.~108]{sayg:P}, что решение уравнения~(\ref{main_equation}) в области $D^+$, удовлетворяющее условиям~(\ref{boundary_condition_1}) и
\begin{equation*}%\label{tau_1}
  \lim\limits_{y\rightarrow0+}y^{1-\alpha}u(x,\,y)=\tau_1(x), \quad 0\leqslant x\leqslant1,
\end{equation*}
записывается в виде
\begin{multline}\label{solution}
  u(x,\,y)=\int_0^y\varphi_0(\eta)G_{\xi}(x,\,y;\,0,\,\eta)d\eta-\\
           -\int_0^y\varphi_1(\eta)G_{\xi}(x,\,y;\,1,\,\eta)d\eta+\Gamma(\alpha)\int_0^1\tau_1(\xi)G(x,\,y,\,\xi,\,0)d\xi;
\end{multline}
где
\begin{multline*}
  G(x,\,y,\, \xi,\,\eta)=\frac{(y-\eta)}{2}^{\beta-1}\sum_{n=-\infty}^{\infty}\left[e_{1,\,\beta}^{1,\,\beta}\left(-\frac{|x-\xi+2n|}{(y-\eta)^{\beta}}\right)-\right.\\
                         \left.-e_{1,\,\beta}^{1,\,\beta}\left(-\frac{|x+\xi+2n|}{(y-\eta)^{\beta}}\right)\right],\quad \beta=\frac{\alpha}{2},
\end{multline*}
\begin{gather*}
    e_{\alpha,\,\beta}^{\mu,\,\delta}(z)=\sum_{n=0}^{\infty}\frac{z^n}{\Gamma(\mu+\alpha n)\Gamma(\delta-\beta n)},\quad \alpha>\beta,\; \alpha>0,\; z\in \mathbb{C};
\\
    e_{1,\,\delta}^{1,\,\delta}(z)=\sum_{k=0}^{\infty}\frac{z^k}{k!\Gamma\left(\frac{(1-k)^{\alpha}}{2}\right)}.
\end{gather*}
Также известно (см., например, [\citen{sayg:G}, \citen{sayg:KR3}]), что функциональное соотношение между $\tau_1(x)$ и~$\nu_1(x)$, принесённое из параболической области $D^+$ на линию $y=0$, имеет вид
\begin{equation}\label{function_relation_1}
    \nu_1(x)=\frac{1}{\Gamma(1+\alpha)}\,\tau_1''(x).
\end{equation}

Используя результаты работы \cite{sayg:RSH}, нетрудно найти функциональное соотношение между $\tau_2(x)$ и~$\nu_2(x)$, принесённое из гиперболической области $D^-$ на линию $y=0$. Оно имеет вид
\begin{equation}\label{function_relation_2}
 k_2\left(I_{0+}^{2-2q,\,q-p,\,q-1}\nu_2(t)\right)(x)=x^{p-1}(A-k_1)\tau_2(x)+Bx^{p+\varepsilon},
\end{equation}
где
$$
    k_1=\dfrac{2^{p-q}\Gamma(2q-1)}{\Gamma(q-\frac{1}{2})},\qquad k_2=-\dfrac{2^{p+3q-1}\Gamma(2-2q)}{\Gamma(\frac{3}{2}-q)}.
$$
Применив к обеим частям~(\ref{function_relation_2}) оператор
$\left(I_{0+}^{2-2q,\,q-p,\,q-1}\right)^{-1}=I_{0+}^{2q-2,\,p-q,\,1-q}$, получим
$$%\begin{multline*}%\label{nu2_and_tau2}
 k_2\nu_2(x)=(A-k_1)\left(I_{0+}^{2q-2,\,p-q,\,1-q}t^{p-1}\tau_2(t)\right)(x)
             +B\left(I_{0+}^{2q-2,\,p-q,\,1-q}t^{p+\varepsilon}\right)(x).
$$%\end{multline*}

Справедливы следующие утверждения.

\smallskip

\begin{lemma}[1]\hypertarget{say:lem1}{Если}  функция $\tau_1(x)$ достигает положительного максимума $($отрицательного минимума$)$ на отрезке~$[0,\,1]$ в точке $x=x_0$ $(0<x_0<1),$ то
$\nu_1(x_0)\leqslant0$ $(\nu_1(x_0)\geqslant0).$ 
\end{lemma}

Доказательство леммы~\hyperlink{say:lem1}{1} непосредственно следует из соотношения~(\ref{function_relation_1}).

\smallskip

\begin{lemma}[2]\hypertarget{say:lem2}{Если} функция $\tau_2(x)$ достигает положительного максимума
(отрицательного минимума) на отрезке~$[0,\,1]$ в точке $x=x_0$ $(0<x_0<1),$
$B=0,$ $A-k_1<0,$ то $\nu_2(x_0)>0$ $(\nu_2(x_0)<0).$
\end{lemma}

Доказательство леммы~\hyperlink{say:lem2}{2} аналогично доказательству леммы~1 из \cite{sayg:KR1}.

\smallskip

\begin{theorem}[1]\hypertarget{say:th1}{Пусть} $A-k_1<0$ и $B=0$. 
Если существует решение исследуемой задачи для уравнения~\eqref{main_equation}{\rm,} то оно единственно{\rm.}
\end{theorem}

\begin{proof} Пусть  $u(x,\,y)$ "--- решение однородной задачи, но  \linebreak $u(x,\,y)\not\equiv0$ в~области
$\widetilde{D}=D^+\cup \overline{AB}\cup\overline{AA_1}\cup\overline{A_2B}$, где $A_1(0,\,1)$, $A_2(1,\,1)$.
Пусть $Q\in \widetilde{D}$ "--- точка положительного максимума. 
В~силу однородных условий $Q$ не может принадлежать $\overline{AA_1}\cup\overline{A_2B}$. 
На основании принципа экстремума для нелокального параболического уравнения~\cite[с.~47]{sayg:N} $Q$ не может принадлежать $D^+\cup \overline{A_1A_2}$
и~поэтому $Q\in I=AB$, т.\,е. $Q(x_0,\,0)$. По лемме~1 $\nu_1(x_0)\leqslant0$, $x_0\in I$.
Так как $\tau_1(x)=\tau_2(x)=\tau_(x)$, то $\max\limits_{\overline{I}}\tau(x)=\tau(x_0)$.
По лемме~2 $\nu_2(x_0)>0$, что противоречит условию~(\ref{connection_2}). Следовательно,
$u(x,\,y)\equiv0$ и, в частности, $\tau_1(x_0)=0$ на $I$. Но тогда на основании равенства
$\tau_1(x)=\tau_2(x)$ из~(\ref{function_relation_2}) следует, что $\nu_2(x)=0$. Откуда получаем,
что $u(x,\,y)\equiv0$ и в области $\overline{D^-}$. Теорема~\hyperlink{say:th1}{1} доказана. 
\end{proof}

\Section{Существование решения задачи}
Существование решения исходной задачи приведём для случая $p=q$ и сведём её к~дифференциальному уравнению дробного порядка. 
Из равенства~(\ref{function_relation_2}) имеем
\begin{equation}\label{tau_2}
 \tau_2(x)=m_1x^{1-p}\left(I_{0+}^{2-2q,\,q-p,\,q-1}\nu_2(t)\right)(x)+m_2x^{1+\varepsilon},
\end{equation}
где
%\begin{equation}\label{m1_and_m2}
    $m_1={k_2}/({A-k_1})$, $m_2={B}/({k_1-A})$, $A<k_1$.
%\end{equation}
Последовательно применяя формулы \cite{sayg:S}
\begin{equation*}
\begin{array}{r}
  x^{\alpha+\beta+\eta}\left(I^{\alpha,\;\beta,\;\eta}_{0+}\varphi(t)\right)(x)=\left(I^{\alpha,\;-\alpha-\eta,\;-\alpha-\beta}_{0+}\varphi(t)\right)(x), \quad \alpha>0,\quad \beta,\eta\in\mathbb{R},\\
  \left(I^{\alpha,\;\delta,\;\gamma}_{0+}\varphi(t)\right)(x)=\left(I^{\alpha,\;\gamma,\;\delta}_{0+}t^{\gamma-\delta}\varphi(t)\right)(x), \quad \alpha>0,\quad \gamma,\delta\in\mathbb{R},\\
\end{array}
\end{equation*}
приведём равенство~(\ref{tau_2}) к~виду \begin{equation}\label{tau_2'}
 \tau_2(x)=m_1\left(I_{0+}^{2-2q,\,p+q-2,\,q-1}t^{p-1}\nu_2(t)\right)(x)+m_2x^{1+\varepsilon}.
\end{equation}

Полагая в соотношении~(\ref{tau_2'}) $p=q$, дважды дифференцируя обе части этого равенства
по переменной $x$ и учитывая~(\ref{operator's_property_a=-b}) и~(\ref{riemann-liouville_integral}),
имеем
\begin{equation*}
 \tau_2''(x)=m_1\left(D_{0+}^{2q}t^{q-1}\nu_2(t)\right)(x)+m_2(1+\varepsilon)\varepsilon x^{\varepsilon-1}.
\end{equation*}
Согласно условиям сопряжения~(\ref{connection_1}) и~(\ref{connection_2})
$\tau_1(x)=\tau_2(x)$ при $0\leqslant x\leqslant1$, а $\nu_1(x)=\nu_2(x)$ при $0<x<1$.
Обозначив $\tau_1(x)=\tau_2(x)=\tau(x)$, $\nu_1(x)=\nu_2(x)=\nu(x)$ и учитывая
соотношение~(\ref{function_relation_1}), приходим к дифференциальному уравнению дробного
порядка~$2q$ ($1<2q<2$) вида
\begin{equation}\label{diff_equ}
 \left(D_{0+}^{2q}t^{q-1}\nu(t)\right)(x)=\lambda_1\nu(x)+\lambda_2x^{\varepsilon-1},
\end{equation}
где
%\begin{equation}\label{lambda_1,2}
 $\lambda_1={\Gamma(1+\alpha)}/{m_1}$, $\lambda_2=-{m_2(1+\varepsilon)\varepsilon}/{m_1}$.
%\end{equation}

Известно (см., например, \cite[с.~102]{sayg:KS1}), что решение дифференциального уравнения дробного порядка $\alpha>0$
\begin{equation}\label{diff_equ_frac}
 \left(D_{0+}^{\alpha}[t^{\alpha(1-m)}\varphi(t)]\right)(x)=a\varphi(x)+bx^{\mu},
\end{equation}
где
$ \alpha>0$, $m>0$, $\mu>-1$, $0<x<d<\infty$,
даётся формулой
\begin{equation}\label{solution'}
 \varphi(x)=\frac{b\Gamma(1+\mu)}{\Gamma(1+\mu+\alpha)}\,x^{\alpha m+\mu}\mathrm{E}_{\alpha,\,m,\,\frac{m+\mu}{\alpha}}\left(ax^{\alpha\mu}\right).
\end{equation}
Здесь $\mathrm{E}_{\alpha,\,m,\,l}(z)$ "--- специальная функция типа Миттаг"--~Леффлера:
\begin{equation*}
\begin{array}{c}
  \displaystyle\mathrm{E}_{\alpha,\,m,\,l}(z)=\sum_{n=0}^{\infty}c_nz^n,\quad c_0=1,\quad c_n=\prod_{i=0}^{n-1}\frac{\Gamma[\alpha(im+l)+1]}{\Gamma[\alpha(im+l+1)+1]}, \\
  (n=1,\,2,\,...) \textrm{ с } \alpha>0,\quad m>0,\quad l\in \mathbb{R},\\
  \alpha(jm+l)\neq0,\,-1,\,-2,\,...\quad (j=0,\,1,\,2,\,...), \\
\end{array}
\end{equation*}
введённая в работе \cite{sayg:KS2}, при $m=1$ с точностью до множителя совпадает с~функцией Миттаг"--~Леффлера (см. \cite[с.~33]{sayg:SKM}, \cite[с.~17]{sayg:D}):
\begin{equation*}
  \displaystyle\mathrm{E}_{\alpha,\,1,\,l}(z)=\Gamma(\alpha l+1)\mathrm{E}_{\alpha,\,\alpha l+1}(z), \qquad 
  \displaystyle\mathrm{E}_{\alpha,\,\beta}(z)=\sum_{n=0}^{\infty}\frac{z^n}{\Gamma(\alpha n+\beta)} \quad (\alpha>0,\;\beta>0).
\end{equation*}

Уравнение~(\ref{diff_equ})  суть не что иное, как уравнение вида~(\ref{diff_equ_frac}) с $\varphi(x)=\nu(x)$,
$\alpha=2q$, $m={1}/{2}+{1}/({2q})$, $a=\lambda_1$, $b=\lambda_2$, $\mu=\varepsilon-1$,
и поэтому решение вида~(\ref{solution'}) для уравнения~(\ref{diff_equ}) даётся формулой
\begin{equation}\label{nu}
    \nu(x)=\frac{\lambda_2\Gamma(\varepsilon)}{\Gamma(\varepsilon+2q)}\,x^{\varepsilon+q}\mathrm{E}_{2q,\,\frac{1}{2}+\frac{1}{2q},\,\frac{1}{2q}\left(\frac{1}{2q}-\frac{1}{2}+\varepsilon\right)}(\lambda_1x^{1+q}).
\end{equation}
Подставляя~(\ref{nu}) в~(\ref{tau_2'}) (с $\tau_2(x)=\tau(x)$, $\nu_2(x)=\nu(x)$, $p=q$),
получаем выражение для $\tau(x)$:
\begin{equation}\label{tau}
    \tau(x)=\frac{m_1\lambda_2\Gamma(\varepsilon)}{\Gamma(\varepsilon+2q)}\left(I_{0+}^{2-2q}t^{2q+\varepsilon-1}\mathrm{E}_{2q,\,\frac{1}{2}+\frac{1}{2q},\,\frac{1}{2q}\left(\frac{1}{2q}-\frac{1}{2}+\varepsilon\right)}(\lambda_1t^{1+q})\right)(x).
\end{equation}

Подставляя~(\ref{tau}) в формулу~(\ref{solution}) (с $\tau_1(x)=\tau(x)$), получаем в явном виде решение
$u(x,\,y)$ исследуемой задачи для уравнения~(\ref{main_equation}).

Используя полученное решение, непосредственно проверяем выполнение краевых условий (\ref{boundary_condition_1}), (\ref{boundary_condition_2}) и условий сопряжения~(\ref{connection_1}), (\ref{connection_2}),
а также принадлежность полученного решения $u(x,\,y)$ поставленной задачи классу функций~(\ref{u''}).

Таким образом доказана следующая теорема.
\smallskip

\begin{theorem}[2] Если $q=p$, ${1}/{2}<q<1,$ $A<k_1,$ то решение задачи~\eqref{boundary_condition_1}{\rm,} \eqref{connection_2} для уравнения~\eqref{main_equation} существует и определяется формулой~\eqref{solution}{\rm,} где $\tau_1(x)=\tau(x)$ и задаётся соотношением~\eqref{tau}\rm. \end{theorem}
