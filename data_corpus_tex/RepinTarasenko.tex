\Section{Введение}

Рассмотрим уравнение
\begin{equation}
       \label{eq1}
\left\{ {\begin{array} {l}
u_{xx}-D_{0+,y}^{\alpha_{0}} u=0 \quad (y>0,\, 0<\alpha_{0}<1), \\
(-y)^{m} u_{xx}-u_{yy}+a_{0} (-y)^{\frac{m}{2}-1}u_{x}=0 \quad (y<0),
\end{array}} \right.
\end{equation}
где $a_{0}$ -- вещественная постоянная, $m>2$ в конечной области $\Omega$, ограниченной отрезками $AA_{0}$, $BB_{0}$, $A_{0}B_{0}$ прямых $x=0$, $x=1$, $y=1$ соответственно, лежащих в полуплоскости $y>0$ и его характеристиками
\[
AC : x-\frac{2}{m+2} \, (-y)^{\frac{m+2}{2}}=0, \quad  BC: x+\frac{2}{m+2} \, (-y)^{\frac{m+2}{2}}=1
\]
в полуплоскости $y<0$.

$D_{0+,y}^{\alpha_{0}}$ -- частная дробная производная Римана-Лиувилля [1, с.33].

Пусть $\Omega_{1}= \Omega \cap (y>0)$, $\Omega_{2}= \Omega \cap (y<0)$, $I \equiv AB$ -- единичный интервал $0<x<1$ прямой $y=0$.

\begin{newthm}{Задача} {Найти решение $u(x,y)$ уравнения (\ref{eq1}) в области $\Omega$, удовлетворяющее краевым условиям
\begin{equation}
       \label{eq2}
u(0,y) = \varphi_{1}(y), \quad  u(1,y) = \varphi_{2}(y), \quad  0 \leqslant y \leqslant 1,
\end{equation}
\begin{equation}
      \label{eq3}{\begin{array}{l}
A \left( I^{ a,b,\beta-1-a}_{0+} u[ \Theta_{0}(t)] \right) (x)+B \left( I^{ a_{1},a+b+\alpha-a_{1}, \beta-a-1}_{0+} u_{y}(t,0) \right) (x)+ \\
 + C \left( I^{ a_{2},a+b+\alpha-a_{2},\beta-a-1}_{0+} u_{y}(t,0) \right) (x)=g(x), \quad x \in I,
\end{array}}
\end{equation}
\noindent а также условиям сопряжения
\begin{equation}
       \label{eq4}
\lim_{y \to +0} y^{1-\alpha_{0}} u(x,y) = \lim_{y \to -0} u(x,y), \quad   x \in \overline{I},
\end{equation}
\begin{equation}
       \label{eq5}
\lim_{y \to +0} y^{1-\alpha_{0}} \left( y^{1-\alpha_{0}}  u(x,y) \right)_{y} = \lim_{y \to -0} u_{y}(x,y), \quad   x \in  I.
\end{equation}
Здесь $\displaystyle \alpha=\frac{m-2a_{0}}{2(m+2)}$, $\displaystyle \beta =\frac{m+2a_{0}}{2(m+2)}$, $\displaystyle \alpha+\beta=\frac{m}{m+2}$.

\noindent $\displaystyle \Theta_{0}(x)=\frac{x}{2}-i \left[ \frac{(m+2)x}{4}\right]^{\frac{2}{m+2}}$ -- точка пересечения характеристик уравнения (\ref{eq1}), выходящих из точек $(x,0)$ $(x \in I)$ с характеристикой $AC$. $(I_{0+}^{\alpha, \beta, \eta} f)(x)$ -- оператор обобщенного дробного интегро-дифференцирования, введенный японским математиком М. Сайго [2] (см. также [1, с.326-327],  [3, с.14], [4, с.12]).
$A$, $B$, $C$ -- вещественные константы, такие, что
\[
A<0, B \geqslant 0, C \geqslant 0 \quad \text{или} \quad A>0, B \leqslant 0, C \geqslant 0.
\]
$a$, $b$, $a_{1}$, $a_{2}$ -- действительные числа, причем
\[
a> max \{-\alpha, \beta-1 \}, \quad 1+a+\alpha-2\beta< a_{1}< a_{2} <2+a+\alpha,
\]
\[
y^{1-\alpha_{0}} \varphi_{1}(y), y^{1-\alpha_{0}} \varphi_{2}(y) \in C([0,1] ), \quad \varphi_{1}(0) =\varphi_{2}(0)=0, \quad g(x) \in C^{2} ([0,1]).
\]
}
\end{newthm}

Будем искать решение поставленной задачи в классе дважды дифференцируемых в области $\Omega$ функций $u(x,y)$ таких, что
\[
y^{1-\alpha_{0}} u(x,y) \in C( \overline{\Omega}_{1}), \quad u(x,y) \in C( \overline{\Omega}_{2}),
\]
\[
y^{1-\alpha_{0}} \left( y^{1-\alpha_{0}}  u_{y} \right)_{y} \in C( \Omega_{1} \cup \{ (x,y): 0<x<1, y=0 \}),
\]
\[
u_{xx} \in C(\Omega_{1} \cup \Omega_{2}), \quad u_{yy} \in C( \overline{\Omega}_{2}).
\]

Отметим, что исследуемая задача является продолжением работ [5], [6].

\bigskip
\Section{Единственность решения задачи}

Пусть существует решение исследуемой задачи.

Введем обозначения
\[
\lim_{y \to 0+} y^{1-\alpha_{0}} u(x,y) = \tau_{1}(x), \quad \lim_{y \to 0-} u(x,y)= \tau_{2}(x),
\]
\[
\lim_{y \to 0+} y^{1-\alpha_{0}} \left( y^{1-\alpha_{0}}  u(x,y) \right)_{y} = \nu_{1}(x), \quad \lim_{y \to 0-} u_{y}(x,y) =\nu_{2}(x).
\]





Известно [7], [8], что решение уравнения (\ref{eq1}) в области $\Omega_{1}$, удовлетворяющее условию (\ref{eq2}) и условию
\[
\lim_{y \to 0+} y^{1-\alpha_{0}} u(x,y) = \tau_{1}(x) \quad (0 \leqslant x \leqslant 1),
\]
дается формулой
\begin{equation}
       \label{eq6}{\begin{array}{l}
\displaystyle u(x,y) = \int\limits_{0}^{y}\frac{\partial G}{\partial \xi}|_{\xi=0} \varphi_{1} (\eta)  \, d \eta \, - \int\limits_{0}^{y} \frac{\partial G}{\partial \xi}|_{\xi=1} \varphi_{2} (\eta) \, d \eta - \int\limits_{0}^{1} G(x,y; \xi,0) \tau_{1}( \xi) \, d \xi ,
\end{array}}
\end{equation}
\noindent где
\[
{\begin{array}{l}
\displaystyle G(x,y; \xi, \eta) = \frac{\Gamma(\alpha_{0})}{2} (y- \eta)^{\delta-1}\sum_{n=- \infty}^{ \infty} \left[ e_{1, \delta}^{1, \delta} \left( - \frac{|x- \xi +2n|}{(y- \eta)^{\delta}} \right) - \right. \\
\displaystyle \quad\quad\quad\quad - \left. e_{1, \delta}^{1, \delta} \left( - \frac{|x+ \xi +2n|}{(y- \eta)^{\delta}} \right) \right],
\end{array}}
\]
\[
e^{\mu, \, \delta}_{\alpha_{0}, \,\beta_{0}}(z)= \sum_{k=0}^{ \infty} \frac{z^{k}}{ \Gamma(\mu+k \alpha_{0}) \Gamma(\delta-k \beta_{0})}, \quad \alpha_{0}>\beta_{0}, \quad \delta=\frac{\alpha_{0}}{2}, \quad \alpha_{0}>0,
\]
\[
e^{1, \, \delta}_{1, \, \delta}(z)= \sum_{k=0}^{ \infty} \frac{z^{k}}{k!  \Gamma(\delta- \delta k)}, \quad \delta<1.
\]

{\bf Замечание.} Решение $u(x,y)$ может быть выражено в терминах специальной функции Райта [7]
\[
\varphi(\gamma, \delta; z)= \sum_{k=0}^{ \infty} \frac{z^{k}}{k!  \Gamma(\delta+ \gamma k)},
\]
поскольку
\[
e_{1, \, \frac{\alpha_{0}}{2}}^{1, \, \frac{\alpha_{0}}{2}}(z)= \varphi \left( -\frac{\alpha_{0}}{2}, \frac{\alpha_{0}}{2}; z \right).
\]

Функциональное соотношение между $\tau_{1}(x)$ и $\nu_{1}(x)$, принесенное из параболической части $\Omega_{1}$ на линию $y=0$ имеет вид (см., например, [8])
\begin{equation}
       \label{eq7}
\nu_{1}(x) = \frac{1}{ \Gamma(1+ \alpha_{0})} \tau_{1}''(x).
\end{equation}

Найдем функциональное соотношение между $\tau_{2}(x)$ и $\nu_{2}(x)$, принесенное на линию $y=0$ из гиперболической части $\Omega_{2}$ области $\Omega$.

Используя формулу решения задачи Коши в области $\Omega_{2}$, выпишем $u[\Theta_{0}(x)]$:
\[
u[ \Theta_{0}(x)] = k_{1} \left( I_{0+}^{\alpha,0, \beta-1} \tau _{2} (t)\right)(x) +
k_{2} \left(I_{0+}^{1- \beta, \alpha+\beta-1,  \beta -1} \nu_{2}(t) \right)(x),
\]
где
\[
k_{1}= \frac{ \Gamma( \alpha+ \beta)}{ \Gamma( \beta)}, \quad k_{2}= \frac{ \Gamma( 2-\alpha- \beta)}{ \Gamma( 1-\alpha)}  \left( \frac{m+2}{4} \right) ^{ \frac{2}{m+2}}.
\]

Подставляя $u[\Theta_{0}(x)]$ в краевое условие (\ref{eq3}) и используя полугрупповое свойство обобщенных операторов [1, с.327]
\begin{equation}
       \label{eq8}
I_{0+}^{\alpha, \beta, \eta} I_{0+}^{\gamma, \delta,\alpha+\eta} f= I_{0+}^{\alpha+\gamma, \beta+\delta, \eta}f, \quad \gamma>0,
\end{equation}
получим
\begin{equation}
       \label{eq9}{\begin{array}{l}
k_{1}A \left( I_{0+}^{a+\alpha, b,  \beta-1-a} \tau_{2}(t) \right) (x) + k_{2}A \left( I_{0+}^{a+1-\beta, b+\alpha+\beta-1, \beta-1-a} \nu_{2}(t) \right) (x)+\\
 + B\left( I_{0+}^{a_{1}, a+b+\alpha-a_{1}, \beta-1-a} \nu_{2}(t) \right) (x)+ C \left( I_{0+}^{a_{2}, a+b+\alpha-a_{2}, \beta-1-a} \nu_{2}(t) \right) (x)=g(x).
\end{array}}
\end{equation}

Подействуем на обе части соотношения (\ref{eq9}) оператором $I_{0+}^{-a-\alpha,-b, \alpha+\beta-1}$.

Непосредственные вычисления с использованием формулы (\ref{eq8}) дают возможность записать
\begin{equation}
       \label{eq10} {\begin{array}{l}
\displaystyle \tau_{2}(x)=k_{3} \left( I_{0+}^{1-\alpha- \beta} \nu_{2}(t) \right) (x) + k_{4} \left( I_{0+}^{a_{1}-a-\alpha} \nu_{2}(t) \right) (x)+\\
\displaystyle  + k_{5}\left( I_{0+}^{a_{2}- a-\alpha} \nu_{2}(t) \right) (x)+ g_{1}(x),
\end{array}}
\end{equation}
где $I_{0+}^{\alpha}$ -- оператор Римана-Лиувилля,
\begin{equation}
       \label{eq11}{\begin{array}{l}
\displaystyle k_{3}=-\frac{k_{2}}{k_{1}}, \quad k_{4}=-\frac{B}{A k_{1}}, \quad k_{5}= - \frac{C}{A k_{1}}, \\
\displaystyle g_{1}(x) =\frac{1}{A k_{1}} \left( I_{0+}^{-a-\alpha,-b, \alpha+\beta-1} g(t)\right)(x).
\end{array}}
\end{equation}

При $g(x)=0$ покажем, что интеграл
\[
I= \int\limits_{0}^{1}\tau_{2}(x) \nu_{2}(x) dx \geqslant 0.
\]

В самом деле, опираясь на (\ref{eq10}), имеем
\[
I=\frac{k_{3}}{ \Gamma(1-\alpha-\beta)}  \int\limits_{0}^{1} \nu_{2}(x) dx \int\limits_{0}^{x} \nu_{2}(t) (x-t)^{-\alpha-\beta} dt +
 \]
 \[
 + \frac{k_{4}}{ \Gamma(a_{1}-a-\alpha)}  \int\limits_{0}^{1} \nu_{2}(x) dx \int\limits_{0}^{x} \nu_{2}(t) (x-t)^{a_{1}-a-\alpha-1} dt +
\]
\[
+ \frac{k_{5}}{ \Gamma(a_{2}-a-\alpha)}  \int\limits_{0}^{1} \nu_{2}(x) dx \int\limits_{0}^{x} \nu_{2}(t) (x-t)^{a_{2}-a-\alpha-1} dt.
\]

Далее, проводя стандартные вычисления (см., например, [9]) получим $I \geqslant 0$, а для области $\Omega_{1}$ -- $I \leqslant 0$. Из этих неравенств имеем $I \equiv 0$. Последнее тождество влечет выполнение тождества $u(x,y) \equiv 0$, что и доказывает единственность решения исходной задачи.


\bigskip
\Section{Существование решения задачи}

Для доказательства существования решения исследуемой задачи достаточно найти $\nu_{1}(x)$ или $\nu_{2}(x)$.

Полагая $\tau_{1}=\tau_{2}(x)=\tau(x)$, $\nu_{1}(x)=\nu_{2}(x)=\nu(x)$ продифференцируем обе части соотношения (\ref{eq10}) дважды по $x$. Получим уравнение
\begin{equation}
       \label{eq12} {\begin{array}{l}
\displaystyle \tau''(x)=k_{3} \left( D_{0+}^{1+2 \beta} \nu \right) (x) + k_{4} \left( D_{0+}^{2+a+\alpha-a_{1}} \nu \right) (x)+\\
\displaystyle  + k_{5}\left( D_{0+}^{2+a+\alpha-a_{2}} \nu \right) (x)+ g''_{1}(x),
\end{array}}
\end{equation}
здесь $(D_{0+}^{\alpha}f)(x)$ -- это дробная производная в смысле Римана-Лиувилля.

В силу (\ref{eq7}) $\tau''(x)=\Gamma(1+\alpha_{0})\nu(x)$ и уравнение (\ref{eq12}) принимает вид
\begin{equation}
       \label{eq13} {\begin{array}{l}
\displaystyle  \left( D_{0+}^{1+2 \beta} \nu \right) (x) -\lambda \left( D_{0+}^{2+a+\alpha-a_{1}} \nu \right) (x)-\\
\displaystyle  -\delta \left( D_{0+}^{2+a+\alpha-a_{2}} \nu \right) (x)-\mu \nu(x)= g_{2}(x),
\end{array}}
\end{equation}
где
\[
\lambda=-\frac{k_{4}}{k_{3}}, \quad \delta=-\frac{k_{5}}{k_{3}}, \quad \mu=-\frac{\Gamma(1+\alpha_{0})}{k_{3}}, \quad g_{2}(x)=-\frac{1}{k_{3}} g''_{1}(x).
\]

В монографии [10, с.301] рассмотрено уравнение дробного порядка
\[
(D_{0+}^{\alpha} y)(x) - \lambda (D_{0+}^{\beta} y)(x) - \delta (D_{0+}^{\gamma} y)(x)=f(x) \quad (x>0, \lambda, \delta, \mu \in R),
\]
где
\[
l-1< \alpha \leqslant l \quad (l \in R), \quad 0< \gamma < \beta< \alpha
\]
и выписано его решение в виде
\[
y(x) =\int\limits_{0}^{x} (x-t)^{\alpha-1} G_{\gamma, \beta, \alpha, \lambda} (x-t)f(t)dt,
\]
\[
G_{\gamma, \beta, \alpha, \lambda} (z)= \sum_{n=0}^{\infty} \left( \sum_{i+v=n}\right)  \frac{\mu^{i} \delta^{v}}{i! v!}\, z^{(\alpha-\beta) n+ \beta i-(\beta-\gamma)v} \, \times
\]
\[
\times _{1}\Psi_{1}    \Biggr[ {\begin {array}{l}  (n+1,1) \\ ((\alpha-\beta) n + \beta i +\alpha+(\beta-\gamma)v, \alpha -\beta)  \end{array}} \Biggr| \lambda z^{\alpha-\beta}  \Biggr].
\]

Для уравнения (\ref{eq13}) это решение принимает вид
\[
\nu(x)=\int\limits_{0}^{x} (x-t)^{2\beta} G_{2+a+\alpha-a_{2}, 2+a+\alpha-a_{1}, 1+2\beta, \lambda}(x-t)g_{2}(t)dt,
\]
где
\[
G_{2+a+\alpha-a_{2}, 2+a+\alpha-a_{1}, 1+2\beta, \lambda} (x)=
\]
\[
= \sum_{n=0}^{\infty} \left( \sum_{i+v=n} \right) \frac{\mu^{i} \delta^{v}}{i! v!}\, z^{(a_{1}+2\beta+1-a+\alpha) n+ (2+a+\alpha-a_{1})i-(a_{2}-a_{1})v} \, \times
\]
\[
\scriptsize   {\times _{1}\Psi_{1}    \Biggr[ {\begin {array}{l}  (n+1,1) \\ ((a_{1}+2\beta+a+\alpha) n + (2+a+\alpha-a_{1}) i + 1+2\beta +(a_{2}-a_{1})v, a_{1}+2\beta-a-\alpha)  \end{array}} \Biggr| \lambda z^{a_{1}+2\beta-1-a-\alpha}  \Biggr].}
\]
