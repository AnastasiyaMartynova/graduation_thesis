Одним из эффективных методов параметрической идентификации систем
с турбулентным трением является метод,
 в основе которого лежит  среднеквадратичное оценивание коэффициентов стохастического разностного уравнения, описывающего результаты наблюдений ординат колебаний механической системы [1,~2].
 Построена линейно"=параметрическая дискретная модель свободных колебаний системы с турбулентным трением, связывающая в рекуррентной форме несколько последовательных мгновенных значений виброграммы свободных
 колебаний, причём коэффициенты модели известным образом связаны с динамическими характеристиками системы.
 В матричной форме линейно"=параметрическая модель колебаний систем
 с турбулентным трением имеет следующий вид:

\begin{equation}
    \label{math/1}
    \left\{
     \begin{array}{l}
        b=F\lambda+\eta, \\
        \eta=P_{\lambda}\varepsilon, 
     \end{array}
     \right.
\end{equation}
где  $P_{\lambda}$ "--- невырожденная матрица линейного
преобразования вектора  $\varepsilon$ случайной помехи в~результатах наблюдений. 
Алгоритм численного метода среднеквадратичного оценивания коэффициентов разностного уравнения
описывается совокупностью рекуррентных формул 
\begin{equation}
    \label{math/2}
    \hat{\lambda}^{( i ) }= \bigl( F^{\top}U^{-1}_{\hat{\lambda}^{( i ) }}F \bigr) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}^{( i ) }}b, 
\end{equation}
где $U^{-1}_{\hat{\lambda}^{( i ) }}$ "--- ленточная пятидиагональная  симметричная матрица размера
$N{\times} N$:
\begin{gather*}
u_{11}=u_{22}=u_{31}=1, \quad 
u_{32}=u_{23}=-\hat{\lambda}^{( i ) }_0 \bigl( 1+\hat{\lambda}^{( i ) }_1 \bigr) , \quad
u_{42}=u_{24}=1+\hat{\lambda}^{( i ) }_1, \\
u_{k k}=
    2+4\hat{\lambda}^{( i ) }_1( k-2 ) +2 \bigl( \hat{\lambda}^{( i ) }_1 \bigr) ^2( k^2-4k+5 ) + \bigl( \hat{\lambda}^{( i ) }_0 \bigr) ^2 \bigl( 1+( k-2 ) \hat{\lambda}^{( i ) }_1 \bigr) ^2, \\
k\equiv 3, 4, \ldots, N;\\
u_{k~k-1}=u_{k-1~k}=
    -\hat{\lambda}^{( i ) }_0 \bigl( 2+2( 2k-5 ) \hat{\lambda}^{( i ) }_1+( 2k^2-10k+13 ) \bigl( \hat{\lambda}^{( i ) }_1 \bigr) ^2 \bigr) ,\\ k\equiv 4,  5,  \ldots,  N;
\\
u_{k~k-2}=u_{k-2~k}= \bigl( 1+( k-3 ) \hat{\lambda}^{( i ) }_1 \bigr) ^2, \quad
 k\equiv 5, 6, \ldots, N, 
\end{gather*}
 где $i\in \mathbb N,$ $\hat{\lambda}^{( i ) }$ "--- среднеквадратичная оценка вектора
коэффициентов,  вычисленная на $i$-той итерации. Остальные элементы
$u_{kj}=0$.
%$P_{\hat{\lambda}^{( 0 ) }}=E$ - единичная матрица.

В данной работе рассматриваются достаточные условия сходимости
итерационной процедуры среднеквадратичного оценивания
коэффициентов линейно"=параметрической дискретной модели колебаний
систем с турбулентным трением,  описываемой формулами~(\ref{math/2}).

Обозначим векторную функцию
$q( \hat{\lambda} ) = \bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}b$, 
где  $q=( q_{0}, q_{1}, \ldots, q_{4} ) \in \mathbb R^5$. Далее
рассмотрим систему нелинейных уравнений
$$
        \begin{cases}
            \hat{\lambda}_{0}=q_{0}( \hat{\lambda}_{0}, \hat{\lambda}_{1}, \ldots, \hat{\lambda}_{4} ) , \\
            \hat{\lambda}_{1}=q_{1}( \hat{\lambda}_{0}, \hat{\lambda}_{1}, \ldots, \hat{\lambda}_{4} ) , \\
            \hspace{.6cm}\vdots \\
            \hat{\lambda}_{4}=q_{4}( \hat{\lambda}_{0}, \hat{\lambda}_{1}, \ldots, \hat{\lambda}_{4} ) , 
        \end{cases}
 $$
 которая в матричной форме имеет вид
\begin{equation}
    \label{math/3}
    \hat{\lambda}=q( \hat{\lambda} ) .
\end{equation}


В $n$-мерном пространстве введём норму вектора $\lambda$ и матрицы
$W$ следующим образом:
$\|\lambda\|=\max_i |\lambda_{i}|$, 
$\|W\|=\max_i \sum_{j=1}^{n}|w_{ij}|$, 
$i\equiv 1, 2, \ldots, n$.

\begin{theorem}
Пусть функции   $q_i( \hat{\lambda} ) $ и  $\frac{\partial q_i( \hat{\lambda} ) }{\partial \lambda_j}$  $(i, j \equiv 0, 1, \ldots, 4)$ определены и~непрерывны
    в известной замкнутой области $G$ действительного  $n$-мерного пространства $E^n,$  причём в~области $G$ выполняется неравенство:
\begin{equation}
\label{math/4}
        \lambda=\max{\hat{\lambda}\in G} \biggl[\Bigl\| \bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}\Bigr\|\|e\|\max_i 
        \Bigl( \sum_{k=1}^{N}\left|\omega ^{0}_{ik}\right|+\sum_{k=1}^{N}\left|\omega ^{1}_{ik}\right| \Bigr) \biggr]<1, 
    \end{equation}
    где $\omega ^{j}_{ik}$ "---  элементы матрицы $U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}$  размера  $N{\times }N;$
    $e=F\hat{\lambda}-b$ "--- вектор остатков. 
    
    Тогда,  если последовательные приближения
    \begin{equation}
        \label{math/5}
        \hat{\lambda}_k=q( \hat{\lambda}_{k-1} ) , \quad  k \in \mathbb N, 
    \end{equation}
    не выходят за пределы области $G{:}$ $\hat{\lambda}_k \in G,$  то
    \begin{enumerate}
        \item[\rm 1)] независимо от выбора начального приближения $\hat{\lambda}_{0}\in G,$  итерационный процесс~\eqref{math/5} сходится,  т.\,е. существует предел $\lim_{k\to\infty}\hat{\lambda}_k=\bar{\lambda};$
        \item[\rm 2)] предельный вектор $\bar{\lambda}$  является единственным решением уравнения~\eqref{math/3} в~области~$G;$
        \item[\rm 3)] имеет место оценка
            \begin{equation}
                \label{math/6}
                \bigl\|\bar{\lambda}-\hat{\lambda}^{( k ) }\bigr\|\leq\frac{\alpha}{1-\alpha}\bigl\|\lambda^{( k ) }-\lambda^{( k-1 ) }\bigr\|,
\end{equation}
 где $0\leq\alpha<1.$
    \end{enumerate}
\end{theorem}

\begin{proof}
    Пусть $\lambda^{( 1 ) }$  и  $\lambda^{( 2 ) }\in G$. Тогда известно,  что существует вектор  $\xi\in G$ такой,  что выполняется неравенство
    \begin{equation}
        \label{math/7}
        \bigl\|q( \lambda^{( 1 ) } ) -q( \lambda^{( 2 ) } ) \bigr\|\leq \|W( \xi ) \|\bigl\|\lambda^{( 1 ) }-\lambda^{( 2 ) }\bigr\| \leq \max_{\hat{\lambda}\in G} \|W( \xi ) \|\bigl\|\lambda^{( 1 ) }-\lambda^{( 2 ) }\bigr\|, 
    \end{equation}
    где  
        \begin{equation}
        \label{math/8}
        W( \xi ) = \Bigl[\frac{\partial q_{i}( \hat{\lambda} ) }{\partial\hat{\lambda}_j}\Bigr]_{\hat{\lambda}=\xi}    \end{equation}
     "--- матрица Якоби системы функций $\bigl\{q_i( \hat{\lambda} ) \bigr\}^{4}_{i=0}$   по переменным $\hat{\lambda}_{0}, \hat{\lambda}_{1}, \ldots, \hat{\lambda}_{4}$, 
     вычисленная в точке $\xi\in G$. 
     %В развернутой форме матрица Якоби размера $5{\times 5$ имеет вид
%    \begin{equation}
%        \label{math/8}
%        W( \hat{\lambda} ) =
%         \left[

%                    \frac{\partial q_{1}}{\partial\hat{\lambda}_0} & \frac{\partial q_{1}}{\partial\hat{\lambda}_1} & \cdots & \frac{\partial q_{1}}{\partial\hat{\lambda}_4} \\
%                    \vdots  &  \vdots  &  \ddots  &  \vdots\\
%                    \frac{\partial q_{4}}{\partial\hat{\lambda}_0} & \frac{\partial q_{4}}{\partial\hat{\lambda}_1} & \cdots& \frac{\partial q_{4}}{\partial\hat{\lambda}_4}
%                \end{array}
%         \right].

    Покажем,  что при выполнении условия~(\ref{math/4})  выполняется условие
$$\max_{\hat{\lambda}\in     G} \bigl\|W( \hat{\lambda} ) \bigr\|\leq\alpha<1, $$
    т.\,е. отображение~(\ref{math/3}) является сжимающим в области  $G$. 
    Действительно,  матрицу Якоби~(\ref{math/8}) можно представить в виде
    $W( \hat{\lambda} ) =\left[
                        \begin{array}{cccc}
                            \frac{\partial q}{\partial\hat{\lambda}_0}& \frac{\partial q}{\partial\hat{\lambda}_1} & \cdots & \frac{\partial q}{\partial\hat{\lambda}_4}
                        \end{array}
                    \right]$,  где  $ \frac{\partial q}{\partial\hat{\lambda}_j}$ "--- вектор-функции. 
Тогда {\small
\begin{multline*} 
\frac{\partial q}{\partial\hat{\lambda}_j}=\frac{\partial ( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}}{\partial\hat{\lambda}_j}F^{\top}U^{-1}_{\hat{\lambda}}b+( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}\frac{\partial U^{-1}_{\hat{\lambda}}}{\partial\hat{\lambda}_j}b= \\
=
        -( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}\frac{\partial F^{\top}U^{-1}_{\hat{\lambda}}F}{\partial\hat{\lambda}_j}( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}b-( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}b=\\
        =
        ( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}F( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}b-
        \\-
        ( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}b= \\
        =
        ( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}( F\hat{\lambda}-b ) .
\end{multline*} }

    Обозначив вектор остатков через  $e=F\hat{\lambda}-b$,  получим
$$\frac{\partial q}{\partial\hat{\lambda}_j}=( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}e,$$
где 
    $\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_0}=U^0_{\hat{\lambda}}$ "---  ленточная трёхдиагональная  симметричная
матрица    размера  $N{\times}N$:
$$u^{0}_{k k}=2\hat{\lambda}_0( 1+( k-2 ) \hat{\lambda}_1 ) ^2, \quad k\equiv 3, 4, \ldots, N;$$
$$u^{0}_{k~k-1}=u^{0}_{k-1~k}=-2-2( 2k-5 ) \hat{\lambda}_1-( 2k^2-10k+13 ) \hat{\lambda}^2_1, \quad k\equiv 4, 5, \ldots, N;$$
$u^{0}_{32}=u^{0}_{23}=-( 1+\hat{\lambda}_1 );$ остальные элементы  $u^{0}_{kj}=0$; 
$\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_1}=U^1_{\hat{\lambda}}$ "--- 
ленточная   пятидиагональная   симметричная     матрица    размера  $N{\times} N$:
$$u^{1}_{32}=u^{1}_{23}=-\hat{\lambda}_0;\quad u^{1}_{42}=u^{1}_{24}=1;$$
$$u^{1}_{k k}=4( k-2 ) +4\hat{\lambda}_1( k^2-4k+5 ) +2( k-2 ) \hat{\lambda}^2_0( 1+( k-2 ) \hat{\lambda}_1 ), \quad k\equiv 3, 4, \ldots, N;$$
$$u^{1}_{k~k-1}=u^{1}_{k-1~k}=-2\hat{\lambda}_0( 2k-5+( 2k^2-10k+13 ) \hat{\lambda}_1 ),\quad  k\equiv 4, 5, \ldots, N;$$
$$u^{1}_{k~k-2}=u^{0}_{k-2~k}=2( k-3 ) ( 1+( k-3 ) \hat{\lambda}_1 ) , \quad k\equiv 5, 6, \ldots, N;$$
остальные элементы  $u^{1}_{k~j}=0$;
 $\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_2}$, $\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_3}$, $\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_4}$ "--- нулевые матрицы размера $N{\times} N$,  поскольку матрица  $U_{\hat{\lambda}}$ зависит только от коэффициентов  $\hat{\lambda}_0$ и   $\hat{\lambda}_1$.

    Тогда матрица Якоби~(\ref{math/8}) примет вид
    \begin{equation}
        \label{math/9}
        W( \xi ) =( F^{\top}U^{-1}_{\hat{\lambda}}F ) ^{-1}F^{\top}V( \hat{\lambda} ) , 
    \end{equation}
    где   $V( \hat{\lambda} ) =\left[
                        \begin{array}{ccccc}
                            U^{-1}_{\hat{\lambda}}U^{0}_{\hat{\lambda}}U^{-1}_{\hat{\lambda}}e & U^{-1}_{\hat{\lambda}}U^{1}_{\hat{\lambda}}U^{-1}_{\hat{\lambda}}e & 0 & 0 & 0
                        \end{array}
                    \right]$ "--- матрица размера $N{\times} 5$, 
и~элементы $v_{ij}$  матрицы $V( \hat{\lambda} ) $  могут быть описаны формулой $$v_{ij}=\sum_{k=1}^{N}\omega^{j}_{ik}e_k ,$$ 
    где  $\omega^{j}_{ik}$ "---  элементы матрицы $U^{-1}_{\hat{\lambda}}\frac{\partial U_{\hat{\lambda}}}{\partial\hat{\lambda}_j}U^{-1}_{\hat{\lambda}}$.

Оценим норму матрицы  $V_{\hat{\lambda}}$:
\begin{multline*}
\bigl\|V_{\hat{\lambda}}\bigr\|=
\max_i \sum_{j=0}^{N}v_{ij}=\max_i \Bigl( \Bigl|\sum_{k=1}^{N}\omega^{0}_{ik}e_k\Bigr|+\Bigl|\sum_{k=1}^{N}\omega^{1}_{ik}e_k\Bigr| \Bigr) \leq \\ \leq 
    \max_i \Bigl( \sum_{k=1}^{N}\left|\omega^{0}_{ik}e_k\right|+\sum_{k=1}^{N}\left|\omega^{1}_{ik}e_k\right| \Bigr) 
    \leq \|e\|\max_i \Bigl( \sum_{k=1}^{N}\left|\omega^{0}_{ik}\right|+\sum_{k=1}^{N}\left|\omega^{1}_{ik}\right| \Bigr) , 
\end{multline*}
    где $\|e\|=\max_k |e_k|$, $k\equiv 1, 2, \ldots, N$; $i\equiv1, 2, \ldots, N.$
    
В соответствии с~(\ref{math/9}) получаем
\begin{multline*}
\bigl\|W( \xi ) \bigr\|=\bigl\|\bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}\bigr\| \cdot 
\bigl\|V( \hat{\lambda} ) \bigr\|\leq \\ 
\leq \bigl\|\bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}\bigr\| \|e\| 
\max_i \Bigl( \sum_{k=1}^{N}\left|\omega^{0}_{ik}\right|+\sum_{k=1}^{N}\left|\omega^{1}_{ik}\right| \Bigr) .
\end{multline*}
Следовательно,  с учётом~(\ref{math/4}) имеем оценку
\begin{multline*}
\max_{\hat{\lambda}\in G} \left\|W( \xi ) \right\|=
\max_{\hat{\lambda}\in G} \Bigl[\bigl\|\bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}\bigr\| \cdot 
\bigr\|V( \hat{\lambda} ) \bigr\|\Bigr]\leq \\
\leq \max_{\hat{\lambda}\in G} \Bigr[\bigr\| \bigr( F^{\top}U^{-1}_{\hat{\lambda}}F \bigl) ^{-1}F^{\top}\bigr\| \|e\|\max_i 
\Bigl( \sum_{k=1}^{N}\left|\omega^{0}_{ik}\right|+\sum_{k=1}^{N}\left|\omega^{1}_{ik}\right| \Bigr) \Bigr]<1.
\end{multline*}
Тогда,  усиливая неравенство~(\ref{math/7}),  получим
    \begin{equation}
        \label{math/10}
        \bigl\|q( \hat{\lambda}^{( 1 ) } ) -q( \hat{\lambda}^{( 2 ) } ) \bigr\|\leq \alpha \bigl\|\hat{\lambda}^{( 1 ) }-\hat{\lambda}^{( 2 ) }\bigr\|, 
    \end{equation}
    где $0\leq\alpha<1 $. Следовательно,  отображение~(\ref{math/3}) является сжимающим в области  $G$.

Покажем,  что при выполнении условия~(\ref{math/10})  итерационный процесс~(\ref{math/5}) сходится. 
Для этого используем обобщённый критерий Коши:
\begin{multline}
\label{math/11}
\bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|=
\bigl\|q( \hat{\lambda}^{( k ) } ) -q( \hat{\lambda}^{( k-1 ) } ) \bigr\|\leq 
\alpha \bigl\|\hat{\lambda}^{( k ) }-\hat{\lambda}^{( k-1 ) }\bigr\| \leq \\
\leq \alpha ^2\bigl\|\hat{\lambda}^{( k-1 ) }-\hat{\lambda}^{( k-2 ) }\bigr\|\leq 
\ldots 
\leq 
\alpha ^k\bigl\|\hat{\lambda}^{( 1 ) }-\hat{\lambda}^{( 0 ) }\bigr\|.
\end{multline}

Рассмотрим норму
\begin{multline}
\label{math/12}
\bigl\|\hat{\lambda}^{( k+p ) }-\hat{\lambda}^{( k ) }\bigr\|
=
\bigl\|( \hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) } ) + \bigl( \hat{\lambda}^{( k+2 ) }-\hat{\lambda}^{( k+1 ) }+\ldots +( \hat{\lambda}^{( k+p ) }-\hat{\lambda}^{( k+p-1 ) } )  \bigr) \bigr\|\leq \\ 
\leq \sum_{i=0}^{p-1}\bigl\|\hat{\lambda}^{( k+i+1 ) }-\hat{\lambda}^{( k+i ) }\bigr\|.
\end{multline}
Отсюда с учётом~(\ref{math/11}) получаем
%\begin{multline*}

    Так как  $0\leq\alpha<1$,  то $\lim_{i\to\infty}\alpha^i=0$\/ и из~(\ref{math/13}) следует,  что
    $\forall \varepsilon>0$ $\exists N=N( \varepsilon )$: $\forall k > N( \varepsilon ) $ и $\forall p>0 \Rightarrow \bigl\|\hat{\lambda}^{( k+p ) }-\hat{\lambda}^{( k ) }\bigr\|<\varepsilon$, 
т.\,е. для последовательности $\bigl\{ \hat{\lambda}^{( k ) } \bigr\}^\infty_{k=0}$  выполнен критерий Коши. Поэтому существует
    $\lim_{k\to\infty}\hat{\lambda}^{( k ) }=\bar{\lambda}$,  причём $\bar{\lambda}\in G$  в силу замкнутости области $G$.

Перейдем к пределу в равенстве~(\ref{math/5}): $\lim_{k\to\infty}\hat{\lambda}^{( k ) }=\lim_{k\to\infty} q( \hat{\lambda}^{( k-1 ) } ) $.
    Учитывая непрерывность функции  $q( \hat{\lambda} ) $ в области $G$,  получим $\lim_{k\to\infty}\hat{\lambda}^{( k ) }=q\bigl( \lim_{k\to\infty} \hat{\lambda}^{( k-1 ) } \bigr) $  или
    \begin{equation}
        \label{math/14}
        \bar{\lambda}=q( \bar{\lambda} ) .
    \end{equation}
    т.\,е. $\bar{\lambda}$ "--- решение уравнения~(\ref{math/3}).

Покажем, что    это решение единственное в области $G$.
Пусть  $\exists\bar{\bar{\lambda}}\neq\bar{\lambda}$, причём 
    \begin{equation}
        \label{math/15}
        \bar{\bar{\lambda}}=q(  \bar{\bar{\lambda}} ) .
    \end{equation}
Вычтем из~(\ref{math/14}) равенство~(\ref{math/15}):
$$\bar{\lambda}-\bar{\bar{\lambda}}=q( \bar{\lambda} ) -q( \bar{\bar{\lambda}} ) .$$
С учётом~(\ref{math/10}) имеем $\bigl\|\bar{\lambda}-\bar{\bar{\lambda}}\bigr\|=\bigl\|q( \bar{\lambda} ) -q(  \bar{\bar{\lambda}} ) \bigr\|\leq\alpha\bigl\|\bar{\lambda}-\bar{\bar{\lambda}}\bigr\|$  или
$( 1-\alpha ) \bigl \|\bar{\lambda}-\bar{\bar{\lambda}}\bigr\|\leq 0$. 
Так как $0\leq\alpha<1$,  то  $0<1-\alpha\leq1$,  а следовательно,
$\bigl\|\bar{\lambda}-\bar{\bar{\lambda}}\bigr\|\leq 0$. 
Однако это возможно только в случае,  если  $\bigl\|\bar{\lambda}-\bar{\bar{\lambda}}\bigr\|= 0$,  т.\,е.  $\bar{\lambda}=\bar{\bar{\lambda}}$.

Поскольку в соответствии с~(\ref{math/11})
\begin{gather*}
\bigl\|\hat{\lambda}^{( k+2 ) }-\hat{\lambda}^{( k+1 ) }\bigr\|\leq
\alpha\bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|, \\
\bigl\|\hat{\lambda}^{( k+3 ) }-\hat{\lambda}^{( k+2 ) }\bigr\|\leq
\alpha\bigl\|\hat{\lambda}^{( k+2 ) }-\hat{\lambda}^{( k+1 ) }\bigr\|\leq
\alpha^2\bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|, \\
\dots\\
\bigl\|\hat{\lambda}^{( k+i ) }-\hat{\lambda}^{( k+i-1 ) }\bigr\|\leq
\alpha^i\bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|, \quad   i\in \mathbb N,
\end{gather*}
то из формулы~(\ref{math/12}) получаем
\begin{multline*}
\bigl\|\hat{\lambda}^{( k+p ) }-\hat{\lambda}^{( k ) }\bigr\|\leq
\sum_{i=0}^{p-1}\bigl\|\hat{\lambda}^{( k+i+1 ) }-\hat{\lambda}^{( k+i ) }\bigr\|\leq
\bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|\sum_{i=0}^{p-1}\alpha^i\leq \\
\leq
\frac{1-\alpha^p}{1-\alpha}
    \bigl\|\hat{\lambda}^{( k+1 ) }-\hat{\lambda}^{( k ) }\bigr\|\leq
    \frac{\alpha( 1-\alpha^p ) }{1-\alpha}\bigl\|\hat{\lambda}^{( k ) }-\hat{\lambda}^{( k-1 ) }\bigr\|.
\end{multline*}
    Отсюда,  переходя к пределу при $p\to\infty$,  получаем~\eqref{math/6},

%$$

что и требовалось доказать.
\end{proof}

\begin{newthm}{Следствие}
    Вектор остатков $e$ в формуле~\eqref{math/4} может быть представлен в следующем виде:

        \label{math/17}
        H=F\bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}-E.
    \end{equation}
\end{newthm}

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}

=

    =
    F\bigl( \lambda+ \bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}P_{\lambda}\varepsilon-\lambda \bigr) -P_{\lambda}\varepsilon= \\
    =
    \bigl( F\bigl( F^{\top}U^{-1}_{\hat{\lambda}}F \bigr) ^{-1}F^{\top}U^{-1}_{\hat{\lambda}}-E \bigr) P_{\lambda}\varepsilon=HP_{\lambda}\varepsilon. \quad \square
\end{multline*}
