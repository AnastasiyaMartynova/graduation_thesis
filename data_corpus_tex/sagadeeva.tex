\enabstract{Sobolev type equations now constitute a vast area of nonclassical equations of mathematical physics.
Those called nonclassical equations of mathematical physics, whose representation in the form of equations or systems of equations
partial does not fit within one of the classical types (elliptic, parabolic or hyperbolic).
In this paper we prove the existence of a unique optimal and hard control over solutions of Showalter--Sidorov problem for nonstationary operator-differential equations unresolved with respect to the time derivative. In this case, one of the operators in the equation
is multiplied by a scalar function of the time-variable, besades stationary equation has a strong continuous degenerate resolving  semigroup of operators.
Apart from the introduction and bibliography article comprises two parts. The first part provides the necessary information regarding the theory of $p$-radial operators, the second contains the proof of main results of this article.
}

\Section[1]{Введение}
Пусть ${\mathfrak X}$, ${\mathfrak Y}$ и $\mathfrak U$ "--- гильбертовы пространства. 
Пусть операторы
${L}\in {\mathcal L}({\mathfrak X};{\mathfrak Y})$, $M\in {\mathcal C}l({\mathfrak X};{\mathfrak Y})$ и $B\in {\mathcal L}({\mathfrak U};{\mathfrak Y})$.
Рассмотрим задачу Шоуолтера"--~Сидорова \cite{SagShul:bib:SvZag}
\begin{equation}\label{eq:SagShul:SS}
P(x(0)-x_0)=0
\end{equation}
для уравнения соболевского типа [\citen{SagShul:bib:DemUsp}--\citen{SagShul:bib:KorEn}]
\begin{equation}\label{eq:SagShul:upr}
L\dot{x}(t)=a(t)Mx(t)+f(t)+Bu(t)\qquad  (\ker L\ne\{0\}),\quad t\in[0,\tau],
\end{equation}
где  вектор-функции  $u:[0,\tau]\to{\mathfrak U}$, $f:[0,\tau]\to{\mathfrak Y}$ и скалярная функция $a:[0,\tau]\to{\mathbb R}_+$ будут определены в дальнейшем.


Нашей задачей является нахождение минимума функционала вида %качества следующего
\begin{multline}\label{eq:SagShul:fk}
J(u)=\alpha\sum_{q=0}^1\int _0^\tau \bigl\|z^{(q)}(t)-z^{(q)}_d(t)\bigr\|^2_{\mathfrak Z}dt+ \\ 
+ (1-\alpha)
\sum_{q=0}^{k}\int _0^\tau\left\langle N_qu^{(q)}(t),u^{(q)}(t)\right\rangle_{\mathfrak U}dt,
\end{multline}
где $\alpha\in(0, 1]$, $k=0, 1,\dots, p+1$, $N_q\in{\mathcal L}(\mathfrak U)$  "--- самосопряжённые и положительно определённые операторы, $z=Cx$, оператор $C\in{\cal  L}({\mathfrak X};{\mathfrak Z})$, а
$z_d=z_d(t,s)$ "--- плановое наблюдение из некоторого гильбертова пространства наблюдений $\mathfrak Z$.
Отметим, что $\alpha\in(0, 1]$ и $(1-\alpha)$ "--- весовые коэффициенты целей оптимального управления, заключающиеся в~достижении плановых показателей наблюдаемой величины без скачкообразных изменений (первое слагаемое в~\eqref{eq:SagShul:fk}) и минимизации расходуемых для этого ресурсов управления (второе слагаемое в~\eqref{eq:SagShul:fk}). 
При $\alpha=1$ в функционале \eqref{eq:SagShul:fk} исчезает второе слагаемое и мы получаем задачу жёсткого управления, т.е. когда  при оптимизации затраты на достижение цели не интересны.


\smallskip

\textbf{1. Относительно $\boldsymbol p$-радиальные операторы.}
Доказательства утверждений этой части можно найти в работе \cite{SagShul:bib:SvFed}.

Обозначим
$$\rho^L(M)=\{\mu\in{\mathbb C}:(\mu L-M)^{-1} \in{\cal L(\mathfrak F;\mathfrak U)}\},\qquad\sigma^L(M)={\mathbb C}\setminus\rho^L(M),$$
$$R_{\mu }^L(M) =(\mu L-M)^{-1}L,\quad L_{\mu }^L(M)=L(\mu L-M)^{-1},\quad\mu\in\rho^L(M),$$
$$R_{(\lambda ,p)}^L(M)\!=\!\prod \limits_{k=0}^p\!R_{\lambda _k}^L(M),\,
L_{(\lambda ,p)}^L(M)\!=\!\prod\limits_{k=0}^p\!L_{\lambda_k}^L(M),\,\,\,\lambda_k\!\in\!\rho^L(M)\,(k=\overline{0,p}).
$$

\smallskip

\begin{definition}[1]{\rm Оператор $M$ называется {\it $p$-радиальным} {\it
относительно оператора} $L$ (или, коротко, $(L,p)${\it -радиальным}),
если
\begin{itemize}
\item[(i)] $\exists \omega\!\in {\mathbb R}$ $\forall\mu >\omega\ \Rightarrow$ $ \mu\!\in\rho
^L(M)$;

\item[(ii)] $\exists K>0$ $ \forall\mu_k >\omega$, $k=\overline{0, p}$, $\forall n\!\in {\mathbb N}$

$\qquad\qquad \max\{ \| (R_{(\mu ,p)}^L(M))^n\|
_{{\cal L}({\mathfrak X})}, \| (L_{(\mu, p)}^L(M))^n \| _{{\cal L}({\mathfrak Y})}\}\le
{\displaystyle\frac {K}{\prod _{k=0}^p (\mu _k-\omega)^n}.}$

\end{itemize}

}\end{definition}

\smallskip

Также введём обозначения
$$\mathfrak X^0=\ker R_{(\mu,p)}^L(M),\,\,
\mathfrak Y^0=\ker L_{(\mu,p)}^L(M),\,\,L_0=L\bigr|_{\mathfrak X^0},\,\,M_0=M\bigr|_{{\rm dom}M\cap\mathfrak X^0}.$$
Через ${\mathfrak X}^1$ $({\mathfrak Y}^1)$ обозначим замыкание линеала
${\rm im} R_{(\mu,p)}^L(M)$ $({\rm im} L_{(\mu ,p)}^L(M))$,
а через $\tilde {\mathfrak X}$ $ (\tilde {\mathfrak Y})$ "--- замыкание линеала
${\mathfrak X}^0\dot + {\rm im} R_{(\mu ,p)}^L(M)$
$({\mathfrak Y}^0 \dot + {\rm im} L_{(\mu ,p)}^L(M))$ в норме пространства
${\mathfrak X}$ $({\mathfrak Y})$.

\smallskip
\begin{definition}[2]{\rm
Сильно непрерывное отображение $V^{\bullet}:{\mathbb R} _+\to {\cal L(V)}$ называется
{\it сильно непрерывной полугруппой разрешающих операторов} (или просто
{\it разрешающей $C_0$-полугруппой})% уравнения (\ref{133})
, если
\begin{itemize}
\item[(i)] $V^sV^t=V^{s+t}$ $\forall s,t>0;$
\item[(ii)] $v(t)=V^tv_0$ есть решение этого уравнения для любого $v_0$ из
плотного в ${\cal V}$ линеала.
\end{itemize}
}\end{definition}

\smallskip

Полугруппу $\{V(t)\in{\cal L(V)}:t\in\overline{\mathbb R}
_+\}$ будем называть {\it экспоненциально ограниченной} с
константами $C,\omega$, если

$\mbox{\phantom{aa}}\qquad \exists C>0\ $ $\exists
\omega\in{\mathbb R}\ $ $\forall t\in\overline{\mathbb R} _+\ \ $ $\
\|V(t)\|_{{\cal L(V)}}\le Ce^{\omega t}. $


\smallskip

\begin{theorem}[1]{\rm{\it Пусть оператор} $M$ $ (L,p)$-{\it радиален.
Тогда существует экспоненциально ограниченная c константами $K,\omega$
из определения {\rm 1} и сильно непрерывная разрешающая
полугруппа для однородного уравнения} \eqref{eq:SagShul:upr} {\it при $a\equiv1$,  рассматриваемого на подпространстве} $\tilde {\mathfrak X}.$% $ (\tilde {\mathfrak F})$.
}\end{theorem}

\smallskip

\begin{remark}[1] {\rm Операторы разрешающей полугруппы для уравнения \eqref{eq:SagShul:upr} при $a\equiv1$ и $t>0$ можно представить в виде
$$
X^t=s - \lim\limits_{k\to\infty}\left (\Bigl(L-\frac{t}{k}M\Bigr)^{-1}L\right)^{k}
=s - \lim\limits_{k\to\infty}\Bigl(\frac kt R^L_{\frac kt}(M)\Bigr)^{k},
$$
принимая во внимание поправки формулы, обсуждаемые в работе \cite{SagShul:bib:SagShul}.
}
\end{remark}
\smallskip

\begin{remark}[2] {\rm Единицей полугруппы $\{X^t\in{\cal L}(\tilde{\mathfrak X}):t\in\overline{\mathbb R} _+\}$
%($\{F(t)\in{\cal L(}\tilde{\mathfrak F}):t\in\overline{\mathbb R} _+\}$)
является проектор $P$ %($Q$)
вдоль ${\mathfrak X}^0$ %(${\mathfrak F}^0$)
на ${\mathfrak X}^1$. %(${\mathfrak F}^1$).
}\end{remark}


\begin{definition}[3] {\rm
Оператор $M$  называется {\it сильно} $(L,p)$-{\it радиальным},
если при любых $\lambda,\mu_0,\mu_1,...,\mu_p>\omega$ выполняются следующие условия:

(i) существует плотный в ${\mathfrak Y}$ линеал
$\stackrel{\circ}{{\mathfrak Y}}$ такой, что для всех $y\in \stackrel{\circ}{{\mathfrak Y}}$

$\mbox{\phantom{aa}}\qquad \displaystyle{ \|M(\lambda L-M)^{-1}L^L_{(\mu,p)}(M)y\| _{{\mathfrak Y}}\le
\frac{\mbox{\rm const}(y)}{(\lambda-\omega)\prod\limits^p_{k=0}(\mu_k-\omega)};}
$

(ii)$\qquad
\displaystyle{
\|R^L_{(\mu,p)}(M)(\lambda L-M)^{-1}\| _{{\cal L}({\mathfrak Y};{\mathfrak X})}\le
\frac{K}{(\lambda-\omega)\prod\limits^p_{k=0}(\mu_k-\omega)}}.
$
}\end{definition}

\smallskip

\begin{theorem}[2]Пусть оператор $M$ сильно
$(L,p)$-радиален$.$ Тогда
\begin{itemize}

\item[{\rm (i)}] ${\mathfrak X}={\mathfrak X}^0 \oplus{\mathfrak X}^1,$
${\mathfrak Y}={\mathfrak Y}^0\oplus {\mathfrak Y}^1;$

\item[{\rm (ii)}] $L_k=L\big|_{{\mathfrak X}^k}\in{\cal L}({\mathfrak X}^k;{\mathfrak Y}^k),$
$\quad M_k=M\big|_{{\rm dom}M_k}\in{\cal C}l({\mathfrak X}^k;{\mathfrak Y}^k),$

${\rm dom}M_k={\rm dom}M\cap{\mathfrak X}^k,$ $k=0, 1;$

\item[{\rm (iii)}] существуют  операторы $M^{-1}_0\in{\cal L}({\mathfrak Y}^0;{\mathfrak X}^0)$ и
$L^{-1}_1\in{\cal L}({\mathfrak Y}^1;{\mathfrak X}^1).$

\end{itemize}

\end{theorem}

\smallskip

\textbf{2. Оптимальное и жёсткое управления решениями задачи Шоуолтера--Сидорова для нестационарного уравнения}

\begin{definition}[4]{\rm
Вектор-функция 
$$x\in H^1({\mathfrak X})=\{x\in L_2(0,\tau;{\mathfrak X}): \dot x\in L_2(0,\tau;{\mathfrak X})\}$$ называется {\it сильным решением уравнения} \eqref{eq:SagShul:upr}, если она почти всюду на $(0,\tau)$ обращает его в тождество. Сильное решение $x=x(t)$ уравнения \eqref{eq:SagShul:upr} называется {\it сильным решением задачи Шоуолтера"--~Сидорова} \eqref{eq:SagShul:SS}, \eqref{eq:SagShul:upr}, если оно удовлетворяет \eqref{eq:SagShul:SS}.
}\end{definition}

\smallskip

Обозначим ${\mathbb N}_0\equiv{\mathbb N}\cup\{0\}$.
Построим гильбертово пространство

\vspace{3mm}
$\phantom{atr}
H^{p+1}({\mathfrak Y})=\{y\in L_2(0,\tau;{\mathfrak Y}): y^{(p+1)}\in L_2(0,\tau;{\mathfrak Y}),\ p\in{\mathbb N}_0\}$ \\ со скалярным произведением
$$
\ [y,z]=
{\displaystyle\sum_{q=0}^{p+1}\int _0^\tau\left\langle y^{(q)},z^{(q)}\right\rangle_{\mathfrak Y}dt}.
$$

\smallskip

\begin{theorem}[3]Пусть оператор $M$ сильно $(L,p)$-радиален$,$ $p\in{\mathbb N}_0,$ а функция  $a\in C^{p+1}([0,\tau);{\mathbb R}_+).$
Тогда для любых $x_0\in{\mathfrak X}$ и $f\in H^{p+1}({\mathfrak Y})$
существует единственное сильное
решение $x \in H^1({\mathfrak X})$
задачи Шоуолтера"--~Сидорова \eqref{eq:SagShul:SS}
для уравнения \eqref{eq:SagShul:upr}$,$ причём %\hspace{2mm}$x(t) =$

\begin{multline}\label{eq:SagShul:reupr}
x(t) = X^{\displaystyle\int _0^t a(\zeta)d\zeta}P x_0 + \int _0^t
{X^{\displaystyle\int _s^t {a(\zeta)d\zeta}}L_1^{-1}Q(f(s)+Bu(s))ds}-
\\
-\sum_{k=0}^p(M_0^{-1}L_0)^kM_0^{-1}(I\!-\!Q)(AD)^{k}\!A\left(f(t)+Bu(t)\right),
\end{multline}
где $$(Ah)(t)=a^{-1}(t)h(t) ,\quad (Dh)(t)={\displaystyle\frac{dh}{dt}(t)}.$$
\end{theorem}

\smallskip

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,c\,т\,в\,о}\/ этой теоремы аналогично доказательству теоремы~4 из работы \cite{SagShul:bib:SagBad} и потому не приводится.

\smallskip

%Выделим в пространстве $H^{1}\!(\mathfrak U)$ замкнутое и выпуклое подмножество $H_{\partial}^{1}(\mathfrak U)$ --- множество допустимых управлений.

\begin{definition}[5]
Вектор-функцию $v\in H_{\partial}^{1}({\mathfrak U})$ назовем оптимальным управлением (жестким управлением)  решениями задачи \eqref{eq:SagShul:SS}, \eqref{eq:SagShul:upr}, если
\begin{equation}\label{eq:SagShul:min}
J(v)=\min\limits_{(x,u)\in{\mathfrak X}\times H^{1}_{\partial}({\mathfrak U})} J(u),\qquad \alpha\in(0, 1)\quad (\alpha=1),
\end{equation}
где пары $(x,u)\in{\mathfrak X}\times H^{1}_{\partial}({\mathfrak U})$ удовлетворяют  \eqref{eq:SagShul:SS}, \eqref{eq:SagShul:upr}, а $H_{\partial}^{1}(\mathfrak U)$ --- некоторое замкнутое и выпуклое подмножество
 в $H^{1}\!(\mathfrak U)$.
\end{definition}

\begin{theorem}[4]Пусть  оператор $M$ сильно $(L,p)$-радиален$,$ $p\in{\mathbb N}_0,$ функция  $a\in C^{p+1}(\overline{\mathbb R} _+;{\mathbb R}_+).$
Тогда при любых $x_0\in{\mathfrak X},$ $f\in H^{p+1}({\mathfrak Y})$ существует единственное оптимальное управление
 $v\in H_{\partial}^{p+1}({\mathfrak U})$ задачи \rm \eqref{eq:SagShul:SS}--\eqref{eq:SagShul:fk}, \eqref{eq:SagShul:min}.
\end{theorem}

\smallskip

\begin{proof}
Зафиксируем $x_0\in\mathfrak X$, $f\in H^1({\mathfrak Y})$, и рассмотрим \eqref{eq:SagShul:reupr} как отображение $G:u\to x(u)$. В условиях предыдущей теоремы отображение  $G:H^{p+1}({\mathfrak U})\to H^1({\mathfrak X})$, определенное формулой \eqref{eq:SagShul:reupr}, непрерывно.
%Пользуясь отображением $G$ из предыдущей леммы,
Теперь перепишем функционал качества \eqref{eq:SagShul:fk} в виде
$$
J(u)=\bigl\|Cx(t;u)-z_d\bigr \|^2_{H^1({\mathfrak Z})}+[\eta,u],
$$
где $\eta^{(k)}(t)=N_ku^{(k)},$ $k=0, 1, \ldots,p+1$. Откуда
$$
J(u)=\pi(u,u)-2\lambda(u)+ \bigl \|z_d-Cx(t;0)\bigr \|^2_{H^1({\mathfrak Z})},
$$
где $$\pi(u,u)= \bigl \|C(x(t;u)-x(t;0)) \bigr \|^2_{H^1({\mathfrak Z})}+[\eta,u]$$ "---
билинейная непрерывная коэрцитивная форма на $H^{1}({\mathfrak U})$, а
$$
\lambda(u)=\left\langle
z_d-Cx(t;0), C(x(t;u)-x(t;0))
\right\rangle_{H^1({\mathfrak Z})}
$$ "---
линейная непрерывная на $H^{1}({\mathfrak U})$ форма. А~значит, утверждение теоремы следует из  \cite[гл.~1]{SagShul:bib:Lions}.
\end{proof}

{\small\sc Следствие.} {\it
Пусть  оператор $M$ сильно $(L,p)$-радиален$,$ $p\in{\mathbb N}_0,$ функция  $a\in C^{p+1}(\overline{\mathbb R} _+;{\mathbb R}_+).$
Тогда при любых $x_0\in{\mathfrak X},$ $f\in H^{p+1}({\mathfrak Y})$ существует единственное жёсткое управление
 $v\in H_{\partial}^{p+1}({\mathfrak U})$ задачи \rm \eqref{eq:SagShul:SS}--\eqref{eq:SagShul:fk}, \eqref{eq:SagShul:min}.
}
