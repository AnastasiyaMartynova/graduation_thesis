\title{Для уравнения смешанного типа единственность решения задачи $T$ 
с сопряжением производной\\ по нормали с дробной производной}

В данной статье рассмотрим уравнение 
\begin{equation} 
U_{xx}+\sgn y\cdot U_{yy}-\lambda U=0,\quad \lambda>0, 
\end{equation} 
в области $G$, ограниченной гладкой кривой $\Gamma$, которaя лежит в 
полуплоскости $y>0$, с концами $A(0;0)$ и $B(1;0)$ (начало кривой в точке 
$B$) и отрезками характеристик $AC:y=-x$, $BC:y=x-1$, $C(1/2;-1/2)$. 
 
Обозначим $G_+=G\cap\{(x,y)\mid y>0\}$, $G_-=G\cap\{(x,y)\mid y<0\}$. 
\medskip 
 
{\bf Задача $T$.} {\it Найти функцию $U(x,y)$ со следующими свойствами\/${:}$ 
 
$1)$ $U(x,y)\in C(\ov G)$; $U_{xx},U_{yy}\in C(G);$ 
 
$2)$ $U(x,y)$ --- решение уравнения (1) в $G_-$ и $G_+;$ 
 
$3)$ $U(x,y)$ удовлетворяет условиям 
\begin{alignat*}{2} 
U(x,y)|_\Gamma &=f(s),&\quad s&\in[0;l],\ \ 
\text{$l$ --- длина кривой $\Gamma$}, \\ 
% 
U(x,y)|_{y=-x}&=\vartheta(x), &\quad x&\in[0;1/2], 
\end{alignat*} 
и условию сопряжения 
\begin{equation} 
v_+(x)=D v_-(x),\quad x\in(0;1), 
\end{equation} 
где $v_+(x)=\lim\limits_{y\to+0}\frac{\partial U(x,y)}{\partial y}$, $D 
v_-(x)$ в характеристических координатах $\xi=x+y$, $\eta=x-y$ имеет вид 
$D v_-(\xi)=\lim\limits_{\eta\to\xi+0}\frac{\partial}{\partial\xi} 
\int\limits^\xi_0(\xi-t)^{-r}U(t,\eta)dt$, $0<r<1$. 
}
 
При $y>0$ уравнение (1) примет вид 
\begin{equation} 
L(U)\equiv U_{xx}+U_{yy}-\lambda U=0,\quad \lambda>0. 
\end{equation} 
 
\begin{lem} 
Пусть непрерывная в $\ov G_+$ функция $U(x,y)$ удовлетворяет уравнению 
$(3)$ в области $G_+$ и принимает в $\ov G_+$ наибольшее положительное 
$($наименьшее отрицательное$)$ значение в точке $(x_0;0)$, $0<x_0<1$. 
Если значения $U(x,y)$ на кривой $\Gamma$ меньше $($больше$)$, чем 
$U(x_0;0)$, то $v_+(x_0)=\lim\limits_{y\to+0} \frac{\partial 
U(x_0;y)}{\partial y}<0$ $(>0)$ при условии, что этот предел существует. 
\end{lem} 
 
\begin{pf} 
Пусть условия леммы выполнены для функции $U(x,y)$. 
 
Неравенство $\lim\limits_{y\to+0}\frac{\partial U(x_0;y)}{\partial y}>0$ 
невозможно по принципу максимального значения (\cite{1}, с.\,492). 
Допустим, что 
$\lim\limits_{y\to+0}\frac{\partial U(x_0;y)}{\partial y}{=}0$. 
Без ограничения общности рассуждений будем считать $U(x_0;0){=}1$. Из 
условия леммы следует $\max U(x,y)|_\Gamma<U(x_0;0)$. Значит, существует 
такое $\varepsilon:0<\varepsilon<1$,
что $\max U(x,y)|_\Gamma\le1-\varepsilon$. 
 
Рассмотрим вспомогательную функцию 
\begin{equation} 
V(x,y)=\frac{U(x,y)}{A(y)}, 
\end{equation} 
где $A(y)=e^d-\varepsilon e^y>0$, $d$ -- диаметр области $G_+$, 
$0<\varepsilon<1$. Получим $L(U)=A(y) L_1(V)$, где 
$$ 
L_1(V)\equiv V_{xx}+V_{yy}+\frac{2 A'(y)}{A(y)} V_y+ 
\bigg(\frac{A''(y)}{A(y)}-\lambda \bigg) V. 
$$ 
 
По условию леммы $L(U)=0$, тогда $L_1(V)=0$. Непосредственные вычисления 
показывают, что $\frac{A''(y)}{A(y)}-\lambda<0$ и $V(x,y)|_\Gamma\le 
\frac{U(x,y)|_\Gamma}{e^d(1-\varepsilon)}\le \frac{1}{e^d}< V(x_0;0)$. 
Значит, $V(x,y)$ удовлетворяет всем условиям данной леммы, и, 
следовательно, невозможно неравенство
$\lim\limits_{y\to+0}\frac{\partial V(x_0;y)}{\partial y}>0$. 
С другой стороны, из (4) следует 
$\lim\limits_{y\to+0}\frac{\partial V(x_0;y)}{\partial y}= 
\frac{\varepsilon e^y}{A^2(y)}>0$. Полученное противоречие 
доказывает первое утверждение леммы. Второе утверждение доказывается 
аналогично. 
\end{pf} 
 
При $y<0$ уравнение (1) в характеристических координатах $\xi=x+y$, 
$\eta=x-y$ примет вид 
\begin{equation} 
U_{\xi\eta}-\frac{\lambda}{4}U=0. 
\end{equation} 
 
Область $G_-$ преобразуется в область $D=\{(\xi;\eta):0<\xi<\eta<1\}$. 
 
\begin{lem} 
Если функция $U(\xi;\eta)\in C(\ov D)$ удовлетворяет уравнению $(5)$ в 
области $D$, то $U(0;0)=U(1;1)=0$ либо $U(0;0)=U(0;1)=0$. 
\end{lem} 
 
{\bf Доказательство.} 
Предположим, что для функции $U(\xi;\eta)$ не выполняется равенство 
$U(0;0)=U(1;1)=0$. Построим вспомогательную функцию 
$$ 
V(\xi;\eta)=U(\xi;\eta)-U(0;0) 
\frac{g(1;1)w(\xi;\eta)-w(1;1)g(\xi;\eta)} 
{g(1;1)-w(1;1)}-U(1;1) 
\frac{g(\xi;\eta)-w(\xi;\eta)}{g(1;1)-w(1;1)}, 
$$ 
где $w(\xi;\eta)=\exp\big(\frac{\sqrt{\lambda}(\xi+\eta)}{2}\big)$, 
$g(\xi;\eta)=\,{}_0F_1(1;\frac{\lambda}{4}\xi\eta)$. Функции 
$w(\xi;\eta)$ и $g(\xi;\eta)$ являются решениями уравнения (5), 
следовательно, функция $V(\xi;\eta)$ также является решением уравнения 
(5) в области $D$. Вычисления показывают $V(0;0)=V(1;1)=0$, что 
доказывает первое утверждение леммы. Второе утверждение лeммы доказывается 
аналогично с использованием вспомогательной функции 
$$ 
V(\xi;\eta)=U(\xi;\eta)-U(0;0) 
\frac{q(0;1)w(\xi;\eta)-w(0;1)q(\xi;\eta)} 
{q(0;1)-w(0;1)}-U(0;1) 
\frac{q(\xi;\eta)-w(\xi;\eta)}{q(0;1)-w(0;1)}, 
$$ 
где $q(\xi;\eta)=\,{}_0F_1(1;\frac{\lambda}{4}(\xi+1)\eta)$. Методом 
Римана--Адамара находим функцию $U(\xi;\eta)\in C(\ov D)$ --- решение 
уравнения (5) в области $D$, удовлетворяющее условиям 
\begin{gather} 
U(\xi;\xi)=\tau(\xi),\quad \xi\in[0;1],\\ 
U(0;\eta)=\vartheta(\tfrac{\eta}{2})=\varphi(\eta),\quad 
\eta\in[0;1].\quad\square 
\end{gather} 
 
\begin{thm} 
Если $\varphi(\eta)\in C^1_{[0;1]}$, $\tau(\xi)\in C_{[0;1]}\cap 
C^1_{(0;1)}$, то существует единственное решение задачи Дарбу для 
уравнения $(5)$ в области $D$ с условиями $(6)$ и $(7)$, непрерывное в 
$\ov D$, определяемое формулой 
\begin{equation} 
U(\xi;\eta)=\tau(\xi)+\int^\eta_0 \varphi'(t)V(0,t;\xi,\eta) 
dt+\frac{\lambda}{4}(\eta-\xi)\int^\xi_0 \tau(t)\,{}_0F_1 
\bigg(2;\frac{\lambda}{4}(t-\xi)(t-\eta)\bigg)dt. 
\end{equation} 
\end{thm} 
 
Функция Римана--Адамара для уравнения (5) имеет вид 
$$ 
V(\xi,\eta;\xi_0,\eta_0)= 
\begin{cases} 
R(\xi,\eta;\xi_0,\eta_0), & \eta>\xi_0; \\ 
R(\xi,\eta;\xi_0,\eta_0)-R(\eta,\xi;\xi_0,\eta_0), 
& \eta<\xi_0, 
\end{cases} 
$$ 
где $R(\xi,\eta;\xi_0,\eta_0)=\,{}_0F_1(1;\frac{\lambda}{4} 
(\xi-\xi_0)(\eta-\eta_0))$ 
--- функция Римана для уравнения (5) (\cite{2}, с.\,4). Используя 
равенство (8) и аппарат специальных функций гипергеометрического типа 
\cite{3}, находим 
\begin{multline} 
D v_-(\xi)=\int^\xi_0 \tau'(s)(\xi-s)^{-r}ds- 
\frac{r\lambda}{4(1-r)}\int^\xi_0 \tau(s)(\xi-s)^{1-r} 
{}_0F_2\bigg(2-r,2;\frac{\lambda}{4}(\xi-s)^2\bigg)ds+ \\ 
% 
+\int^\xi_0 \varphi'(s)(\xi-s)^{-r}\sum^\infty_{n=0} 
\frac{1}{(1)_n n!}\bigg[\frac{\lambda}{4} 
\xi(\xi-s)\bigg]^n F\bigg(-r,-n;1-r; 
\frac{\xi-s}{\xi}\bigg)ds - \\ 
% 
-\int^\xi_0 \varphi'(s)(\xi-s)^{-r}{}_0F_1 
\bigg(1-r;\frac{\lambda}{4}\xi(\xi-s)\bigg)ds - \\
% 
-r\xi^{-1-r}\int^\xi_0 \varphi'(s)s\sum^\infty_{n=0} 
\frac{1}{(2)_n n!}\bigg[\frac{\lambda}{4}(\xi-s)s \bigg]^n 
F\bigg(1+n,r;2+n;\frac{s}{\xi} \bigg)ds. 
\end{multline} 
 
\begin{lem} 
Если функция $U(\xi;\eta)\in C(\ov D)$ такова, что $U(\xi;\xi)=\tau(\xi)$ 
достигает наибольшего положительного $($наименьшего отрицательного$)$ 
значения в точке $\xi_0:0<\xi_0<1$, при этом $U(0;\eta)\equiv0$, $0< 
r\le(1+\frac{\lambda}{4}\exp(\frac{\lambda}{4}))^{-1}$, то $D v_-(\xi_0)>0$ 
$(<0)$. 
\end{lem} 
 
\begin{pf} 
Из равенства (9) по условию леммы следует 
\begin{equation} 
D v_-(\xi)=\int^\xi_0 \tau'(s)(\xi-s)^{-r}ds- 
\frac{r\lambda}{4(1-r)}\int^\xi_0 \tau(s)(\xi-s)^{1-r} 
{}_0F_2\bigg(2-r,2;\frac{\lambda}{4}(\xi-s)^2 \bigg)ds. 
\end{equation} 
В равенстве (10) применим формулу интегрирования по частям к первому 
слагаемому, учитывая, что $\tau'(s)=(\tau(s)-\tau(\xi))'$. Получим 
$$ 
\int^\xi_0 \tau'(s)(\xi-s)^{-r}ds=\tau(\xi)\xi^{-r}-r 
\int^\xi_0 (\tau(s)-\tau(\xi))(\xi-s)^{-r-1}ds. 
$$ 
Во втором слагаемом равенства (10) положим 
$\tau(s)=\tau(s)-\tau(\xi)+\tau(\xi)$. Тогда 
\begin{multline*} 
D v_-(\xi)=\tau(\xi)\xi^{-r}\bigg[1-\frac{r\lambda}{4(1-r)} 
\sum^\infty_{n=0}\frac{1}{(2)_n(2-r)_n(2-r+2n)} 
\bigg(\frac{\lambda}{4}\xi^2\bigg)^n\bigg]- \\ 
% 
-\frac{r\lambda}{4(1-r)}\int^\xi_0(\tau(s)-\tau(\xi))(\xi-s)^{1-r} 
{}_0F_2\bigg(2-r,2;\frac{\lambda}{4}(\xi-s)^2\bigg)ds- 
r\int^\xi_0(\tau(s)-\tau(\xi))(\xi-s)^{-r-1}ds. 
\end{multline*} 
Функция $\tau(\xi)$, как непрерывная на отрезке $[0;1]$, принимает на нем
свои наибольшее и наименьшее значения. Пусть $\tau(\xi)$ 
принимает наибольшее положительное значение в точке $0<\xi_0<1$. Положим 
в последнем равенстве $\xi=\xi_0$. В силу того что 
$0<\frac{1}{(2)_n}<\frac{1}{n!}$, $0<\frac{1}{(2-r)_n}<1$ и 
$0<\frac{1}{2-r+2n}<1$, получаем 
$$ 
\tau(\xi_0)\xi^{-r}_0\bigg[1-\frac{r\lambda}{4(1-r)} 
\sum^\infty_{n=0}\frac{1}{(2)_n(2-r)_n(2-r+2n)} 
\bigg(\frac{\lambda}{4}\xi^2\bigg)^n\bigg]> 
\tau(\xi_0)\xi^{-r}_0\bigg[1-\frac{r\lambda}{4(1-r)} 
\exp\bigg(\frac{\lambda}{4} \bigg)\bigg]. 
$$ 
Из условия леммы следует 
$1-\frac{r\lambda}{4(1-r)}\exp(\frac{\lambda}{4})\ge0$. Тогда $D 
v_-(\xi_0)>0$. Аналогично доказывается вторая часть утверждения леммы. 
\end{pf} 
 
\begin{thm} 
Если $f(s)\in C_{[0;s]}$, $\vartheta(x)\in C^1_{[0;1]}$, 
$0<r\le(1+\frac{\lambda}{4}\exp(\frac{\lambda}{4}))^{-1}$ и существует 
решение задачи T, то оно единственно. 
\end{thm} 
 
{\bf Доказательство.}
Предположим, что существуют два различных решения задачи $T$: $U_1(x;y)$ 
и $U_2(x;y)$. Тогда функция $U=U_1-U_2$ будет решением уравнения (1) с 
однородными краевыми условиями. Непрерывная функция $U(x;y)$ достигает в 
$\ov G_+$ наибольшего и наименьшего значений. Так как $U|_\Gamma=0$ и по 
принципу максимального значения (\cite{1}, с.\,492) $U(x;y)$ не может 
достигать наибольшего (наименьшего) значения внутри области $G_+$, то она 
достигает их на $y=0$. Пусть $U(x;0)=\tau(x)$ достигает наибольшего 
(наименьшего) значения в точке $x_0\in(0;1)$. Тогда по лемме 1 \ 
$v_+(x_0)<0$ ($>0$), по лемме 3 \ $D v_-(x_0)>0$ ($<0$). Это противоречит 
условию сопряжения (2). Значит, функция $U(x;0)=\tau(x)$ не может 
достигать наибольшего (наименьшего) значений внутри $(0;1)$, т.\,е. 
$\tau(x)\equiv\const$. Так как $\tau(0)=\tau(1)=0$, то $\tau(x)\equiv0$ 
на $[0;1]$. Из формулы~(8) получаем $U(x;y)\equiv0$ в $\ov G_-$, тогда $D 
v_-(x)\equiv0$ и в силу условия сопряжения (2) $v_+(x)\equiv0$. 
Следовательно, $U(x;y)\equiv0$ в $\ov G_+$, а значит, $U_1(x;y)\equiv 
U_2(x;y)$ в области $G$. Полученное противоречие завершает 
доказательство.
