В [1] и [2] было показано, что интерполируя пару
пространств Бесова $(B^{s_0}_{p_0}(R_n),B^{s_1}_{p_1}( R_n))_{\theta,q}$,
$0<p_i<\infty$, $s_0\ne s_1$, $p_0\ne p_1$, получим пространства типа
$BL^{s,k}_{p,q}$. Пространства этого типа получаются также при
интерполяции пространств Лизоркина--Трибеля
$(F^{s_0}_{p_0,q_0}, F^{s_1}_{p_1,q_1})_{\theta, q}$,
$0<p_i<\infty$. В данной работе эти результаты распространяются
на случай $p_0=\infty $. Ряд известных пространств
отождествляется с $F^s_{\infty,q}(R_n)$ или
$B^s_\infty(R_n)$, например, пространства Г\"ельдера--Зигмунда
$G^s=B^s_\infty$, $bmo=F^s_{\infty,2}$ и $L_p=F^0_{p,2}$.
Здесь получены интерполяционные теоремы для пар пространств вида
$(G,F)$, $(G,B)$, $(bmo, F)$, $(G,L_p)$.

\section{Основные определения и обозначения}
 
Пусть $S'$ --- множество всех комплекснозначных умеренных 
распределений на $R_n$, а $Ff$ --- преобразование Фурье распределения
$f\in S'$. Рассмотрим функциональные пространства нескольких типов.
 
(i) {\it Пространства Лебега с весом} $L_p(\omega)$. Пусть
$0<p\le \infty$, $\omega \ne 0$ --- п.\,в.-весовая функция. Тогда
$\nor{f}{L_p(\omega)} =\nor{f(x) \omega (x)}{L_p}$.
 
(ii) {\it Пространства Лоренца с весом}
$L_{p,q}(\omega, \lambda \mu)$. Пусть
$0<p,q\le\infty$, $\omega (x)$ и $\lambda (x)$ --- измеримые
функции, $\mu$ --- мера, и $\lambda \mu$ --- мера, которая
определена равенством $\lambda\mu (E):=\int\limits_E
\lambda (x)d\mu (x)$.
Тогда
\begin{align*}
\nor{f}{L_{p,q}(\omega,\lambda\mu)}&:=\bigg(
\int_0^\infty(t^{1/p}(f(x) \omega (x))^{*(\lambda
\mu)})^q\frac{dt}{t}\bigg)^{1/q},\quad q<\infty, \\
%
\nor{f}{L_{p,\infty}(\omega,\lambda\mu)}&:=
\sup\limits_{t\in R_+}(t^{1/p}(f(x) \omega
(x))^{*(\lambda\mu)});
\end{align*}
здесь $(g(x))^{*(\lambda\mu)}$ --- невозрастающая
равноизмеримая относительно меры $\lambda\mu $ перестановка
функции $g(x)$, зависящая от аргумента $t\in (0,\infty)$.
 
(iii) {\it Пространства Г\"ельдера--Зигмунда} $G^s(R_n)$. Для
вещественного $s>0$ положим $s=[s]+\{ s\}$, где $[s]$ целое и
$0\le \{ s\} <1$. Пусть
$$
G^s(R_n)=\bigg \{ f\mid f\in C^{[s]}(R_n):
\nor{f}{G^s}= \nor{f}{C^{[s]}}+
\sum_{|\alpha|=[s]}\sup\limits_{x\ne y} \frac{| D^{\alpha }f(x){-}
D^{\alpha}f(y)|}{|x-y|^{\{ s\}}}<\infty,\ x\in R_n,\ y\in R_n\bigg\}
$$
при не целом $s$. Далее, если $f(x)$ --- произвольная функция
на $R_n$ и $h\in R_n$, то для целого $s>0$ положим
$$
G^s(R_n)=\Big\{f\mid f\in C^{s-1}:
\nor{f}{G^s}=\nor{f}{C^{s-1}}+
\sum_{|\alpha |=s-1}\sup\limits_{0\ne h\in R_n}|h|^{-1}
\nor{\Delta^2_hD^\alpha f}{C(R_n)}<\infty\Big\}.
$$
 
(iv) {\it Пространства Бесова $B^s_p$ и
Лизоркина--Трибеля} $F^s_{p,q}$. Если $0<p<\infty$,
$-\infty<s<\infty$, $0<q\le\infty$, то
$$
F^s_{p,q}(R_n)=\{f\mid f\in
S'(R_n):\nor{f}{F^s_{p,q}}=\nor{2^{sj}F^{-1}\phi_jFf}
{L_p[\ell_q]}<\infty\},
$$
где $j=0,1,2,\dotsc$, $\{ \phi _j\}^{\infty}_{j=0}\in \varPhi$, $\varPhi$
--- специальный класс последовательностей функций ([3], с.\,57);
положим $F^s_{\infty,q}(R_n):=(F^{-s}_{1,q'}(R_n))'$ при $1<q\le \infty$,
$B^s_p(R_n):=F^s_{p,p}(R_n)$ при $0<p\le \infty$. 
 
(v) {\it Пространства} $L^{s,k}_{p,q}(Z_+\times R_n)$.
Пусть $-\infty<s<\infty$, $-\infty <k<\infty$, $0<p<\infty$, $0<q\le
\infty$, $Z_+=\{0,1,2,\dotsc\}$. Тогда
$$
\Li :=\{ (f)^{\infty}_{j=0}|f_j\in S': \nor{(f_j)}{\Li}=
\nor{(f_j)}{L_{p,q}(2^{j(s-k/p)},2^{jk}\nu \times m_n)}<\infty\},
$$
где $\nu$ --- мера на $Z_+$ такая, что $\nu (n)=1$.
 
(vi) {\it Пространства} $B\Li (R_n)$. Пусть
$-\infty<s<\infty$, $-\infty<k<\infty$, $0<p<\infty$, $0<q\le\infty$.
Тогда
$$
B\Li =\{f\mid f\in S'(R_n):\nor{f}{B\Li} =
\nor{F^{-1}\phi_jFf}{\Li }<\infty \}.
$$
 
(vii) {\it Пространство} $bmo(R_n)$ (неоднородное BMO). Если
$f\in L^{\mathrm{loc}}_1(R_n)$, $Q$ --- куб
в $R_n$ и $|Q|$ --- объем $Q$, то
$$
f_Q=\frac{1}{|Q|}\int_{Q}f(x)dx
$$
и
$$
bmo(R_n)=\bigg\{f\mid f\in
L^{\mathrm{loc}}_1:\nor{f}{bmo(R_n)}=\sup\limits_{|Q|\le1}\frac{1}{|Q|}
\int_Q |f(x)-f_Q|dx+
\sup\limits_{|Q|>1} \frac{1}{|Q|}\int\limits_Q |f(x)|dx<\infty \bigg\}.
$$
 
\section{Интерполяция в классах пространств Лизоркина-Трибеля и
Бесова в случае $p_0=\infty$}
 
В [1] доказано, что
\begin{equation}
\label{*}
(B^{s_0}_{p_0}(R_n),B^{s_1}_{p_1}(R_n))_{\theta,q}=B\Li(R_n)
\end{equation}
при $-\infty<s_0,s_1,k<\infty$, $0<p_0,p_1<\infty$,
$0<q\le \infty$, $p_0\ne p_1$, $s_0\ne s_1$, $0<\theta <1$,
$s=(1-\theta)s_0+\theta s_1$, $1/p=(1-\theta)/p_0+\theta/p_1$,
$k$ --- угловой коэффициент прямой, проходящей через точки
$(1/p_i,s_i)$, $i=0, 1$. Кроме того,
\begin{equation}
\label{**}
(F^{s_0}_{p_0,q_0}(R_n),F^{s_1}_{p_1,q_1}(R_n))_{\theta,q}=B\Li(R_n)
\end{equation}
при дополнительных условиях $1<p_0,p_1< \infty, 1\le q\le\infty$.
 
Основная цель этого параграфа --- доказать, что формула (\ref{*})
cправедлива при $0<p_0,p_1\le \infty$, а формула (\ref{**}) --- при
$1\le p_0,p_1<\infty$, $1\le q\le \infty$.
 
\begin{lemnonum} При $k\ne 0$ и $q\in [1,\infty ]$ справедливы вложения
$$
BL^{s,k}_{1,1}\subset F^s_{1,q}\subset BL^{s,k}_{1,\infty}.
$$
\end{lemnonum}
 
\begin{cornonum} Пусть $-\infty<s_0,s_1,k<\infty$, $0\le q_0,q_1,q\le
\infty$, $0<p_1<\infty$, $s_0\ne s_1$, $0<\theta <1$,
$s=(1-\theta)s_0+\theta s_1$, $1/p=(1-\theta)+\theta/p_1$, $k$ --- угловой
коэффициент прямой, проходящей через точки $A(1,s_0)$ и
$B(1/p_1,s_1)$. Тогда
$$
(F^{s_0}_{1,q_0}(R_n),F^{s_1}_{p_1,q_1}(R_n))_{\theta,q}=B\Li(R_n).
$$
\end{cornonum}

Теперь можно сформулировать интерполяционную теорему для
пространств $B^s_p(R_n)$ и $F^s_{p,q}(R_n)$ в предельном случае
$p_0=\infty$.
 
\begin{thm}
Пусть $-\infty<s_0,s_1<\infty$, $0<p_1<\infty$, $0<\theta<1$,
$1\le q_1\le\infty$, $1<q_0\le \infty$, $0<q\le\infty$, $s_0\ne s_1$,
$k=p_1(s_1-s_0)$ --- угловой коэффициент прямой, проходящей через
точки $(0,s_0)$ и $(1/p_1,s_1)$. Тогда
$$
(B^{s_0}_{\infty}(R_n),B^{s_1}_{p_1}(R_n))_{\theta,q}=B\Li (R_n).
$$
Если, кроме того, $1\le p_1<\infty$, то
$$
(F^{s_0}_{\infty,q_0}(R_n),F^{s_1}_{p_1,q_1}(R_n))_{\theta,q}=B\Li (R_n).
$$
\end{thm}

\section{Интерполяционные теоремы для пространств Г\"ельдера--Зигмунда,
bmo, Лебега}
 
Как следствия полученных результатов можно сформулировать
следующие утверждения.

\begin{thm} 
Пусть $\mathrm a)$~$s_0>0$, $s_0\ne s_1$, $0<p_1<\infty$,
$0<q\le\infty$, $0<\theta <1$, $s=(1-\theta)s_0+\theta s_1$,
$1/p=\theta/p_1$, $k=p_1(s_1-s_0)$, тогда
$$
(G^{s_0}(R_n),B^{s_1}_{p_1}(R_n))_{\theta,q}=B\Li (R_n);
$$
% vvv pay attn in eng
{\em б)} $(G^{s_0}(R_n),F^{s_1}_{p_1,q_1}(R_n))_{\theta,q}=B\Li(R_n)$
при дополнительных условиях $1\le p_1<\infty$, $1\le q_1\le\infty$.
\end{thm} 

\begin{thm} 
Если $-\infty<s_1<\infty$, $s_1\ne 0$, $1\le p_1<\infty$,
$1\le q_1\le \infty$, $0<q\le \infty$, $0<\theta <1$,
$s=\theta s_1$, $1/p=\theta/p_1$, $k=p_1s_1$, то
$$
(bmo(R_n), F^{s_0}_{p_1,q_1}(R_n))_{\theta,q}=B\Li(R_n).
$$
\end{thm} 

\begin{thm}  
Если $0<s_0$, $1<p_1<\infty$, $0<q\le \infty$, $0<\theta<1$,
$s=(1-\theta)s_0$, $1/p=\theta /p_1$, $k=-p_1s_0$, то
$$
(G^{s_0}(R_n), L_{p_1}(R_n))_{\theta,q}=B\Li (R_n).
$$
\end{thm}
