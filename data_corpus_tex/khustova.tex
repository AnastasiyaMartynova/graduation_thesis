\enabstract{Fundamental solution of the model equation of anomalous diffusion with Riemann--Liouville operator is constructed. Using the properties of the integral transformation with Wright function in kernel, we give estimates for the fundamental solution. When the considered equation transformes into the diffusion equation of fractional order, constructed fundamental solution goes into the corresponding fundamental solution of the diffusion equation of fractional order. General solution of the model equation of anomalous diffusion of fractional order is constructed.

\Section[n]{Введение} 
Пусть $D_{ay}^{\gamma}$ "--- оператор интегро"=дифференцирования в смысле Римана"--~Лиувилля дробного
порядка $\gamma$ с~началом в~точке $a$ и с концом в~точке $y$,
который определяется следующим образом~\cite[c.~28]{add:1} \cite[с.~14]{Pskhu}:   
\begin{gather*}
D_{ay}^{\gamma}\,g(y)=\frac{\text{sign}(y-a)}{\Gamma(-\,\gamma)}
\int _{a}^{y}\frac{g(t)}{|y-t|^{\gamma+1}}dt, \quad  \gamma<0;
\\
D_{ay}^{\gamma}\,g(y)=g(y),  \quad  \gamma=0;
\\ 
D_{ay}^{\gamma}\,g(y)=\text{sign}^{n}(y-a)\,\frac{d^n}{dy^n}\,
D_{ay}^{\gamma-n}\,g(y),  \quad  n-1<\gamma\leqslant n,  \; n\in \mathbb{N}.
\end{gather*} 
Здесь $\Gamma(s)$ "--- гамма-функция Эйлера.


В области $\Omega=\{(x,y): r_{1}<x<r_{2},\ 0< y < T \}$ 
рассмотрим уравнение
\begin{equation} \label{MS1}
{\bf L}\,u(x,y) \equiv |x|^{-b}\,\frac{\partial}{\partial x}\Bigl(|x|^{b}\frac{\partial}{\partial x}\,u(x,y)\Bigr)-
D^{\alpha}_{0y}u(x,y)=f(x,y),
\end{equation}
где $|b|<1,$ $0<\alpha\leqslant 1.$ 

Пусть $\Omega^{+}=\Omega\cap\{x>0 \},$ $\Omega^{-}=\Omega\cap\{x<0\},$ 
$\bar{\Omega}$ "--- замыкание области $\Omega.$ 


\smallskip
{\small\sc Определение.}
\textit{Регулярным   решением}    уравнения   \eqref{MS1}   в  области $\Omega$  назовем 
функцию  $u=u(x,y),$ удовлетворяющую уравнению  \eqref{MS1}  в области 
$\Omega^{+}\cup\Omega^{-},$ и такую, что $y^{1-\alpha}u\in{C(\bar{\Omega})},$ 
$|x|^b u_{x}\in{C(\Omega)},$ $u_{xx},$ $D_{0y}^{\alpha} u\in{C(\Omega^{+}\cup\Omega^{-})}.$

\smallskip

Уравнение \eqref{MS1} при  $\alpha=1,$ $f(x,y)\equiv 0$ обращается в уравнение
\begin{equation} \label{MS1_alpha=1}
u_{xx}(x,y)+\frac{b}{x} u_{x}(x,y) - u_y(x,y)=0,
\end{equation}
которое при $x>0$ было объектом исследования работы \cite{Tersenov}.
Уравнение \eqref{MS1_alpha=1} с помощью замены $|\xi|=\left[ |x|/(1-b)\right]^{1-b}$  сводится к уравнению
$$ 
|\xi|^{q} \, u_{\xi\xi}(\xi,y)- u_y(\xi,y)=0, \quad  q=2b/(b-1),
$$ 
и оно при $q<1$ было исследовано в работе \cite{Pagani}, а при 
$q=1$ "--- в работах  \cite{Kepinski, Arena}.

В случае, когда $b=0,$ $0<\alpha<2,$
 уравнение \eqref{MS1} совпадает с 
диффузионно"=волновым уравнением, исследованным в работах многих авторов. Например, в
работах \cite{Mainardi, MainardiApplied1996} в терминах функции Райта построено фундаментальное решение 
диффузионно"=волнового уравнения с оператором Римана"--~Лиувилля, в работе~\cite{Pskhu2}  построено фундаментальное решение 
многомерного диффузионно"=волнового уравнения с оператором Джрбашяна"--~Нерсесяна.
В ~работах \cite{VorKilbas, Gekk} методом интегральных преобразований  
исследована задача Коши для более общих диффузионных и волновых уравнений дробного порядка
с производной Римана"--~Лиувилля. Решения исследуемых задач выписаны в
терминах $H$-функции Фокса и функции типа Райта.
В работах \cite{Kochubei_1, Kochubei_2} исследовалась задача Коши для уравнения
диффузии дробного порядка с регуляризованной дробной производной. 

В работе  А.~В.~Псху \cite{Pskhu}  методом функции Грина построены решения первой, второй и смешанной краевых задач в прямоугольной области и решена задача Коши для диффузионно"=волнового уравнения. 
Более полную библиографию работ, посвященных таким уравнениям, можно найти также в~\cite{Pskhu}.

Уравнение вида  \eqref{MS1}, а именно уравнение 
$$
D^{2/d_w}_{0t}P(r,t)=\frac{1}{r^{d_s-1}}\frac{\partial}{\partial r}\Bigl(r^{d_s-1}\frac{\partial P(r,t)}{\partial r}\Bigr),
$$
где $d_w$ и  $d_s$ характеризуют фрактальную размерность среды,
$P(r,t)$ "--- плотность пространственного распределения частиц в момент времени $t,$
было предложено в~работе  \cite{MetzlerGlockleNonnenmacher} для описания процессов переноса в средах, имеющих фрактальную размерность.

 Интерес к изучению уравнения \eqref{MS1} вызван также его приложениями при решении  задач физики, астрономии и других прикладных наук
[\citen{GionaRoman}--\citen{Uchaikin-2}].

\Section{Некоторые специальные функции} Приведем некоторые свойства двух специальных функций, которые понадобятся для дальнейшего изложения.

\smallskip

{\small\sc Определение.}
Функция $I_{\nu}(z),$ определяемая рядом 
\begin{equation}\label{I_{nu}(z)}
I_{\nu}(z)=\sum\limits_{n=0}^{\infty}\frac{1}{n! \Gamma(\nu+n+1)}
\Bigl(\frac{z}{2} \Bigr)^{\nu+2n}, 
\end{equation}
называется {\it модифицированной цилиндрической или бесселевой функцией первого рода порядка~$\nu$}  \cite{Kuzn, BatemanErdelyi}.

\smallskip

При $|\arg{z}|\leqslant {\pi}/{2} -\varepsilon,$  где $\varepsilon$ "--- произвольное малое 
положительное число, и $|z|\to \infty$ справедлива 
асимптотическая формула
\begin{equation}\label{AsimptI}
I_{\nu}(z)= \frac{\;\;e^{z}}{\sqrt{2\pi z}} 
\,\big[ \,1+O\left(z^{-1}\right)\big].
\end{equation}


Имеют место следующие  формулы дифференцирования:
\begin{gather}\label{Iz}
\frac{d}{dz}\left[z^{\nu}I_{\nu}(z)\right]=
z^{\nu}I_{\nu-1}(z),
\\
\label{Iz^-1}
\frac{d}{dz}\left[z^{-\nu}I_{\nu}(z)\right]=
z^{-\nu}I_{\nu+1}(z),
\end{gather}
откуда вытекают  рекуррентные формулы
\begin{gather}
\label{I_(nu-1)(z)}
z I'_{\nu}(z)+\nu I_{\nu}(z)=z I_{\nu-1}(z),
\\
\label{I_(nu+1)(z)}
z I'_{\nu}(z)-\nu I_{\nu}(z)=z I_{\nu+1}(z).
\end{gather}

Модифицированная цилиндрическая функция $I_{\nu}(z)$ порядков $\nu=1/2 $  и $\nu=-1/2$ 
выражается через элементарные функции
\begin{equation}\label{I_1/2andI_-1/2}
I_{{1}/{2}}(z)=\frac{e^{z}-e^{-z}}{\sqrt{2\pi z}},\quad 
I_{-{1}/{2}}(z)=\frac{e^{z}+e^{-z}}{\sqrt{2\pi z}}.
\end{equation}

\smallskip
{\small \sc Определение.}
Функция $\phi\,(\rho,\delta;z),$ определяемая рядом 
$$
\phi\,(\rho,\delta;z)=\sum\limits_{n=0}^{\infty}\frac{z^{n}}{n! \Gamma(\rho n+\delta)}, \quad  \rho>-1,
$$
называется {\it функцией  Райта} \cite{Wright1, GorenfloLuchkoMainardi}.

\smallskip

При $\rho=-1/2$ и  $\delta=1/2$ функция $\phi\,(\rho,\delta;z)$ выражается через экспоненциальную функцию по формуле
\begin{equation}\label{phi(-1/2,1/2,z)}
\sqrt{\pi} 
\phi\Bigl(-\frac{1}{2}, \frac{1}{2}; - z\Bigr)=e^{-{z^2}/{4}}.
\end{equation}

\Section{Интегральное преобразование с функцией Райта в ядре}
В работе \cite[с.~72]{Pskhu} определено интегральное преобразование для функции $v(y),$ заданной на положительной полуоси 
\begin{equation}
\label{IntRight}
A^{\alpha, \mu}\,v(y)\equiv (A^{\alpha, \mu} v)(y)=
\int _{0}^{\infty} v(t)\,y^{\mu-1}\,
\phi\left(-\alpha,\mu;- t y^{- \alpha} \right)dt, \quad  0<\alpha<1.
\end{equation}
В случае, когда $\mu=0,$ введено обозначение $A^{\alpha, 0} v(y)=A^{\alpha} v(y).$
Если преобразование $A^{\alpha,  \mu}$ применяется к функции, зависящей от нескольких переменных,
то в случае необходимости с помощью нижнего индекса обозначается переменная, по которой проводится преобразование. Например, $A^{\alpha, \mu}_{y} v(x,y).$

Интеграл  \eqref{IntRight} будет сходиться, если функция $v(y)$ интегрируема на любом конечном отрезке положительной полуоси и выполняются асимптотические неравенства
$$
\begin{array}{ll}
|v(y)|<c y^{\lambda}, &  \lambda>-1, \; \mu\neq 0; \; \lambda>-2, \; \mu=0, \;\; y\to 0,
\\
|v(y)|<c \exp\big(k\,y^{\lambda}\big), &  \lambda<1/(1-\alpha), \;\; y\to \infty,
\end{array}
$$
где $c$ и  $k$ "---  положительные постоянные.

Приведем некоторые свойства преобразования $A^{\alpha,\, \mu}$ \cite[c.~78--83]{Pskhu}.

\begin{itemize}
\item[$1^\circ.$] {\it Пусть $v(y)$ непрерывна в точке $y=0$ и дифференцируема при  $y>0.$ 
Тогда 
$$
D_{0y}^{\alpha} A^{\alpha,  \mu} v(y)=A^{\alpha, \mu} v'(y)+\frac{y^{\mu-1}}{\Gamma(\mu)}\,v(0).
$$
В частности$,$
\begin{equation}\label{IntRight3}
D_{0y}^{\alpha} A^{\alpha} v(y)=A^{\alpha} v'(y).
\end{equation}
 }
 
\item[$2^\circ.$] {\it Пусть $0\leqslant \mu \leqslant \alpha$ и $\lim\limits_{y\to 0}D_{0y}^{-\mu/\alpha}\,v(y)=v_0<\infty.$ Тогда
\begin{equation}\label{IntRight4}
\lim\limits_{y\to 0}D_{0y}^{\alpha-1} A^{\alpha, \mu} v(y)=v_0.
\end{equation}
}

\item[$3^\circ.$] {\it Если $u(y) \leqslant v(y)$ и $\mu \geqslant 0,$ то
\begin{equation}\label{IntRight5}
A^{\alpha,  \mu} u(y) \leqslant A^{\alpha, \mu}\,v(y).
\end{equation}
}

\end{itemize}



Для степенной функции и функции Райта справедливы следующие формулы~\mbox{\cite[c.~74, 84]{Pskhu}}:
\begin{gather}
\label{Ay^{delta-1}}
A^{\alpha,\, \mu}\,y^{\delta-1}=
y^{\alpha\delta+\mu-1}\frac{\Gamma(\delta)}{\Gamma(\alpha\delta+\mu)}, \quad  \delta>0, \; \mu\neq 0; \; \; \delta\neq 0, \; \mu= 0,
\\
\label{ARight}
A^{\alpha,\, \mu}\,y^{\delta-1}\phi\,(\rho,\delta;-\,c\,y^{\rho})=
y^{\alpha\delta+\mu-1} \phi\,(\alpha\rho,\alpha\delta+\mu;-\,c\,y^{\alpha\rho}), \quad  \delta>\rho.
\end{gather}
Здесь $c$ "--- положительная постоянная.

\Section{Фундаментальное решение} 

{\small \sc Определение.}
Функцию $w=w(x,y;\xi,\eta)$ назовем {\it фундаментальным решением} уравнения \eqref{MS1}, если
\begin{itemize}
\item[1)] для любых фиксированных $\xi$ и $\eta$ функция $w(x,y;\xi,\eta)$ как функция переменных $x$ и $y,$ $x\neq 0,$ $y>\eta,$
удовлетворяет уравнению
$$
|x|^{-b} \frac{\partial}{\partial x}\Bigl(|x|^{b}\frac{\partial}{\partial x}
w(x,y;\xi,\eta)\Bigr)-
D^{\alpha}_{\eta y}\,w(x,y;\xi,\eta)=0;
$$
\item[2)] для любой функции $h(x)\in C[x_1;x_2]$ 
выполняется соотношение
\begin{equation}
\label{Glava-2-1-(4)}
\lim_{\eta \to y} \int _{x_1}^{x_2}
|\xi|^{b} h(\xi) D^{\alpha-1}_{y\eta}w(x,y;\xi,\eta) d\xi=h(x), \quad  x_1 < x < x_2.
\end{equation}


\end{itemize}

\smallskip

Обозначим через 
\begin{equation}\label{Formula_G}
\Gamma(x,\xi,y)=A_{y}^{\alpha} g(x,\xi,y), 
\end{equation}
где
\begin{equation}\label{Formula_g}
g(x,\xi,y)=\frac{ |x|^{\beta} |\xi|^{\beta}}{4y} \exp \Bigl({-\frac{x^2+\xi^2}{4y}}\Bigr)
\left[I_{\beta}\Bigl(\frac{|x\xi|}{2y} \Bigr)+I_{-\beta}\Bigl(\frac{|x\xi|}{2y} \Bigr) \right],\quad 
\beta=\frac{1-b}{2}.
\end{equation}

Справедливы следующие свойства функции $\Gamma(x,\xi,y).$

\smallskip
\hypertarget{sv1o}{}
{\small \sc Свойство  $1^\circ.$}
{\it Для функции $\Gamma(x,\xi,y)$ при  $|x\xi|\leqslant 2y$ имеют место следующие оценки:
\begin{gather*}
\left|\Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot y^{\alpha\beta-1}, 
\\
\left|\frac{\partial^{n+1}}{\partial x^{n+1} } \Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot |x|^{2\beta-n-1} |\xi|^{2\beta} y^{-\alpha\beta-1}, \quad  \beta\neq 1/2,
\\
\left|\frac{\partial^{2n}}{\partial x^{2n} } \Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot y^{-\alpha(2n+1)/2+\alpha-1}, \quad  \beta=1/2,
\\
\left|\frac{\partial^{2n+1}}{\partial x^{2n+1} }\,\Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot |\xi|y^{-\alpha(2n+1)/2-1}, \quad  \beta=1/2,
\\
\left|D^{\alpha}_{0y}\Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot y^{\alpha\beta-\alpha-1},
\end{gather*}
а при $|x\xi| > 2y$ "--- оценки
$$
\left|\frac{\partial^{n}}{\partial x^{n} }\,\Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot |x|^{\beta+(2n-1)/2}\,|\xi|^{\beta-1/2}A^{\alpha}_{y}\,y^{-(2n-1)/2-1}\exp \Bigl( {-\frac{(x-\xi)^2}{4y}}\Bigr),
$$
$$
\left|D^{\alpha}_{0y}\,\Gamma(x,\xi,y)\right|\leqslant \text{\rm const}\cdot |x|^{\beta+3/2}|\xi|^{\beta-1/2}\,
A^{\alpha}_{y}y^{-5/2}\exp \Bigl({-\frac{(x-\xi)^2}{4y}}\Bigr),
$$
где $n=0, 1, 2, \dots$}


\smallskip

\begin{proof} Оценки при $|x\xi|\leqslant2y$ следуют из представлений \eqref{I_{nu}(z)}, \eqref{Formula_G}, \eqref{Formula_g}, формул дифференцирования 
\eqref{Iz}, \eqref{Iz^-1} для функции $I_{\nu}(z)$,  неравенства \eqref{IntRight5},
 формулы \eqref{Ay^{delta-1}} и $0<e^{-z}<1$ при $z>0.$ Для вывода оценок при $|x\xi|> 2y$ вместо \eqref{I_{nu}(z)} нужно использовать асимптотическую формулу \eqref{AsimptI}.
 \end{proof}

{\small\sc Свойство $2^\circ.$}
{\it Функция $\Gamma(x,\xi,y)$ при  $x\neq 0,$ $y>0$  и фиксированном $\xi$  является решением уравнения}
\begin{equation}\label{GammaOdn}
{\bf L}\,\Gamma(x,\xi,y)=0.
\end{equation}

\smallskip

\begin{proof}
Действительно, обозначим 
$$
g_1(x,\xi,y)=\frac{ x^{\beta} \xi^{\beta}}{4y}\exp\Bigl({-\frac{x^2+\xi^2}{4y}}\Bigr)
I_{\beta}\Bigl(\frac{x\xi}{2y} \Bigr),
$$
$$

I_{-\beta}\Bigl(\frac{x\xi}{2y} \Bigr).
$$
Считая для упрощения доказательства, что $x>0,$ $\xi>0,$  продифференцируем равенство \eqref{Formula_G} по $x$, используя формулы 
\eqref{Iz} и \eqref{Iz^-1}
при $\nu=\beta$ и $\nu=-\beta$ соответственно.

\end{equation}
где 
$$

$$
\begin{equation}\label{g_1Partial_x}

\begin{equation}\label{g_2Partial_x}

Умножим \eqref{g_1Partial_x} на $x^{1-2\beta}$ и продифференцируем полученное равенство по $x,$
используя формулу  \eqref{Iz^-1} при $\nu=\beta-1.$  Воспользуемся затем формулой 
\eqref{I_(nu-1)(z)} при $\nu=\beta$ и умножим полученное равенство на $x^{2\beta-1}.$
Аналогично поступим и с равенством \eqref{g_2Partial_x}. Умножим его на

{\it  В области $\Omega_y$ функция $$\Gamma(x,\xi,y-\eta)=A_{t}^{\alpha} g(x,\xi,t)\big|_{t=y-\eta}$$ при фиксированных $x$ и $y$  как функция переменных $\xi$ и $\eta,$ $\xi\neq 0,$ $0<\eta<y,$ является решением сопряженного уравнения}
$$
{\bf L}^{*}\Gamma(x,\xi,y-\eta)\equiv |\xi|^{-b}\frac{\partial}{\partial \xi}\Bigl(|\xi|^{b}\frac{\partial}{\partial \xi}
\Gamma(x,\xi,y-\eta)\Bigr)-
D^{\alpha}_{y\eta}
\Gamma(x,\xi,y-\eta)=0.
$$

\smallskip

Свойство \hyperlink{sv3o}{$3^\circ$} является следствием свойства  \hyperlink{sv2o}{$2^\circ$}.


\smallskip
\hypertarget{sv4o}{}

{\small\sc Свойство  $4^\circ.$} {\it Для любой функции $h(x)\in C[\,x_1;x_2 ]$ 
выполняется соотношение }
\begin{equation}\label{SvoystvoOfGamma}
\lim_{\eta \to y} \int _{x_1}^{x_2} |\xi|^{b}\,h(\xi) D^{\alpha-1}_{y\eta}\Gamma(x,\xi,y-\eta) d\xi=h(x), \quad  x_1 < x < x_2.
\end{equation}

\smallskip

\proof
Обозначив через $t=y-\eta,$ перепишем левую часть равенства 
\eqref{SvoystvoOfGamma} в виде
$$
\lim_{\eta \to y} \int _{x_1}^{x_2}
|\xi|^{b} h(\xi) D^{\alpha-1}_{y\eta} \Gamma(x,\xi,y-\eta) d\xi=
\lim_{t \to 0} \int _{x_1}^{x_2}
|\xi|^{b} h(\xi) D^{\alpha-1}_{0t}\,\Gamma(x,\xi,t)  d\xi.
$$
Тогда в силу формулы \eqref{IntRight4} можно записать
\begin{multline}
\lim_{t \to 0} \int _{x_1}^{x_2}
|\xi|^{b}\,h(\xi)\,D^{\alpha-1}_{0t} \Gamma(x,\xi,t) d\xi=
\\
=\lim_{t \to 0}\left[
\int _{x_1}^{x_2}
|\xi|^{b} g(x,\xi,t) [h(\xi)-h(x)] d\xi +h(x)\int _{x_1}^{x_2}
|\xi|^{b} g(x,\xi,t) d\xi  \right]=
\\
=\lim_{t \to 0}\left[J_1(x,t)+J_2(x,t)\right].
\end{multline}
Разбивая промежуток интегрирования на части, представим $J_1(x,t)$
  в виде суммы трех слагаемых:
\begin{multline}
J_1(x,t)=\int _{x_1}^{x-\varepsilon}
|\xi|^{b} g(x,\xi,t) [h(\xi)-h(x)] d\xi+
\\
\label{3Slagaemye}
+\int _{x-\varepsilon}^{x+\varepsilon}
|\xi|^{b} g(x,\xi,t) [h(\xi)-h(x)] d\xi+
\int _{x+\varepsilon}^{x_2}
|\xi|^{b} g(x,\xi,t) [h(\xi)-h(x)] d\xi, 
\end{multline}
где $\varepsilon$ "--- произвольное малое положительное число.


Из \eqref{I_{nu}(z)}, \eqref{AsimptI} и \eqref{Formula_g} следуют оценки
\begin{equation}\label{|x*xi|<2y}
|g(x,\xi,t)|\leqslant \mathop{\rm const} \cdot t^{\beta-1}, \quad  |x\xi| \leqslant 2t,
\end{equation}
\begin{equation}\label{|x*xi|>2y}
|g(x,\xi,t)|\leqslant \mathop{\rm const}  \cdot x^{\beta-1/2} \xi^{\beta-1/2} t^{-1/2} \exp \Bigl({-\frac{(x-\xi)^2}{4t}}\Bigr), 
\quad  |x\xi|> 2t.
\end{equation}

В силу оценки \eqref{|x*xi|>2y} первое и третье слагаемые в \eqref{3Slagaemye} стремятся к нулю при $t \to 0.$

Обозначим через $\omega(\varepsilon)=\sup|h(x)-h(\xi)|,$ $\,\xi\in [x-\varepsilon, x+\varepsilon].$ 
%Так как функция $h(x)\in C[x-\varepsilon, x+\varepsilon],$ функция %то
%$\omega(\varepsilon)\to 0$ при $\varepsilon\to 0.$
Функция %то
${\omega(\varepsilon)\to 0}$ при $\varepsilon\to 0$,
так как функция $h(x)\in C[x-\varepsilon, x+\varepsilon]$.
В силу произвольности выбора $\varepsilon$ и оценок  \eqref{|x*xi|<2y} и  \eqref{|x*xi|>2y} легко заметить, что    
второе слагаемое в \eqref{3Slagaemye} тоже стремится к нулю.

Преобразуем интеграл $J_2(x,t).$  Для этого, учитывая \eqref{AsimptI} и \eqref{Formula_g}, запишем
\begin{multline}
\lim_{t \to 0} J_2(x,t)=
\\
=\frac{|x|^{\beta-1/2} h(x)}{2\sqrt{\pi}} \lim_{t \to 0}
\biggl[ \int _{x_1}^{x}
\frac{|\xi|^{1/2-\beta}}{\sqrt{t}}\exp\Bigl({-\frac{(x-\xi)^2}{4t}}\Bigr)d\xi+
\\
+
\int _{x}^{x_2}
\frac{|\xi|^{1/2-\beta}}{\sqrt{t}}\exp\Bigl({-\frac{(x-\xi)^2}{4t}}\Bigr)d\xi  \biggr].
\end{multline}
В последних двух интегралах сделаем замену переменной интегрирования по формулам  $\xi=x-2\sqrt{t}\,s$ и  $\xi=x+2\sqrt{t}\,s$ соответственно. 
В результате будем иметь
\begin{multline}
\lim_{t \to 0} J_2(x,t)=
\\
=\frac{|x|^{\beta-1/2} h(x)}{\sqrt{\pi}} \lim_{t \to 0}
\biggl[ \int _{0}^{\frac{ x-x_{_1}}{2\sqrt{t}}}
\frac{e^{-s^2}ds}{|x-2\sqrt{t} s|^{\beta-1/2}}+
 \int _{0}^{\frac{ x_{_2}-x}{2\sqrt{t}}}
\frac{e^{-s^2}ds}{|x+2\sqrt{t} s|^{\beta-1/2}} \biggr].
\end{multline}
Переходя к пределу в последнем равенстве и учитывая известную формулу
$$2\int _{0}^{+\infty}e^{-s^2}ds=\sqrt{\pi},$$ 
получим:
$$
\lim_{\eta \to y} \int _{x_1}^{x_2}
|\xi|^{b}\,h(\xi)\,D^{\alpha-1}_{y\eta}\,\Gamma(x,\xi,y-\eta) \,d\xi=\lim\limits_{t \to 0}\,J_2(x,t)=h(x). \quad \square
$$

\smallskip

Из свойств \hyperlink{sv3o}{$3^\circ$}  и \hyperlink{sv4o}{$4^\circ$}  следует, что функция $\Gamma(x,\xi,y),$
определяемая равенством \eqref{Formula_G}, является фундаментальным решением уравнения \eqref{MS1}.

При $\beta=1/2$ $(b=0)$ из \eqref{I_1/2andI_-1/2}, \eqref{Formula_G} и  \eqref{Formula_g}  следует
$$
\Gamma(x,\xi,y-\eta)=A_{t}^{\alpha}\,g(x,\xi,t)\big|_{t=y-\eta},\quad 
g(x,\xi,t)=\frac{1}{2\sqrt{\pi t}}  \exp \Bigl({-\frac{(x-\xi)^2}{4t}}\Bigr).
$$
Учитывая \eqref{phi(-1/2,1/2,z)}, последнее равенство можно записать в виде
$$
g(x,\xi,t)=\frac{1}{2\sqrt{t}}\phi\Bigl(-\frac{1}{2}, \frac{1}{2};- \frac{\,|x-\xi|}{\sqrt{t}}\Bigr).
$$
Тогда, используя формулу \eqref{ARight} при $\mu=0,$ $\delta=1/2,$ $\rho=-1/2$ и  $c=|x-\xi|,$  получим функцию
$$
\Gamma(x,\xi,y-\eta)=\frac{\,(y-\eta)^{\sigma-1}}{2} \phi\Bigl(-\sigma, \sigma;- \frac{|x-\xi|}{(y-\eta)^{\sigma}}\Bigr), 
\quad  \sigma=\frac{\alpha}{2},
$$
которая совпадает  с фундаментальным решением уравнения диффузии дробного порядка, приведенным в \cite[с.~127]{Pskhu}.

\Section{Общее представление решения}
Имеет место следующая теорема об общем представлении решения уравнения \eqref{MS1}.

\smallskip

\hypertarget{kh:thm}{}

{\small \sc Теорема.} {\it Пусть $y^{1-\alpha}f(x,y)\in C(\bar{\Omega}),$ $\varphi(x)\in C[r_1;r_2],$ а функция $v=v(x,y;\xi,\eta)$ удовлетворяет следующим условиям\/{\rm:}

\begin{itemize}
\item[$1)$] 
в области $\Omega_y\setminus\{\xi=0\}$ функция $v$ является решением уравнения
\begin{equation}\label{Sopryazh_q}
{\bf L}^{*}v(x,y;\xi,\eta) =q(x,y;\xi,\eta),
\end{equation}
где $\eta^{1-\alpha}q\in L(\Omega_y);$

\item[$2)$]  для любой функции $h(x)\in C[x_1;x_2],$ $r_1\leqslant x_1<x_2 \leqslant r_2,$ выполняется соотношение
\begin{equation}\label{SvoystvoOfv}
\lim_{\eta \to y} \int _{x_1}^{x_2}
|\xi|^{b}\,h(\xi)\,D^{\alpha-1}_{y\eta}\,v(x,y;\xi,\eta) d\xi=h(x), \quad x_1 < x < x_2;
\end{equation}

\item[$3)$] функция $v$  непрерывна в $\bar{\Omega}\times\bar{\Omega}_y 
\setminus \{y=\eta\}$ 
вместе с  $|\xi|^b v_{\xi},$ $D_{y\eta}^{\alpha}v$ и $y^{1-\alpha}v,$
и для любых точек 
$(x,y) \in \Omega$ и $(\xi,\eta) \in \Omega_y$ выполняется неравенство
$$
|v(x,y;\xi,\eta)|\leqslant \text{\rm const} \cdot (y-\eta)^{\alpha\beta-1}.
$$

\end{itemize}

Если функция $u(x,y)$ является регулярным решением уравнения \eqref{MS1},
имеет непрерывную
и интегрируемую производную с весом $|x|^{b}\,u_x(x,y)$ вплоть до участков границы $x=r_1$
и $x=r_2$ и удовлетворяет условию 
\begin{equation}\label{Glava-2-1-(5)}
\Gamma(\alpha)\lim_{y \to 0} y^{1-\alpha}\,u(x,y)=\varphi(x), \ \ r_1 < x < r_2,
\end{equation}
то для любой точки $(x,y)\in \Omega$ имеет место соотношение
\begin{multline}
\label{G4}
u(x,y)=\int _{r_1}^{r_2} |\xi|^{b} v(x,y;\xi,0)\,\varphi(\xi) d\xi+
\\
+\int _{0}^{y} 
\Bigl[|r_{2}|^{b} v(x,y;r_2,\eta) u_{\xi}(r_2,\eta)-|r_{1}|^{b} v(x,y;r_1,\eta) u_{\xi}(r_1,\eta)- 
\\
 -|r_{2}|^{b} v_{\xi}(x,y;r_2,\eta) u(r_2,\eta)+|r_{1}|^{b} v_{\xi}(x,y;r_1,\eta) u(r_1,\eta) \Bigr] d\eta+
\\
+ \int _{r_1}^{r_2} \int _{0}^{y}
|\xi|^{b}\bigl[u(\xi,\eta) q(x,y;\xi,\eta)-v(x,y;\xi,\eta)f(\xi,\eta)\bigr] d\eta\,d\xi.
\end{multline}
}

\smallskip

Предварительно,  воспользовавшись оператором
$$
\bigl( S_{st}^{\delta} g \bigr)(\eta) \equiv S_{st}^{\delta} g(\eta)=
\frac{\text{sign}(t-s)}{\pi}\int _{s}^{t}\frac{g(\xi)}{\xi-\eta}
\Bigl| \frac{\xi-s}{s-\eta} \Bigr|^{\delta}d\xi,
$$ 
приведем лемму \cite[c. 118]{Pskhu}, которая понадобится при доказательстве теоремы.


\smallskip

{\small \sc Лемма.} {\it Пусть $0\leqslant \eta < y_{\varepsilon} < y,$ 
$y_{\varepsilon}=y-\varepsilon.$ 
Если $$g(\eta) \leqslant \text{\rm const}\cdot (y-\eta)^{\alpha\beta-1},$$ 
то для любого $\theta\in[0; 1]$ справедлива оценка
$$
|S_{y_{\varepsilon}y}^{\delta}g(\eta)|\leqslant \text{\rm const}\cdot 
\varepsilon^{\alpha\beta+\delta-\theta}(y_{\varepsilon}-\eta)^{\theta-\delta-1}.
$$
}

\smallskip

{\it  Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о\/ т\,е\,о\,р\,е\,м\,ы.} 
Рассмотрим выражение
\begin{equation} \label{GGrina}
\int _{r_1}^{r_2}\int _{0}^{y_{\varepsilon}}
|\xi|^{b}\big[\,v(x,y;\xi,\eta)\,{\bf L}u(\xi,\eta)-u(\xi,\eta)\,{\bf L}^{*}v(x,y;\xi,\eta)\,\big]\, d\eta\,d\xi,
\end{equation}
где $y_{\varepsilon}=y-\varepsilon,$ $\varepsilon>0.$

В работе \cite[c. 121]{Pskhu} доказано равенство
\begin{multline}
\int _{0}^{y_{\varepsilon}}\left(v\,D^{\alpha}_{0\eta}u-
u\,D^{\alpha}_{y\eta}v\right)d\eta=
u(\xi,y_{\varepsilon})\,D^{\alpha-1}_{yy_{\varepsilon}}v(x,y;\xi,y_{\varepsilon})-
\\ \label{Pskhu1}
-\varphi(\xi)v(x,y;\xi,0)+R\,(x,y,y_{\varepsilon},\xi),
\end{multline}
где 
$$
R(x,y,y_{\varepsilon},\xi)=
\sin\pi(\alpha-1) \cdot \int _{0}^{y_{\varepsilon}}
S^{1-\alpha}_{y_{\varepsilon}y}v\cdot D^{\alpha}_{0\eta}u\,d\eta
-\sin\pi\alpha\cdot\varphi(\xi) S^{1-\alpha}_{y_{\varepsilon}y}v\big|_{\eta=0}.
$$

Учитывая \eqref{Pskhu1} и соотношение 
$$v \frac{\partial}{\partial \xi}
\Bigl(|\xi|^{b} \frac{\partial u}{\partial \xi}\Bigr)-
u \frac{\partial}{\partial \xi}
\Bigl(|\xi|^{b} \frac{\partial v}{\partial \xi}\Bigr)=
\frac{\partial}{\partial \xi}
\Bigl(|\xi|^{b} v \frac{\partial u}{\partial \xi}-
|\xi|^{b} u \frac{\partial v}{\partial \xi}\Bigr),
$$
преобразуем \eqref{GGrina}:
\begin{multline}
\int _{r_1}^{r_2}\int _{0}^{y_{\varepsilon}}
|\xi|
^{b}\big[ v(x,y;\xi,\eta) {\bf L}u(\xi,\eta)-u(\xi,\eta) {\bf L}^{*}v(x,y;\xi,\eta) \big]d\eta\,d\xi=
\\
=\int _{0}^{y_{\varepsilon}} 
|\xi|^{b}\big[ v(x,y;\xi,\eta) u_{\xi}(\xi,\eta)  -v_{\xi}(x,y;\xi,\eta) u(\xi,\eta) \big]\bigg|^{\xi=r_2}_{\xi=r_1}  d\eta+
\\
+\int _{r_1}^{r_2} |\xi|^{b}\big[ v(x,y;\xi,0) \varphi(\xi) %- \\
\label{Formula1}
-u(\xi,y_{\varepsilon})  D^{\alpha-1}_{yy_{\varepsilon}}v(x,y;\xi,y_{\varepsilon})-
R (x,y,y_{\varepsilon},\xi)\,\big]d\xi.
\end{multline}

Из выше приведенной леммы следует, что 
$$
\lim\limits_{\varepsilon\to 0} 
\int _{r_1}^{r_2}|\xi|^{b}
R\,(x,y,y_{\varepsilon},\xi) d\xi=0.$$
Поэтому из \eqref{MS1}, \eqref{Sopryazh_q}, \eqref{SvoystvoOfv} и \eqref{Formula1}, устремляя  
$\varepsilon$ к нулю, получим соотношение \eqref{G4}. $\square$

\Section[n]{Заключение} В работе в терминах интегрального преобразования с функцией Райта в ядре построено и исследованы свойства фундаментального решения модельного уравнения аномальной диффузии дробного порядка, которое может быть использовано при моделировании задач переноса в средах с фрактальной структурой. Показано, что когда исследуемое уравнение переходит в уравнение диффузии дробного порядка, построенное  фундаментальное решение переходит в 
фундаментальное решение  соответствующего уравнения. 
Построено общее представление решения рассматриваемого уравнения в прямоугольной области.
