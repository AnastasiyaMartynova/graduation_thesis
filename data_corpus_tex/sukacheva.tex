\engabstract{The homogeneous thermoconvection problem of the non-compressible viscoelastic Kelvin-Voight fluid of the non-zero order is considered. The conducted research is based on the results of the semilinear Sobolev type equations theory, because the first initial value problem for the corresponding system of the differential equations in private derivatives is reduced to the abstract Cauchy problem for the specified equation. The concepts of the $p$-sectorial operator and the resolving semigroup of operators of the Cauchy problem for the corresponding linear homogeneous Sobolev type equation are used. The existence and uniqueness theorem of the solution which is a quasi-stationary semi-trajectory is proved. The complete description of the phase space is obtained.}

\Section[n]{Введение}
Система уравнений
\begin{equation}
\left \{
\begin{array}{l}
  
(1- \lambda {\nabla}^2)\boldsymbol{v}_t=
\nu {\nabla}^2\boldsymbol{v}-(\boldsymbol{v} \cdot \nabla)\boldsymbol{v}+ \sum\limits_{l=1}^{k} \beta_l {\nabla}^2 { w}_l - g \boldsymbol{q} \theta -\boldsymbol{p} +\boldsymbol{f},
\\
  0=\nabla (\nabla \cdot \boldsymbol{v}),\quad  \frac{\partial  { w}_l}{\partial t} = {v}+ {\alpha}_l { w}_l,\quad 
{\alpha}_l \in {\mathbb R}_-,\quad 
l\in K, \vspace{2mm}
\\ 
   \theta_t=
\text{\ae} {\nabla}^2 \theta -\boldsymbol{v} \cdot \nabla \theta +\boldsymbol{v} \cdot
\boldsymbol{q}
\end{array} \right. 
\label{suk:eq:1}
\end{equation}
моделирует эволюцию скорости $\boldsymbol{v}=(v_1, v_2 \ldots, v_n)$, 
$v_i=v_i(x, t)$, градиента давления $\boldsymbol{p}=(p_1, p_2, \ldots, p_n)$, $p_i=p_i(x, t)$ и температуры
  $\theta =\theta (x, t)$ несжимаемой вязкоупругой жидкости Кельвина"--~Фойгта  порядка $k>0$~[\citen{suk:cite:1}].  
 Параметры $\lambda \in \mathbb R$, $\nu \in {\mathbb R}_{+}$ и $\text{\ae} \in {\mathbb R}_{+}$
характеризуют упругость, вязкость и теплопроводность жидкости соответственно; $g \in {\mathbb R}_{+} $ "---  ускорение свободного
падения; вектор $\boldsymbol{q} =(0, \ldots, 0, 1) $ "---  орт в~${\mathbb R}^n$. 
Параметры ${\beta}_l \in {\mathbb R}_{+}$ определяют время ретардации (запаздывания) давления.
 Свободный член $\boldsymbol{f}=(f_1, f_2, \ldots, f_n)$, $f_i= f_i(x, t)$ отвечает внешнему воздействию на жидкость,  ${w}_{l}={w}_{l}(x, t)$, $l\in K$ "--- некоторые функции, $K=\{1, 2, \dots, k\}$.

 Рассмотрим разрешимость первой начально-краевой задачи
\begin{equation}
\begin{array}{ccll}
 \boldsymbol{v}(x,0)=\boldsymbol{v}_0(x), &  { w}_l(x,0)={ w}_{l_0}(x), &  \theta (x,0)={\theta }_0(x), \;  \forall x \in \Omega ;\\
\boldsymbol{v}(x,t)=0,  &  w_l(x,t)=0,&   \theta (x,t)=0, \; \forall (x,t) \in \partial \Omega \times {\mathbb R}_{+},\; l\in K   \hspace{-2mm}
\end{array} 
\label{suk:eq:2}
\end{equation}
для однородной системы \eqref{suk:eq:1} ($f \equiv 0$).
Здесь $\Omega \subset {\mathbb R}^n$, $n\in \{2, 3, 4\}$ "---  ограниченная область с границей $\partial \Omega$ класса $C^{\infty}.$ 
Ранее задача \eqref{suk:eq:1}, \eqref{suk:eq:2} в случае, когда $k=0$, $f=f(x)$,  изучалась Г.\,А.~Свиридюком~[\citen{suk:cite:2}].% Для
%несжимаемой вязкоупругой жидкости Кельвина-Фойгта поряка $k>0$  с
%нулевым свободным членом $f(x,t)$ указанная задача рассматривается
%впервые.

Статья состоит из трёх частей. 
В~первой части приводятся известные результаты из теории полулинейных уравнений соболевского типа, основанные на понятии $p$-секториального оператора и~полугрупповом подходе~[\citen{suk:cite:3, suk:cite:4}]. 
Во второй части проводится редукция однородной задачи \eqref{suk:eq:1}, \eqref{suk:eq:2} к задаче Коши для полулинейного уравнения соболевского типа. 
В~третьей части устанавливается существование квазистационарных полутраекторий и описывается фазовое пространство исходной задачи.

\Section {Полулинейные уравнения соболевского типа}
Пусть ${\cal U}$ и ${\cal F}$  "---  банаховы пространства,
оператор  $L \in {\cal L}({\cal U}; {\cal F}),$
причём $~\ker L \ne \{ 0 \}$; оператор  $M \hm \in {\cal C}l({\cal U};{\cal F})$.  
Обозначим через ${\cal U}_M= \{ u \in {\rm dom\,} M: \| u \| = {\|M u \|}_{\cal F}+ {\| u \|}_{\cal U}\}.$ 
Пусть оператор $F \in {\cal C}^{\infty }({\cal U}_M;{\cal F}).$

Рассмотрим задачу Коши
\begin{equation}
u(0)=u_0 
\label{suk:eq:3}
\end{equation}
для {\it полулинейного уравнения соболевского типа}
\begin{equation}
L \dot u = M u + F(u).
\label{suk:eq:4}
\end{equation}

 {\it Локальным решением} (далее просто  "---  {\it решением}) задачи \eqref{suk:eq:3}, \eqref{suk:eq:4} назовём вектор"=функцию 
 $u \in {\cal C}^{\infty }((0, T); {\cal U}_M),$ удовлетворяющую уравнению \eqref{suk:eq:4} и такую, что $u(t) \to u_0$ при $t \to 0+$.

Будем рассматривать задачу \eqref{suk:eq:3}, \eqref{suk:eq:4} при условии, что  оператор $M$ сильно $(L,~p)$-секториален~[\citen{suk:cite:3, suk:cite:4}].
Известно, что при этом условии решение задачи \eqref{suk:eq:3}, \eqref{suk:eq:4}  может быть не единственным~[\citen{suk:cite:5}]. 
Мы ограничиваемся поиском только таких решений уравнения \eqref{suk:eq:4}, которые являются {\it квазистационарными полутраекториями}.

\begin{definition}[1]
 Пусть  ${\cal U}={\cal U}_0\oplus {\cal U}_1,$ причём  $\ker L \subset {\cal U}_0.$ 
 Решение $u=v+w,$ где $v(t)\in {\cal U}_0,$   $w(t) \in {\cal U}_1$ при всех $t \in (0,T),$   назовём    {\it квазистационарной полутраекторией}, если $L \dot v \equiv 0.$
\end{definition}

 %\begin{remark}[1] В динамическом случае понятие квазистационарной полутраектории
%совпадает с понятием квазистационарной траектории [5].
%\end{remark}

Также хорошо известно~[\citen{suk:cite:6, suk:cite:7, suk:cite:8}], что решения задачи \eqref{suk:eq:3}, \eqref{suk:eq:4} существуют не для всех $u_0 \in  {\cal{U}}_M.$
Поэтому введём

 \begin{definition}[2] Множество ${\cal{B}}\subset{\cal U}_M$ назовём {\it фазовым пространством} уравнения \eqref{suk:eq:4}, если для любой точки $u_0
\in {\cal B}$ существует единственное решение задачи \eqref{suk:eq:3}, \eqref{suk:eq:4}, причём $u(t) \in{\cal{B}}.$
\end{definition}

 %\begin{remark}[2] В неавтономном случае вместо понятия  {\it
%фазового пространства} уравнения (4) [6, 7] целесообразно
%использовать понятие расширенного фазового пространства.
%\end{remark}

В силу того, что оператор $M$ сильно $(L,p)$-секториален, пространства ${\cal U}$ и ${\cal F}$ расщепляются в~прямые суммы ${\cal U}={\cal U}^0 \oplus {\cal U}^1 ,$ ${\cal F}={\cal F}^0 \oplus {\cal F}^1 ,$ где
${\cal U}^0$, ${\cal F}^0 $  "---  ядра, а ${\cal U}^1$, ${\cal F}^1 $  "---  образы аналитических разрешающих полугрупп $U^t$,~$F^t$
%$$U^t= \frac{1}{2\pi i}\int_{\Gamma} R^{L}_{\mu }(M)e^{\mu t}d\mu ,\quad
%F^t= \frac{1}{2\pi i}\int_{\Gamma} L^{L}_{\mu }(M)e^{\mu t}d\mu
%\eqno{(5)}$$
%($\Gamma \subset {\mathbb Rho}^L (M)$  "---   контур такой,
%что $\arg \mu \to \pm \Theta
%$ при $|\mu |\to + \infty$)
линейного однородного уравнения
\begin{equation}
L \dot u = Mu.
\label{suk:eq:5}
\end{equation}
Обозначим через $L_k (M_k)$ сужение оператора $L(M)$ на ${\cal U}^k$ $({\cal U}^k \cap {\rm dom\,} M)$, $k\hm \in \{0, 1\}.$ 
Тогда $L_k: {\cal U}^k \to {\cal F}^k ,$ $M_k: {\cal U}^k  \cap {\rm dom\,} M\to {\cal F}^k$, $k\in \{0, 1\},$  причём
$M_0$ и~$L_1$  являются линейными непрерывными операторами и~имеют ограниченные обратные операторы.

В силу этих результатов~[\citen{suk:cite:3, suk:cite:4}] приведём задачу \eqref{suk:eq:3}, \eqref{suk:eq:4} к~эквивалентной системе, которую назовём {\it нормальной формой} задачи \eqref{suk:eq:3}, \eqref{suk:eq:4}:
\begin{equation}
 R \dot u^0 =u^0+G(u),\quad u^0(0)=u_0^0, \quad \dot u^1=Su^1+H(u),\quad u^1(0)=u_0^1.
\label{suk:eq:6}
\end{equation}
Здесь $u^k \in {\cal U}^k$, $k\in \{0, 1\}$, $u=u^0+u^1$, операторы $R=M_0^{-1}L_0$, $S=L_1^{-1}M_1$, $ G=M_0^{-1}(I-Q)F$, $H=L_1^{-1}QF.$
%($Q \in {\cal{L}}({\cal F})$ {\it
%--- } проектор, расщепляющий пространство ${\cal F}$ требуемым
%образом).

Далее будем изучать такие  квазистационарные полутраектории,  для  которых $R \dot u^0 \equiv 0 .$
Для этого предположим, что  оператор  $R$   "---  {\it бирасщепляющий}~[\citen{suk:cite:9}], т.\,е. его ядро $\ker R$ и образ ${\rm im}\, R$ дополняемы в~пространстве ${\cal U}$. 
Обозначим ${\cal U}^{00}=\ker R$,  а   через   ${\cal U}^{01}={\cal U}^{0} \ominus {\cal U}^{00}$   обозначим   некоторое   дополнение   к~подпространству ${\cal U}^{00}$. 
Тогда первое уравнение \eqref{suk:eq:6} примет следующий	 вид:
$$R \dot u^{01} =u^{00}+u^{01}+G(u), \quad u=u^{00}+u^{01}+u^1 .$$
%где $u=u^{00}+u^{01}+u^1 .$


\begin{theorem}[1]{\it Пусть  оператор  $M$  сильно  $(L,p)$-секториален, а~оператор    $R$  "---   бирасщепляющий.    
Пусть    существует квазистационарная полутраектория   \eqref{suk:eq:4}. 
Тогда она удовлетворяет соотношениям
\begin{equation}
0 =u^{00}+u^{01}+G(u),\quad u^{01}={\rm const} .
\label{suk:eq:7}
\end{equation}
}
\end{theorem}

Теорема 1 даёт  необходимые условия существования квазистационарной полутраектории уравнения \eqref{suk:eq:4}. 
Рассмотрим теперь достаточные  условия.
Известно, что при условии сильной $(L,p)$-секториальности оператора $M$ оператор  $S$  секториален.  
Следовательно,  он порождает  на  ${\cal U}^1$  аналитическую   полугруппу,   которую  мы обозначим через 
$\{ U_1^t: t \in \bar {\mathbb R}_{+} \},$  так  как оператор $U_1^t$ есть  сужение  оператора  $U^t$  на  ${\cal U}^1.$
Из того, что  ${\cal U}={\cal U}^0 \oplus {\cal U}^1$ следует, что существует проектор $P\in {\cal L}({\cal U}),$ соответствующий данному
расщеплению. Оказывается, что $P\in {\cal L}({\cal U}_M),$  и~тогда ${\cal U}_M={\cal U}^0_M \oplus {\cal U}^1_M,$
причём вложение ${\cal U}^k_M \subset {\cal U}^k$,  $k\in\{0, 1\}$ плотно  и непрерывно~[\citen{suk:cite:3, suk:cite:4}].

\begin{theorem}[2]{\it Пусть  оператор  $M$  сильно $(L,p)$-секториален, оператор $R$  "---  бирасщепляющий, оператор $F \in {\cal C}^ \infty ({\cal U}_M;{\cal F}).$  
Пусть выполняются следующие условия:
\begin{itemize}
\item[${\rm (A1)} $] в некоторой окрестности ${\cal O}_{u_0} \subset {\cal U}_M$ точки $u_0$ выполнено соотношение
\begin{equation}
0=u^{01}_0+(I-P_R)(G(u^{00}+u_0^{01}+u^1)); 
\label{suk:eq:8}
\end{equation}

\item[${\rm   (A2)} $] проектор $P_R \in {\cal L} ({\cal U}_M^0),$ и~оператор $I+P_RG'_{u_0}:{\cal U}_M^{00} \to {\cal U}_M^{00}$  "--- 
топлинейный изоморфизм $( {\cal U}_M^{00}={\cal U}_M \cap {\cal U}^{00});$

\item[{\rm (A3)}] для аналитической полугруппы $\{ U^t_1:t \in \bar {\mathbb R}_{+} \}$ выполнено соотношение
\begin{equation}
\int_0^{\tau} {\| U_1^t \|}_{{\cal L}({\cal U}^1;\,{\cal U}_M^1)} dt< \infty
\quad \forall \tau \in {\mathbb R}_{+}. 
\label{suk:eq:9}
\end{equation}
\end{itemize}

Тогда   существует  единственное   решение   задачи  \eqref{suk:eq:3}, \eqref{suk:eq:4}, являющееся квазистационарной полутраекторией уравнения
\eqref{suk:eq:4}.}
\end{theorem}

Приступим  к   проверке  условий  \eqref{suk:eq:9}  и  \eqref{suk:eq:10}.  
Построим пространство   ${\cal U}_{\alpha}={\cal U}_1 \times {\stackrel{\circ}{W_2^1}} (\Omega ) .$   
Данное  пространство, очевидно,  будет интерполяционным пространством  для  пары ${[{\cal U}, {\cal U}_M]}_{\alpha }$, причём $\alpha =1/2 .$   
Как отмечено выше, полугруппа $ \{ U^t: t \in {\mathbb R}_{+} \} $ продолжается до группы $\{ V^t_1: t \in \mathbb R \}$ на ${\cal U}_1^1 ,$ где
$V_1^t$  "---   сужение  оператора   $V^t$   на  ${\cal U}_1^1 .$
Поскольку  ${\cal U}_M^1={\cal U}_M \cap {\cal U}_1^1$  (по построению) и~оператор $B_1$ непрерывен  (теорема~4),  то  в~силу равномерной ограниченности полугруппы $\{ U^t: t \in {\mathbb R}_{+} \}$ имеем 
\begin{equation}
\int_0^{\tau} {\| V_1^t \|}_{{\cal L}({\cal U}_1^1;\,{\cal U}_M^1)} dt \leq
{\rm const} {\| B_1 \|}_{{\cal L} ({\cal U}_1; \,{\cal F}_1)}
\int_0^{\tau} {\| V_1^t \|}_{{\cal L}({\cal U}_1^1)} dt < \infty
\quad \forall \tau \in {\mathbb R}_{+}. 
\label{suk:eq:13}
\end{equation}
Далее, в силу неравенства Соболева \cite[гл. 9]{suk:cite:10}, полугруппа $\{ W^t: t \in { \bar {\mathbb R}}_{+} \} $ удовлетворяет оценке
\begin{equation}
\int_0^{\tau} {\| W^t \|}_{{\cal L} ({\rm dom~} B_2;\,{\stackrel{\circ}{W_2^1} (\Omega)})} dt < \infty.
\label{suk:eq:14}
\end{equation}
Положим ${\cal U}_{\alpha}^1={\cal U}_{\alpha} \cap {\cal U}^1$, где ${\cal U}^1={\cal U}_1^1 \times {\cal U}_2$. Тогда из \eqref{suk:eq:13} и \eqref{suk:eq:14} вытекает 

\smallskip

\begin{lemma}[3]{\it В условиях леммы $1$ выполняется соотношение \eqref{suk:eq:14}.}
 \end{lemma}

\smallskip

Наконец,  выполняя  требование   \eqref{suk:eq:10},  найдём оператор $H.$ 
Оператор $H$ естественно представить в виде $H = H_1 \otimes H_2,$ где $H_1 =A^{-1}_{11}(I-Q_0-Q_1)F_1,$ а 
$H_2 \equiv F_2$ ($A_{11}$  "---  сужение оператора $A_1$ на ${\cal U}^1_1$). 
Включение \mbox{$H \in {\cal C}^{\infty}({\cal U}_M^1;{\cal U}_{\alpha}^1) $} показывается аналогично тому, как было показано включение $F \in {\cal C}^{\infty}({\cal U}_M;{\cal F})$.

Таким образом, все условия теоремы 2 выполнены. Поэтому справедлива
\begin{theorem}[8]{\it Пусть  ${\lambda}^{-1} \not\in \sigma (A) \cup \sigma (A_{\sigma})$. Тогда при  любом $u_0$ таком, что $u_0\hm\in {\cal A},$ и~некотором  $T \in {\mathbb R}_{+}$ существует  единственное решение \mbox{$u \hm \in {\cal C}^{\infty} ( (0,T);$} ${\cal U}_M  )$ задачи
\eqref{suk:eq:1},  \eqref{suk:eq:4}, являющееся квазистационарной полутраекторией, причём $u(t) \in {\cal A}$.}
\end{theorem}
