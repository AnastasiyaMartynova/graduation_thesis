\section{Предварительные сведения}

В работе
получены условия, при которых решение $x$ задачи
\begin{gather*}
(t-a)^{\mu_1}(b-t)^{\mu_2}
\Ddot x(t)=(Tx)(t)+f(t),\quad t\in[a,b],\\ x(a)=0,\quad x(b)=0,
\end{gather*}
существует, единственно и удовлетворяет неравенству
\begin{equation}\label{0}
%$$
x(t)<0\ \ \text{при всех} \ \ t\in(a,b)
%$$
\end{equation}
при каждой почти всюду (п.\,в.) неотрицательной $f\in L[a,b]$, $f\not\equiv0$.

Работа \cite{Br1} была посвящена необходимым условиям для
выполнения неравенства \re{0}. Здесь результаты
\cite{Br1} дополняются  общим достаточным условием и эффективными
коэффициентными признаками.  Используется терминология и
обозначения из \cite{Br1}.

Рассмотрим краевую задачу
\begin{equation}\label{1}
\begin{array}{c}
(\L x)(t)\eqdef\pi(t)\Ddot x(t)-(Tx)(t)=f(t),\quad t\in[a,b],\\[2mm]
x(a)=\alpha,\quad x(b)=\beta,
\end{array}
\end{equation}
где $\pi(t)=(t-a)^{\mu_1}(b-t)^{\mu_2}$,
$\mu_1,\mu_2\in[0,1]$; $f$ принадлежит банахову  пространству
$\SL$ суммируемых на $[a,b]$ вещественных функций; $T$ ---
линейный ограниченный оператор из банахова пространства
непрерывных вещественных функций $\SC$ в $\SL$. Если хотя бы
одно из чисел $\mu_1$, $\mu_2$ отлично от нуля, то уравнение $\L
x=f$ называется сингулярным \cite{Kig}, \cite{Lab}.

Будем искать решение задачи \re{1} в банаховом пространстве $D$
функций, определенном в \cite{Br1}.  Отметим, что
в пространстве $\D$ задача
\begin{gather*}
(\L_0 x)(t)\eqdef\pi(t)\Ddot x(t)=f(t),\quad t\in[a,b],\\
x(a)=\alpha,\quad x(b)=\beta,
\end{gather*}
при  любых $f\in\SL$, $\aa, \bb\in\R$ имеет единственное решение
$$
x(t)=(G_0f)(t)+\aa\dfrac{t-a}{b-a}+\bb\dfrac{b-t}{b-a},\quad t\in[a,b],
$$
где $G_0:\SL\to\SC$ --- линейный интегральный оператор
$$
(G_0f)(t)=\int_a^bG_0(t,s)f(s)\, ds,\quad t\in[a,b],
$$
явный вид которого приведен в \cite{Br1}.

Если при всех $f\in\SL$, $\aa,\bb\in\R$ задача \re{1} имеет
единственное решение $x\in\D$, то оператор~$G$, ставящий в
соответствие каждой функции $f\in\SL$ решение задачи \re{1} при $\aa=0$
и $\bb=0$, называется оператором Грина этой задачи. Как всякий
линейный ограниченный оператор  из $\SL$ в $\SC$, оператор Грина
$G$ имеет интегральное представление \cite{AMR}
$$
(Gf)(t)=\int_a^bG(t,s)f(s)\, ds,\quad t\in[a,b].
$$
Ядро этого представления --- функция $G(t,s)$ --- называется
функцией Грина задачи \re{1}. В случае однозначной разрешимости
при всех правых частях решение задачи \re{1} имеет представление
$$
x=Gf+\aa x_a+\bb x_b,
$$
где $G$ --- оператор Грина, $x_a$ и $x_b$ --- решения задач
\begin{alignat}{3}
\label{oza}
\L x_a&=0,&\quad x_a(a)&=1,&\quad x_a(b)&=0, \\
%
\label{ozb}
\L x_b&=0,&\quad x_b(a)&=0,&\quad x_b(b)&=1.
\end{alignat}

Будем называть
оператор Грина $G:\SL\to\SC$ {\it строго антитонным}, если
выполнено неравенство $(Gf)(t)<0$ при всех $t\in(a,b)$
для каждой функции $f\in\SL$, $f(t)\ge0$ при п.\,в.  $t\in[a,b]$,
$f\not\equiv0$.
Отметим, что оператор Грина $G_0$ строго антитонен.


Достаточные условия строгой антитонности оператора Грина
получены, например, в работах \cite{AD}--\cite{AAB}. В данной статье,
используя в основном идеи работы \cite{Kob}, уточняем и
обобщаем результаты работ \cite{Kob}, \cite{Lih} об условиях строгой
антитонности оператора Грина несингулярной краевой задачи. Здесь
будут получены эффективные условия, гарантирующие
однозначную разрешимость сингулярной задачи \re{1} и строгую
антитонность ее оператора Грина.

Будем предполагать, что линейный ограниченный
оператор $T:\SC\to\SL$ изотонен.
Такой оператор имеет представление в виде интеграла Стильтьеса
\begin{equation*}%\label{st}
(Tx)(t)=\int_a^bx(s)\,d_sr(t,s),
\end{equation*}
где функция $r(t,s)$ измерима в
квадрате $[a,b]\times[a,b]$; $\int\limits_a^b|r(t,s)|\,dt<\infty$ при
п.\,в. $s\in[a,b]$; функции $r(t,\cdot)$ имеют ограниченное
изменение на $[a,b]$ при каждом $t\in[a,b]$;  $\int\limits_a^b
\Vr\limits_{s\in[a,b]} r(t,s)\, dt<\infty$, причем при всех
$t\in[a,b]$ функции $r(t,\cdot)$ не убывают на $[a,b]$ (\cite{KVP}, с.\,317).


\section{Основные результаты}

\begin{thm}\label{t1}
Если оператор $B\eqdef G_0+G_0TG_0$ антитонен, то задача $\eqref{1}$
однозначно разрешима и ее оператор Грина $G$ строго антитонен,
причем $G_0\le G\le B$.
\end{thm}

\begin{thm}\label{t2}
Оператор $B$ антитонен тогда и только тогда, когда выполнены
неравенства
$$
\int_a^b\frac{s-a}{\pi(s)}F(s)\,ds\le b-a,\quad
\int_a^b\frac{b-s}{\pi(s)}G(s)\,ds\le b-a,
$$
где
$$
F(s)\eqdef\ii r(s,\te)\,d\te-(b-a)r(s,a+0),\quad
G(s)\eqdef (b-a)r(s,b-0)-\ii r(s,\te)\,d\te.
$$
\end{thm}

\begin{note}
Если линейный оператор $T$ определен равенством
\begin{equation}\label{ttt}
(Tx)(t)=\sum_{i=1}^kp_i(t)x_{h_i}(t),\quad t\in[a,b],
\end{equation}
где $p_i\in\SL$, $p_i(t)\ge0$ при $t\in[a,b]$,
$i=1,\dotsc,k$,
$$
x_{h_i}(t)\eqdef
\begin{cases}
x(h_i(t)),&\text{если $h_i(t)\in[a,b]$;}\\
0,&\text{если $h_i(t)\notin[a,b]$,}
\end{cases}
$$
$h_i:[a,b]\to\R$, $i=1,\dotsc,k$, --- измеримые функции, то
$$
F(s)=\sum_{i=1}^k\w p_i(s)(b-h_i(s)),\quad
G(s)= \sum_{i=1}^k\w p_i(s)(h_i(s)-a),
$$
где
$$
\w p_i(s)\eqdef
\begin{cases}
p_i(s),&\text{если $h_i(s)\in(a,b)$;}\\
0,&\text{если $h_i(s)\notin(a,b)$,}
\end{cases}
\quad s\in[a,b],\ \ i=1,\dotsc,k.
$$
\end{note}

\begin{cor}\label{s1}
Если оператор $T:\SC\to \SL$ имеет вид \re{ttt},
где $p_i\in\SL$, $p_i(t)\ge0$ при $t\in[a,b]$, $i=1,\dotsc,k$, и
$$
h_i(t)-a\le M(t-a),\quad b-h_i(t)\le M(b-t),\quad
t\in[a,b],\ \ i=1,\dotsc,k,
$$
при такой постоянной $M>1$, что
$$
\ii(s-a)^{1-\mu_1}(b-s)^{1-\mu_2}\sum_{i=1}^kp_i(s)\,ds\le\frac{b-a}{M},
$$
то задача \re{1} однозначно разрешима, ее оператор Грина $G$
строго антитонен и $G_0\le G\le B$.
\end{cor}

\begin{thm}\label{t3}
Если выполнены неравенства
$$
\int_a^b\frac{s-a}{\pi(s)}\w F(s)\,ds\le b-a,\quad
\int_a^b\frac{b-s}{\pi(s)}\w G(s)\,ds\le b-a,
$$
где
$$
\w F(s)\eqdef\ii r(s,\te)\,d\te-(b-a)r(s,a),\quad
\w G(s)\eqdef (b-a)r(s,b)-\ii r(s,\te)\,d\te,
$$
то решения задач $\eqref{oza}$ и $\eqref{ozb}$ положительны в интервале
$(a,b)$.
\end{thm}

\begin{note}
Из теорем \ref{t2} и \ref{t3} следует, что в условиях теоремы
\ref{t3} задача \re{1} имеет единственное решение при
всех $f\in\SL$, $\alpha,\beta\in\R$, и справедлива теорема о
дифференциальном неравенстве:  {\it
если $x,y\in\D$, $x\ne y$ и
\begin{gather*}
(\L x)(t)\le (\L y)(t) \ \ \text{при п.\,в.}\ \ 
t\in[a,b],\\ x(a)\ge y(a),\quad x(b)\ge y(b),
\end{gather*}
то
$$
x(t)>y(t)\ \ \text{при} \ \ t\in(a,b).
$$
}
\end{note}

\begin{cor}\label{s2}
Если оператор $T:\SC\to \SL$ имеет вид \re{ttt},
где $p_i\in\SL$, $p_i(t)\ge0$ при $t\in[a,b]$, $i=1,\dotsc,k$, и
выполнены неравенства
\begin{gather*}
\ii(s-a)^{1-\mu_1}(b-s)^{-\mu_2}\sum_{i=1}^k\wh
p_i(s)(b-h_i(s))\,ds\le b-a,\\
%
\ii(s-a)^{-\mu_1}(b-s)^{1-\mu_2}\sum_{i=1}^k\wh
p_i(s)(h_i(s)-a)\,ds\le b-a,
\end{gather*}
где
$$
\wh p_i(s)\eqdef
\begin{cases}
p_i(s),&\text{если $h_i(s)\in[a,b]$;}\\
0,&\text{если $h_i(s)\notin[a,b]$,}
\end{cases}
\quad s\in[a,b],\ \ i=1,\dots,k,
$$
то задача \re{1} однозначно разрешима, ее оператор Грина
строго антитонен и  решения задач \re{oza}, \re{ozb} положительны
в $(a,b)$.
\end{cor}



\section{Доказательства основных результатов}


\begin{pf*}{Доказательство теоремы \protect{\ref{t2}}}
Линейный ограниченный оператор $B:\SL\to\SC$ имеет
представление $(Bf)(t)=\int\limits^b_a B(t,s)f(s)\,ds$, $t\in[a,b]$, где
\begin{gather}\label{B}
B(t,s)=G_0(t,s)+\frac{1}{\pi(s)(b-a)}\bigg[
\frac{b-t}{b-a}\int_a^t\frac{\t-a}{\pi(\t)}K(s,\t)\,d\t+\notag \\
+\frac{t-a}{b-a}\int_t^b\frac{b-\t}{\pi(\t)}K(s,\t)\,d\t
\bigg],\quad t,s\in(a,b), \\
%
K(s,\t)=\int_s^br(\t,\te)\,d\te(s-a)+
\int_a^sr(\t,\te)\,d\te(s-b),\ \ s,\t\in(a,b).\notag
\end{gather}
Функция $B(t,s)$ непрерывна при $t,s\in(a,b)$.
Оператор $B$ антитонен тогда и только тогда, когда $B(t,s)\le0$
при всех $t,s\in(a,b)$.
Рассмотрим случай $a<s\le t<b$. Тогда $B(t,s)\le0$ в том и
только том случае, когда
$$
\int_a^t\frac{\t-a}{\pi(\t)}\frac{K(s,\t)}{s-a}\,d\t+
\frac{t-a}{b-t}\int_t^b\frac{b-\t}{\pi(\t)}\frac{K(s,\t)}{s-a}\,d\t
\le b-a,\quad t,s\in(a,b).
$$
Функция $K(s,\t)/(s-a)$ не возрастает по $s$ при каждом
$\t\in[a,b]$, причем
$\lim\limits_{s\to a+0}\frac{K(s,\t)}{s-a}=F(\t)$.
Таким образом, $B(t,s)\le0$ при $a<s\le t<b$ тогда и
только тогда, когда
$$
H(t)\eqdef\int_a^t\frac{\t-a}{\pi(\t)}F(\t)\,d\t+
\frac{t-a}{b-t}\int_t^b\frac{b-\t}{\pi(\t)}F(\t)\,d\t
\le b-a,\quad t\in(a,b).
$$
Кроме того, из предыдущего следует, что в условиях теоремы
$B(t,s)=0$ при некотором $s\in(a,t)$ в том и только том случае,
когда
\begin{equation}\label{rem1}
r(\t,\te)\equiv r_1(\t)
\ \ \text{при} \ \ \te\in(a,s)\ \ \text{и при п.\,в.}\ \  \t\in[a,b].
\end{equation}
Следовательно, в этом случае $B(t,\te)=0$ при всех
$\te\in(a,s]$.

Функция $H(t)$ не убывает,
причем $H(t)\le b-a$ при всех $t\in(a,b)$ тогда и только тогда,
когда $\int\limits^b_a\frac{\t-a}{\pi(\t)}F(\t)\,d\t\le b-a$.

Аналогично заключаем, что $B(t,s)\le0$ при всех $a<t\le
s<b$ тогда и только тогда, когда
$\int\limits^b_a\frac{b-\t}{\pi(\t)}G(\t)\,d\t\le b-a$.
Кроме того, в условиях теоремы $B(t,s)=0$ при некотором $s\in(t,b)$ в
том и только том случае, когда
\begin{equation}\label{rem2}
r(\t,\te)\equiv r_2(\t)\ \ \text{при} \ \ \te\in(s,b)\ \ \text{при п.\,в.}
\ \ \t\in[a,b].
\end{equation}
Следовательно, в этом случае
$B(t,\te)=0$ при всех $\te\in[s,b)$.
\end{pf*}

Всюду ниже через $\r(A)$ обозначаем спектральный
радиус комплексного расширения линейного оператора $A$,
$I$ --- тождественный оператор.


\begin{lem}\label{l1}
$\r(TG_0)=\r(G_0T)$.
\end{lem}

\begin{pf}
Здесь рассматриваем комплексные расширения
операторов в комплексификациях пространств (\cite{KA}, с.\,477).
Пусть $\l$ --- комплексное число, $\l\ne0$.  Задача
\begin{equation}\label{w1}
\l\pi\Ddot x=Tx+f,\quad x(a)=0,\ \ x(b)=0,
\end{equation}
эквивалентна уравнению
\begin{equation}\label{w2}
\l x=G_0Tx+G_0f
\end{equation}
в пространстве $\SC$ и уравнению
\begin{equation}\label{w3}
\l z=TG_0 z+f,
\end{equation}
где $z\in\SL$ связан с $x$ равенством $x=G_0z$.

Если оператор $\l I-G_0T:\SC\to\SC$ обратим, то уравнение
\re{w2}, следовательно, и задача \re{w1} имеют единственное
решение при всех $f\in\SL$. Поэтому и уравнение \re{w3} имеет
единственное решение при всех $f\in\SL$. Таким образом, оператор
$\l I-TG_0:\SL\to\SL$ обратим.

Если оператор $\l I-TG_0:\SL\to\SL$ обратим, то \re{w3}, а
следовательно, \re{w1} и \re{w2} имеют единственное решение
при любых $f\in\SL$. Обратимость оператора $\l I-G_0T:\SC\to\SC$
в этом случае следует из леммы 2 работы \cite{MR}. Для удобства
приведем доказательство. Положим в \re{w2} $f\equiv Tg$, где
$g\in\SC$, и определим $y\eqdef x+g$. Тогда $y$ --- единственное
решение уравнения $\l y= G_0Ty+\l g$ при любой $g\in\SC$. Таким
образом, оператор $\l I-G_0T$ обратим,
ненулевые точки спектров операторов $TG_0$ и $G_0T$ совпадают,
поэтому совпадают и спектральные радиусы этих операторов.
\end{pf}


\begin{lem}\label{l2}
Если $B\eqdef G_0+G_0TG_0\le0$, то $\r(TG_0)=\r(G_0T)<1$.
\end{lem}

Отметим, что в работах \cite{Kob}, \cite{Lih} условия
$B\le0$ и $\r(G_0T)<1$ рассматривались как независимые.

\begin{pf}
Обозначим $\l\eqdef\r(G_0T)=\r(TG_0)$.
Изотонный оператор
$$
K\eqdef-TG_0:\SL\to\SL
$$
слабо вполне непрерывен (\cite{Zab}, с.\,135).  Таким образом, $K^2$ ---
вполне непрерывный оператор (\cite{DSh}, c.\,549).  Из утверждений
книг (\cite{DSh}, VII.4.6) и (\cite{F}, с.\,401) следует, что
$\l=\r(K)$ --- позитивное собственное число оператора $K$,
т.\,е. существует такая функция $v\in\SL$, $v(t)\ge0$ при п.\,в.
$t\in[a,b]$, $v\noeq0$, что $\l v=Kv$.  Тогда
$Bv=G_0(v+TG_0v)=G_0(v-Kv)=(1-\l)G_0v$. Отсюда следует, что
если $\l>1$, то $(Bv)(t)>0$, $t\in(a,b)$, что противоречит
антитонности оператора~$B$.

Предположим, что $\l=1$.  Тогда  $B$ --- антитонный
интегральный оператор, непрерывное в $(a,b)\times(a,b)$ ядро
которого определено равенством \re{B}, и существует такая п.\,в.
неотрицательная функция $v\in\SL$, $v\noeq0$, что $(Bv)(t)=0$
при всех $t\in[a,b]$. Пусть $E\subset[a,b]$ --- множество
ненулевой меры, на котором $v(t)>0$. Пусть $t_1\in(a,b)$ ---
такая точка, что для любой ее окрестности $\omega$ мера Лебега
пересечения $\omega\cap E$ положительна. Очевидно, что по
крайней мере две такие точки существуют. Тогда $B(t,t_1)=0$ при
всех $t\in[a,b]$, в противном случае $(Bv)(t)<0$ в некоторой
точке $t\in[a,b]$. Отсюда и из \re{rem1}, \re{rem2} следует, что
при некоторых функциях $C_1(t)$, $C_2(t)$ выполнены равенства
$r(t,s)=C_1(t)$ при $s\in(a,t_1)$ и $r(t,s)=C_2(t)$ при
$s\in(t_1,b)$. Так как есть другие точки, обладающие тем же
свойством,  что и $t_1$, то $r(t,s)=C(t)$ при всех $t\in(a,b)$
и при некоторой функции $C(t)$. Тогда оператор $TG_0$ нулевой и
число $\l=1$ не является  его спектральным радиусом.
Таким образом, $\l<1$.
\end{pf}


\begin{pf*}{Доказательство теоремы \protect{\ref{t1}}}
Задача $\L x=f$, $x(a)=0$, $x(b)=0$ эквивалентна уравнению
\begin{equation}\label{zc}
x=G_0Tx+G_0f
\end{equation}
в пространстве $\SC$. Так как $\r(G_0T)<1$ по лемме \ref{l2},
то это уравнение имеет единственное решение, удовлетворяющее
равенству $G_0Tx=(G_0T)^2x+G_0TG_0f$, и, следовательно,
равенству $x=(G_0T)^2x+Bf$. Учитывая, что
$\r((G_0T)^2)=(\r(G_0T))^2<1$, получаем
\begin{equation}\label{ex}
x=(I-(G_0T)^2)^{-1}Bx=(I+(G_0T)^2+(G_0T)^4+\dotsb)Bf.
\end{equation}
Отсюда
$G=(I+(G_0T)^2+(G_0T)^4+\dotsb)B$.
Из этого равенства и из того, что операторы $I$ и $G_0T$
изотонны, а $B$ антитонен, следует, что  $G$ --- антитонный
оператор и $G\le B$. Из равенства \re{zc} при $x=Gf$ теперь
следует, что $G-G_0=G_0TG$ --- изотонный оператор. Таким
образом, $G_0\le G$.

Предположим, что оператор $G$ не является строго
антитонным. Пусть $f\not\equiv0$, $f(t)\ge0$ при п.\,в.
$t\in[a,b]$ и $x(t_0)=0$ в некоторой точке $t_0\in(a,b)$.
Рассмотрим второе слагаемое в разложении \re{ex}:
$((G_0T)^2Bf)(t_0)=(G_0\psi)(t_0)=0$, где $\psi\eqdef
TG_0TG_0f+TG_0TG_0TG_0f$.  Так как оператор $G_0$ строго
антитонен, то $\psi\equiv0$.  Таким образом,
\begin{equation}\label{eqv}
TG_0TG_0f\equiv0,
\end{equation}
т.\,к. в противном случае эта функция была бы собственной функцией
оператора $-TG_0$, спектральный радиус которого, как следует из
леммы \ref{l2}, меньше единицы.

Если $Tx\equiv p_a x(a)+p_bx(b)$, $p_a,p_b\in\SL$, то $G=G_0$.
Если $Tx\not\equiv p_a x(a)+p_bx(b)$, то $(TG_0f)(t){\le}0$ при
п.\,в. $t\in[a,b]$ и $TG_0f\not \equiv0$,
т.\,к. $(G_0f)(t)<0$ при $t\in(a,b)$.
Таким же образом получаем $(TG_0)^2f\not
\equiv0$, что противоречит \re{eqv}.
\end{pf*}


\begin{pf*}{Доказательство теоремы \protect{\ref{t3}}}
В условиях теоремы выполнены условия теоремы \ref{t2} и,
следовательно, условия леммы \ref{l2}. Поэтому $\r(G_0T)<1$.
Покажем, что решение задачи \re{ozb} положительно при
$t\in(a,b]$.  Решение этой задачи удовлетворяет уравнению $x=G_0Tx+y$ в
пространстве $\SC$, где $y\eqdef\frac{t-a}{b-a}$,
$t\in[a,b]$. Учитывая, что $\r(G_0T)<1$, получаем
$$
x=(I-G_0T)^{-1}y=(I+G_0T+(G_0T)^2+\dotsb)y=
(I+(G_0T)^2+(G_0T)^4+\dotsb)(I+G_0T)y.
$$

Имеем
\begin{gather*}
(Ty)(t)=\ii\dfrac{t-a}{b-a}\,d_sr(t,s)=r(t,b)-\frac{1}{b-a}\ii
r(t,s)\,ds,\quad t\in[a,b],\\
z(t)\eqdef y(t)+(G_0Ty)(t)=\dfrac{t-a}{b-a}+
\dfrac{t-b}{b-a}\int_a^t\frac{s-a}{\pi(s)}(Ty)(s)\,ds+
\frac{t-a}{b-a}\int_t^b\frac{s-b}{\pi(s)}(Ty)(s)\,ds,\quad
t\in[a,b].
\end{gather*}

Таким образом, $z(t)\ge0$, $t\in[a,b]$,
тогда и только тогда, когда выполнено неравенство
$$\w H(t)\eqdef\frac{b-t}{t-a}\int_a^t\frac{s-a}{\pi(s)}\w G(s)\,ds+
\int_t^b\frac{b-s}{\pi(s)}\w G(s)\,ds
\le b-a,\quad t\in(a,b).
$$
Функция $\w H(t)$ не возрастает на $(a,b)$, поэтому последнее
неравенство выполнено тогда и только тогда, когда
$$
\int_a^b\frac{b-s}{\pi(s)}\w G(s)\,ds\le b-a.
$$
Итак, в условиях теоремы
\begin{equation}\label{e70}
z(t)\ge0,\ \ t\in[a,b],\ \ \text{причем} \ \ Tz\noeq0,
\end{equation}
т.\,к. в противном случае $T(I+G_0T)y\equiv0$ и
$Ty$ --- собственная функция оператора $-G_0T$, спектральный
радиус которого меньше единицы (если же $Ty\equiv0$, то $y$ ---
решение задачи \re{ozb}). Так как оператор $G_0$ строго
антитонен, то из
\re{e70} и равенства
$$
x=(I+(G_0T)^2+(G_0T)^4+\dotsb)z
$$
следует, что $x(t)>0$, $t\in(a,b]$, при $Tx\not\equiv p_a
x(a)+p_bx(b)$, $p_a,p_b\in\SL$. Если же $Tx\equiv p_a
x(a)+p_bx(b)$, то
$$
x(t)=(G_0T+(G_0T)^2+\dotsb)y(t)+z(t)>0,\quad t\in(a,b].
$$

Доказательство положительности решения задачи \re{oza}
аналогично.
\end{pf*}

\begin{pf*}{Доказательство следствий \protect{\ref{s1}},
\protect{\ref{s2}}}
В условиях следствия \ref{s1} выполнены условия теорем
\ref{t1}, \ref{t2}. В условиях следствия \ref{s2} выполнены
условия теорем \ref{t1}, \ref{t2}, \ref{t3}.
\end{pf*}
