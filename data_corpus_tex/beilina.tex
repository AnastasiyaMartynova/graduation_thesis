\Section[n]{Введение} К настоящему времени появилось значительное количество работ, посвященных исследованию обратных задач с~интегральным условием переопределения. Однако в подавляющем большинстве изучены задачи для параболических уравнений
[\citen{Cannon-Lin-Wang:1991}--\citen{Prilepko-Tkachenko:zhvm2003}]. Более подробная библиография и классификация задач приведена
в работах \cite{Kozhanov:smj-2005,Prilepko-Tkachenko:zhvm2003}.

В предлагаемой работе рассмотрена обратная задача нахождения неизвестной функции, входящей в правую часть гиперболического
уравнения, с~интегральным условием переопределения.

\smallskip

\Section{Постановка задачи} В прямоугольнике \mbox{$Q_T\,{=}\,\{(x,t)\,{:}\, 0<x<l, 0<t<T\}$} рассмотрим следующую обратную задачу: найти пару
функций $(u(x,\,t)$, $p(t))$, удовлетворяющих уравнению
    \begin{equation}
    \label{nvb:eq1}
    u_{tt}(x,\,t)-u_{xx}(x,\,t)+c(x,\,t)u(x,\,t)=p(t)f(x,\,t)+G(x,\,t,\,u(x,\,t)),
    \end{equation}
начальным условиям
  \begin{equation}\label{nvb:eq2}
  u(x,\,0)=\varphi(x), \quad u_t(x,\,0)=\psi(x),
  \end{equation}
граничным условиям
    \begin{equation}\label{nvb:eq3}
    u_x(0,\,t)=u_x(l,\,t)=0
    \end{equation}
и интегральному условию переопределения
    \begin{equation}\label{nvb:eq4}
    \int_0^l K(x,\,t)u(x,\,t) dx=0.
    \end{equation}
Функции  $\varphi(x)$, $\psi(x)$ заданы на отрезке $[0;\,l]$, а $f(x,\,t)$, $K(x,\,t)$, $G(x,\,t,\,u)$, $c(x,\,t)$ "--- в области
$\bar Q_T$.

Введём понятие обобщённого решения задачи \eqref{nvb:eq1}--\eqref{nvb:eq4}. Заметим, что условие  \eqref{nvb:eq4} эквивалентно
следующему условию:
    \begin{multline}
    \label{nvb:eq5}
    p(t)=\biggl( \int _0^l K(x,\,t) f(x,\,t) dx\biggr)^{-1}\biggl[ \int _0^l\bigl[K(x,\,t)c(x,\,t)-K_{tt}(x,\,t) \bigr]u(x,t) \,
    dx+\\
    + \int _0^l K_x(x,\,t) u_x(x,\,t)dx -2\int _0^l K_t(x,\,t) u_t(x,\,t) dx- \\
    -\int _0^l K(x,\,t) G(x,\,t,\,u(x,\,t))dx \biggl].
    \end{multline}

Действительно, дифференцируя \eqref{nvb:eq4} дважды по $t$ и учитывая, что $u(x,\,t)$ удовлетворяет уравнению \eqref{nvb:eq1} и
условию \eqref{nvb:eq3}, получим \eqref{nvb:eq5}. Обратное показывается прямым вычислением.

Обозначим
      \begin{equation*}
      \hat{W}_2^1(Q_T)=\{ v(x,\,t): v(x,\,t)\in W_2^1(Q_T),\, v(x,\,T)=0\}.
      \end{equation*}
Умножим уравнение \eqref{nvb:eq1} на функцию $v(x,\,t) \in \hat{W}_2^1(Q_T)$ и проинтегрируем по прямоугольнику $Q_T$. После
интегрирования по частям получим
    \begin{multline}
    \label{nvb:eq6}
    \int _0^T\int _0^l \left[ - u_{t}(x,t)v_t(x,t)+u_x(x,t)v_x(x,t) +c(x,t)u(x,t)v(x,t)\right]dxdt=\\
    =\int _0^T\int _0^l p(t)f(x,t)v(x,t)\,dxdt+\int _0^T\int _0^l G(x,t,u(x,t)) v(x,t)\,dxdt +\\
    +\int _0^l \psi(x)v(x,0)\,dx.
    \end{multline}

\begin{definition}
  Пару функций $(u(x,\,t)$, $p(t))$ будем называть обобщённым решением задачи \eqref{nvb:eq1}--\eqref{nvb:eq4},
  если $u(x,\,t) \in W_2^1(Q_T)$, $u(x,\,0)\,{=}\,\varphi(x)$,  \mbox{$ p(t)\,{\in}\, L_2 (0,\, T)$,} и $(u(x,\,t)$, $p(t))$ удовлетворяет
   \eqref{nvb:eq5} (в смысле равенства функций в $L_2$) и тождеству \eqref{nvb:eq6} для любой функции $v(x,t) \in\hat{W}_2^1(Q_T)$.
\end{definition}

\smallskip

\Section{Разрешимость поставленной задачи}

\begin{theorem}
 Если $K(x,\,t)\in C^1(\bar Q_T),$ $K_{tt}(x,\,t)\in C^1(\bar Q_T),$  $c(x,\,t)\in C^1(\bar Q_T),$
 $f(x,\,t)\in C(\bar Q_T),$  $\varphi(x)\in W_2^1(0,\,l),$  $\psi(x)\in L_2 (0,\,l),$ $G(x,\,t,\,u)\in C(Q_T\times \mathbb R^1),$
 $ \displaystyle \int _0^l K(x,\,t) f(x,\,t) dx\not=0$ и для любых $(x,\,t)$ выполняется условие Липшица  $|G(x,\,t,\,u_1)-G(x,\,t,\,u_2)|
 \leq L|u_1-u_2|,$ то существует единственное обобщённое решение \rm \eqref{nvb:eq1}--\eqref{nvb:eq4}.
\end{theorem}

 \begin{proof}  Из условий теоремы следует, что найдутся такие положительные константы $c_0$, $ c_1$, $ c_2$, $ f_1$, $ g_1$,
 $h_0$, $ a_1$, $ a_2$, что выполняются следующие неравенства:
$$\max\left|G(x,\,t,\,0)\right|\leq g_1, \quad  \left|f(x,\,t)\right|\leq f_1, \quad \max\left| K(x,\,t)c(x,\,t)-K_{tt}(x,\,t)\right|\leq a_1,$$
$$\max\left| K(x,\,t); K_t(x,\,t); K_x(x,\,t) \right|\leq a_2,\quad c_0\leq \left|c(x,\,t)\right|\leq c_1,\quad   \left|c_t(x,\,t)\right|\leq  c_2,$$ 
$$ \int _0^l K(x,\,t) f(x,\,t) dx=h_0>0.$$

Для доказательства теоремы построим последовательность приближённых решений, а затем покажем, что эта последовательность сходится к~обобщённому решению поставленной задачи.

Приближённые решения $(u^m(x,\,t),\,p^m(t))$ будем искать из следующих соотношений:
     \begin{multline}
     \label{nvb:eq7}
    \int _0^T \int _0^l \left[ - u_t^m(x,\,t)v_t(x,\,t)+u_x^m(x,\,t)v_x(x,\,t) + c(x,\,t)u^m(x,\,t)v(x,\,t)\right]
    dxdt=\\
    =\int _0^T\int _0^l p^m(t)f(x,\,t)v(x,\,t)dxdt+\int _0^T\int _0^l G(x,\,t,\,u^m(x,\,t)) v(x,\,t)dxdt +\\
    +\int _0^l \psi(x)v(x,\,0)dx.
    \end{multline}
    \begin{equation}
    \label{nvb:eq8}
    u^0(x,\,t)=0, \quad  u^m(x,\,0)=\varphi_m, \quad m=1,\,2, \ldots,
    \end{equation}
где  $\varphi_m \xrightarrow{L_2}{\varphi(x)}$,
    \begin{multline}
    \label{nvb:eq9}
    p^m(t)=\left( \int _0^l K(x,\,t) f(x,\,t) dx\right)^{-1}\left[ \int _0^l\bigl[K(x,\,t)c(x,\,t)-K_{tt}(x,\,t) \bigr] \times \right. \\ \times u^{m-1}(x,t)      dx
    + \int _0^l K_x(x,\,t) u_x^{m-1}(x,\,t)dx - 2\int _0^l K_t(x,\,t) u_t^{m-1}(x,\,t) dx - \\
    \left.-\int _0^l K(x,\,t) G(x,\,t,\,u^{m-1})dx\right].
      \end{multline}

Покажем, что для каждого $m$ существует единственная функция $u^m(x,t)$, удовлетворяющая тождеству \eqref{nvb:eq7} и условию \eqref{nvb:eq8}, если  $ p^m(t)$ известно. Для этого заметим, что тождество  \eqref{nvb:eq7}  и равенство \eqref{nvb:eq8} определяют обобщённое решение из $W_2^1(Q_T)$ второй начально--краевой задачи с однородными граничными условиями
      \begin{gather}
      \label{nvb:eq10}
      u^m_{tt}(x,\,t)-u^m_{xx}(x,\,t)+c(x,\,t)u^m(x,\,t)=H(x,\,t)+G(x,\,t,\,u^m(x,\,t)),\\
      \notag
      H(x,\,t)=p^m(t)f(x,\,t),\\
      \label{nvb:eq11}
      u^m(x,\,0)=\varphi^m(x), \quad u^m_t(x,\,0)=\psi(x),\\
       \label{nvb:eq12}
       u^m_x(0,\,t)=u^m_x(l,\,t)=0.
      \end{gather}

Применяя стандартные методы \cite{lad} и условие Липшица, которому удовлетворяет функция $ G(x,\,t,\,u(x,\,t))$, нетрудно
доказать однозначную разрешимость задачи \eqref{nvb:eq10}--\eqref{nvb:eq12}. Следовательно, можно утверждать, что для любого
$p^m(t)$ существует единственная функция $u^m(x,\,t)$, удовлетворяющая тождеству \eqref{nvb:eq7}. Так как $ p^m(t)$ находятся
явным образом из \eqref{nvb:eq9}, то можно считать, что последовательность $\left\lbrace u^m(x,\,t),\,p^m(t)\right\rbrace $
построена.

Для обоснования сходимости этой последовательности рассмотрим разности
    \begin{equation*}
        z^m(x,\,t) = u^m (x,\,t)- u^{m-1}(x,\,t), \ r^m(t) = p^m(t) - p^{m-1}(t).
    \end{equation*}

Заметим, что  для $z^m(x,\,t)$ справедливо тождество
    \begin{multline*}
%    \label{nvb:eq24}
     \int _0^T\int _0^l \left[ z_{tt}^m(x,\,t)v(x,\,t)+z_x^m(x,\,t)v_x(x,\,t) + \right.\\
     \left.+ c(x,\,t)z^m(x,\,t)v(x,\,t)\right]
     dxdt
    =\int _0^T\int _0^l r^m(t)f(x,\,t)v(x,\,t)dxdt+\\
    +\int _0^T\int _0^l \left[ G(x,\,t,\,u^m)-G(x,\,t,\,u^{m-1})\right]  v(x,\,t)dxdt.
    \end{multline*}
Полагая $ v(x,\,t)=z^m_t(x,\,t)$ и интегрируя по частям интегралы в левой части, получим
    \begin{multline*}
    \int _0^l  \left[ \left( z^{m}_{t}(x,\tau)\right)^2+  \left( z^{m}_x(x,\tau)\right)^2 +c(x,\tau)\left( z^{m}(x,\tau)\right)^2\right]
    dx=\\
%\label{nvb:eq25}
    =2\int _0^\tau\int _0^l r^m(t)f(x,t)z^{m}_{t}(x,t) dx dt +2\int _0^\tau\int _0^l c_t(x,t)\left(z^{m}_{t}(x,t)\right)^2 dx
    dt+\\
    +2\int _0^\tau\int _0^l \left[  G(x,t,u^m)-G(x,t,u^{m-1})\right] z^{m}_{t}(x,t) dx dt.
    \end{multline*}

Применяя к правой части неравенство Юнга, элементарные неравенства, а также условия теоремы, нетрудно получить оценку
    \begin{multline}\label{nvb:eq26}
    m\int _0^l  \left[ \left( z^{m}_{t}(x,\tau)\right)^2{+} \left( z^{m}_x(x,\tau)\right)^2 {+}\left( z^{m}(x,\tau)\right)^2\right] dx \leq \\
    \leq \delta \left(1{+}L^2\right)  \int _0^\tau\int _0^l \left( r^m(t)\right)^2 dx dt{+}\\
 +    M_3\int _0^\tau\int _0^l \left( \left(z^{m}_{t}(x,t)\right)^2 +\left(z^{m}(x,t)\right)^2
+\left(z^{m}_{x}(x,t)\right)^2 \right] dx dt,
    \end{multline}
где $M_3=\max\left\lbrace ({f_1^2+1})/{\delta}; 2c_2\right\rbrace.$ Применим к \eqref{nvb:eq26} неравенство Гронуолла
\cite{gor}, а затем интегрируя по  $\tau$ от $0$ до $T$, приходим к оценке
    \begin{equation}\label{nvb:eq27}
    \|z^m\|_{W_2^{1}(Q)} \leq \sqrt{N_1\delta} \|r^m(t)\|_{L_2(0,T)},
    \end{equation}
где $N_1=\exp({\frac{M_3T}{m}})(1+L^2)$, $\delta>0$ произвольно.

Оценим теперь $r^m(t)$, для которого справедливо равенство
    \begin{multline}
      r^m(t)=\left( \int _0^l K(x,t) f(x,t) dx\right)^{-1}\left[ \int _0^l\left[K(x,t)c(x,t)-K_{tt}(x,t) \right]z^{m-1}(x,t) \,
      dx+\right.\\
     \label{nvb:eq28}
       +\left. \int _0^l K_x(x,t) z_x^{m-1}(x,t) dx -2\int _0^l K_t(x,t) z_t^{m-1}(x,t) dx-\right.\\
     \left.-\int _0^l K(x,t) \left[ G(x,t,u^{m-1})-G(x,t,u^{m-2})\right]  dx\right].
       \end{multline}

Возводя равенство \eqref{nvb:eq28} в квадрат, учитывая условия теоремы, а также то, что $z^{m-1}(x,\,t)\in W_2^{1}(Q_T)$, нетрудно получить
неравенство
    \begin{equation} \label{nvb:eq29}
     \|r^m(t))\|_{L_2(0,T)} \leq  \sqrt{N_2} \|z^{m-1}(x,t)\|_{W^1_2(Q_T)},
    \end{equation}
где $N_2=\max\left\lbrace 4a_2^2l; 4(a_1^2l+a_2^2lL^2)\right\rbrace.$

Из \eqref{nvb:eq27} и \eqref{nvb:eq29} вытекает, что
    \begin{gather} \label{nvb:eq30}
     \|z^m(x,t))\|_{W^1_2(0,T)} \leq \sqrt{N_3} \|z^{m-1}(x,t)\|_{W^1_2(Q_T)},\\
    \label{nvb:eq31}
     \|r^m(t))\|_{L_2(0,T)} \leq \sqrt{N_3} \|r^{m-1}(t))\|_{L_2(0,T)},
    \end{gather}
где $N_3=N_1N_2\delta$.

Пользуясь произволом $\delta$, выберем его так, чтобы $\sqrt{N_3}=q<1$. Тогда неравенства \eqref{nvb:eq30} и \eqref{nvb:eq31}
означают, что  последовательность $(u^m(x,t),\, p^m(t))$ фундаментальна.

Так как $W_2^{1}(Q_T)$ и $L_2(0,T)$ "--- полные пространства, то фундаментальная последовательность $(u^m(x,t),\,p^m(t))$
сходится к элементу $(u(x,t),\, p(t))$, где $u(x,t) \in W_2^{1}(Q_T)$, $p(t) \in L_2(0,T)$. Но тогда, переходя к пределу в
\eqref{nvb:eq7} и \eqref{nvb:eq9}, получим соответственно тождества \eqref{nvb:eq6} и \eqref{nvb:eq5}, так как из сильной
сходимости следует слабая.

Таким образом, пара функций $(u(x,t),\,p(t))$, полученная в результате предельного перехода в $(u^m(x,t),\,p^m(t))$ и
эквивалентных преобразований, является обобщённым решением задачи \eqref{nvb:eq1}--\eqref{nvb:eq4}.

Единственность задачи \eqref{nvb:eq1}--\eqref{nvb:eq4} непосредственным образом следует из оценок \eqref{nvb:eq30} и
\eqref{nvb:eq31}. Действительно, полагая, что существуют два различных решения $(u_1,\,p_1)$ и $(u_2,p_2)$, приходим к оценкам
$$
        (1-q)\|u(x,t)\|_{W_2^1(Q_T)}\leq 0,\quad 
        (1-q)\|p(t)\|_{L_2(0,T)}\leq 0,
$$
где $(u,\,p)= (u_1-u_2,\,p_1-p_2)$. Но так как $q<1$, то $\|u(x,\,t)\|_{W_2^1(Q_T)}=0$ и $\|p(t)\|_{L_2(0,\,T)}=0$.
\end{proof}
