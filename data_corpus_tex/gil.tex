\Section[n]{Введение} Рассматривается оператор свёртки
\begin{equation}
\label{1.3}
M_{\theta}^{\overline{\beta}}{\varphi}=m_{\theta}^{\overline{\beta}}*\varphi
\end{equation}
с ядром, имеющим степенные особенности на конечном
объединении сфер:
\begin{multline}
\label{1.4}
m_{\theta}^{\overline{\beta}}(y)=\theta_1(|y|)(r_1^2-|y|^2+i0)^{\beta_1-1}\times\ldots\times\theta_{s-1}(|y|)
\times\\
\times
r_{s-1}^2-|y|^2+i0)^{\beta_{s-1}-1}\theta_{s}(|y|)(1-|y|^2)^{\beta_{s}-1}_+,
\end{multline}
где ${\overline{\beta}}=(\beta_1,\ldots,\beta_s)$, $\beta_j>0$,
$1\leq j\leq s$, $s\geq2$, $0<r_1< r_2<\ldots< r_{s-1}<r_s=1$. Здесь
$\theta_j(r)$ "--- гладкие функции, $\theta_j(r_j)\neq0$, $1\leq
j\leq s$.

Операторы вида \eqref{1.3} возникают при решении задачи Коши для
волнового уравнения (см. [\citen{b1}--\citen{b13}]). $(H^p-H^q)$"=оценки для этих операторов,
$0<p\leq q<\infty$, были получены в [\citen{b100}].

Методом аппроксимативных обратных
операторов (АОО) строится обращение потенциалов \eqref{1.4} с
плотностями из пространства $L^p$, $1\leq p\leq 2$ и из пространства Харди $H^1$ в неэллиптическом случае,
когда их символы вырождаются на множестве меры нуль в ${\mathbb R^n}$. 
Даётся также описание образов $M_{\theta}^{\overline{\beta}}(L^p)$ и $M_{\theta}^{\overline{\beta}}(H^1)$ в~терминах обращающих конструкций.

\Section{Предварительные сведения}
Введём следующие обозначения:
$\displaystyle (Ff)(\xi)=\widehat{f}(\xi)=\int_{\mathbb
R^n}f(x)e^{i{\xi}x}dx$ "--- преобразование Фурье функции $f$;
$(F^{-1}f)(\xi)=\tilde{f}(\xi)=(2\pi)^{-n}(Ff)(-\xi)$ "--- обратное
преобразование Фурье; $R^0({\mathbb R^n})=\{f:f(x)=\widehat{\varphi}(x), \varphi(x)\in L^1({\mathbb R^n})\}$ "--- винеровское кольцо; $\mathcal S$ "--- класс Шварца быстро убывающих
гладких функций; $\mathcal S'$ "--- пространство обобщённых функций
медленного роста; $C_0({\mathbb R^n})=\{f:f\in C({\mathbb R^n}), f(\infty)=0\}$ "---
пространство непрерывных функций, исчезающих на бесконечности;
$W_\varepsilon \varphi= w_\varepsilon *\varphi$
"--- интеграл Гаусса"--~Вейерштрасса, где $w_\varepsilon(x)= (4\pi
\varepsilon)^{-n/2} e^{-|x|^2/(4\varepsilon)}$
($\hat{w}_\varepsilon (\xi)= e^{-\varepsilon|\xi|^2}$); $\Phi$,
$\Psi$ "--- пространства П.\,И.~Лизоркина:  $\Psi=\{\psi\in {\mathcal
S}:\ (D^\nu \psi)(0)=0,\ |\nu|=0, 1, \dots\}$, $\Phi=\{\varphi\in
{\mathcal S}:\hat{\varphi}\in \Psi\}$.

 Через
$H^1=H^1(\mathbb R^n)$ обозначим множество всех
${\mathcal S}'$-распределений таких, что
$$f^+(x)=\sup\limits_{0<\varepsilon<\infty}{|(f*\varphi_\varepsilon)(x)}|\in{L^1},$$
где $\varphi\in{{\mathcal S}}$ и $ \displaystyle  \int_{\mathbb
R^n}\varphi(x)dx\neq0$,
$\varphi_\varepsilon(x)=\varepsilon^{-n}\varphi(\frac{x}\varepsilon)$
и
$(f*\varphi_\varepsilon)(x)=\langle{f},\varphi_\varepsilon(x-{}\cdot{})\rangle.$
Положим
$\|f\|_{H^1}=\|f^+\|_{L^1}$
(см. \cite[с.\,269]{b3}).

 Пусть $V$ "--- произвольное
замкнутое множество в $\mathbb R^n$. Через $\Psi_V$ обозначим
класс всех функций из $\mathcal{S}$, которые исчезают вместе со
всеми своими производными на $V$:
\[
    \Psi_V=\{\psi(\xi)\in\mathcal{S}: D^k\psi(\xi)=0, \xi\in V,
    |k|=0,1,2,\dots\}.
\]

Через $\Phi_V$ обозначим класс прообразов Фурье функций из
$\Psi_V$: $\Phi_V=F^{-1}(\Psi_V)$.

Пространства $\Psi_V$ и $\Phi_V$ были введены и изучены С.\,Г.~Самко (см. {\cite[\S\,3]{b14}}).

\begin{theorem}[{1\cite[с.\,50]{b14}}]
    Пусть $f\in L^1({\mathbb R^n}).$ Если
    $\frac{\partial^m f}{\partial x_{k_1}\ldots\partial x_{k_m}}\in L^p({\mathbb R^n})$
    при каком-нибудь $1<p\leq 2$ для всех $m=1, 2, \ldots, n$ и любых $k_1, \ldots, k_m,$ то $f(x)\in R^0({\mathbb R^n})$.
\end{theorem}

\begin{theorem}[2~{\rm \cite{b100}}] Пусть $\beta_j>0$, $1\leq j\leq s${\rm ,}
$\beta_0=\min\{\beta_1,\ldots,\beta_s\}$. Имеют место следующие соотношения{\rm :}
\begin{itemize}
\item[\rm  1)] оператор $M_{\theta}^{\overline{\beta}}$ ограничен из $L^p$ в $L^q${\rm ,} $1<p\leq q<\infty${\rm ,}
 тогда и только тогда{\rm ,} когда
$$\frac1p+\frac1q\leq1, \; \frac{1}p-\frac{n}q\leq{\beta_0}\quad\mbox{или}\quad
\frac1p+\frac1q\geq1, \; \frac{n}p-\frac1q\leq{\beta_0+(n-1)};$$
\item[\rm 2)] оператор $M_{\theta}^{\overline{\beta}}$ ограничен из $L^1$ в $L^q$ тогда и только тогда$,$ когда
$${1-\beta_0}<\frac 1 q\leq1;$$
\item[\rm 3)] оператор $M_{\theta}^{\overline{\beta}}$ ограничен из $H^1$ в $H^1.$
\end{itemize}
\label{teor:2}
\end{theorem}

\Section{Основные результаты}
Воспользуемся идеей обращения потенциалов с символами
$\widehat{m^\beta}(\xi)=
(2\pi)^{n/2}2^{\beta-1}|\xi|^{1-\beta-n/2}J_{\beta-1+n/2}(|\xi|)$
из \cite{b5}.

Обращение потенциала $f=M_{\theta}^{\overline{\beta}}\varphi$,
$\varphi\in L^p$, $1\leq p\leq 2$, в неэллиптическом случае, когда
$
 {\mathrm { mes}} \Bigl\{\xi:\ \widehat{m_{\theta}^{\overline{\beta}}}(\xi)=0\Bigr\}=0,
$
 будем строить в виде
\begin{equation}\label{2.16}
 T_\theta^{\overline{\beta}}f=\lim_{\varepsilon\to 0}^{(L^{p})}
 \lim_{\delta\to 0}^{(L^2)}
 T_{\theta,\varepsilon,\delta}^{\beta}f,
\end{equation}
где
\begin{equation}\label{1.10}
 T_{\theta,\varepsilon,\delta}^{\overline{\beta}}f=
 F^{-1}\left({\overline{\widehat{m_{\theta}^{\overline{\beta}}}(\xi)}\
 e^{-\varepsilon|\xi|^2}}
\left/
\left( {|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^2+i\delta}\right)\right)\right.*f,
\end{equation}
\begin{multline*}
  \widehat{m_{\theta}^{\overline{\beta}}}(\xi)=\int _{0}^1 \rho^{n-1}
\theta_1(\rho)(r_1^2-\rho^2+i0)^{\beta_1-1}\times\ldots
\\
\ldots\times\theta_{s-1}(\rho)
(r_{s-1}^2-\rho^2+i0)^{\beta_{s-1}-1}\theta_{s}(\rho)(1-\rho^2)^{\beta_{s}-1}_+
\,d\rho \int _{S^{n-1}}
     e^{i(\rho\xi\cdot \sigma)}\,d\sigma.
\label{4.79}
\end{multline*}
Следующая теорема даёт обращение потенциалов
$M_{\theta}^{\overline{\beta}}\varphi$, $\varphi\in L^p$, $1\leq p\leq 2$.

\begin{theorem}[3]Пусть $\varphi\in L^p${\rm ,} $1\leq p\leq 2${\rm .} Тогда
    \begin{equation}\label{2.20}
        \left(T_\theta^{\overline{\beta}}
       M_\theta^{\overline{\beta}}\varphi\right)(x)=\varphi(x),
    \end{equation}
где $T_\theta^{\overline{\beta}}$ "--- оператор \eqref{2.16}.
\end{theorem}

\begin{proof}
Из теоремы 1 следует, что оператор \eqref{1.10} "--- оператор свёртки с суммируемым ядром. Рассуждения будем основывать на равенстве
\begin{equation}\label{4.83}
 (T_{\theta,\varepsilon,\delta}^{\overline{\beta}}M_{\theta}^{\overline{\beta}}\varphi)(x)=(W_\varepsilon
 \varphi)(x)-i\delta
 (N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}
 \varphi)(x),\quad  \varepsilon, \; \delta>0,
\end{equation}
где
\begin{equation}\label{4.84}
 N_{\theta,\varepsilon,\delta}^{\overline{\beta}}f
 =F^{-1}\left(n_{\theta,\varepsilon,\delta}^{\overline{\beta}}(\xi)\right)*f,\qquad
 n_{\theta,\varepsilon,\delta}^{\overline{\beta}}(\xi)={
 e^\frac{-\varepsilon|\xi|^2}2}\left/
 \left({|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^2+i\delta}\right)\right..\nonumber
\end{equation}
 Для функций $\varphi\in\Phi$ равенство \eqref{4.83}
проверяется переходом к образам Фурье. Это равенство
распространяется по ограниченности на все пространство $L^p$, $1< p\leq 2$, с
учетом того, что операторы в обеих частях \eqref{4.83} ограничены
из $L^p$ в $L^1_{\gamma}$, при некотором $\gamma>0$, где
$$L^1_{\gamma}=\left\{f(x): \int_{\mathbb R^n}\frac{|f(x)|dx}{(1+|x|)^\gamma}<\infty \right\}.$$
В случае $p=1$ равенство \eqref{4.83} выполняется в смысле $\Phi'$:
\begin{equation}\label{1.11}
 \langle T_{\theta,\varepsilon,\delta}^{\overline{\beta}}M_{\theta}^{\overline{\beta}}\varphi, \omega \rangle=
 \langle W_\varepsilon
 \varphi-i\delta
 N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}
 \varphi,\omega\rangle,\quad \omega\in \Phi.
\end{equation}
Из \eqref{1.11} следует, что
\begin{equation}\label{1.12}
\left(T_{\theta,\varepsilon,\delta}^{\overline{\beta}}M_{\theta}^{\overline{\beta}}\varphi\right)(x)=
 \left( W_\varepsilon\varphi\right)(x)-i\delta
 \left(N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}
 \varphi\right)(x)+P(x),
\end{equation}
где $P(x)$ "--- некоторый многочлен.
Так как функции
$\left(T_{\theta,\varepsilon,\delta}^{\overline{\beta}}M_{\theta}^{\overline{\beta}}\varphi\right)(x)$,
  $\left( W_\varepsilon\varphi\right)(x)$ и $\left(N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}
 \varphi\right)(x)$ принадлежат пространству $L^1$, $P(x)\equiv0$ в \eqref{1.12}.

С учётом того, что $W_\varepsilon \varphi\rightarrow \varphi$ при
$\varepsilon\rightarrow0$ по $L^p$-норме или почти всюду, формула \eqref{2.20} будет следовать из
равенства
\begin{equation}\label{4.86}
  \lim\limits_{\delta\to
  0}\delta^2\left\|(N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}\varphi)(x)\right\|^2_{2}=0.
\end{equation}
Докажем \eqref{4.86}. Применяя равенство Парсеваля, получаем
\begin{equation}\label{4.861}
 \delta^2\left\|(N_{\theta,\varepsilon,\delta}^{\overline{\beta}}W_{\frac\varepsilon 2}\varphi)(x)\right\|_2^2=
 \frac{1}{(2\pi)^{n}}\int_{{\mathbb R^n}} \frac{\delta^2 e^\frac{-\varepsilon|\xi|^2}2}{|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^4+\delta^2}
  \left|(\widehat{W_{\frac\varepsilon 2}\varphi})(\xi)\right|^2 \, d\xi \to 0
\end{equation}
при $\delta\to 0$, с учётом того, что
\[
 \lim\limits_{\delta\to
  0}\frac{\delta^2 e^\frac{-\varepsilon|\xi|^2}2}{|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^4+\delta^2}
  \left|(\widehat{W_{\frac\varepsilon 2}\varphi})(\xi)\right|^2=0,\quad
  \xi\not\in \{\xi:\ \widehat{m_{\theta}^{\overline{\beta}}}(\xi)=0\}.
\]
Предельный переход \eqref{4.861} обосновывается применением
мажорантной теоремы Лебега с учетом оценки
\[
\left({\delta^2
e^\frac{-\varepsilon|\xi|^2}2}\left/\left({|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^4+\delta^2}
\right)\right)\right.
\left|(\widehat{W_{\frac\varepsilon 2}\varphi})(\xi)\right|^2
\leqslant e^\frac{-\varepsilon|\xi|^2}2
\left|(\widehat{W_{\frac\varepsilon 2}\varphi})(\xi)\right|^2 \in
L_1.
\]
Здесь существенным являлся тот факт, что
$W_{\frac\varepsilon 2}\varphi\in L^2$, если $\varphi\in L^p$, $1\leq p\leq 2$. Кроме того, учтено, что функция
$${\overline{\widehat{m_{\theta}^{\overline{\beta}}}(\xi)}
 e^\frac{-\varepsilon|\xi|^2}2}\left/
 {\left(\left|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)\right|^{_{_2}}+i\delta\right)}\right.$$ 
является 2-мультипликатором.
\end{proof}
\smallskip

Описание образа $M_\theta^{\overline{\beta}}(L^p)$, $1\leq p\leq 2$, даёт следующая теорема.
\smallskip
\begin{theorem}[4]Пусть ${\overline{\beta}}=(\beta_1,\ldots,\beta_s)${\rm ,} $\beta_j>0${\rm ,}
$1\leq j\leq s$ и $1\leq p\leq 2${\rm .} Тогда
$$\label{2.21}
    M_\theta^{\overline{\beta}}(L^p)=\left\{f\in L^q: T_\theta^{\overline{\beta}} f\in
    L^p\right\},$$
где $T_\theta^{\overline{\beta}}$ "--- оператор \eqref{2.16}{\rm ,} $q$ "--- произвольное число{\rm ,}  $1\leq q\leq 2${\rm ,} такое{\rm ,} что
оператор $M_\theta^{\overline{\beta}}$ ограничен из $L^p$ в $L^q${\rm .}
\end{theorem}

\begin{proof}
Вложение
\begin{equation}\label{4.87}
  M_\theta^{\overline{\beta}}(L^p)\subset\left\{f\in L^q: T_\theta^{\overline{\beta}} f\in
    L^p\right\}\end{equation}
вытекает из теоремы 2.

Докажем вложение
$$\label{4.70}
  M_\theta^{\overline{\beta}}(L^p)\supset\left\{f\in L^q: T_\theta^{\overline{\beta}} f\in
    L^p\right\},$$
обратное к \eqref{4.87}.
 Пусть функция $\omega\in \mathcal{S}$ такова, что
$\widehat{\omega}(\xi)=0$ в некоторой окрестности множества
$V=\{\xi:\ \widehat{m_{\theta}^{\overline{\beta}}}(|\xi|)=0\}$
(следовательно, $\omega\in\Phi_V$).

Имеем
\begin{equation}\label{4.88}
 \langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f,
 \omega\rangle=\langle T_\theta^{\overline{\beta}}f,
 \overline{M_\theta^{\overline{\beta}}}\omega\rangle=\langle \lim\limits_{\varepsilon\to 0}^{(L^p)}
 \lim\limits_{\delta\to 0}^{(L^2)} T_{\theta,\varepsilon,\delta}^{\overline{\beta}}f,
 \overline{M_\theta^{\overline{\beta}}}\omega\rangle,
\end{equation}
где $\overline{M_\theta^{\overline{\beta}}}$ "--- оператор свёртки с символом $\overline{\widehat{m_{\theta}^{\overline{\beta}}}(|\xi|)}$. С учётом
\eqref{4.88} и того факта, что сходимость по $L^p$ норме предполагает сходимость в $\Phi_V'$, получаем
\begin{multline}\label{4.89}
\langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f,
 \omega\rangle=\lim\limits_{\varepsilon\to 0}\langle \lim\limits_{\delta\to
0}^{(L^2)} T_{\theta,\varepsilon,\delta}^{\overline{\beta}}f,
 \overline{M_\theta^{\overline{\beta}}}\omega\rangle=\lim\limits_{\varepsilon\to 0}\lim\limits_{\delta\to
0}\langle T_{\theta,\varepsilon,\delta}^{\overline{\beta}}f,
 \overline{M_\theta^{\overline{\beta}}}\omega\rangle=
\\[10pt]
=\lim\limits_{\varepsilon\to 0}\lim\limits_{\delta\to 0}\langle f,
 \overline{T_{\theta,\varepsilon,\delta}^{\overline{\beta}}}\
 \overline{M_\theta^{\overline{\beta}}}\omega\rangle,
\end{multline}
где $\overline{T_{\theta,\varepsilon,\delta}^{\overline{\beta}}}$ "---
мультипликаторный оператор с символом \begin{equation}
 \left({{{\widehat{m_{\theta}^{\overline{\beta}}}(\xi)}}\
 e^{-\varepsilon|\xi|^2}}\right)
\left/\left({|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^2-i\delta}\right)\right..\nonumber
\end{equation}
 Далее имеем
\begin{equation}\label{4.90}
(\overline{T_{\theta,\varepsilon,\delta}^{\overline{\beta}}}\
 \overline{M_\theta^{\overline{\beta}}}\omega)(x)=(W_\varepsilon
 \omega)(x)+i\delta
 (\overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon 2}
 \omega)(x),\quad  \varepsilon,\delta>0,
\end{equation}
где $\overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}$  "---
ограниченный в $L^2$ оператор, порождаемый 2-мультипликатором
$$
{e^\frac{-\varepsilon|\xi|^2}2}\left/\left({|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^2-i\delta}\right)\right..
$$
Равенство \eqref{4.90} проверяется переходом к преобразованиям
Фурье.

С учётом \eqref{4.89} и \eqref{4.90} получаем
\begin{equation}
  \langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f,
 \omega\rangle=\lim\limits_{\varepsilon\to 0}\langle f,
W_\varepsilon \omega\rangle +\lim\limits_{\varepsilon\to
0}\lim\limits_{\delta\to 0}\langle f,
 i\delta \overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon 2}
 \omega\rangle.\label{4.91}
\end{equation}

Докажем, что
\begin{equation}\label{4.92}
 \lim\limits_{\delta\to 0}\langle f,
 i\delta \overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon 2}
 \omega\rangle =0.
\end{equation}
Так как
$(\overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon
2} \omega)(x) \in \Phi_V$, 
\[
 \langle f, i\delta \overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon 2}\omega\rangle =
 (2\pi)^{-n}\langle \widehat{f}, i\delta F(\overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon
 2}\omega)\rangle,
\]
где преобразование Фурье $\widehat{f}$, понимаемое в  смысле $\Phi'$, совпадает с преобразованием Фурье в пространстве $L^{q'}$,
$\frac1q+\frac1{q'}=1$ (в соответствии с теоремой Хауссдорфа"--~Юнга).

Применяя неравенство Гёльдера, будем иметь
\begin{eqnarray}
 \bigl|\langle \widehat{f}, i\delta F(\overline{N_{\theta,\varepsilon,\delta}^{\overline{\beta}}}W_{\frac\varepsilon 2}\omega)\rangle\bigr|\leqslant\
 \delta \|\widehat{f}\|_{q'}
 \biggl\{\int_{{\mathbb R^n}} \frac{ e^{-\varepsilon q|\xi|^2}
 \left| \widehat{\omega}(\xi) \right|^q}{|\widehat{m_{\theta}^{\overline{\beta}}}(\xi)|^{2q}} d\xi\biggr\}^{\frac1q}.\label{4.93}
\end{eqnarray}
Заметим, что интеграл в правой части \eqref{4.93} конечен, так как
$\widehat{\omega}(\xi)=0$ в некоторой окрестности множества $V$.

Переходя в \eqref{4.93} к пределу  при $\delta\to 0$, получаем
\eqref{4.92}.

Из \eqref{4.91} и \eqref{4.92} следует, что
\begin{equation}\label{4.94}
  \langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f,
 \omega\rangle=\lim\limits_{\varepsilon\to 0}\langle f,
W_\varepsilon \omega\rangle= \langle f, \omega\rangle.
\end{equation}

Переходя к завершающему этапу доказательства, для заданной функции
$\omega\in \mathcal{S}$ выберем последовательность
$\{\omega_N\}$, $\omega_N\in \Phi_V$, такую, что
$\widehat{\omega_N}(\xi)$ обращается в нуль в некоторой
окрестности множества $V$ и
$$\lim\limits^{(L^{q'})}_{N\to \infty}\omega_N=\omega,\quad 1<q\leq2 \qquad\mbox{и}\qquad
  \lim\limits^{(C_0)}_{N\to \infty}\omega_N=\omega$$ (возможность
выбора такой последовательности доказана в \cite[\S 3]{b14}).

Из \eqref{4.94} вытекает, что
$
  \langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f,
 \omega_N\rangle= \langle f, \omega_N\rangle.
$
Переходя в этом равенстве к пределу при $N\to\infty$ на основании
мажорантной теоремы Лебега, получаем
$
\langle f,\omega\rangle= \langle M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}
f,
 \omega\rangle,$ $\omega\in {\mathcal S},
$
откуда следует, что
$
f(x)=(M_\theta^{\overline{\beta}}T_\theta^{\overline{\beta}}f)(x)\nonumber
$
для почти всех $x\in {\mathbb R^n}$.
\end{proof}

\Section{Обращение и описание потенциалов $\boldsymbol{M_\theta^{\overline{\beta}}\varphi}$ с $\boldsymbol{H^1}$-плотностями}
Обращение потенциала
$f=M_{\theta}^{\overline{\beta}}\varphi$,
$\varphi\in H^1$ в неэллиптическом случае будем строить в виде
\begin{equation}\label{2.016}
 T_\theta^{\overline{\beta}}f=\lim_{\varepsilon\to 0}^{(H^{1})}
 \lim_{\delta\to 0}^{(L^2)}
 T_{\theta,\varepsilon,\delta}^{\overline{\beta}}f,
\end{equation}
где $T_{\theta,\varepsilon,\delta}^{\overline{\beta}}$ определяется равенством \eqref{1.10}.

\begin{theorem}[5]
    Пусть $\varphi\in H^1${\rm .} Тогда
    \begin{equation}\nonumber
        M_\theta^{\overline{\beta}}(H^1)=\left\{f\in H^1: T_\theta^{\overline{\beta}} f\in
    H^1\right\},
    \end{equation}
где $T_\theta^{\overline{\beta}}$ "--- оператор \eqref{2.016}{\rm .}
\end{theorem}

Доказательство теоремы 5 проводится по той же схеме, что и доказательство теорем 3 и 4.
