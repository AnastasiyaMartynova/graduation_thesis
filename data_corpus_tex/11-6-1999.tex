\maketitle 
\label{begin} 
 
Пусть $R$ --- класс регулярных в единичном круге $E=\{z:|z|<1\}$ 
функций вида $f(z)=z+\sum\limits^\infty_{k=2}a_k z^k$. Обозначим через 
$S^0_n(a,b)$ и $S^*_n(a,b)$ классы функций $f(z)\in R$, удовлетворяющих 
соответственно соотношениям подчиненности 
\begin{align} 
(zf'(z))'/f'_n(z)&\prec (1+az)/(1+bz),\quad z\in E,\\ 
\intertext{и} 
zf'(z)/f_n(z)&\prec (1+az)/(1+bz),\quad z\in E, 
\end{align} 
где $a$ и $b$ --- произвольно заданные числа, $-1\le b\le 0$, $b<a\le 
1$, 
\begin{equation} 
f_n(z)=\frac{1}{n}\sum^{n-1}_{m=0}\varepsilon^{-m}f(\varepsilon^m z), 
\quad \varepsilon=\exp(2\pi i/n),\quad n=1,2,\dotsc 
\end{equation} 
Через $K_n(a,b)$ обозначим класс функций $g(z)\in R$, для каждой из 
которых существует такая функция $f(z)\in S^0_n(a,b)$, что 
\begin{equation} 
\re\{(zg'(z))'/f'_n(z)\}>0,\quad z\in E. 
\end{equation} 
Имеем $S^0_n(a,b)\subset S^0_n(1,-1)\equiv S^0_n$ --- класс функций, 
выпуклых относительно $n$ симметричных точек в $E$ \cite{1}, 
$S^*_n(a,b)\subset S^*_n(1,-1)\equiv S^*_n$ --- класс функций, 
звездообразных относительно $n$ симметричных точек в $E$ \cite{2}, 
\cite{3}. Функции $g(z)\in K_n(a,b)$ почти выпуклы относительно $n$ 
симметричных точек в $E$ \cite{1}; $K$ --- класс всех почти выпуклых функций 
в $E$ (напр., \cite{4}, с.\,583--584). 
 
Для $\delta>0$ определим $\delta$-окрестности функции $f(z)\in R$ 
следующим образом: 
$$ 
N^\nu_\delta(f)=\bigg\{g\in R:g(z)=z+\sum^\infty_{k=2}b_k z^k,\  
\sum^\infty_{k=2}k^\nu|a_k-b_k|\le \delta\bigg\},\quad 
\nu=1,2. 
$$ 
Заметим, что $N^2_\delta(f)\subset N^1_\delta(f)$. Понятие 
$\delta$-окрестности $N^1_\delta(f)$ введено в \cite{5}. 
 
В работах \cite{5}--\cite{9} рассматривался вопрос о включении 
$N^1_\delta(f)\subset S^*_n(a,b)$ для $f(z)\in S^0_n(a,b)$ при 
определенных значениях параметров $n$, $a$ и $b$. В данной статье этот 
вопрос исследуется при всех допустимых значениях параметров, и 
рассматривается вопрос о включении $N^2_\delta(f)\subset K_n(a,b)$, 
$f(z)\in S^0_n(a,b)$. Основным результатом статьи является 
 
\begin{thm} 
Если $f(z)\in S^0_n(a,b)$, то $N^1_{\delta^*_n}(f)\subset S^*_n(a,b)$, 
где 
$$ 
\delta^*_n= 
\begin{cases} 
\displaystyle\frac{a-b}{1-b}\int^1_0(1-t^n) 
(1-bt^n)^{(a-b)/bn-1}dt, & b\ne 0;\\ 
\displaystyle a\int^1_0(1-t^n)\exp(-at^n/n)dt, & b=0. 
\end{cases} 
$$ 
Постоянную $\delta^*_n$ нельзя заменить б\'ольшей. 
\end{thm} 
 
Доказательство теоремы 1 основывается на идее С.\,Рушевейха \cite{5}, 
развитой Дж.\,Броуном для обоснования включения $N^1_\delta(f)\subset 
S^*_1(1,b)$, $f(z)=z+a_{m+1}z^{m+1}+\dotsb\in S^0_1(1,b)$ \cite{8}, и 
на следующих леммах. 
 
\begin{lem} 
Если $f(z)\in R$, $|(f(z)*\psi_\theta(z))/z|\ge c>0$ в $E$, где 
\begin{equation} 
\psi_\theta(z)=(a-b)^{-1}e^{-i\theta}\bigg[ 
(1+ae^{i\theta})\frac{z}{1-z^n}-(1+be^{i\theta}) 
\frac{z}{(1-z)^2}\bigg],\quad 0\le\theta<2\pi, 
\end{equation} 
то $N^1_\delta(f)\subset S^*_n(a,b)$, $\delta=c(a-b)/(1-b)$. 
\end{lem} 
 
\begin{pf} 
Для $f(z)\in R$ соотношение подчиненности (2) равносильно условию 
$$ 
zf'(z)/f_n(z)\ne (1+ae^{i\theta})/(1+be^{i\theta}),\quad 
z\in E,\ \; 0\le\theta<2\pi, 
$$ 
которое с учетом равенств 
\begin{equation} 
f(z)* z/(1-z^n)=f_n(z),\quad 
f(z)* z(1-z)^{-2}=zf'(z) 
\end{equation} 
и определения $\psi_\theta(z)$ преобразуется к виду 
$(f(z)*\psi_\theta(z))/z\ne 0$. Следовательно, 
\begin{equation} 
f(z)\in S^*_n(a,b)\Longleftrightarrow 
(f(z)* \psi_\theta(z))/z\ne 0,\quad z\in E. 
\end{equation} 
По формуле (5) получим разложение 
$\psi_\theta(z)=z+\sum\limits^\infty_{k=2}c_k z^k$, где 
$$ 
c_k= 
\begin{cases} 
-k(1+b e^{i\theta})/(a-b)e^{i\theta}, & k\ne m n+1;\\ 
{[}1-k+(a-k b)e^{i\theta}]/(a-b)e^{i\theta}, & k=m n+1, 
\end{cases} 
\qquad m=1,2,\dotsc, 
$$ 
и заметим, что $|c_k|\le(1-b)k/(a-b)$, $k=2,3,\dotsc$, поскольку $-1\le 
b\le 0$, $b<a\le 1$. 
 
Пусть $g(z)=z+\sum\limits^\infty_{k=2}b_k z^k\in N^1_\delta(f)$, где 
функция $f(z)=z+\sum\limits^\infty_{k=2}a_k z^k$ удовлетворяет условиям 
леммы, $\delta=c(a-b)/(1-b)$. Тогда 
\begin{multline*} 
\biggl|\frac{g(z)*\psi_\theta(z)}{z} \biggr|\ge 
\biggl|\frac{f(z)*\psi_\theta(z)}{z} \biggr|- 
\biggl|\frac{(g(z)-f(z))*\psi_\theta(z)}{z} \biggr|> \\ 
>c-\sum^\infty_{k=2}|c_k(a_k-b_k)|\ge c-\frac{1-b}{a-b} 
\sum^\infty_{k=2}k|a_k-b_k|\ge 0, 
\end{multline*} 
откуда $(g(z)*\psi_\theta(z))/z\ne 0$, $z\in E$. Это в соответствии с 
(7) означает, что $g(z)\in S^*_n(a,b)$. 
\end{pf} 
 
\begin{notenonum} 
Импликация (7) при $n=1$, $b=-1$ и $a=1-2\alpha$, $0\le\alpha<1$, 
составляет содержание известной \cite{10} теоремы о свертке функций, 
звездообразных в $E$ порядка $\alpha$ (по Робертсону). 
\end{notenonum} 
 
\begin{lem} 
Если $f(z)\in S^0_n(a,b)$ и $f_n(z)$ определяется по формуле $(3)$, то 
\begin{equation} 
|f'_n(z)|\ge 
\begin{cases} 
(1-br^n)^{(a-b)/nb}, & b\ne 0;\\ 
\exp(-ar^n/n), & b=0, 
\end{cases} 
\end{equation} 
$|z|=r$, $0<r<1$. 
\end{lem} 
 
\begin{pf} 
Учитывая вытекающее из (3) равенство 
$f_n(\varepsilon^k z){=}\varepsilon^k f_n(z)$ с $k{=}1,2,\dotsc,n{-}1$, 
однолистность и выпуклость в $E$ функции $(1+az)/(1+bz)$, убеждаемся, 
что (1) влечет 
\begin{equation} 
1+z f''_n(z)/f'_n(z)\prec (1+az)/(1+bz),\quad z\in E. 
\end{equation} 
Отсюда в силу одного обобщения леммы Шварца (\cite{4}, с.\,323) имеем 
неравенство 
$$ 
|z f''_n(z)/f'_n(z)|\le |z|^n|a-b(1+z f''_n(z))/f'_n(z)|, 
$$ 
из которого следует, что значения функционала $w=1+z f''_n(z)/f'_n(z)$ 
при заданном $z\in E$, $|z|=r>0$, лежат в круге 
\begin{equation} 
\Delta_r=\{w:|w-w_0|\le\rho\}, 
\end{equation} 
где $w_0=(1-ab r^{2n})/(1-b^2r^{2n})$, $\rho=(a-b)r^n/(1-b^2r^{2n})$. 
Из (10) находим 
$$ 
\re\{z f''_n(z)/f'_n(z)\}\ge -(a-b)r^n/(1-br^n), 
$$ 
откуда 
$$ 
\frac{\partial}{\partial r}\ln|f'(r e^{i\varphi})|\ge 
-(a-b)r^{n-1}/(1-b r^n). 
$$ 
Интегрируя это неравенство по $r$ от $0$ до $r$, приходим к оценке 
(8). 
\end{pf} 
 
\begin{lem} 
Если $f(z)\in S^0_n(a,b)$, то $f(z)*\psi_\theta(z)\in K$ при каждом 
$\theta\in[0,2\pi)$, где $\psi_\theta(z)$ определяется по формуле 
$(5)$. 
\end{lem} 
 
\begin{pf} 
Если $b=-1$ и $\theta=0$, то согласно (5)\; $\psi_\theta(z)=z/(1-z^n)$ 
и, следовательно, $f(z)*\psi_\theta(z)=f_n(z)$. Но ввиду (9)\; 
$f_n(z)\in S^0_1(a,b)$, так что $f(z)*\psi_\theta(z)\in 
S^0_1(a,b)\subset K$. 
 
Пусть $\theta\in[0,2\pi)$, $\theta\ne 0$ при $b=-1$. Используя формулу 
(5) и равенства (6), получим 
\begin{equation} 
\frac{(a-b)e^{i\theta}}{1+b e^{i\theta}} 
\frac{(f(z)*\psi_\theta(z))'}{f'_n(z)}= 
\frac{1+a e^{i\theta}}{1+b e^{i\theta}}- 
\frac{(z f'(z))'}{f'_n(z)}.
\end{equation} 
Фиксируем $\theta$ и положим $\theta_0=\arg\{e^{i\theta}/(1+b 
e^{i\theta})\}$. Тогда для любого действительного $\alpha$ из (11) 
будем иметь 
\begin{equation} 
e^{i(\theta_0+\alpha)} 
\frac{(f(z)*\psi_\theta(z))'}{f'_n(z)}=e^{i\alpha} 
\frac{|1+b e^{i\theta}|}{a-b} 
\biggl[\frac{1+a e^{i\theta}}{1+b e^{i\theta}}- 
\frac{(z f'(z))'}{f'_n(z)}\biggr]. 
\end{equation} 
 
На основании (1) заключаем, что значения 
$w=(z f'(z))'/f'_n(z)$ для $f(z)\in S^0_n(a,b)$ и $z\in E$ лежат в 
$\Delta$, где $\Delta$ --- круг в $\Delta_1$ (см. (10)) при $b\ne -1$ и 
$\Delta$ --- полуплоскость $\re w>(1-a)/2$ при $b=-1$. Граница $\Delta$ 
--- окружность (при $b=-1$ --- прямая) 
$$ 
\Gamma=\{w:w=(1+ae^{it})/(1+be^{it}),\ 0\le t<2\pi\}. 
$$ 
Так как $(1+ae^{i\theta})/(1+be^{i\theta})\in\Gamma$ и 
$0\notin\Delta\cup\Gamma$, то существует такое $\alpha=\alpha(\theta)$, 
что 
\begin{equation} 
\re\biggl\{e^{i\alpha}\bigg[\frac{1+ae^{i\theta}}{1+be^{i\theta}}- 
\frac{(z f'(z))'}{f'_n(z)}\bigg] 
\biggr\}>0,\quad z\in E. 
\end{equation} 
 
Сопоставив (12) и (13), приходим к неравенству 
$$ 
\re\{e^{i(\theta_0+\alpha)}(f(z)*\psi_\theta(z))'/ 
f'_n(z)\}>0,\quad z\in E, 
$$ 
означающему вместе с (9), что $f(z)*\psi_\theta(z)\in K$ (\cite{4}, 
с.\,583--584). 
\end{pf} 
 
\begin{lem} 
Если $f(z)\in S^0_n(a,b)$, то
\begin{equation} 
\bigg|1+ae^{i\theta}-(1+be^{i\theta}) 
\frac{(z f'(z))'}{f'_n(z)}\bigg|\ge 
\frac{(a-b)(1-r^n)}{1-b r^n}, 
\end{equation} 
$|z|=r$, $0<r<1$. 
\end{lem} 
 
\begin{pf} 
При $b=-1$ и $\theta=0$ неравенство (14) тривиально. Пусть 
$0\le\theta<2\pi$, $\theta\ne 0$ при $b=-1$. Положим 
\begin{equation} 
Q(w)=(1+be^{i\theta})\biggl(\frac{1+ae^{i\theta}}{1+be^{i\theta}}-w 
\biggr),\quad w\in\Delta_r, 
\end{equation} 
где $\Delta_r$ определяется по (10). Функция $Q(w)$ регулярна и 
$Q(w)\ne 0$ в $\Delta_r$. Поэтому к $Q(w)$ примен\'им принцип минимума 
модуля для регулярных функций (\cite{11}, с.\,192), в силу которого 
\begin{multline*} 
|Q(w)|\ge\min_{w\in\partial\Delta_r}|Q(w)|= 
\min_{0\le\varphi<2\pi}|1+ae^{i\theta}-(1+be^{i\theta}) 
(w_0+\rho e^{i\varphi})|= \\ 
=\frac{a-b}{1-b^2r^{2n}}\left( 
\sqrt{1+2br^{2n}\cos\theta+b^2r^{4n}}-r^n 
\sqrt{1+2b\cos\theta+b^2}\,\right). 
\end{multline*} 
Находя минимум правой части относительно $\theta$, получаем 
$$ 
|Q(w)|\ge (a-b)(1-r^n)/(1-br^n), 
$$ 
что вместе с (15) и (1) влечет неравенство (14). 
\end{pf} 
 
\begin{pf*}{Доказательство теоремы 1} 
Пусть $f(z)\in S^0_n(a,b)$. Тогда из (11) в силу леммы 4 имеем 
$$ 
|(f(z)*\psi_\theta(z))'|\ge|f'_n(z)| 
(1-r^n)/(1-br^n),\quad |z|=r. 
$$ 
По лемме 2 отсюда получаем 
\begin{equation} 
|(f(z)*\psi_\theta(z))'|\ge 
\begin{cases} 
(1-r^n)(1-br^n)^{(a-b)/nb-1}, & b\ne 0;\\ 
(1-r^n)\exp(-ar^n/n), & b=0. 
\end{cases} 
\end{equation} 
 
Ввиду леммы 3 функция $f(z)*\psi_\theta(z)$ однолистна в $E$, поскольку 
класс $K$ состоит из однолистных функций в $E$ (напр., \cite{4}, 
с.\,584). При однолистном отображении $w=f(z)*\psi_\theta(z)$ круга $E$ 
образом окружности $|z|=r$ является замкнутая кривая Жордана, 
заключающая внутри точку $w=0$. Отметим на этой кривой точку, ближайшую 
к $w=0$, и соединим ее с $w=0$ прямолинейным отрезком. Соответствующую 
отрезку кривую $|z|\le r$ обозначим через $L$. Тогда 
\begin{equation} 
|f(z)*\psi_\theta(z)|=\int_L|(f(z)*\psi_\theta(z))'|\,|dz|\ge 
\int^{|z|}_0|(f(z)*\psi_\theta(z))'|d|z|. 
\end{equation} 
Из (16) и (17) выводим оценку 
\begin{equation} 
\biggl|\frac{f(z)*\psi_\theta(z)}{z} \biggr|\ge 
\begin{cases} 
\nu(r)=\displaystyle\frac{1}{r}\int^r_0 
\frac{(1-t^n)dt}{(1-bt^n)^{1-(a-b)/bn}}, & b\ne 0;\\ 
\mu(r)=\displaystyle\frac{1}{r}\int^r_0 
(1-t^n)\exp(-at^n/n)dt,& b=0. 
\end{cases} 
\end{equation} 
Функции $\nu(r)$ и $\mu(r)$ убывают при $0<r<1$, поэтому из (18) имеем 
$$ 
|(f(z)*\psi_\theta(z))/z|\ge (1-b)\delta^*_n/(a-b), 
$$ 
где $\delta^*_n$ то же, что и в теореме. Теперь включение 
$N^1_{\delta^*_n}(f)\subset S^*_n(a,b)$ следует из леммы 1. 
 
Простыми вычислениями убедимся, что при $\theta=-\pi$, $z^n=-r^n$ имеет 
место равенство $(f(z)*\psi_\theta(z))/z=\nu(r)$ для функции  
$$ 
f(z)=\int^z_0(1+b\xi^n)^{(a-b)/bn}d\xi\in S^0_n(a,b), 
$$ 
если $b\ne 0$, и равенство $(f(z)*\psi_\theta(z))/z=\mu(r)$ для функции 
$$ 
f(z)=\int^z_0\exp(a\xi^n/n)d\xi\in S^0_n(a,b), 
$$ 
если $b=0$. Следовательно, постоянную $\delta^*_n$ нельзя заменить 
б\'ольшей. 
\end{pf*} 
 
\begin{thm} 
Если $f(z)\in S^0_n(a,b)$, то $N^2_{\delta_n}(f)\subset K_n(a,b)$, где 
$\delta_n=(1-a)(1-b)^{(a-b)/nb-1}$ при $b\ne 0$ и 
$\delta_n=(1-a)\exp(-a/n)$ при $b=0$. 
\end{thm} 
 
\begin{pf} 
Пусть $g(z)\in N^2_{\delta_n}(f)$. Тогда $|(z f'(z))'-(z 
g'(z))'|<\delta_n$ в $E$, так что 
\begin{equation} 
\re\frac{(z g'(z))'}{f'_n(z)}\ge \re 
\frac{(z f'(z))'}{f'_n(z)}-\frac{\delta_n}{|f'_n(z)|}. 
\end{equation} 
 
Из условия (1) по принципу подчиненности и лемме Шварца получим 
неравенство 
$$ 
\re\{(z f'(z))'/f'_n(z)\}\ge (1-ar)/(1-br),\quad |z|=r<1, 
$$ 
справедливое для любой функции $f(z)\in S^0_n(a,b)$. Правая часть в 
этом неравенстве убывает в промежутке $0<r<1$. Поэтому 
\begin{equation} 
\re\{(z f'(z))'/f'_n(z)\}\ge (1-a)/(1-b),\quad z\in E. 
\end{equation} 
 
В силу леммы 2 имеем в $E$ оценку 
\begin{equation} 
\frac{1}{|f'_n(z)|}\le 
\begin{cases} 
(1-b)^{-(a-b)/nb}, & b\ne 0;\\ 
\exp(a/n), & b=0. 
\end{cases} 
\end{equation} 
Из (19)--(21) с учетом определения $\delta_n$ находим неравенство (4), 
означающее, что $g(z)\in K_n(a,b)$. Следовательно, 
$N^2_{\delta_n}(f)\subset K_n(a,b)$. 
\end{pf}
