\Section{Плоская задача армированных сред в криволинейной ортогональной системе координат}
Уравнения равновесия в~криволинейных координатах $u$, $v$ запишутся в виде
\cite{FeodorovaNA:Lyv}
\begin{equation}
\label{FeodorovaNA:or4}
\begin{array}{c} \displaystyle
 \frac{\partial(H_2\sigma_{11})}{\partial
u}+\frac{\partial(H_1\sigma_{12})}{\partial v}+\frac{\partial
H_1}{\partial v}\sigma_{12}-\frac{\partial H_2}{\partial
u}\sigma_{22}+H_1H_2\Phi_1=0, \vspace{1mm} \\
\displaystyle
\frac{\partial(H_2\sigma_{12})}{\partial
u}+\frac{\partial(H_1\sigma_{22})}{\partial v}+\frac{\partial
H_2}{\partial u}\sigma_{12}-\frac{\partial H_1}{\partial
v}\sigma_{22}+H_1H_2\Phi_2=0.
\end{array}
\end{equation}
В уравнениях (\ref{FeodorovaNA:or4}) $\sigma_{ij}(u,v)$ "--- компоненты тензора напряжений;
$\Phi_1$, $\Phi_2$ "--- контравариантные компоненты вектора массовой силы;
$H_1=H_1(u,v),$ $H_2=H_2(u,v)$ "--- дифференциальные коэффициенты Ламе (в дальнейшем "--- коэффициенты Ламе), 
представляющие собой в общей ортогональной системе координат заданные функции координат $u$, $v$. 
Система записана относительно физических компонент тензоров напряжений.
Связь между физическими компонентами тензора напряжений $\sigma_{ij}(u,v)$ и~физическими компонентами тензора деформаций $\varepsilon_{ij}(u,v)$
формулируется на основе структурной модели армированного материала~\cite{FeodorovaNA:N}. 
В~условиях термоупругого ортотропного деформирования  в~криволинейных ортогональных координатах она
представляется такими же формулами, что и структурная модель в
декартовых координатах:
\begin{equation}
\label{FeodorovaNA:orz} 
\begin{array}{c} \displaystyle
\sigma_{ij}=\Omega \sigma_{ij}^{c}+\sum_{k=1}^{2}\sigma_{k}\omega_{k}l_{ki}l_{kj},\quad \Omega=1-(\omega_1(u,v)+\omega_2(u,v)),\\
l_{k1}=\cos(\varphi_k),\quad l_{k2}=\sin(\varphi_k),\quad k\in \{1, 2\}.
\end{array}
\end{equation}
В (\ref{FeodorovaNA:orz}) напряжения в связующем находятся по
формулам
\begin{equation}
\label{FeodorovaNA:orz1}
\begin{array}{c}  
\sigma_{ii}^{c}=\frac{E}{1-\nu^2}\bigl(\varepsilon_{ii}+\nu\varepsilon_{jj}- \alpha_c(1+\nu)T\bigr),\;
\sigma_{ij}^{c}=\frac{E}{1+\nu}\varepsilon_{ij}, \quad 
j=3-i,\; i\in \{1, 2\}.
\end{array}
\end{equation}
 Здесь $\sigma_{k}(u,v)$  "--- напряжения в~волокнах; 
$\omega_k(u,v)$, $\varphi_k(u,v)$ "--- интенсивность и угол армирования волокон $k$-того семейства; 
$T$ "--- температура; 
$E$, $\nu$, $\alpha_c$ "--- соответственно модуль Юнга, коэффициент Пуассона и~коэффициент температурного расширения связующего материала. 
В~дальнейшем будем пользоваться следующими обозначениями для констант:
$m_1={(1-\nu^2)}^{-1}$, $m_2={(1+\nu)}^{-1}$, $m_3=Em_1$, $m_4=Em_2.$ 
Условие постоянства сечений волокон $k$-того
семейства имеет вид
\begin{equation}
\label{FeodorovaNA:orw} \frac{\partial}{\partial u}(H_2\omega_k
\cos \varphi_k)+\frac{\partial}{\partial v}(H_1\omega_k \sin
\varphi_k)=0.
\end{equation}

Если углы армирования совпадают с направлениями координатных линий,
то их значения $\varphi_1=0$, $\varphi_2={\pi}/{2},$ и~уравнения (\ref{FeodorovaNA:orw}) можно проинтегрировать:
$$\omega_1(u,v)=\frac{H_2(u^0,v)\omega^0_1(v)}{H_2(u,v)},\quad 
\omega_2(u,v)=\frac{H_1(u,v^0)\omega^0_2(u)}{H_1(u,v)},$$ 
где $\omega^0_1(v)$, $\omega^0_2(u)$ "--- известные функции, заданные на линиях $u=u^0=\rm const,$ $v=v^0=\rm const,$ 
удовлетворяющие условиям постоянства расхода арматуры. 
Общие ограничения для интенсивностей армирования задаются в виде \mbox{$0<\omega_k<0,7.$}
Интенсивность прослоек связующего изменяется в интервале $0<\Omega<1.$

 После выполнения ковариантного дифференцирования уравнения совместности Сен"--~Венана \cite{FeodorovaNA:Dem}
 и выражения символов Кристоффеля через коэффициенты Ламе получим для любой
ортогональной криволинейной системы координат уравнение совместности
 деформаций относительно физических компонент тензора деформаций:
\begin{multline}
\label{FeodorovaNA:or10d}
 C_1\frac{\partial^2\varepsilon_{22}}{\partial u^2}+
C_2\frac{\partial^2\varepsilon_{11}}{\partial
v^2}+C_3\frac{\partial^2\varepsilon_{12}}{\partial u\partial v}+
C_4\frac{\partial \varepsilon_{11}}{\partial u}+ C_5\frac{\partial
\varepsilon_{11}}{\partial v}+
 +C_6\frac{\partial \varepsilon_{22}}{\partial u}+
C_7\frac{\partial \varepsilon_{22}}{\partial v} +\\
+C_8\frac{\partial
\varepsilon_{12}}{\partial u}+C_9\frac{\partial \varepsilon_{12}}{\partial v}+
C_{10}\varepsilon_{11}+C_{11}\varepsilon_{22}+C_{12}\varepsilon_{12}=0.
\end{multline}
Коэффициенты $C_{s}$ в уравнении
(\ref{FeodorovaNA:or10d}) связаны с коэффициентами Ламе следующими зависимостями:
\begin{eqnarray*}
& C_1=H_1^2,\quad C_2=H_2^2,\quad C_3=-2H_1H_2,\quad C_4=-H_2\frac{\partial H_2}{\partial u},
\\
& C_5=H_1^2\Bigl(-\frac{2}{H_1}\frac{\partial H_1}{\partial v}-\frac{1}{H_2}\frac{\partial H_2}{\partial v}+\frac{4}{H_1}\frac{\partial H_1}{\partial v}\Bigr),\\
& C_6=H_2^2\Bigl(-\frac{2}{H_2}\frac{\partial H_2}{\partial u}-\frac{1}{H_1}\frac{\partial H_1}{\partial u}+\frac{4}{H_2}\frac{\partial H_2}{\partial u}\Bigr),\quad 
C_7=-H_1\frac{\partial H_1}{\partial v},\\
&C_8=2H_1H_2\Bigl(\frac{1}{H_2}\frac{\partial H_2}{\partial v}-\frac{1}{H_1H_2}
\Bigl(H_1\frac{\partial H_2}{\partial v}+H_2\frac{\partial H_1}{\partial v}\Bigr)\Bigr),\\
&C_9=2H_1H_2\Bigl(\frac{1}{H_1}\frac{\partial H_1}{\partial u}-\frac{1}{H_1H_2}\Bigl(H_1\frac{\partial H_2}{\partial
u}+H_2\frac{\partial H_1}{\partial u}\Bigr)\Bigr),\\
&\!\!\!\!C_{10}=2\Bigl(\frac{H_2}{H_1}\frac{\partial H_1}{\partial
u}\frac{\partial H_2}{\partial u}+H_1\frac{\partial^2
H_1}{\partial v^2}-\frac{H_2}{H_1}\frac{\partial H_2}{\partial
u}\frac{\partial H_1}{\partial u}-\frac{H_1}{H_2}\frac{\partial
H_2}{\partial v}\frac{\partial H_1}{\partial v}\Bigr),\\
&\!\!\!\!C_{11}=2\Bigl(\frac{H_1}{H_2}\frac{\partial H_1}{\partial
v}\frac{\partial H_2}{\partial v}+H_2\frac{\partial^2
H_2}{\partial u^2}-\frac{H_1}{H_2}\frac{\partial H_1}{\partial
v}\frac{\partial H_2}{\partial v}-\frac{H_2}{H_1}\frac{\partial
H_1}{\partial
u}\frac{\partial H_2}{\partial u}\Bigr),\\
&\!\!\!\!\!C_{12}=2\Bigl(-\frac{\partial H_1}{\partial u
}\frac{\partial H_2}{\partial v}+\frac{\partial H_1}{\partial
v}\frac{\partial H_2}{\partial u}-H_2\frac{\partial^2
H_1}{\partial v\partial u}-\frac{\partial H_1}{\partial
u}\frac{\partial H_2}{\partial v}- \frac{\partial H_1}{\partial
v}\frac{\partial H_2}{\partial u}\\
&-H_1\frac{\partial^2
H_2}{\partial v\partial u}+\frac{
\partial H_2}{\partial v}\Bigl(\frac{H_1}{H_2}\frac{\partial H_2}{\partial u}+\frac{\partial H_1}{\partial u}\Bigr)
+\frac{\partial H_1}{\partial u}
\Bigl(\frac{\partial H_2}{\partial
v}+\frac{H_2}{H_1} \frac{\partial H_1}{\partial v}\Bigr)\Bigr).
\end{eqnarray*}
На основе уравнений (\ref{FeodorovaNA:or4})--(\ref{FeodorovaNA:or10d}) строим систему разрешающих уравнений относительно компонент тензора деформаций. 
В~классической плоской задаче теории упругости \cite{FeodorovaNA:Tim} решение строится через функцию напряжений, которая удовлетворяет дифференциальному уравнению четвертого порядка. 
При этом вводятся ограничения на объёмные силы "--- это либо сила тяжести, либо объёмные силы, обладающие потенциалом. 
Преимущества постановки задачи в деформациях заключаются в том, что  объёмные силы могут быть произвольными, а~исходная разрешающая система является системой более низкого порядка
(второго вместо четвёртого), что важно в~дальнейшем для построения численного алгоритма при аппроксимации дифференциального оператора при построении разностной схемы. 
Более низкий порядок дифференциального оператора дает возможность надёжного численного счёта при наличии градиентов решений (дырки в пластине). 
Поэтому в~работе выбрана постановка задачи в деформациях. 
В~дальнейшем в~работе при присоединении граничных условий (как статических, так и~кинематических) будет показана возможность сведения условий на границе к~формулировкам в~деформациях.

Зададимся некоторыми криволинейными ортогональными системами координат и сформулируем в них полученное выше уравнение~(\ref{FeodorovaNA:or10d}).

\Section {Пример биполярной системы координат}
Пусть $(x,y)$ "--- декартовы координаты. 
Будем рассматривать биполярные координаты $(u,v),$ образующие взаимно ортогональную сетку кривых $u=\rm const$, $v=\rm const.$
Выразим прямоугольные координаты $x$ и $y$ через биполярные $u$ и~$v$ с~помощью конформного отображения~\cite{FeodorovaNA:Lavr}:



$$x+iy=a\th \frac{u+iv}{2}.
$$




\noindent
Разделяя действительную и мнимую части, имеем



\begin{equation}
\label{FeodorovaNA:or1} x=\frac{a\sh u}{\ch u+\cos
v}, \quad y=\frac{a\sin v}{\ch u+\cos v}.
\end{equation}




\noindent 
Параметр $a=\rm const$ задает координаты точек $x=\pm a,$ которые называются полюсами. 
Координатные линии $u=u_0=\rm const$ представляют собой эксцентрические окружности с~центрами на оси $OX$:
\begin{equation}
\label{FeodorovaNA:or2} 
(x-a\cth u_0)^2+y^2=\frac{a^2}{\sh^2 u_0}.
\end{equation}


\begin{wrapfigure}{R}{3.4cm}
\includegraphics[height=3.2cm]{bp.eps}
 \caption{} \label{FeodorovaNA:R4}
 \vspace{-5mm}
\end{wrapfigure}

\noindent
Координатные линии $v=v_0=\rm const$ "--- дуги окружностей с центрами на оси $OY$, проходящие через полюсы:
\begin{equation}
\label{FeodorovaNA:or3} 
x^2+(y+a\ctg v_0)^2=\frac{a^2}{\sin^2 v_0}.
\end{equation}



\noindent
Рассматриваемая структура армирования изображена на рис.~\ref{FeodorovaNA:R4}.


Вычислим углы армирования $\varphi_1$, $\varphi_2$ в биполярной системе координат. 
Для этого продифференцируем (\ref{FeodorovaNA:or2}) по $x$:
$2(x-a\cth u_0)+2yy^{\prime}_{x}=0.
$ 
Отсюда получим
$$
\tg \varphi_1=y^{\prime}_x=-\frac{(x-a \cth u_0)}{y},
$$
и после замены по формулам (\ref{FeodorovaNA:or1}):
$$
\tg \varphi_1=-\frac{\sh u-\cth u_0(\ch u+\cos v)}{\sin v}.
$$

Дифференцируя (\ref{FeodorovaNA:or3}) по $x,$ получим
$$x+(y+a\ctg v_0)y^{\prime}_x=0.$$
С учётом (\ref{FeodorovaNA:or1}) получаем значение тангенса угла армирования
второго семейства волокон:
\begin{eqnarray*}
y^{\prime}=-\frac{x}{y+a\ctg v_0}=-\frac{a\sh u}{a\sin v+a\ctg
v_0(\ch u+\cos v)}=\tg \varphi_2.
\end{eqnarray*}

В уравнения структурной модели (\ref{FeodorovaNA:orz}) входят значения квадратов синусов и~косинусов углов армирования $\varphi_1$, $\varphi_2$ и~их
произведений, находим их через полученные тангенсы по известным формулам тригонометрии. 
Подстановка коэффициентов Ламе, соответствующих биполярной системе координат, в~(\ref{FeodorovaNA:or10d}) и~дальнейшие преобразования приводят 
к~следующему виду уравнения совместности относительно физических компонент тензора деформаций:
\begin{multline}
\label{FeodorovaNA:or11}
\frac{\partial^2\varepsilon_{22}}{\partial u^2}+
\frac{\partial^2\varepsilon_{11}}{\partial v^2}-
2\frac{\partial^2\varepsilon_{12}}{\partial u\partial v}+
\frac{\sh u}{\ch u+\cos v}\frac{\partial\varepsilon_{11}}{\partial u}+ 
\frac{\sin v}{\ch u+\cos v}\frac{\partial\varepsilon_{11}}{\partial v}-
\\
- \frac{\sh u}{\ch u+\cos v}\frac{\partial\varepsilon_{22}}{\partial u}-
\frac{\sin v}{\ch u+\cos v}\frac{\partial\varepsilon_{22}}{\partial v}-
2\frac{\sin v}{\ch u+\cos v}\frac{\partial\varepsilon_{12}}{\partial u}+\\
+2\frac{\sh u}{\ch u+\cos v}\frac{\partial\varepsilon_{12}}{\partial v}+ 
2\frac{1+\ch u\cos v}{(\ch u+\cos v)^2}\varepsilon_{11}+
2\frac{1+\ch u\cos v}{(\ch u+\cos v)^2}\varepsilon_{22}+
\\
+4\frac{\sh u\sin v}{(\ch u+\cos v)^2}\varepsilon_{12}=0.
\end{multline}

\Section{Разрешающая система дифференциальных уравнений плоской
задачи армированных сред в деформациях в криволинейной системе
координат}
 Для формулировки разрешающей системы в деформациях к~полученному уравнению совместности деформаций добавляем два уравнения равновесия
(\ref{FeodorovaNA:or4}), сформулированные в деформациях после подстановки в~них (\ref{FeodorovaNA:orz}), (\ref{FeodorovaNA:orz1}).
Деформации в~волокне находим по структурной модели \cite{FeodorovaNA:N}:
\begin{eqnarray}
\label{FeodorovaNA:or16}
\varepsilon_{11}l_{k1}^{2}+\varepsilon_{22}l_{k2}^{2}+2\varepsilon_{12}l_{k1}l_{k2}=\varepsilon_{k}^{0}.
\end{eqnarray}
В формулах (\ref{FeodorovaNA:or16}) использованы  следующие обозначения:
$\varepsilon_{k}^{T}=\alpha_{k}^{a}T$, 
$\varepsilon_{k}^{0}=\varepsilon_{k}+\varepsilon_{k}^{T},$
 $\alpha_k^a$ "---
коэффициенты линейного расширения материала $k$-того семейства волокон $(k\in\{1, 2\})$. 
В~формулы  входит напряжение в~волокне, оно вычисляется по закону Гука
$\sigma_k=E_k\varepsilon_k+E_k\varepsilon_k^{T},$ где $E_k$ "--- модуль
Юнга материала $k$-того семейства волокон.
 После преобразований с учетом условия постоянства сечений волокон (\ref{FeodorovaNA:orw})   имеем следующий вид уравнений равновесия в~деформациях:
 
 \vspace{-5mm}
 
\begin{multline*}
a_{11}\frac{\partial
\varepsilon_{11}}{\partial u}+a_{12}\frac{\partial
\varepsilon_{11}}{\partial v}+a_{13}\frac{\partial
\varepsilon_{22}}{\partial u}+a_{14}\frac{\partial
\varepsilon_{22}}{\partial v}+a_{15}\frac{\partial
\varepsilon_{12}}{\partial u} a_{16}\frac{\partial
\varepsilon_{12}}{\partial v}+\\ 
+F_1(H_1,H_2,\omega_1,\omega_2,\varphi_1,\varphi_2,\varphi_{1,u},\varphi_{1,v},\varphi_{2,u}\varphi_{2,v})+H_1H_2\Phi_1=0,
\end{multline*} \vspace{-10mm}
\begin{multline}
a_{21}\frac{\partial \varepsilon_{11}}{\partial
u}+a_{22}\frac{\partial \varepsilon_{11}}{\partial
v}+a_{23}\frac{\partial \varepsilon_{22}}{\partial
u}+a_{24}\frac{\partial \varepsilon_{22}}{\partial
v}+a_{25}\frac{\partial \varepsilon_{12}}{\partial u}
a_{26}\frac{\partial \varepsilon_{12}}{\partial
v}+\\+F_2(H_1,H_2,\omega_1,\omega_2,\varphi_1,\varphi_2,\varphi_{1,u},\varphi_{1,v},\varphi_{2,u}\varphi_{2,v})+H_1H_2\Phi_2=0.
\label{FeodorovaNA:or18}  
\end{multline}

\noindent
Коэффициенты $a_{sr}$ в
(\ref{FeodorovaNA:or18}) определяются следующими зависимостями:
\begin{eqnarray*}
a_{11}=H_2(\Omega
m_3+E_1\omega_1\cos^4\varphi_1+E_2\omega_2\cos^4\varphi_2),\\
a_{12}=H_1(E_1\omega_1\sin\varphi_1\cos^3\varphi_1+E_2\omega_2\cos^3\varphi_2\sin\varphi_2),\\
a_{13}=H_2(\Omega\nu
m_3+E_1\omega_1\cos^2\varphi_1\sin^2\varphi_1+E_2\omega_2\cos^2\varphi_2\sin^2\varphi_2),\\
a_{14}=H_1(E_1\omega_1\sin^3\varphi_1\cos\varphi_1+E_2\omega_2\sin^3\varphi_2\cos\varphi_2),\\
a_{15}=2H_2(E_1\omega_1\cos^3\varphi_1\sin\varphi_1+E_2\omega_2\cos^3\varphi_2\sin\varphi_2),\\
a_{16}=2H_1(E_1\omega_1\cos^2\varphi_1\sin^2\varphi_1+E_2\omega_2\cos^2\varphi_2\sin^2\varphi_2+\Omega
m_4/2),\\
a_{21}=H_2(E_1\omega_1\sin\varphi_1\cos^3\varphi_1+E_2\omega_2\sin\varphi_2\cos^3\varphi_2),\\
a_{22}=H_1(\nu
m_3\Omega+E_1\omega_1\sin^2\varphi_1\cos^2\varphi_1+E_2\omega_2\sin^2\varphi_2\cos^2\varphi_2),\\
a_{23}=H_2(E_1\omega_1\cos\varphi_1\sin^3\varphi_1+E_2\omega_2\cos\varphi_2\sin^3\varphi_2),\\
a_{24}=H_1(\Omega
m_3+E_1\omega_1\sin^4\varphi_1+E_2\omega_2\sin^4\varphi_2),\\
a_{25}=2H_2(\Omega
m_4/2+E_1\omega_1\sin^2\varphi_1\cos^2\varphi_1+E_2\omega_2\cos^2\varphi_2\sin^2\varphi_2),\\
a_{26}=2H_1(E_1\omega_1\sin^3\varphi_1\cos\varphi_1+E_2\omega_2\sin^3\varphi_2\cos\varphi_2).
\end{eqnarray*}

Найденная совокупность трех уравнений (\ref{FeodorovaNA:or10d}), (\ref{FeodorovaNA:or18}) и~представляет
собой исходную разрешающую систему уравнений относительно трёх
компонент тензора деформаций
$\varepsilon_{11}$, $\varepsilon_{12}$, $\varepsilon_{22}.$

\Section{Тип разрешающей системы  плоской задачи в деформациях в~криволинейной системе координат}
Полученная выше система имеет следующую особенность "---  дифференциальные
уравнения, входящие в систему, имеют разный порядок: первое уравнение
(\ref{FeodorovaNA:or10d}) содержит вторые производные от неизвестных функций,
два вторых уравнения "--- первые производные.

 Для установления типа системы и дальнейшей постановки краевой задачи применим детерминантный метод \cite{FeodorovaNA:pet, FeodorovaNA:Bi2},
предварительно продифференцировав уравнения (\ref{FeodorovaNA:or18}) по любой из
независимых переменных. После дифференцирования порядок системы не
меняется (максимальный порядок производной, входящей в систему
второй), но система становится системой типа Коши"--~Ковалевской  \cite{FeodorovaNA:Bi1} и, следовательно,
возможно построение характеристического определителя.
 Запишем полученную систему в виде
\begin{equation}
\label{FeodorovaNA:or7x}
 A^{11}\frac{\partial^2\overline{\varepsilon}}{\partial u^2}+
2A^{12}\frac{\partial^2\overline{\varepsilon}}{\partial u \partial v}+
A^{22}\frac{\partial^2\overline{\varepsilon}}{\partial v^2}+\overline{F}=0,
\end{equation}
где
$\overline{\varepsilon}=(\varepsilon_{11}, \varepsilon_{22}, \varepsilon_{12}).$
Вектор $\overline{F}$ содержит первые производные искомых
неизвестных и правые части уравнений (\ref{FeodorovaNA:or10d}), (\ref{FeodorovaNA:or18}).
Коэффициентами системы (\ref{FeodorovaNA:or7x}) являются квадратные матрицы
третьего порядка $A^{11}$, $A^{22}$, $A^{12},$ они имеют такой вид:
\[A^{11}=
\left(
\begin{array}{ccc}
0&C_1& 0\\
a_{11}&a_{13} &a_{15} \\
a_{21}&a_{23} &a_{25}
\end{array}
\right), \;
%\]
%\[
 2A^{12}= \left(
\begin{array}{ccc}
0&0& C_3\\
a_{12}&a_{14} &a_{16} \\
a_{22}&a_{24} &a_{26}
\end{array}
\right), \;
%\]
%\[
 A^{22}= \left(
\begin{array}{ccc}
C_2&0&0\\
0&0&0\\
0&0&0
\end{array}
\right).
\]

Для анализа типа систем согласно \cite{FeodorovaNA:Bi2} построим
характеристическое уравнение вида
$$P(\lambda)=\det(A^{11}\lambda^2+2A^{12}\lambda+A^{22})=0.$$
После подстановки матриц $A^{11},A^{12},A^{22}$ найдем корни
$\lambda$ характеристического уравнения \[ \left|
\begin{array}{ccc}
C_2&C_1\lambda^2& C_3\lambda\\
(a_{11}\lambda^2+a_{12}\lambda)&(a_{13}\lambda^2+a_{14}\lambda) &(a_{15}\lambda^2+a_{16}\lambda) \\
(a_{21}\lambda^2+a_{22}\lambda)&(a_{23}\lambda^2+a_{24}\lambda)
&(a_{25}\lambda^2+a_{26}\lambda)
\end{array}
\right|=0.
\]
Очевидно, что $\lambda_1=\lambda_2=0.$ Далее при анализе
характеристического уравнения ограничимся случаем, когда углы
армирования совпадают с направлениями координатных линий, т.\,е. $\varphi_1=0$, $\varphi_2={\pi}/{2}.$ 
Заметим, что в каждой точке $(u, v)$ коэффициенты Ламе равны между собой $(H_1=H_2),$ поскольку ортогональные системы координат вводятся  аналитическими функциями
комплексного переменного; следовательно, выполняются условия
Коши"--~Римана \cite{FeodorovaNA:Lavr}. 
Исходный характеристический определитель упрощаем и приравниваем нулю:
\begin{equation} 
\label{FeodorovaNA:or75}
\left|
\begin{array}{ccc}
1&\lambda^2& -2\lambda\\
\lambda(\Omega m_3+E_1\omega_1)&\lambda \Omega m_3 &\Omega m_4 \\
\Omega\nu m_3 &\Omega m_3+E_2\omega_2&\Omega m_4\lambda
\end{array} \right|=0.
\end{equation}
Соотношение \eqref{FeodorovaNA:or75} представляет собой неполное алгебраическое уравнение четвёртого порядка (биквадратное уравнение относительно $\lambda$).
Коэффициенты этого уравнения зависят от технических характеристик материалов связующего и~арматуры, интенсивностей армирования $\omega_1(u,v)$, $\omega_2(u,v),$ для которых справедливо физическое ограничение $0<\omega_1(u, v),\, \omega_2(u, v)<0{,}7.$ 
Для широкого класса известных материалов, задавшись значениями коэффициентов Пуассона $\nu$, модулями Юнга связующего и~волокон $E$, $E_1$, $E_2$  и~учитывая ограничения для интенсивностей $\omega_1(u,v)$, $\omega_2(u,v),$ решаем биквадратное уравнение путём замены и~сведения к квадратному. 
При этом устанавливаем, что дискриминант квадратного уравнения строго больше нуля, а~корни квадратного уравнения отрицательны. 
Следовательно, биквадратное уравнение (\ref{FeodorovaNA:or75}) имеет четыре чисто мнимых попарно сопряжённых корня.
% Как известно, корни
%уравнения четвертой степени выражаются через свои коэффициенты в
%виде некоторой комбинации радикалов.
Поэтому исходная система дифференциальных уравнений в частных производных (\ref{FeodorovaNA:or10d}), (\ref{FeodorovaNA:or18}) является системой
эллиптического типа \cite{FeodorovaNA:Bi2}.

Рассмотрим предельный случай: $\omega_1=\omega_2=0,$ нет
армирующих семейств волокон. Тогда характеристический определитель
примет вид
\[ \left|
\begin{array}{ccc}
1&\lambda^2& -2\lambda\\
\lambda\frac{E}{1-\nu^2}&\lambda\frac{E\nu}{1-\nu^2} &\frac{E}{1+\nu} \\
\frac{E\nu}{1-\nu^2} &\frac{E}{1-\nu^2}&\lambda\frac{E}{1+\nu}
\end{array} \right|=0.
\]
После преобразований получим уравнение
$$\lambda^4+2\lambda^2+1=0.$$
Оно имеет четыре чисто мнимых
попарно сопряженных корня ($\lambda=\pm i $).
Это подтверждает известный в литературе факт об эллиптичности
 классической задачи теории упругости изотропного тела.

\Section {Граничные условия на криволинейном контуре в~криволинейной системе координат}
Полное напряжение на наклонной площадке с ортом $\overline{\nu}$ представляется в виде векторной суммы
$
\overline{P}_{\nu}=\overline{\sigma}_{\nu}+\overline{\tau}_{\nu},
$
где 
$\overline{\sigma}_{\nu}=\sigma_{\nu}\overline{\nu}$ "--- нормальное напряжение, 
$\overline{\tau}_{\nu}$ "--- полное касательное напряжение.
Разложение $\overline{P}_{\nu}$ в плоской декартовой системе координат с ортами $\overline{\imath}$, $\overline{\jmath}$ имеет вид
$\overline{P}_{\nu}=X_{\nu}\overline{\imath}+Y_{\nu}\overline{\jmath}.$

Если орт нормали $\overline{\nu}$ к контуру  определен как вектор $\overline{\nu}=l\overline{\imath}+m\overline{\jmath},$ то орт касательной
$\overline{s}$ на этой площадке  $\overline{s}=m\overline{\imath}-l\overline{\jmath}.$ В скалярной форме
напряжения на площадке с ортом $\overline{\nu}$ задаются соотношениями
\begin{equation}
\label{FeodorovaNA:or19}
\sigma_{\nu}=(\overline{P}_{\nu},\overline{\nu})=X_{\nu} l+Y_{\nu} m,
\quad
\tau_{\nu}=(\overline{P}_{\nu},\overline{s})=X_{\nu}m-Y_{\nu}l.
\end{equation}
Величины $l$, $m$ "--- направляющие косинусы углов, которые образует $\overline{\nu}$ c ортами осей координат. Значения $X_{\nu}$, $Y_{\nu}$
выражаем через компоненты напряжений:
\begin{equation}
\label{FeodorovaNA:or20}  X_{\nu}=\sigma_{11}l+\sigma_{12}m,
\quad 
Y_{\nu}=\sigma_{12}l+\sigma_{22} m.
\end{equation}

Подставим (\ref{FeodorovaNA:or20}) в (\ref{FeodorovaNA:or19}), получим значение нормального
и касательного напряжений на граничном контуре:
\begin{equation}
\label{FeodorovaNA:or21}
\sigma_{\nu}=\sigma_{11}l^2+\sigma_{22}m^2+2\sigma_{12}lm, \quad 
\tau_{\nu}=(\sigma_{11}-\sigma_{22})lm+\sigma_{12}(m^2-l^2).
\end{equation}
 Если векторная функция описывается в криволинейных координатах $u$, $v,$ то применяют локальный базис из векторов, касательных к координатным линиям либо перпендикулярных к ним.
%(существует три способа выбора локального базиса, Корн).
Функции
$\frac{1}{\sqrt{g_{11}}}\frac{\partial x}{\partial u}$, $\frac{1}{\sqrt{g_{22}}}\frac{\partial y}{\partial v}$
являются направляющими косинусами орта $i$-той координатной линии по отношению к осям $OX$, $OY.$ 
Поэтому статические граничные условия (\ref{FeodorovaNA:or21}) на контуре $\Gamma_s$ при заданных $p_{n}$, $p_{\tau}$
относительно физических компонент тензора напряжений $\sigma_{ij}$ преобразуем к следующему виду:

\begin{equation}
\label{FeodorovaNA:or22} 
\begin{array}{c}
\displaystyle \sigma_{11}\frac{1}{\sqrt{g_{11}}}\frac{\partial
x}{\partial u}+\sigma_{22}\frac{1}{\sqrt{g_{22}}}\frac{\partial
y}{\partial
v}+2\sigma_{12}\frac{1}{\sqrt{g_{11}}}\frac{1}{\sqrt{g_{22}}}\frac{\partial
y}{\partial v}=p_{n}, \vspace{1mm}\\
\displaystyle
(\sigma_{11}-\sigma_{22})\frac{1}{\sqrt{g_{11}}}\frac{1}{\sqrt{g_{22}}}\frac{\partial
x}{\partial u}\frac{\partial y}{\partial
v}+\sigma_{12}\Bigl(\frac{1}{g_{11}}\Bigl(\frac{\partial x}{\partial u}\Bigr)^2-\frac{1}{g_{22}}\Bigl(\frac{\partial y}{\partial v}\Bigr)^2\Bigr)=p_{\tau}.
\end{array}
\end{equation}
В (\ref{FeodorovaNA:or22}) $g_{ii}$ "--- компоненты метрического тензора. 
Для формулировки граничных условий в~деформациях в~(\ref{FeodorovaNA:or22}) подставим выражения для напряжений через деформации по формулам~(\ref{FeodorovaNA:orz}).

Пусть на граничном контуре $\Gamma_u$ заданы кинематические условия
для перемещений $u_1$, $u_2$:
\begin{equation}
\label{FeodorovaNA:or24 } u_1(\Gamma_u)=u_1^0(s),\quad u_2(\Gamma_u)=u_2^0(s).
\end{equation}
В соотношениях (\ref{FeodorovaNA:or24 })  $ u_1^0(s)$, $u_2^0(s)$ "--- известные функции. 
Дифференцируем заданные функции по формулам Коши в криволинейных
 координатах:
\begin{gather*}
\varepsilon_{11}=\frac{1}{H_1}\frac{\partial u_1}{\partial u}+\frac{1}{H_1H_2}\frac{\partial H_1}{\partial
v}u_2,\quad \varepsilon_{22}=\frac{1}{H_2}\frac{\partial u_2}{\partial v}+\frac{1}{H_1H_2}\frac{\partial H_2}{\partial
u}u_1,\\
\varepsilon_{12}=\frac{H_2}{H_1}\frac{\partial}{\partial u}\Bigl(\frac{u_2}{H_2}\Bigr)+\frac{H_1}{H_2}\frac{\partial}{\partial
v}\Bigl(\frac{u_1}{H_1}\Bigr),
\end{gather*}
где $u_i-$ компоненты вектора смещений.
В результате  получим
значения для компонент тензора деформаций на рассматриваемом
контуре. Когда на одной части контура заданы статические граничные
условия, а на другой "--- кинематические, комбинируем описанные выше случаи и устанавливаем  формулировку граничных условий в деформациях.

 Изложенный выше подход позволил получить разрешающие системы дифференциальных
уравнений для структур армирования по ортогональным траекториям,
изображенным на рис.~2.
