Рассмотрим задачи Дирихле 
$$
(-\Delta_D+q_j)u(x)\equiv-\Delta u(x)+ 
q_j(x)u(x)=\lambda u(x) \ \; (\forall x\in \Omega),\quad
u(x)=0 \ \; (\forall x\in S), 
$$
для которых $\Omega$ --- трехмерная ограниченная область с границей $S$ 
класса $C^2$ 
за 
исключением конечного числа угловых точек; $q_j$ ($j=1,2$) --- 
действительные 
непрерывные в $\Omega$ функции; $\lambda$ --- комплексный параметр. 
 
Пусть $\mu_j(q_j)$ --- собственные числа оператора $-\Delta_D+q_j$, 
перечисленные в 
порядке возрастания их величин с учетом кратности; $u_i(q_j)$ 
($u_i(q_j)\in\cal L_2(\Omega)$) --- 
отвечающие им собственные ортонормированные функции. 
 
Целью данной работы является доказательство следующего утверждения. 
 
\begin{thmnonum} 
Пусть потенциалы $q_j$ в $\Omega$ таковы, что 
\begin{gather} 
\mu_i(q_1),\mu_i(q_2)\sim C_1 i^{2/3}+ 
C_2 i^{1/3}+o(i^{1/3});\\ 
% 
p_0=\ov{\lim_{t\to\infty}}\,i^\varepsilon 
|\mu_i(q_2)-\mu_i(q_1)|<\infty 
\ \text{ при }\ \varepsilon>\frac{5}{9};\\ 
% 
\bigg\|\frac{\partial u_i(q_1)}{\partial\nu}(x)
\bigg\|_{\cal L_2(S)}, \;
\bigg\|\frac{\partial u_i(q_2)}{\partial\nu}(x)
\bigg\|_{\cal L_2(S)}\le C_3 i^{1/3};\\ 
% 
q_0=\ov{\lim_{i\to\infty}}\,i^\delta\bigg\| 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y)- 
\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
\bigg\|_{\cal L_2(S\times S)}<\infty 
\ \text{ при }\ \delta>\frac{1}{3}. 
\end{gather} 
Тогда $q_1=q_2$. 
\end{thmnonum} 
 
Введем действующие в $\cal L_2(S)$ операторы $N(\lambda,q_j)$, 
$$ 
N(\lambda,q_j)f=\frac{\partial u_j}{\partial\nu}\bigg|_S\,, 
$$ 
где $\nu$ --- внутренняя нормаль к $S$, а $u_j$ --- решение задачи 
$$
(-\Delta_D+q_j)u(x)=\lambda u(x) \ \; (\forall x\in\Omega),\quad
u(x)=f(x) \ \; (\forall x\in S,\ \lambda\ne\mu_i(q_j), 
\ i=\ov{1,\infty}). 
$$
 
В [1] показано, что оператор $N(\lambda,q_j)$ имеет представление 
$$ 
N(\lambda,q_j)f=\int_S\bigg(\,\sum^\infty_{i=1} 
\frac{\partial u_i(q_j)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_j)}}{\partial\nu}(y) 
(\mu_i(q_j)-\lambda)^{-1}\bigg)f(x)dx. 
$$ 
Используя ядро 
$$ 
^k N(\lambda,q_j)=\sum^\infty_{i=1} 
\frac{\partial u_i(q_j)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_j)}}{\partial\nu}(y) 
(\mu_i(q_j)-\lambda)^{-1} 
$$ 
этого интегрального оператора, находим 
\begin{gather*} 
^k N(\lambda,q_1)-\,^k N(\lambda,q_2)= 
\sum^\infty_{i=1}\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y) 
(\mu_i(q_1)-\lambda)^{-1}- \\ 
% 
-\sum^\infty_{i=1}\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
(\mu_i(q_2)-\lambda)^{-1}=\\ 
% 
= 
\sum^\infty_{i=1}\bigg( 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y) 
(\mu_i(q_2)-\lambda)- 
\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
(\mu_i(q_1)-\lambda)\bigg)\times \\ 
% 
\times(\mu_i(q_1)-\lambda)^{-1} 
(\mu_i(q_2)-\lambda)^{-1}=\\ 
% 
=\sum^\infty_{i=1}\bigg( 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y) 
(\mu_i(q_2)-\mu_i(q_1)+\mu_i(q_1)-\lambda)-\\ 
% 
-\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
(\mu_i(q_1)-\lambda)\bigg) 
(\mu_i(q_1)-\lambda)^{-1}(\mu_i(q_2)-\lambda)^{-1} 
\equiv\Sigma'+\Sigma'', 
\end{gather*} 
где 
\begin{align*} 
\Sigma'&=\sum^\infty_{i=1}(\mu_i(q_2)-\mu_i(q_1)) 
(\mu_i(q_1)-\lambda)^{-1} 
(\mu_i(q_2)-\lambda)^{-1} 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y);\\ 
% 
\Sigma''&=\sum^\infty_{i=1}\bigg( 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y)- 
\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y)\bigg) 
(\mu_i(q_2)-\lambda)^{-1}. 
\end{align*} 
Теперь для операторной нормы имеем 
\begin{gather*} 
\|N(\lambda,q_1)-N(\lambda,q_2)\|\le 
\|N(\lambda,q_1)-N(\lambda,q_2)\|_2=\\ 
% 
=\bigg(\int_S|^k[N(\lambda,q_1)-{}^k N(\lambda,q_2)]|^2 
dx\,dy\bigg)^{1/2}=\\ 
% 
=\bigg(\int_S|^k N(\lambda,q_1)-\,^k N(\lambda,q_2)|^2 
dx\,dy\bigg)^{1/2}= 
\bigg(\int_S|\Sigma'+\Sigma''|^2dx\,dy \bigg)^{1/2}\le \\ 
% 
\le\bigg(\int_S|\Sigma'|^2dx\,dy \bigg)^{1/2}+ 
\bigg(\int_S|\Sigma''|^2dx\,dy \bigg)^{1/2}= 
\|\Sigma'\|_{\cal L_2(S)}+\|\Sigma''\|_{\cal L_2(S)}\le \\ 
% 
\le\sum^\infty_{i=1}|\mu_i(q_1)-\mu_i(q_2) 
\|\mu_i(q_1)-\lambda|^{-1}|\mu_i(q_2)-\lambda|^{-1} 
\bigg\|\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\bigg\|^2_{\cal L_2(S)}+\\ 
% 
+\sum^\infty_{i=1}\bigg\| 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y)- 
\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
\bigg\|_{\cal L_2(S\times S)} 
|\mu_i(q_2)-\lambda|^{-1}\equiv S_1+S_2. 
\end{gather*} 
 
В [2] доказано, что при $|n-t_k|>c_1t^\beta_k>0$, где 
$\frac{1}{3}\le\beta\le1$, справедливы 
оценки 
\begin{equation} 
\begin{split} 
|\mu_n(q_j)-\mu_{t_k}(q_j)|&\ge c_2\max 
\{t_k,n\}^\beta>0,\\ 
|\mu_n(q_2)-\mu_{t_k}(q_1)|&\ge c_3\max 
\{t_k,n\}^\beta>0. 
\end{split} 
\end{equation} 
Полагая $\lambda=N^2-1+2N i\equiv\mu_{i_0}(q_1)+2N i$, 
$$ 
\max\{\tfrac{5}{6}-\tfrac{\varepsilon}{2};\tfrac{1}{3}\} 
<\beta<\min\{1;\varepsilon\},\quad 
\max\{1-\delta;\tfrac{1}{3}\}<\wt\beta<\min 
\{1;\tfrac{1}{3}+\delta\}, 
$$ 
имеем в силу (3) 
\begin{gather*} 
S_1\le c_4\sum^\infty_{i=1} 
|\mu_i(q_1)-\mu_i(q_2)|i^{2/3} 
|\mu_i(q_1)-\mu_{i_0}(q_1)-2N i|^{-1} 
|\mu_i(q_2)-\mu_{i_0}(q_1)-2N i|^{-1}\equiv \\ 
% 
\equiv\sum_{|i-i_0|\le C i^\beta_0}+ 
\sum_{|i-i_0|\ge C i^\beta_0>0}. 
\end{gather*} 
 
В силу (2) и (5) (применительно ко второму слагаемому) 
\begin{gather*} 
\sum_{|i-i_0|\le C i^\beta_0}+\sum_{|i-i_0|\ge C i^\beta_0>0}\le 
c_5\bigg(\frac{p_0i^{-\varepsilon}_0i^{2/3}_0i^\beta_0} 
{i^{1/3}_0i^{1/3}_0}+\sum_{|i-i_0|\ge C i^\beta_0>0} 
\frac{p_0i^{-\varepsilon}i^{2/3}}{i^{2\beta}}\bigg)\le \\ 
% 
\le c_6 p_0\bigg(\frac{1}{i_0^{\varepsilon-\beta}}+ 
\frac{1}{i_0^{2\beta+\varepsilon-5/3}}\bigg)\to 0 
\end{gather*} 
при $i_0\to\infty$. Аналогично 
\begin{gather*} 
S_2=\sum^\infty_{i=1}\bigg\| 
\frac{\partial u_i(q_1)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_1)}}{\partial\nu}(y)- 
\frac{\partial u_i(q_2)}{\partial\nu}(x) 
\frac{\ov{\partial u_i(q_2)}}{\partial\nu}(y) 
\bigg\|_{\cal L_2(S\times S)} 
|\mu_i(q_2)-\mu_{i_0}(q_1)-2N i|^{-1}\equiv \\ 
% 
\equiv 
\sum_{|i-i_0|\le C_0 i^{\wt \beta}_0}+ 
\sum_{|i-i_0|\ge C_0 i^{\wt \beta}_0>0}, 
\end{gather*} 
и в силу (4), (5) (применительно ко второму слагаемому) 
\begin{gather*} 
\sum_{|i-i_0|\le C_0 i^{\wt \beta}_0}+ 
\sum_{|i-i_0|\ge C_0 i^{\wt \beta}_0>0}\le c_7\bigg( 
\frac{q_0i_0^{-\delta}i_0^{\wt\beta}}{i_0^{1/3}}+ 
\sum_{|i-i_0|\ge C_0 i^{\wt \beta}_0>0} 
\frac{q_0i^{-\delta}}{i^{\wt\beta}}\bigg)\le \\ 
% 
\le c_8 q_0\bigg(\frac{1}{i_0^{\delta-\wt\beta+1/3}}+ 
\frac{1}{i_0^{\delta+\wt\beta-1}}\bigg)\to 0 
\end{gather*} 
при $i_0\to\infty$. Подытожив, приходим к равенству 
\begin{equation} 
\lim_{i_0\to\infty}\|N(\lambda,q_1)-N(\lambda,q_2)\|=0. 
\end{equation} 
 
Введем функцию рассеяния 
$$ 
S(\lambda,\theta,\omega;q_j)=\int_S(N(\lambda,q_j) 
\varphi_{\lambda,\omega})(x)\ov{\varphi_{\lambda,-\theta}}d S_x, 
$$ 
где $S_x$ --- элемент площади поверхности, 
$\varphi_{\lambda,\omega}(x)=\exp(i\sqrt{\lambda}\,\omega x)$, 
$\|\omega\|=\sqrt{\omega^2_1+\omega^2_2+\omega^2_3}$. Имеем 
\begin{gather*} 
|S(\lambda,\theta,\omega;q_1)- 
S(\lambda,\theta,\omega;q_2)|=\bigg|\int_S 
(N(\lambda,q_1)\varphi_{\lambda,\omega})(x) 
\ov{\varphi_{\lambda,-\theta}})(x)d S_x -
\int_S(N(\lambda,q_2)\varphi_{\lambda,\omega})(x) 
\ov{\varphi_{\lambda,-\theta}(x)}d S_x\bigg|=\\ 
% 
=\bigg|\int_S[(N(\lambda,q_1)\varphi_{\lambda,\omega})(x) 
-(N(\lambda,q_2)\varphi_{\lambda,\omega})(x)]
\ov{\varphi_{\lambda,-\theta}(x)}d S_x\bigg|= \\ 
% 
=\bigg|\int_S([N(\lambda,q_1)-N(\lambda,q_2)]
\varphi_{\lambda,\omega}(x)]\ov{\varphi_{\lambda,-\theta}(x)}
d S_x\bigg|=\\
%
=\langle[N(\lambda,q_1)-N(\lambda,q_2)] 
\varphi_{\lambda,\omega},\ \varphi_{\lambda,-\theta}\rangle \le 
\|[N(\lambda,q_1)-N(\lambda,q_2)]\varphi_{\lambda,\omega} 
\|_{\cal L_2(S)}\|\varphi_{\lambda,-\theta}\|_{\cal L_2(S)} \le \\ 
% 
\le\|N(\lambda,q_1)-N(\lambda,q_2)\|\, 
\|\varphi_{\lambda,\omega}\|_{\cal L_2(S)} 
\|\varphi_{\lambda,-\theta}\|_{\cal L_2(S)}. 
\end{gather*} 
Отсюда в силу (6) и равномерной ограниченности 
$\varphi_{\lambda,\omega}$ 
получаем 
\begin{equation} 
\lim_{i_0\to\infty} 
|S(\lambda,\theta,\omega;q_1)- 
S(\lambda,\theta,\omega;q_2)|=0. 
\end{equation} 
Доказано [1], что 
\begin{multline*} 
S(\lambda,\theta,\omega;q_j)=-\frac{\lambda}{2} 
(\theta-\omega)^2\int_\Omega\exp(-i\sqrt{\lambda} 
(\theta-\omega)x)dx+\\ 
% 
+\int_\Omega\exp(-i\sqrt{\lambda}(\theta-\omega)x)q_j(x)dx- 
\int_S(R(\lambda)q_j\varphi_{\lambda,\omega})(x) 
\ov{q_j\varphi_{\lambda,-\theta}(x)}d S_x, 
\end{multline*} 
где $R(\lambda)=(-\Delta_D+q_j-\lambda)^{-1}$; 
$$ 
\lim_{N\to\infty} 
(St_N,\theta_N,\omega_N;q_j)=-\frac{|\xi|^2}{2} 
\int_\Omega\exp(-i x\xi)dx+\int_\Omega 
\exp(-i x\xi)q_j(x)dx, 
$$ 
где при $(\xi,\eta)=0$, $\xi\ne0$ положено 
$$ 
c_N=\bigg(1-\frac{\xi^2}{4N^2}\bigg)^{1/2},\ \; 
\theta_N=c_N\eta+\frac{\xi}{2N},\ \; 
\omega_N=c_N\eta-\frac{\xi}{2N},\ \; 
\sqrt{t_N}=N+i. 
$$ 
 
Если $N\to\infty$, то $\mu_{i_0}(q_1)\to\infty$, поскольку 
$\mu_{i_0}(q_1)=N^2-1$, и из (1) 
следует, что $i_0\to\infty$. Таким образом, в силу (7) 
\begin{equation} 
\lim_{N\to\infty}|S(t_N,\theta_N,\omega_N;q_1)- 
S(t_N,\theta_N,\omega_N;q_2)|=0. 
\end{equation} 
Кроме того, ввиду существования пределов 
$\lim\limits_{N\to\infty}S(t_N,\theta_N,\omega_N;q_1)$ и 
$\lim\limits_{N\to\infty}S(t_N,\theta_N,\omega_N;q_2)$ имеем 
$$ 
\lim_{N\to\infty}|S(t_N,\theta_N,\omega_N;q_1)- 
S(t_N,\theta_N,\omega_N;q_2)|= 
|\lim_{N\to\infty}S(t_N,\theta_N,\omega_N;q_1)- 
\lim_{N\to\infty}S(t_N,\theta_N,\omega_N;q_2)|. 
$$ 
Теперь (8) дает 
$\lim\limits_{N\to\infty}S(t_N,\theta_N,\omega_N;q_1)= 
\lim\limits_{N\to\infty}S(t_N,\theta_N,\omega_N;q_2)$, 
т\,.е. $\int\limits_\Omega\exp(-i x\xi)q_1(x)dx= \linebreak
\int\limits_\Omega\exp(-i x\xi)q_2(x)dx$. 
 
Таким образом, показано, что преобразования Фурье для функций $q_1$ и 
$q_2$ 
совпадают, откуда следует, что $q_1=q_2.\quad\square$
