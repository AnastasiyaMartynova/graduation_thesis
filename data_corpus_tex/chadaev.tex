Рассмотрим нелинейное уравнение дробного порядка
\begin{equation}
\label{chad:eq1}
D_{0x}^\alpha y(t)=f\left(x, \, y(x), \, D_{0x}^\beta y(t)\right), 
\end{equation}
где $D_{0x}^\alpha$ "--- оператор дробного дифференцирования порядка $\alpha$,  причём
$n-1<\alpha\leq n$,  $n\in \mathbb N$,  $0\leq \beta <\min\{1, \, \alpha\}$,  $x\in ]0,  \,  l]$~\cite{chadaev:1, chadaev:2}.

Пусть $C_\gamma^\alpha\left[ 0, \,  l\right]$ "--- класс функций  $y\left(x\right)$,  имеющих суммируемую дробную производную порядка $\alpha$ во всех точках $x\in (0,  \,  l]$ и~представимых в~виде
\begin{equation}
\label{chad:eq2}
y\left(x\right)=x^{-\gamma}\varphi\left(x\right), 
\end{equation}
где $\varphi\left(x\right)\in C\left[ 0, \,  l\right]$ \cite{chadaev:3}.

\begin{newthm}{Задача типа Коши}
Найти решение уравнения \eqref{chad:eq1},  удовлетворяющее условиям
\begin{equation}
\label{chad:eq3}
%\begin{array}{c}
D_{0x}^{\alpha-k} y(t)\big|_{x=0} =b_k,  \quad k=1,  2,  \dots,  n-1; \quad
\underset{x \to 0}{\lim} x^{n-\alpha} y(x)=y_0 < \infty
%\end{array}
\end{equation}
в классе $C_\gamma^\alpha\left[ 0, \,  l\right]$.
\end{newthm}

Сформулируем и докажем теорему о существовании	 и  единственности решения этой задачи. Обозначим через $D$ множество точек $(x, \, y, \, z)$ из области $G$,  лежащей в~$\mathbb R^3$:
\begin{multline*}
%\label{chad:eq4}
D=\Bigl\{(x, y, z)\in G\, :\, 0\, <x\leq l, \; \Bigm|x^{n-\alpha}y(x)-\frac{b_n}{\Gamma(\alpha -n+1)}\Bigm|\leq a,  
\\ 
\Bigm|x^{n-\alpha+\beta}z(x)-\frac {b_n}{\Gamma(\alpha -n-\beta+1)}\Bigm|\leq b\Bigl\}, 
\end{multline*}
где $
\sum_{k=0}^{n-1} \frac{l^{n-k}b_k}{\Gamma (\alpha -k+1)}<a$, 
$
\sum_{k=0}^{n-1} \frac{l^{n-k}b_k}{\Gamma (\alpha -k-\beta +1)}<b$;
$\Gamma(\alpha)$ "--- гамма"=функция Эйлера;
$z(x)=D_{0x}^\beta y(t)$;
$a$,  $b$,  $l$,  $b_0$ "--- постоянные.


\begin{newthm}{Теорема} 
Пусть $f(x, \, y, \, z)$ "--- вещественнозначная,  непрерывная в области $G$ функция,  удовлетворяющая условию Липшица по $z{:}$
\begin{equation}
\label{chad:eq5}
|f(x, y_1, z_1)-f(x, y_2, z_2)|\leq N \bigl(|y_1-y_2|+|z_1-z_2|\bigr)
\end{equation}
и ограничению
\begin{equation}
\label{chad:eq6}
\max_{0<x\leq l}|f(x, y, z)|=b_0<\infty.
\end{equation}                                    
Тогда решение задачи  \eqref{chad:eq3} для уравнения \eqref{chad:eq1} в области $D\subset G$ существует,  принадлежит классу $C_{n-\alpha}^\alpha[0, l]$ и единственно.
\end{newthm}

\begin{proof}
Для любой функции,  представимой в виде \eqref{chad:eq2},  имеем
\begin{equation}
\label{chad:eq7}
\Gamma (n-\alpha)D_{0x}^{\alpha-n}y(t)\!=\!\int_0^x \frac{y(t)dt}{(x-t)^{\alpha+1-n}}=
x^{n-\alpha-\gamma}\int_0^1 \frac{\varphi(xt)dt}{t^\gamma(1-t)^{\alpha+1-n}}.
\end{equation}  
Поэтому
\begin{equation}
\label{chad:eq8}
\lim_{x \to 0} x^{\alpha+\gamma-n}D_{0x}^{\alpha-n}y(t)=\frac{\varphi(0)\Gamma(1-\gamma)}{\Gamma(n+1-\alpha-\gamma)}.
\end{equation}  
Из \eqref{chad:eq7} и \eqref{chad:eq8} можно заключить,  что необходимым условием разрешимости задачи для уравнения \eqref{chad:eq1} в~классе  $C_\gamma^\alpha[0, l]$ является условие $\alpha+\gamma<n$,   если $y_0=0$,  и~условие  $\alpha+\gamma=n$ "--- для произвольного $y_0$.

Применяя к уравнению \eqref{chad:eq1} оператор $D_{0x}^{-\alpha}$~{\rm\cite{chadaev:3}},  имеем
\begin{equation*}
%\label{chad:eq9}
y(x)=\sum_{k=1}^n \frac{x^{\alpha-k}}{\Gamma(\alpha-k+1)}D_{0x}^{\alpha-k}y(t)\Bigr|_{x=0}+\frac{ 1}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}f\bigl(t, y(t), D_{0t}^\beta y(s)\bigr)dt.
\end{equation*}  
Учитывая \eqref{chad:eq3},  получим $b_n=y_0\Gamma(\alpha-n+1)$,  тогда
\begin{equation}
\label{chad:eq10}
y(x)=\sum_{k=1}^n b_k\frac{x^{\alpha-k}}{\Gamma(\alpha-k+1)}+\frac{1}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}f\bigl(t, y(t), D_{0t}^\beta y(s)\bigr)dt.
\end{equation}

Для метода последовательных приближений,  полагая
\begin{equation*}
%\label{chad:eq11}
y_0(x)=\sum_{k=1}^n b_k\frac{x^{\alpha-k}}{\Gamma(\alpha-k+1)} , 
\end{equation*}
имеем
\begin{equation}
\label{chad:eq12}
y_m(x)=y_0(x)+\frac{1}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}f\bigl(t, y_{m-1}(t), D_{0t}^\beta y_{m-1}(s)\bigr)dt, \quad m\in \mathbb N.
\end{equation}
При $0<x\leq l$ необходимо,  чтобы точки $(x, y_m, z_m)$ оставались в $D$. Из условия \eqref{chad:eq6} и уравнения \eqref{chad:eq12} вытекает оценка
$$
x^{n-\alpha}y_m(x)-\sum_{k=1}^n b_k\frac{x^{n-k}}{\Gamma(\alpha-k+1)}\leq b_0\frac{x^n}{\Gamma(\alpha+1)},
$$
следовательно,
\begin{equation}
\label{chad:eq13}
\Bigl|x^{n-\alpha}y_m(x)-\frac{b_n}{\Gamma(\alpha-n+1)}\Bigl|\leq\sum_{k=0}^{n-1} b_k\frac{x^{n-k}}{\Gamma(\alpha-k+1)}\leq \sum_{k=0}^{n-1}b_k\frac{l^{n-k}}{\Gamma(\alpha-k+1)}<a.
\end{equation}
Для $z_m(x)=D_{0x}^\beta y_m(t)$ имеем
\begin{equation}
\label{chad:eq14}
\Bigl|x^{n-\alpha+\beta}z_m(x)-\frac{b_n}{\Gamma(\alpha-n-\beta+1)}\Bigl|<\sum_{k=0}^{n-1} \frac{b_kl^{n-k}}{\Gamma(\alpha-n-\beta+1)}<b.
\end{equation}

Из \eqref{chad:eq13}, \eqref{chad:eq14} очевидно,  что точки $(x, y_m, z_m)\in D$. 
Из уравнения \eqref{chad:eq12} и условия \eqref{chad:eq6} получим оценку
\begin{equation}
\label{chad:eq15}
\Bigl|y_1(x)-y_0(x)\Bigl|\leq\frac{b_0x^\alpha}{\Gamma(\alpha+1)}\leq \frac{b_0l^\alpha}{\Gamma(\alpha+1)}.
\end{equation}
Используя оценку \eqref{chad:eq15} и условие~\eqref{chad:eq5},  получим
\begin{multline*}
%\label{chad:eq16}
\Bigl|y_2(x)-y_1(x)\Bigl|=\Biggl|\int_0^x\frac{(x-t)^{\alpha-1}}{\Gamma(\alpha)}\Bigl[f(t, y_1(t), D_{0t}^\beta \,  y_1(s))-f(t, y_0(t), D_{0t}^\beta y_0(s))\Bigl]dt\Biggl|\leq
\\
\leq\frac{N}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}\Bigl[|y_1(t)-y_0(t)|+|D_{0t}^\beta y_1(s)-D_{0t}^\beta y_0(s)|\Bigl]dt\leq
\\
\leq\frac{Nb_0}{\Gamma(\alpha)}\int_0^x\frac{(x-t)^{\alpha-1}}{\Gamma(\alpha+1)}\Bigl(t^\alpha+D_{0t}^\beta s^\alpha\Bigl)dt %\leq \\ 
\leq Nb_0x^{2\alpha}\Biggl(\frac{1}{\Gamma(2\alpha+1)}+\frac{x^{-\beta}}{\Gamma(2\alpha+1-\beta)}\Biggl)\leq \\ \leq Nb_0l^{2\alpha}\Biggl(\frac{1}{\Gamma(2\alpha+1)}+\frac{l^{-\beta}}{\Gamma(2\alpha+1-\beta)}\Biggl).
\end{multline*}  


Продолжая процесс итерации,  получим
\begin{equation*}
%\label{chad:eq17}
\Bigl|y_{m+1}(x)-y_m(x)\Bigl|\leq N^mb_0l^{(m+1)\alpha}\sum_{k=0}^{b}C_m^k\frac{l^{-k\beta}}{\Gamma((m+1)\alpha+1-k\beta)}=T(m, l).
\end{equation*}

Таким образом, показано,  что для достаточно малых значений $l$ 
%\begin{equation*}
%\label{chad:eq18}
$$\lim_{x \to \infty}T(m, l)=0, $$
%\end{equation}
а это значит,  что последовательность функций $y_m(x)$ сходится к~функции $y(x)$. 
Эта функция при $0<x\leq l$ принадлежит классу $C_{n-\alpha}^\alpha[0, l]$ и~удовлетворяет неравенству
\begin{equation*}
%\label{chad:eq19}
\Bigl|y(x)-y_0x^{\alpha-n}\Bigl|\leq ax^{\alpha-n}, 
\end{equation*}
которое следует из \eqref{chad:eq13} при $m\rightarrow \infty$.
%

%\begin{equation}
%\label{chad:eq20}
%Nl^\alpha\Biggl(\frac{1}{\Gamma(\alpha+1)}+\frac{l^{-\beta}}{\Gamma(\alpha+1-\beta)}\Biggl)<1.
%\end{equation}
%Подставляя $y(x)$ и $r(x)$ в \eqref{chad:eq10} и почленно вычитая,  имеем
%\begin{multline}
%\label{chad:eq21}
%\Bigl|y(x)-r(x)\Bigl|=\Biggl|\int_0^x\frac{(x-t)^{\alpha-1}}{\Gamma(\alpha)}\Bigl[f(t, y(t), D_{0t}^\beta y(s))-f(t, y(t), D_{0t}^\beta r(s))\Bigl]dt\Biggl|\leq
%\\
%\leq \frac {N}{\Gamma(\alpha)}\int_0^x(x-t)^{\alpha-1}\Bigl[\bigl|y(x)-r(x)\bigl|+\bigl|D_{0t}^\beta \bigl[ y(x)-r(x)\bigl]\bigl|\Bigl]dt.
%\end{multline} 
%Допустим,  что $\max\limits_{0<x\leq l}|y(x)-r(x)|=B$,  тогда из \eqref{chad:eq21} имеем
%$$
%B\leq NBl^\alpha\Biggl(\frac{1}{\Gamma(\alpha+1)}+\frac{l^{-\beta}}{\Gamma(\alpha+1-\beta)}\Biggl).
%$$
%Это противоречит \eqref{chad:eq20},  что доказывает единственность решения задачи \eqref{chad:eq3} для уравнения \eqref{chad:eq1} [4].
\end{proof}
