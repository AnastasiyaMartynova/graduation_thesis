\Section[n]{Введение}
За последние несколько десятилетий в математической литературе появился 
ряд работ, посвященных исследованию нелокальных
задач для дифференциальных уравнений. Одной из первых среди них была статья А.\,В.~Бицадзе и~А.\,А.~Самарского~\cite{bits}, 
в~которой были предложены новые постановки задач для эллиптических уравнений и которая стала отправной точкой большинства исследований в~этом направлении.

Среди нелокальных задач большой интерес представляют задачи с интегральными условиями, которые являются обобщением нелокальных условий, заданных в~виде линейной комбинации.
Разрешимость нелокальных задач с нелокальными, в~том числе интегральными, условиями
и~качественные свойства решений для эллиптических уравнений рассмотрены в~работах А.\,Л.~Скубачевского~\cite{skub}, А.\,К.~Гущина 
и~В.\,П.~Михайлова~\cite{mih, gusсh} и~ряда других авторов.

В предлагаемой работе рассматривается нелокальная задача с интегральными условиями для уравнения Лапласа второго порядка в прямоугольной области. 

\smallskip

\Section{Постановка задачи} 
В прямоугольнике $D=\{(x,y): 0<x<a,\ 0<y<b\}$
рассмотрим уравнение 
\begin{equation}\label{nvb:eq1}
 	u_{xx}(x,y)+u_{yy}(x,y)=f(x,y)
	\end{equation}
и поставим для него задачу со следующими условиями:
\begin{equation}\label{nvb:eq2}
	 u(x,b)=r(x), \quad u(a,y)=s(y),
	\end{equation}
	\begin{equation}\label{nvb:eq3}
 	 u(0,y)=\alpha\int_0^aN(x,y)u(x,y)dx, \quad u(x,0)=\beta\int_0^bM(x,y)u(x,y)dy.
	\end{equation}
Здесь $r(x),$ $s(y),$ $M(x,y),$ $N(x,y)$ и $f(x,y)$ "--- заданные в $\overline{D}$ функции, $\alpha$, $\beta$ "--- числа.

\smallskip

\Section{Разрешимость поставленной задачи} Сформулируем и докажем следующую теорему.

\begin{theorem}
Если 
$f(x,y)\in C(\overline{D}),$ 
$M(x,y)\in C(\overline{D}),$ 
$N(x,y)\in C(\overline{D}),$ 
$N_x(x,y)\in C(\overline{D}),$ 
$M_x(x,y)\in C(\overline{D}),$ 
$$\textstyle ab(\alpha^2N^2_1+\beta^2M^2_1)+\frac{2}{3}(\alpha^2N^2_1a^2+\beta^2M^2_1b^2)<1,$$ где $\max\limits_D |N(x,y)|\leq N_1,$ $\max\limits_D |M(x,y)|\leq M_1,$ и выполняются условия согласования $M(0,y)=N(x,0),$ $r(a)=s(b),$ то существует решение $u(x,y)\in C^2(D)\cap C(\overline{D})$ задачи \eqref{nvb:eq1}--\eqref{nvb:eq3}.
\end{theorem}
 \begin{proof}
Будем искать решение задачи \eqref{nvb:eq1}--\eqref{nvb:eq3} в виде суммы $u(x,y)=v(x,y)+w(x,y),$ где $v(x,y)$ --- решение задачи (A):
\begin{gather*}
 	v_{xx}(x,y)+v_{yy}(x,y)=0;
\\
	 v(x,b)=r(x), \quad v(a,y)=s(y);
\\
 	 v(0,y)=\alpha\int_0^aN(x,y)v(x,y)dx, \quad v(x,0)=\beta\int_0^bM(x,y)v(x,y)dy,
	\end{gather*}
 а $w(x,y)$--- решение задачи (B): 
\begin{gather*}
 	w_{xx}(x,y)+w_{yy}(x,y)=f(x,y);
\\
	 w(x,b)=0, \quad w(a,y)=0; \qquad w(0,y)=0, \quad w(x,0)=0.
	\end{gather*}


Для нахождения решения задачи (А) и функции $v(x, y)$ рассмотрим вспомогательные задачи, считая временно известными функции $p(x)$ и $q(y)$:

{\small \sc Задача I.}~~\,
$
\Delta v_1(x,y)=0;\, v_1(x,0)=p(x), \, v_1(x,b)=0; \, v_1(0,y)=0,\,  v_1(a,y)=0;
$

{\small \sc Задача II.}~\,
$
\Delta v_2(x,y)=0; \, v_1(x,0)=0, \, v_2(x,b)=r(x); \, v_2(0,y)=0, \, v_2(a,y)=0;
$

{\small \sc Задача III.}
$
\Delta v_3(x,y)=0; \, v_3(x,0)=0, \, v_3(x,b)=0; \, v_3(0,y)=q(y), \, v_3(a,y)=0;
$

{\small \sc Задача IV.}\,
$
\Delta v_4(x,y)=0; \, v_4(x,0)=0, \, v_4(x,b)=0; \, v_4(0,y)=0, \, v_4(a,y)=s(y).
$

Тогда $v(x,y)=v_1(x,y)+v_2(x,y)+v_3(x,y)+v_4(x,y).$ Решая задачи (I)--(IV) методом Фурье, находим, что
\begin{multline}\label{nvb:eq4}
v(x,y)= 
 \sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n b}{a}}\left[p_n\sh\frac{\pi n(b-y)}{a}+ r_n\sh\frac{\pi ny}{a}\right]\sin\frac{\pi n x}{a}+\\
 +\sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n a}{b}}\left[q_n\sh\frac{\pi n(a-x)}{b}+ s_n\sh\frac{\pi nx}{b}\right]\sin\frac{\pi n y}{b},
\end{multline}
где $ p_n,$ $r_n,$ $q_n,$ $s_n$ "--- коэффициенты разложения соответствующих функций в ряд Фурье по системе синусов.

Для отыскания функций $p(x)$ и $q(y)$ применим к \eqref{nvb:eq4} нелокальные условия \eqref{nvb:eq3}, получим
\begin{multline}\label{nvb:eq5}
q(y)= \alpha\int_0^aN(x,y)\sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n b}{a}}\left[p_n\sh\frac{\pi n(b-y)}{a}+ r_n\sh\frac{\pi ny}{a}\right]\sin\frac{\pi n x}{a}dx+\\
 +\alpha\int_0^aN(x,y)\sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n a}{b}}\left[q_n\sh\frac{\pi n(a-x)}{b}+ s_n\sh\frac{\pi nx}{b}\right]\sin\frac{\pi n y}{b}dx,
\end{multline}
\begin{multline}\label{nvb:eq6}
p(x)= \beta\int_0^bM(x,y)\sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n b}{a}}\left[p_n\sh\frac{\pi n(b-y)}{a}+ r_n\sh\frac{\pi ny}{a}\right]\sin\frac{\pi n x}{a}dy+\\
 +\beta\int_0^bM(x,y)\sum\limits_{n=1}^\infty\frac{1}{\sh\frac{\pi n a}{b}}\left[q_n\sh\frac{\pi n(a-x)}{b}+ s_n\sh\frac{\pi nx}{b}\right]\sin\frac{\pi n y}{b}dy.
\end{multline}

Ряды, стоящие в правой части \eqref{nvb:eq5} и \eqref{nvb:eq6}, сходятся равномерно в прямоугольнике $D$ по признаку Абеля. Меняя порядок суммирования и интегрирования, имеем
\begin{gather}\label{nvb:eq7}
 q(y)-\int_0^aK_1(y,\xi)p(\xi)d\xi -\int_0^b K_2(x,\xi)q(\xi)d\xi=f(y),
\\
\label{nvb:eq8}
 p(x)-\int_0^aK_3(y,\xi)p(\xi)d\xi -\int_0^b K_4(x,\xi)q(\xi)d\xi=g(x),
\end{gather}
где 
$$K_1(y,\xi)=\frac{2\alpha}{a}\sum\limits_{n=1}^\infty\int_0^a\frac{N(x,y)}{\sh\frac{\pi n b}{a}}\sh\frac{\pi n(b-y)}{a}\sin\frac{\pi n x}{a}dx\sin\frac{\pi n \xi}{a},$$
$$K_2(y,\xi)=\frac{2\alpha}{b}\sum\limits_{n=1}^\infty\int_0^a\frac{N(x,y)}{\sh\frac{\pi n a}{b}}\sh\frac{\pi n(a-x)}{b}\sin\frac{\pi n y}{a}dx\sin\frac{\pi n \xi}{b},$$
$$K_3(x,\xi)=\frac{2\beta}{a}\sum\limits_{n=1}^\infty\int_0^b\frac{M(x,y)}{\sh\frac{\pi n b}{a}}\sh\frac{\pi n(b-y)}{a}\sin\frac{\pi n x}{a}dy\sin\frac{\pi n \xi}{a},$$
$$K_4(x,\xi)=\frac{2\beta}{b}\sum\limits_{n=1}^\infty\int_0^b\frac{M(x,y)}{\sh\frac{\pi n a}{b}}\sh\frac{\pi n(a-x)}{b}\sin\frac{\pi n y}{a}dy\sin\frac{\pi n \xi}{b}.$$
 
Возьмём $c=\max\{a,b\}$ и продолжим ядра $K_i$ нулем до квадрата $(0,c)\times(0,c)$. Тогда система \eqref{nvb:eq7}--\eqref{nvb:eq8} примет вид
\begin{gather}\label{nvb:eq9}
 q(\eta)-\int_0^cK_1(\eta,\xi)p(\xi)d\xi -\int_0^c K_2(\eta,\xi)q(\xi)d\xi=f(\eta),
\\
\label{nvb:eq10}
 p(\eta)-\int_0^cK_3(\eta,\xi)p(\xi)d\xi -\int_0^c K_4(\eta,\xi)q(\xi)d\xi=g(\eta).
\end{gather}
Выясним, при каких условиях на $a$, $b$, $\alpha$, $\beta$, $N(x,y)$ и $ M(x,y) $ система интегральных уравнений \eqref{nvb:eq9}, \eqref{nvb:eq10} разрешима.

% Рассмотрим ядро $K_1(y,\xi).$ Так как $\frac{\sin\frac{\pi n x}{a}}{\sh\frac{\pi n b}{a}}\leq 1$ для всех $n$ и $x\in(0,a),$ то справедлива оценка
% \begin{equation}\label{nvb:eq11}
%  K_1(y,\xi)\leq \alpha N_1.
% \end{equation}
Для ядер $K_i$ справедливы следующие оценки:
\begin{gather*}
\textstyle
 \| K_1(\eta,\xi)\|^2_{L_2(D)}\leq ab\alpha^2N^2_1,\quad \| K_2(\eta,\xi)\|^2_{L_2(D)}\leq \frac{2}{3}a^2\alpha^2N^2_1,
\\
\textstyle
 \| K_3(\eta,\xi)\|^2_{L_2(D)}\leq ab\beta^2M^2_1, \quad \| K_4(\eta,\xi)\|^2_{L_2(D)}\leq \frac{2}{3}b^2\beta^2M^2_1.
\end{gather*}
 
Следовательно, если выполняется условие 
\begin{equation*}
\textstyle
 ab\alpha^2N^2_1+\frac{2}{3}a^2\alpha^2N^2_1+ab\beta^2M^2_1+\frac{2}{3}b^2\beta^2M^2_1<1,
\end{equation*}
то система интегральных уравнений \eqref{nvb:eq9}, \eqref{nvb:eq10} однозначно разрешима. 

Таким образом, классическое решение задачи (A) имеет вид \eqref{nvb:eq4}, где $p(x)$ и $q(y)$ "--- решение системы интегральных уравнений \eqref{nvb:eq9}, \eqref{nvb:eq10}.

Решение задачи (B) находится стандартным способом:

$$w(x,y)=\sum\limits_{n=1}^\infty Y_n(y)\sin\frac{\pi n}{a}x,$$ где $Y_n(y)$ "--- решение задачи
\begin{equation*}
 Y_n''(y)-\frac{\pi n}{a}Y_n(y)=\frac{2}{a}\int_0^a f(x,y)\sin\frac{\pi n}{a}x dx;\quad 
 Y_n(0)=Y_n(b)=0.
\end{equation*}

Итак, построено классическое решение задачи \eqref{nvb:eq1}--\eqref{nvb:eq3}.
\end{proof}
