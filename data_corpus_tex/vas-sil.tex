Находятся комплексные потенциалы, коэффициенты интенсивности напряжений в~вершине трещины, приводятся их графики.}

Пусть в кусочно"=однородном упругом изотропном теле, составленном из разных по упругим свойствам верхней и~нижней полуплоскостей, на линии раздела сред $y=0$  расположена полубесконечная открытая трещина \mbox{$[0, +\infty)$,} 
верхний берег которой на участке $[0, l]$ подкреплён абсолютно жесткой прямолинейной накладкой, присоединённой к телу без натяга, а остальная часть этого берега и весь нижний берег трещины свободны от напряжений:
\begin{gather*}
u^{+}(x)+i v^{+}(x)=i\varepsilon x,\quad  x\in(0, l); \qquad
\tau_{xy}^{+}(x)+i\sigma_y^{+}(x)=0,\quad x\in(l, +\infty);\\
\tau_{xy}^{-}(x)+i\sigma_y^{-}(x)=0,\quad x\in(0, +\infty),
\end{gather*}
где  $u+iv$ "--- вектор смещений,  $\tau_{xy}+i\sigma_{y}$ "--- вектор напряжений, $\varepsilon$ "--- неизвестный угол поворота накладки, а верхними индексами плюс и минус помечены значения функций на верхнем и нижнем берегах трещины. Вдоль луча \mbox{$(-\infty, 0]$}  полуплоскости жёстко соединены друг с другом, и в точке  $x=x_0$ ($x_0<0$) действует сосредоточенная сила  $X_0+iY_0$. На бесконечности напряжения и~вращение исчезают. На накладку никакие внешние силы не действуют.

\smallskip
{\it Требуется найти комплексные потенциалы, описывающие плоское напряженное состояние тела, и~исследовать поведение напряжений вблизи вершины трещины.}

\smallskip

Воспользуемся формулами Колосова"--~Мусхелишвили для кусочно"=однородной плоскости \cite{vas-sil:bib:Cherepanov}:
\begin{equation}
\begin{array}{l}
 \sigma_{x}+\sigma_{y}=4\mbox{Re\,}\Phi_*(z),\quad  z=x+iy,\\
 \sigma_y-i\tau_{xy}=\Phi_*(z)+\Omega_*(\overline{\strut z})+(z-\overline{\strut z})\overline{\strut\Phi'_*(z)},\\
 2\mu_j(u+iv)'_x=\kappa_j\Phi_*(z)-\Omega_*(\overline{\strut z})-(z-\overline{\strut z})\overline{\strut \Phi'_*(z)},
\end{array} \label{eq:vas-sil:Kol-Muskh}
\end{equation}
$$
\Phi_*(z)=
\left\{
   \begin{array}{cl}
        \Phi(z), &\text{$\mbox{Im}z>0$,}  \\
        \alpha_1\Phi(z)+ \alpha_2\Omega(z), & \text{$\mbox{Im}z<0$,}
   \end{array}
\right.     
      \Omega_*(z)=
\left\{
   \begin{array}{cl}
        \Omega(z), & \text{$\mbox{Im}z>0$,}  \\
        \alpha_3\Omega(z)+ \alpha_4\Phi(z), & \text{$\mbox{Im}z<0$,}
   \end{array}
\right. 
$$   
$$
\alpha_1=(1+\mu_*\kappa_1)/(1+\kappa_2),\quad  \alpha_2=(1-\mu_*)/(1+\kappa_2),\quad \alpha_3=1-\alpha_2,\quad \alpha_4=1-\alpha_1,$$
$$\mu_*=\mu_2/\mu_1,\quad  \kappa_j=(3-\nu_j)/(1+\nu_j),\quad  j=1, 2,$$
где $\Phi(z)$, $\Omega(z)$ "--- кусочно"=голоморфные функции (комплексные потенциалы) с линией разрыва $[0,+\infty)$, 
а~$\mu_1$, $\nu_1$ и $\mu_2$, $\nu_2$ "--- модули сдвига и~коэффициенты Пуассона верхней $(j=1)$ и нижней $(j=2)$ полуплоскостей соответственно. 
В~точках $z=0$ и $z=l\pm i0$ функции $\Phi(z)$, $\Omega(z)$  могут иметь интегрируемые особенности, на $\infty$ они исчезают, а~в~точке $z=x_0$ имеют простые полюсы с~вычетами $P_1=-(X_0+iY_0)/[2\pi (1+\mu_*\kappa_1)]$  и  \mbox{$P_2=\kappa_2(X_0+iY_0)/[2\pi (\mu_*+\kappa_2)]$} соответственно \cite{vas-sil:bib:Silvestrov}. 
На луче $(0,+\infty)$  функции  $\Phi(z)$, $\Omega(z)$ удовлетворяют краевым условиям
\begin{equation}
\begin{array}{l}
   \kappa_1\Phi^{+}(x)-\Omega^{-}(x) =2i\mu_1\varepsilon,\quad  x\in(0, l); \\ \Phi^{+}(x)+\Omega^{-}(x) =0,\quad  x\in(l,+\infty);\\
 \alpha_1\Phi^{-}(x)+\alpha_2\Omega^{-}(x)+ \alpha_3\Omega^{+}(x)+\alpha_4\Phi^{+}(x)=0,\quad  x\in(0,+\infty),
\end{array} \label{eq:vas-sil:BoundaryCond}
\end{equation}
откуда находим \cite{vas-sil:bib:Hvochinskaya}:
\begin{gather*}
\Phi(z)=H_1(z/l), \quad   \Omega(z)=H_2(z/l);
\\
H_j(\zeta)=\bigl[A_0{+}C_2(\zeta{-}t_0)^{-1}{+}\varepsilon \mu_1 J_2(\zeta)\bigr]\chi_{j1}(\zeta)+  
\bigl[C_1(\zeta{-}t_0)^{-1}{+}\varepsilon\mu_1 J_1(\zeta)\bigr]\chi_{j2}(\zeta),\\
A_0=-\frac{X_0+iY_0}{2\pi lc_3(\alpha_1+\alpha_3)},\quad  C_j=(-1)^j\frac{P_1\chi_{2j}(t_0)-P_2\chi_{1j}(t_0)}{l\det X^{+}(t_0)},\quad  t_0=x_0/l,
\\
J_j(\zeta)=\frac{(-1)^j}{\kappa_1\pi}\int_0^1 \frac{\chi_{2j}^{+}(t)+\alpha_4\alpha_3^{-1}\chi_{1j}^{+}(t)}{(t-\zeta)\det X^{+}(t)} dt,\quad  j=1, 2,
\end{gather*}
\begin{multline*}
\varepsilon\mu_1=-\biggl(
\int _0^1 t\,\mbox{Re\,}\left[\left(A_0+C_2(t-t_0)^{-1}\right)\chi_{11}^{+}(t)+C_1(t-t_0)^{-1}\chi_{12}^{+}(t)\right]\,dt\biggr)\times \\
\times\biggl(\int _0^1 t\,\mbox{Re\,}\left[\chi_{11}^{+}(t)J_2(t)+\chi_{12}^{+}(t)J_1(t)\right]\,dt\biggr)^{-1},
\end{multline*}
\begin{gather*}
c_3=\frac{c_1\xi_2e^{i\pi a}\Gamma(c)\Gamma(b-a)}{\Gamma(c-a)\Gamma(b)}+\frac{c_2\xi_1e^{i\pi (a+1-c)}\Gamma(2-c)\Gamma(b-a)}{\Gamma(b+1-c)\Gamma(1-a)},\\
c_1=\frac{(1-\xi_1)\Gamma(2-c)e^{-i\pi c}}{\Gamma(a+1-c)\Gamma(1-b)},\quad  c_2=\frac{(1-\xi_2)\Gamma(c)}{\Gamma(a)\Gamma(c-b)}, \\
\xi_{1, 2}=\frac{1}{2}\Bigl(-\alpha\pm\sqrt{\alpha^2-4m\kappa_1^{-1}}\Bigr),\quad  \alpha=\frac{\alpha_2\kappa_1+\alpha_4}{\alpha_3\kappa_1},\quad   m=\frac{\alpha_1}{\alpha_3},\\
a=1-\frac{\ln\xi_1}{2\pi i},\quad  b=\frac{3}{2}+\frac{1}{2\pi i}(\ln m-\ln\xi_1),\quad   c=1+\frac{1}{2\pi i}(\ln\xi_2-\ln\xi_1)\\
(0<\arg\xi_1\le\pi,\ \ \pi\le\arg\xi_2<2\pi,\ \ \mbox{Im}\xi_1\ge0,\ \ \mbox{Im}\xi_2\le0),
\end{gather*}

\noindent
где  $\Gamma(\zeta)$ "--- гамма-функция Эйлера и  $\chi_{ij}(\zeta)$ "--- элементы канонической матрицы ${\mathbf X}(\zeta)$  однородной векторной краевой задачи Римана \eqref{eq:vas-sil:BoundaryCond} при  $l=1$. Для изучения напряжений вблизи вершины трещины $z=0$ достаточно знать выражение этой матрицы при $\mbox{Re\,}z<1$:
\begin{gather*}
{\mathbf X}(\zeta)=\zeta^{\lambda}
\begin{pmatrix} c_1\xi_2 & c_2\xi_1 \\ c_1m & c_2m \end{pmatrix}
\begin{pmatrix} \eta_{11}(\zeta) & \eta_{12}(\zeta) \\ \eta_{21}(\zeta) & \eta_{22}(\zeta) \end{pmatrix},\quad 
\lambda=-\frac{\ln\xi_1}{2\pi i}, 
\\
\begin{array}{ll}
\eta_{11}(\zeta)=F(a,b;c;\zeta),\quad  \eta_{12}(\zeta)=a(1-\zeta)F(a+1,b;c;\zeta), & \\
\eta_{21}(\zeta)=\zeta^{1-c}F(a-c+1,b-c+1;2-c;\zeta), & \\
\eta_{22}(\zeta)=(a-c+1)\zeta^{1-c}(1-\zeta)F(a-c+2,b-c+1;2-c;\zeta) \\ 
\hspace{8cm}\text{при $|\zeta|<1,$ $\zeta\not\in[0, 1];$} \vspace{1mm} \\
\eta_{11}(\zeta)=a_*(\zeta-1)^{-a}F(a,c-b;c;\zeta_*), &\\
\eta_{12}(\zeta)=aa_*(\zeta-1)^{-a}F(a+1,c-b;c;\zeta_*), & \\
\eta_{21}(\zeta)=-c_*\zeta^{1-c}(\zeta-1)^{c-a-1}F(a+1-c,1-b;2-c;\zeta_*), &  \\
\eta_{22}(\zeta)=(c-a-1)c_*\zeta^{1-c}(\zeta-1)^{c-a-1}F(a+2-c,1-b;2-c;\zeta_*), & \\
a_*=e^{i\pi a},\quad  \zeta_*=\zeta/(\zeta-1), \quad   c_*=e^{i\pi(a-c)} \\
\hspace{8cm}\text{при $\mbox{Re\,}\zeta<1/2$, $\zeta\not\in[0, 1/2].$}
\end{array}
\end{gather*}
Здесь $F(a,b;c;\zeta)$ "--- гипергеометрическая функция Гаусса, а у многозначных функций $\zeta^p$  и $(\zeta-1)^q$  берутся ветви, однозначные в плоскости с разрезами $[0,+\infty)$  и $[1,+\infty)$  соответственно, определяемые условиями $0<\arg\zeta<2\pi$  и $0<\arg(\zeta-1)<2\pi$.


Из приведённых формул и формул \eqref{eq:vas-sil:Kol-Muskh} следует, что вблизи точки  $z=0$ на действительной отрицательной полуоси $x<0$ вектор напряжений $\sigma_{y}-i\tau_{xy}$  имеет асимптотику
\begin{gather*}
\sigma_{y}(x)-i\tau_{xy}(x)=
\frac{K_{\mathtt{I}}-iK_{\mathtt{I\negmedspace I}}}{\sqrt{2\pi}|x|^{\gamma_0-i\delta_0}}+
\frac{K_{\mathtt{I\negmedspace I\negmedspace I}}-iK_{\mathtt{I\!V}}}{\sqrt{2\pi}|x|^{1-\gamma_0-i\delta_0}}+\textsf{O}(1),\quad  x\to 0{-}0,\\
K_{\mathtt{I}}-iK_{\mathtt{I\negmedspace I}}=\sqrt{2\pi}(\xi_1+m)D_2e^{-i\pi(\gamma_0-i\delta_0)},\\ 
K_{\mathtt{I\negmedspace I\negmedspace I}}-iK_{\mathtt{I\!V}}=\sqrt{2\pi}(\xi_2+m)D_1e^{-i\pi(1-\gamma_0-i\delta_0)},\quad  \gamma_0-i\delta_0={\ln\xi_2}/({2\pi i}), \\ 
D_j=c_j[A_0-C_2t_0^{-1}+\varepsilon\mu_1J_2(0)+(\lambda_j-1)(C_1t_0^{-1}-\varepsilon\mu_1J_1(0)]l^{\lambda_j},\\ \lambda_j=(2\pi i)^{-1}\ln\xi_j,\quad  j=1, 2
\end{gather*}
в случае $\gamma_0\not=1/2$ и асимптотику
\begin{gather*}
\sigma_{y}(x)-i\tau_{xy}(x)=\frac{(K_{\mathtt{I}}-iK_{\mathtt{I\negmedspace I}})|x|^{i\delta_2}+(K_{\mathtt{I\negmedspace I\negmedspace I}}-iK_{\mathtt{I\!V}})|x|^{i\delta_1}}{\sqrt{2\pi |x|}}+\textsf{O}(1),\quad  x\to 0{-}0,\\
K_{\mathtt{I}}-iK_{\mathtt{I\negmedspace I}}=-\sqrt{2\pi}(\xi_1+m)D_2e^{-\pi\delta_2},
\\
 K_{\mathtt{I\negmedspace I\negmedspace I}}-iK_{\mathtt{I\!V}}=-\sqrt{2\pi}(\xi_2+m)D_1e^{-\pi\delta_1},\quad  \delta_{1,2}={\ln|\xi_{1,2}|}/({2\pi}) 
 \end{gather*}
в случае $\gamma_0=1/2$. 
Таким образом, интенсивность напряжений вблизи точки  $z=0$ в любом случае определяется четырьмя коэффициентами  $K_{\mathtt{I}}$, $K_{\mathtt{I\negmedspace I}}$, $K_{\mathtt{I\negmedspace I\negmedspace I}},$ $ K_{\mathtt{I\!V}}$, и~напряжения в~вершине трещины всегда имеют как степенную, так и осциллирующую особенность. 
Для значений упругих параметров $\kappa_1=1,8,$ $\kappa_2=2,2$ и $\mu_*=2$, когда $\gamma_0=0,79373$, графики зависимости коэффициентов $K_{\mathtt{I}}$, $K_{\mathtt{I\negmedspace I}}$, $K_{\mathtt{I\negmedspace I\negmedspace I}},$ $ K_{\mathtt{I\!V}}$ и~угла поворота накладки $\varepsilon$ от расстояния $d=|x_0|$ между точкой приложения сосредоточенной силы и~вершиной трещины и от направления $\phi$ силы приведены на рис.~\hyperlink{vas:pict:1}{1}--\hyperlink{vas:pict:4}{4}. 
В~первом случае сила \mbox{$X_0+iY_0=-P$} $(P>0)$ постоянна и~направлена вдоль действительной оси, а~во втором случае сила $X_0+iY_0=P(\cos\phi+i\sin\phi)$ составляет угол $\phi$ с~осью  $x$, $x_0=0,2$. 
В~обоих случаях принято $l=1$.
