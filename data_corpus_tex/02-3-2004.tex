\section{Введение}  
 
Допустим, $r\in N$ и $W^r$ --- класс непрерывных $2\pi$-периодических 
функций, для которых существует абсолютно непрерывная $(r-1)$-я 
производная $f^{(r-1)}(x)$ и 
$\|f^{(r)}(x)\|=\sup\limits_x|f^{(r)}(x)|\le1$. Обозначим 
$R_m(W^r)=\sup\limits_{f\in W^r}\|f(x)-S_{m-1}(x,f)\|$, 
где $S_{m-1}(x,f)$ --- частичные суммы $(m-1)$-го порядка ряда Фурье 
функции $f(x)$. 
 
А.Н.\,Колмогоров \cite{1} доказал, что 
$$ 
R_m(W^r)=m^{-r}(4\pi^{-2}\ln m+O(1)). 
$$ 
Исследования А.Н.\,Колмогорова для более общих классов были продолжены, 
например, в \cite{2}--\cite{6}. 
 
В данной работе решается дискретный аналог задачи А.Н.\,Колмогорова. 
 
Если $q\in N$, $x_n=\frac{2\pi n}{q}$ ($n\in\Bbb Z$), $\bold 
e_k=\{e_k(x_n)=e^{ikx_n}\}^{q-1}_{n=0}$, $k=-[\frac{q-1}{2}],\dotsc, 
[\frac{q}{2}]$, то произвольную дискретную функцию 
$f:\{x_0,\dotsc,x_{q-1}\}\to\Bbb R$ можно 
разложить по ортогональной системе $\{\bold e_k\}$ 
($k=-[\frac{q-1}{2}],\dotsc,[\frac{q}{2}]$) в следующий дискретный ряд 
Фурье: 
$$ 
f(x_n)=\sum^{[q/2]}_{k=-[\frac{q-1}{2}]}c_k e_k(x_n),\quad 
c_k=\frac{1}{q}\sum^{q-1}_{n=0}f(x_n)e^{-ikx_n}. 
$$ 
Допустим, $\bold f=\{f(x_n)\}^{q-1}_{n=0}$, тогда $S_{m-1}(\bold f; 
x_n)=\sum\limits^{m-1}_{k=-(m-1)}c_k e_k(x_n)$ --- частичная сумма 
дискретного ряда Фурье. 
 
Обозначим через $W^r_q$ класс $2\pi$-периодических дискретных функций 
$f$, заданных на равномерной сетке $\{x_n\}_{n\in\Bbb Z}$ и таких, что 
$$ 
\|\Delta^r_{2\pi/q}f\|=\max_n|\Delta^r_{2\pi/q}f(x_n)|\le 
\big(2\pi/q\big)^r. 
$$ 
 
Асимптотические свойства $C_m(W^r_q)=\sup\limits_{f\in 
W^r_q}\|f-S_{m-1}(\bold f;x_n)\|$ при $m,q\to\infty$ зависят от предела 
отношения порядка 
сумм Фурье к числу точек равномерной сетки и, в частности, от того, 
является указанный предел числом рациональным или иррациональным. 
Оказывается, что в отличие от непрерывного случая здесь естественным 
образом появляются функции римановского типа. 
 
 
\section{Дискретные константы Лебега} 
 
Для решения поставленной задачи необходимо знать поведение констант 
\begin{gather*}  %%%% return align in eng.
\displaybreak[3]
L_m(q)=\frac{1}{q}\sum^{q-1}_{n=0} 
\frac{|\sin(m x_n/2)|}{\sin(x_n/2)},\\ 
% 
\ov L_m(q)=\frac{1}{q}\sum^{q-[q/m]}_{n=[q/m]} 
\frac{|\cos(\pi m(n+1/2)/q)|}{\sin(\pi(n+1/2)/q)},\quad 
m,q\in N,\ \ m<q. 
\end{gather*} 
Константы $L_m(q)$ изучены в \cite{7}, где доказана 
 
\begin{thm} 
Пусть $m_n,q_n\in N$, $m_n\to\infty$, $\frac{m_n}{q_n}\to\alpha$ 
$(n\to\infty)$. Тогда 
 
\noindent{}{\em a)} если $\alpha$ --- иррациональное число или $\alpha=0$, то 
$$ 
\lim_{n\to\infty}\frac{L_{m_n}(q_n)}{\ln(m_n+1)}=\frac{4}{\pi^2}; 
$$ 
если $\alpha=s/p$, $(s,p)=1$, то 
$$ 
\underset{n\to\infty}{\varliminf}{} 
\frac{L_{m_n}(q_n)}{\ln(m_n+1)}\ge\frac{2}{\pi p}\ctg 
\frac{\pi}{2p},\quad \underset{n\to\infty}{\varlimsup}{} 
\frac{L_{m_n}(q_n)}{\ln(m_n+1)}\le\frac{4}{\pi^2},
$$ 
 
\noindent{}{\em б)} для любого $\gamma\in[\frac{2}{\pi p}\ctg \frac{\pi}{2p};\ 
\frac{4}{\pi^2}]$ существуют $m'_n,q'_n\in N$ такие , что 
$\frac{m'_n}{q'_n}\to\frac{s}{p}$ и $\lim\limits_{n\to\infty} 
\frac{L_{m'_n}(q'_n)}{\ln(m'_n+1)}=\gamma$. 
\end{thm} 
 
Для константы $\ov L_m(q)$ докажем следующую теорему. 
 
\begin{thm} 
Пусть $m_n,q_n\in N$, $m_n<q_n$, $\tau_n=\min(m_n,q_n-m_n)+1\to\infty$, 
$\frac{m_n}{q_n}\to\alpha$ при $n\to\infty$. Тогда 
 
{\em a)} для иррациональных $\alpha$ и $\alpha=0;1$ имеем 
$\lim\limits_{n\to\infty} \frac{\ov L_{m_n}(q_n)}{\ln\tau_n}= 
\frac{4}{\pi^2}$, 
 
{\em б)} если $\alpha=\frac{s}{p}$, $(s,p)=1$, то 
\begin{align*} 
\underset{n\to\infty}{\varliminf}{} 
\frac{\ov L_{m_n}(q_n)}{\ln\tau_n}=h&= 
\begin{cases} 
\frac{2}{\pi p}\ctg\frac{\pi}{2p} & \text{при}\ s+p\equiv0 
\ (\Mod2);\\ 
\frac{4}{\pi^2} & \text{при}\ s+p\equiv1\ (\Mod2), 
\end{cases} 
\\ 
% 
\underset{n\to\infty}{\varlimsup}{} 
\frac{\ov L_{m_n}(q_n)}{\ln\tau_n}=H&= 
\begin{cases} 
\frac{4}{\pi^2} & \text{при}\ s+p\equiv0\ (\Mod2);\\ 
\frac{2}{\pi p\sin\frac{\pi}{2p}} & \text{при}\ s+p\equiv1\ (\Mod2). 
\end{cases} 
\end{align*} 
Кроме того, для любого $\gamma\in[h;H]$ существуют $m'_n,q'_n\in N$ 
такие, что $m'_n/q'_n\to s/p$ $(n\to\infty)$ и 
$\lim\limits_{n\to\infty} \frac{\ov 
L_{m'_n}(q'_n)}{\ln\tau'_n}=\gamma$. 
\end{thm} 
 
{\bf Доказательство.} Обозначив $\frac{|\cos\pi 
m(n+1/2)/q|}{\sin\pi(n+1/2)/q}=t_n$, запишем $\ov L_m(q)$ в виде 
\begin{equation} 
\ov L_m(q)=\frac{1}{q}\sum^{q-1}_{n=1}t_n-\frac{1}{q} 
\sum^{[q/m]-1}_{n=1}t_n-\frac{1}{q}\sum^{q-1}_{n=q-[q/m]+1} 
t_n\equiv G_1-G_2-G_3. 
\end{equation} 
Докажем, что 
\begin{equation} 
G_2+G_3=\frac{2}{\pi}\ln\frac{q}{m}+O(1). 
\end{equation} 
Имеем $G_2=\frac{1}{q}\sum\limits^{[q/(2m)]}_{n=1}t_n+O(1)$. Применяя 
равенство $\cos x\ge1-\frac{2}{\pi}x$, $x\in[0,\pi/2]$, получим 
\begin{equation} 
G_2\ge\frac{1}{\pi}\ln\frac{q}{m}+O(1). 
\end{equation} 
Так как 
\begin{equation} 
\frac{1}{\sin x}=\frac{1}{x}+O(1),\quad x\in(0;\pi/2], 
\end{equation} 
то $G_2\le\frac{1}{\pi}\ln\frac{q}{m}+O(1)$. Отсюда и из (3) следует 
\begin{equation} 
G_2=\frac{1}{\pi}\ln\frac{q}{m}+O(1). 
\end{equation} 
Так как $G_3=\frac{1}{q}\sum\limits^{[q/m]-1}_{n=1}t_{n-1}$, то 
аналогично получим $G_3=\frac{1}{\pi}\ln\frac{q}{m}+O(1)$, откуда с 
учетом (5) следует (2). Используя разложение 
$$ 
|\cos x|=1+\frac{8}{\pi}\sum^\infty_{v=1} 
\frac{(-1)^v\sin^2vx}{4v^2-1} 
$$ 
и формулу 
$$ 
\frac{\sin^2nx}{\sin x}=\sum^n_{k=1}\sin(2k-1)x, 
$$ 
получим для $G_1$ следующее выражение: 
\begin{equation} 
G_1=\frac{2\ln q}{\pi}+\frac{8}{\pi q}\sum^\infty_{v=1} 
\frac{(-1)^v}{4v^2-1}\sum^{vm}_{k=1}\sin^{-1}(2k-1) 
\frac{\pi}{q}+O(1). 
\end{equation} 
Из (1), (2), (6) следует 
$$ 
\ov L_m(q)=\frac{2\ln m}{\pi}+\frac{8}{\pi q}\sum^\infty_{v=1} 
\frac{(-1)^v}{4v^2-1}\sum^{vm}_{k=1}\sin^{-1}(2k-1) 
\frac{\pi}{q}+O(1). 
$$ 
Используя формулы $\sin^{-1}x=\ctg\frac{x}{2}-\ctg x$, 
$\sum\limits^{pq}_{k=1}\ctg(2k-1)\frac{\pi}{2q}=0$, $p\in N$, представим 
\begin{equation} 
\ov L_m(q)=\frac{2\ln m}{\pi}-\frac{8}{\pi^2}\sum^\infty_{v=1} 
\frac{(-1)^v\ln(q\|\frac{vm}{q}\|+1)}{4v^2-1}+ 
\frac{16}{\pi^2}\sum^\infty_{v=1} 
\frac{(-1)^v\ln(2q\|\frac{vm}{2q}\|+1)}{4v^2-1}+O(1), 
\end{equation} 
где $\|\cdot\|$ --- расстояние до ближайшего целого. 
 
Докажем пункт a). Пусть $\frac{m_n}{q_n}\to\alpha\ne0;1$ ($n\to\infty$). 
Тогда $\|\frac{vm_n}{q_n}\|\to\|v\alpha\|$ и $v\alpha$ не целое для 
иррациональных $\alpha$. Разделив (7) на $\ln m_n$, используя равномерную 
сходимость рядов и переходя к пределу при $n\to\infty$, получим 
$$ 
\lim_{n\to\infty}\frac{\ov L_{m_n}(q_n)}{\ln m_n}=\frac{2}{\pi}+ 
\frac{8}{\pi^2}\sum^\infty_{v=1}\frac{(-1)^v}{4v^2-1}=\frac{4}{\pi^2}. 
$$ 
Далее получим неравенство 
\begin{equation} 
\frac{2\tau}{\pi q}\ctg\frac{\pi\tau}{2q}\ln\tau+O(1)\le\ov L_m(q) 
\le\frac{2\tau\ln\tau}{\pi q\sin\pi\tau/q}+O(1), 
\end{equation} 
где $\tau=\min(m,q-m)+1$. 
 
Пусть $q\ge2m$. В силу (4) имеем 
$$ 
\ov L_m(q)=\frac{1}{\pi}\sum^{[q/2]}_{n=[q/m]}\tau_n+ 
\frac{1}{\pi}\sum^{[q/2]}_{n=[q/m]}\tau_{n-1}+O(1) 
$$ 
при $\tau_n=\frac{|\cos\pi m(n+1/2)/q|}{n+1/2}$. Замечая, что 
$\sign\cos x=(-1)^{k+1}$, $\frac{\pi}{2}+k\pi\le 
x\le\frac{\pi}{2}+(k+1)\pi$, получим 
$$ 
\ov L_m(q)=\frac{1}{\pi}\sum^{[\frac{m-1}{2}]}_{k=0}(-1)^{k+1} 
\sum^{[a^-_{k+1}]}_{n=[a^-_k]+1}\tau_n+\frac{1}{\pi} 
\sum^{[\frac{m-1}{2}]}_{k=0}(-1)^{k+1} 
\sum^{[a^+_{k+1}]}_{n=[a^+_k]+1}\tau_{n-1}+O(1), 
$$ 
где $a^\pm_k=\frac{q}{m}(k+\frac{1}{2})\pm\frac{1}{2}$. Имеем 
\begin{align*} 
\ov L_m(q)&\ge\frac{m}{\pi q}\sum^{[\frac{m-1}{2}]}_{k=0} 
\frac{(-1)^{k+1}}{k+3/2}(A^-_k+A^+_k)+O(1),\\ 
% 
\ov L_m(q)&\le\frac{m}{\pi q}\sum^{[\frac{m-2}{2}]}_{k=0} 
\frac{(-1)^{k+1}}{k+1/2}(A^-_k+A^+_k)+O(1),
\end{align*} 
где $A^\pm_k=\sum\limits^{[a^\pm_{k+1}]}_{n=[a^\pm_k]+1} \cos \pi 
m(n\mp\frac{1}{2})/q$. 
 
Пусть $\{a\}$ --- дробная часть $a$. Замечая, что 
$$ 
A^\pm_k=\frac{(-1)^{k+1}(\cos(\frac{1}{2}-\{a^\pm_{k+1}\}) 
\frac{\pi m}{q}+\cos(\frac{1}{2}-\{a^\pm_k\})\frac{\pi m}{q})} 
{2\sin\frac{\pi m}{2q}} 
$$ 
и 
$$ 
2\cos\frac{\pi m}{q}\le\cos\bigg(\frac{1}{2}-\{a^\pm_{k+1}\}\bigg) 
\frac{\pi m}{q}+\cos\bigg(\frac{1}{2}-\{a^\pm_k\}\bigg) 
\frac{\pi m}{q}\le2, 
$$ 
получим 
\begin{align*} 
\ov L_m(q)&\ge\frac{2m}{\pi q}\ctg\frac{\pi m}{2q} 
\sum^{[\frac{m-1}{2}]}_{k=0}\frac{1}{k+3/2}+O(1),\\ 
% 
\ov L_m(q)&\le\frac{2m}{\pi q\sin\frac{\pi m}{2q}} 
\sum^{[\frac{m-1}{2}]}_{k=0}\frac{1}{k+1/2}+O(1), 
\end{align*} 
откуда следует (8) при $q\ge2m$. 
 
Пусть теперь $q<2m$. Так как 
$$ 
\ov L_m(q)=\frac{1}{q}\sum^{q-1}_{n=1}t_n+O(1)=\frac{1}{q} 
\sum^{q-1}_{n=1}\frac{|\sin\pi(q-m)(n+\frac{1}{2})/q|} 
{\sin\pi(n+\frac{1}{2})/q}+O(1), 
$$ 
то совершенно аналогично получим 
$$
\frac{2(q-m)}{\pi q}\ctg\frac{\pi(q-m)}{2q}\ln(q-m)+O(1)\le 
\ov L_m(q)\le\frac{2(q-m)\ln(q-m)} 
{\pi q\sin\pi\frac{q-m}{2q}}+O(1). 
$$
Неравенство (8) доказано. Из (8) следует утверждение п.\,a) для 
$\alpha=0$ и $\alpha=1$. 
 
Докажем пункт б). Пусть $\frac{m_n}{q_n}=\frac{s}{p}+\varepsilon_n$, где 
$\varepsilon_n\to0$ ($n\to\infty$) такое, что существует (в силу 
(8)) $\lim\limits_{n\to\infty}\frac{\ov L_{m_n}(q_n)}{\ln m_n}$. 
 
Допустим $\frac{m_{n_k}}{q_{n_k}}$ --- подпоследовательность, для которой 
$\lim\limits_{k\to\infty}\frac{\ln(|\varepsilon_{n_k}|+q^{-1}_{n_k})} 
{\ln m_{n_k}}=\beta$. Не нарушая 
общности, будем считать $n_n=n$. Ясно, что $\beta\in[-1;0]$. Пусть для 
фиксированного целого $k$ \ $|\varepsilon_n|<\frac{1}{2kp}$. Тогда 
\begin{align} 
\lim_{n\to\infty}\frac{\ln(\|\frac{kpm_n}{q_n}\|+\frac{kp}{q_n})} 
{\ln m_n}&=\beta, \\ 
% 
\lim_{n\to\infty}\frac{\ln(2\|\frac{kpm_n}{2q_n}\|+\frac{kp}{q_n})} 
{\ln m_n}&= 
\begin{cases} 
\beta, & \text{если}\ ks\equiv0\ (\Mod2);\\ 
0, & \text{если}\ ks\equiv1\ (\Mod2). 
\end{cases} 
\end{align} 
Так как 
$$ 
o\le\ln\bigg(\bigg\|\frac{kpm_n}{q_n}\bigg\|+ 
\frac{kp}{q_n}\bigg)-\ln\bigg(\bigg\| 
\frac{kpm_n}{q_n}\bigg\|+\frac{1}{q_n}\bigg)\le\ln kp, 
$$ 
то, замечая, что $\frac{vs}{p}$ не целое при $v\equiv0\ (\Mod p)$, и 
применяя (8), из (7) получим 
$$ 
\lim_{n\to\infty}\frac{\ov L_{m_n}(q_n)}{\ln m_n}= 
\begin{cases} 
\frac{4}{\pi^2}+\frac{8\beta}{\pi^2}\sum\limits^\infty_{k=1} 
\frac{(-1)^k}{4k^2p^2-1}, & 
\text{если}\ s+p\equiv1\ (\Mod 2);\\ 
\frac{4}{\pi^2}+\frac{8\beta}{\pi^2}\sum^\infty_{k=1} 
\frac{1}{4k^2p^2-1}, & 
\text{если}\ s+p\equiv0\ (\Mod2), 
\end{cases} 
$$ 
отсюда 
\begin{align*} 
\underset{n\to\infty}{\varliminf}{} 
\frac{\ov L_{m_n}(q_n)}{\ln m_n}&\ge 
\begin{cases} 
\frac{4}{\pi^2}, & \text{если}\ s+p\equiv1\ (\Mod2);\\ 
\frac{4}{\pi^2}-\frac{8}{\pi^2}\sum\limits^\infty_{k=1} 
\frac{1}{4k^2p^2-1}, & 
\text{если}\ s+p\equiv0\ (\Mod2), 
\end{cases} 
\\ 
\underset{n\to\infty}{\varlimsup}{} 
\frac{\ov L_{m_n}(q_n)}{\ln m_n}&\le 
\begin{cases} 
\frac{4}{\pi^2}+\frac{8}{\pi^2}\sum\limits^\infty_{k=1} 
\frac{(-1)^{k-1}}{4k^2p^2-1}, & 
\text{если}\ s+p\equiv1\ (\Mod2);\\ 
\frac{4}{\pi^2}, & \text{если}\ s+p\equiv0\ (\Mod2). 
\end{cases} 
\end{align*} 
Для завершения доказательства первой части п.\,б) осталось заметить, 
что 
\begin{align} 
\sum^\infty_{k=1}\frac{1}{4k^2p^2-1}&=\frac{1}{2}- 
\frac{\pi}{4p}\ctg\frac{\pi}{2p},\\ 
% 
\sum^\infty_{k=1}\frac{(-1)^k}{4k^2p^2-1}&=\frac{1}{2}- 
\frac{\pi}{4p\sin\frac{\pi}{2p}}. 
\end{align} 
 
Докажем заключительную часть п.\,б). Рассмотрим последовательности 
$m_n=ns$, $q_n=np$. Заметив, что при $v=kp$, где $k\in\Bbb N$, 
$$ 
\bigg\|\frac{vm_n}{q_n}\bigg\|=0,\quad \bigg\|\frac{vm_n}{2q_n} 
\bigg\|= 
\begin{cases} 
0, & ks\equiv0\ (\Mod2);\\ 
\frac{1}{2}, & ks\equiv1\ (\Mod2), 
\end{cases} 
$$ 
и поступив так же, как при доказательстве п.\,a), из (7) получим 
$$ 
\lim_{n\to\infty}\frac{\ov L_{ns}(np)}{\ln(ns)}= 
\begin{cases} 
\frac{2}{\pi}+\frac{8}{\pi^2}\sideset{}{'}\sum\limits^\infty_{v=1} 
\frac{(-1)^v}{4v^2-1}, & 
\text{если}\ s\equiv0\ (\Mod2);\\ 
%
\frac{2}{\pi}+\frac{8}{\pi^2}\sideset{}{'}\sum\limits^\infty_{v=1} 
\frac{(-1)^v}{4v^2-1}
+\frac{16}{\pi^2}\sum\limits^\infty_{k=1} 
\frac{(-1)^p}{4(2k-1)^2p^2-1}, & 
\text{если}\ s\equiv1\ (\Mod2), 
\end{cases} 
$$ 
где \ ${}'$ \ при сумме означает, что $v\equiv0\ (\Mod p)$. Так как 
$\frac{8}{\pi^2}\sum\limits^\infty_{v=1}\frac{(-1)^v}{4v^2-1}= 
\frac{4}{\pi^2}-\frac{2}{\pi}$, то, применяя (11), (12), получим 
\begin{equation} 
R(p)=\lim_{n\to\infty}\frac{\ov L_{ns}(np)}{\ln(ns)}= 
\begin{cases} 
\frac{2}{\pi p\sin \frac{\pi}{2p}}, & 
\text{если}\ s+p\equiv1\ (\Mod2);\\ 
\frac{2\ctg\frac{\pi}{2p}}{\pi p}, & 
\text{если}\ s+p\equiv0\ (\Mod 2). 
\end{cases} 
\end{equation} 
Отсюда видно, что $R(p)\to\frac{4}{\pi^2}$ при $p\to\infty$. 
Следовательно, для любого рационального $s/p$ существует 
последовательность $\frac{m_n}{q_n}\to\frac{s}{p}$ ($n\to\infty$) 
такая, что 
\begin{equation} 
\lim_{n\to\infty}\frac{\ov L_{m_n}(q_n)}{\ln m_n}=\frac{4}{\pi^2}. 
\end{equation} 
Из (13), (14) следует утверждение п.\,б) для $\gamma=h$ и $\gamma=H$. 
 
Если $\gamma\in(h;H)$, то возьмем $q'_n\in\Bbb N$, $q'_n\to\infty$ 
($n\to\infty$) и положим $m'_n=[\frac{sq'_n-(q'_n)^{1-t}}{p}]$, где 
$$ 
t= 
\begin{cases} 
\frac{\gamma-h}{H-h}, & \text{если}\ s+p\equiv1\ (\Mod2);\\ 
\frac{H-\gamma}{H-h}, & \text{если}\ s+p\equiv0\ (\Mod2). 
\end{cases} 
$$ 
Ясно, что $\frac{m'_n}{q'_n}=\frac{s}{p}+\varepsilon_n$, где 
$p^{-1}(q'_n)^{-t}\le|\varepsilon_n|\le p^{-1}(q'_n)^{-t}+ 
(q'_n)^{-1}$, откуда $\lim\limits_{n\to\infty} 
\frac{\ln(|\varepsilon_n|+1/q'_n)} {ln m'_n}=-t$, что доказывает 
утверждение для $\gamma\in(h;H)$ согласно (7), (9), (10). Теорема 
доказана, т.\,к. если $\alpha\ne0;1$, то 
$\lim\limits_{n\to\infty}\frac{\ln(q_n-m_n)}{\ln m_n}=1$. 
 
 
\section{Основной результат} 
 
Обозначим 
$\tau_n=\min(2m_n-1,q_n-2m_n+1)+1$, 
$Y_n=\frac{C_{m_n}(W^r_q)}{m^{-r}_n\ln\tau_n}$,
$g(\alpha)=\big(\frac{\alpha\pi}{2\sin\frac{\alpha\pi}{2}}\big)^r$, 
где $m_n,q_n\in\Bbb N$. 
 
Основным результатом данной работы является 
 
\begin{thm} 
Пусть $q_n\ge2m_n$, $\tau_n\to\infty$, $\frac{2m_n-1}{q_n}\to\alpha$ при 
$n\to\infty$, $r\equiv i\ (\Mod2)$, $i=0$ или $i=1$. Тогда 
 
{\em a)} если $\alpha$ иррационально или $\alpha=0;1$, то 
$\lim\limits_{n\to\infty}Y_n=\frac{4}{\pi^2}g(\alpha)$, 
 
{\em б)} если $\alpha=\frac{s}{p}$, где $s,p\in\Bbb N$, $(s,p)=1$, то 
$\underset{n\to\infty}{\varliminf}{}Y_n=g(\alpha)h_i$, 
$\underset{n\to\infty}{\varlimsup}{}Y_n=g(\alpha)H_i$, где 
\begin{align*} 
& h_0=\frac{2}{\pi p}\ctg\frac{\pi}{2p},\quad 
H_0=\frac{4}{\pi^2},\\ 
h_1&= 
\begin{cases} 
h_0 & \text{при}\ s+p\equiv0\ (\Mod2);\\ 
H_0 & \text{при}\ s+p\equiv1\ (\Mod2), 
\end{cases} 
\\ 
H_1&= 
\begin{cases} 
H_0 & \text{при}\ s+p\equiv0\ (\Mod2);\\ 
\frac{2}{\pi p\sin\frac{\pi}{2p}} & \text{при}\ s+p\equiv1\ (\Mod2). 
\end{cases} 
\end{align*} 
Кроме того, для любого $\gamma\in[h_i;H_i]$, где $i=0$ при $r\equiv0\ 
(\Mod2)$ и $i=1$ при $r\equiv1\ (\Mod2)$, существует последовательность 
$\frac{m'_n}{q'_n}\to\frac{s}{p}$ $(n\to\infty)$ такая, при которой 
$\lim\limits_{n\to\infty}Y_n=\gamma g(\alpha)$. 
\end{thm} 
 
\begin{pf} 
Докажем, что равномерно по $m$ и $q$ имеет место формула 
\begin{equation} 
C_m(W^r_q)=\bigg(\frac{\pi}{q\sin\frac{\pi m}{q}}\bigg)^r 
(L'_{2m-1}(q)+O(1)), 
\end{equation} 
где 
$$ 
L'_{2m-1}(q)= 
\begin{cases} 
L_{2m-1}(q) & \text{при}\ r\equiv0\ (\Mod2);\\ 
\ov L_{2m-1}(q) & \text{при}\ r\equiv1\ (\Mod2). 
\end{cases} 
$$ 
Так как класс $W^r_q$ инвариантен относительно сдвига аргумента, то в 
определении $C_m(W^r_q)$ можно вместо нормы $\|f-S_{m-1}(\bold f)\|$ 
взять отклонение $|f(0)-S_{m-1}(\bold f,0)|$. 
 
Применяя $r$ раз преобразование Абеля, получим для дискретных 
коэффициентов Фурье $2\pi$-периодической функции следующее выражение: 
$$ 
c_k=\frac{1}{q(1-e^{ix_k})^r}\sum^{q-1}_{n=0} 
\Delta^r_{2\pi/q}f(x_n)e^{-kx_n i},\quad k\ne0. 
$$ 
Следовательно, подставляя значения $f(0)$, $S_{m-1}(\bold f,0)$ и $c_k$, 
получим 
\begin{equation} 
C_m(W^r_q)=\sup_{f\in W^r_q}\frac{2}{q}\bigg|\sum^{q-1}_{n=0} 
\Delta^r_{2\pi/q}f(x_n)\sum^{[q/2]}_{k=m} 
\frac{\cos(\frac{\pi k(2n+r)}{q}+\frac{r\pi}{2})} 
{(2\sin\frac{\pi k}{2})^r}\bigg|+O(q^{-r)}. 
\end{equation} 
Допустим сначала, что $r\equiv0\ (\Mod2)$. Из (16) следует 
\begin{equation} 
C_m(W^r_q)\le\frac{2^{r+1}\pi^r}{q^{r+1}}\sum^{q-1}_{n=0}\bigg| 
\sum^{[q/2]}_{k=m}\frac{\cos kx_n}{(2\sin x_k/2)^r}\bigg| 
+O(q^{-r}). 
\end{equation} 
 
Обозначим 
\begin{gather*} 
a_k=(2\sin x_k/2)^{-r},\quad \Delta(k)=a_k-a_{k+1},\quad 
\Delta^2(k)=\Delta(k)-\Delta(k+1),\\ 
% 
B_m(n)=\sum^{[q/2]}_{k=m}a_k\cos kx_n,\quad 
D_j(x)=\frac{1}{2}\sum^j_{n=1}\cos nx,\quad 
F_k(x)=\sum^k_{n=0}D_n(x). 
\end{gather*} 
 
Применяя дважды преобразование Абеля, получим 
\begin{multline} 
B_m(n)=-a_m D_{m-1}(x_n)+a_{[q/2]}D_{[q/2]}(x_n)+ \\ 
% 
+\sum^{[q/2]-2}_{k=m}\Delta^2(k) F_k(x_n)-\Delta(m) 
F_{m-1}(x_n)+\Delta([q/2]-1)F_{[q/2]-1}(x_n)\equiv \\ 
% 
\equiv-a_m\frac{\sin(m-\frac{1}{2})x_n}{2\sin x_n/2}+G(n). 
\end{multline} 
Так как $F_k(x)\ge0$ и $\sum\limits^{q-1}_{n=0}F_k(x_n)= 
\frac{(k+1)q}{2}$, то 
$$ 
\frac{1}{q}\sum^{q-1}_{n=0}\sum^{[q/2]-2}_{k=m}\Delta^2(k) 
F_k(x_n)=\frac{1}{2}(m\Delta(m)-([q/2]-1)\Delta([q/2]-1)+ 
a_m-a_{[q/2]-1}). 
$$ 
Кроме того, $\Delta(k)=O(\frac{a_k}{k})$, $D_{[q/2]}(x_n)=O(1)$. 
Следовательно, будем иметь 
\begin{equation} 
\frac{1}{q}\sum^{q-1}_{n=0}|G(n)|=O(a_m). 
\end{equation} 
Из (17)--(19) следует оценка сверху. 
Теперь докажем оценку снизу. Пусть 
\begin{align*} 
A&=\{([o;[\tfrac{q}{4}]-1]\cup[\tfrac{3q}{4}+2;q-1])\cap 
\Bbb Z\},\\ 
% 
B&=\{[0;q-1]\cap(\Bbb Z\setminus A)\} 
\end{align*} 
--- множество целых точек. Положим 
$$ 
\varphi(x_n)=\bigg(\frac{2\pi}{q} \bigg)^r 
\begin{cases} 
\sign\frac{\sin(m-\frac{1}{2})x_n}{\sin x_n/2}, & n\in A;\\ 
\frac{-\sum\limits_{j\in A}\varphi(x_j)}{|B|}, & n\in B, 
\end{cases} 
$$ 
где $|B|$ --- число элементов $B$. Убедимся, что из условия 
$\sum\limits^{q-1}_{n=0}\varphi(x_n)=0$ следует существование функции 
$f\in W^r_q$ такой, при которой $\Delta^r_{2\pi/q}f(x_n)=\varphi(x_n)$. 
Возьмем $f_1(x_n)=-\sum\limits^n_{k=0}\varphi(x_k)+C_1$, 
$n=1,\dotsc,q-1$, $f_1(o)=C_1$ и выберем $C_1$ так, чтобы 
$\sum\limits^{q-1}_{n=0}f_1(x_n)=0$. Ясно, что функцию $f_1$ можно 
продолжить как периодическую с периодом $2\pi$ и 
$\Delta_{2\pi/q}f(x_n)=\varphi(x_n)$. Далее возьмем 
\begin{gather*} 
f_2(x_n)=-\sum^n_{k=0}f_1(x_k)+C_2,\quad n=1,\dotsc,q-1,\\ 
f_2(o)=C_2,\quad \sum^{q-1}_{n=0}f_2(x_n)=0 
\end{gather*} 
и т.\,д. Продолжая процесс, можно построить $\bold 
f=\{f(x_n)\}^{q-1}_{n=0}$ с 
периодом $2\pi$ так, чтобы $\Delta^r_{2\pi/q}f(x_n)=\varphi(x_n)$. 
Поэтому из (16) следует 
$$ 
C_m(W^r_q)\ge\frac{2}{q}\bigg|\sum^{q-1}_{n=0}\varphi(x_n) 
B_m(n)\bigg|+O\bigg(\frac{a_m}{q^r} \bigg). 
$$ 
Заметив, что $|\varphi(x_n)|\le(\frac{2\pi}{q})^r$ и 
$\frac{\sin(m-\frac{1}{2})x_n}{\sin x_n/2}=O(1)$ при $n\in B$, из (18), 
(19) имеем 
\begin{multline*} 
C_m(W^r_q)\ge\frac{a_m}{q}\bigg(\frac{2\pi}{q} \bigg)^r\bigg( 
\sum^{q-1}_{n=0}\frac{|\sin(m-\frac{1}{2})x_n|}{\sin x_n/2} - \\ 
% 
-2\sum_{n\in B}\frac{|\sin(m-\frac{1}{2})x_n|}{\sin x_n/2}\bigg) 
+O\bigg(\frac{a_m}{q^r} \bigg)=\bigg(\frac{2\pi}{q} \bigg)^r 
a_m(L_m(q)+O(1)). 
\end{multline*} 
Для четных $r$ формула (16) установлена. 
 
Пусть теперь $r\equiv1\ (\Mod2)$. Обозначим 
$$ 
\varphi_0(x_n)=\bigg(\frac{2\pi}{q} \bigg)^r\sign 
\sum^{[q/2]}_{k=m}\frac{\sin(2n+r)\pi k/q} 
{(2\sin\frac{\pi k}{q})^r}. 
$$ 
Ясно, что $\sum\limits^{q-1}_{n=0}\varphi_0(n)=0$. Поэтому существует 
функция $f\in W^r_q$ такая, что $\Delta^r_{2\pi/q}f(x_n)=\varphi_0(x_n)$. 
Следовательно, из (16) имеем 
\begin{equation} 
C_m(W^r_q)=\frac{2}{q}\bigg(\frac{2\pi}{q} \bigg)^r\sum^{q-1}_{n=0} 
|A_m(n)|+O\bigg(\frac{a_m}{q^r} \bigg), 
\end{equation} 
где $A_m(n)=\sum\limits^{[q/2]}_{k=m} 
\frac{\sin2\pi k(n+\frac{1}{2})/q}{(2\sin\frac{\pi k}{q})^r}$. 
 
Применяя дважды преобразование Абеля, получим 
\begin{multline} 
A_m(n)=a_m\frac{\cos(2m-1)(n+\frac{1}{2})\pi/q} 
{2\sin(n+\frac{1}{2})/q}- \\ 
% 
-a_{[q/2]}\frac{\cos([\frac{q}{2}]+\frac{1}{2})(2n+1)\pi/q} 
{2\sin\pi(n+\frac{1}{2})/q}+ 
\frac{\Delta(m)\sin(2n+1)\pi m/q} 
{4\sin^2\pi(n+\frac{1}{2})/q} - \\ 
% 
-\Delta([\tfrac{q}{2}]-1) 
\frac{\sin[\frac{q}{2}](2n+1){\pi}/{q}} 
{4\sin^2\pi(n+\frac{1}{2})/q}-\sum^{[q/2]-2}_{k=m} 
\frac{\sin(k+1)(2n+1){\pi}/{q}} 
{4\sin^2\pi(n+\frac{1}{2})/q}\Delta^2(k)\equiv \\ 
% 
\equiv a_m\frac{\cos(2m-1)(n+\frac{1}{2}){\pi}/{q}} 
{2\sin\pi(n+\frac{1}{2})/q}+Q_m(n). 
\end{multline} 
Заметив, что 
\begin{gather*} 
\frac{\cos([\frac{q}{2}]+\frac{1}{2})(2n+1){\pi}/{q}} 
{2\sin(n+\frac{1}{2}){\pi}/{q}}= 
\begin{cases} 
0, & \text{если}\ q\equiv1\ (\Mod2);\\ 
\frac{(-1)^n}{2}, & \text{если}\ q\equiv0\ (\Mod2), 
\end{cases} 
\\ 
\sum^{q-[\frac{q}{2m-1}]-3}_{n=[\frac{q}{2m-1}]+2} 
\frac{|\sin(k+1)(2n+1){\pi}/{q}|} 
{\sin^2\pi(n+\frac{1}{2})/q}=O(mq), 
\end{gather*} 
получим 
\begin{equation} 
\frac{2}{q}\sum^{q-[\frac{q}{2m-1}]-3}_{n=[\frac{q}{2m-1}]+2} 
|Q_m(n)|=O(a_m). 
\end{equation} 
С другой стороны, 
\begin{equation} 
\frac{1}{q}\bigg(\sum^{[\frac{q}{2m-1}]+1}_{n=0} 
|A_m(n)|+\sum^{q-1}_{n=q-[\frac{q}{2m-1}]-2}|A_m(n)|\le 
\frac{2}{q}\bigg(\bigg[\frac{q}{2m-1}\bigg]+2\bigg) 
\sum^{[q/2]}_{k=m}\frac{1}{(2\sin\frac{\pi k}{q})^r}=O(a_m). 
\end{equation} 
Применяя (21)--(23), из (20) получим формулу (15) для нечетных $r$. 
 
Так как $L_{2m_n-1}(q_n)=L_{q_n-(2m_n+1)}(q_n)+O(1)$ 
при $2m_n-1\ge q_n/2$ 
и $\lim\limits\begin{Sb}n\to\infty \\ \frac{2m_n-1}{q_n}\to\alpha 
\end{Sb}\frac{\ln(q_n-2m_n+1)}{\ln(2m_n-1)}{=}1$ при $\alpha\ne0;1$, то 
для завершения доказательства теоремы 3 осталось применить
теоремы 1\linebreak и ~2. 
\end{pf}
