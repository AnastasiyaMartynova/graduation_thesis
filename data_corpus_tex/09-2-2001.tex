\section{Введение} 
 
В данной работе построен базисный аналог обобщенной $H$-функции с 
ядром, зависящим от дробно-рациональной функции от $q$-гамма-функций 
\cite{1}--\cite{6}. Этот аналог может включать, например, $I_q$-функцию 
\cite{7}, $H_q$-функцию \cite{8}, \cite{9}, $G_q$-функцию \cite{8}, 
\cite{10}, $I$-функцию \cite{11}--\cite{13},\linebreak $H$-функцию Фокса 
\cite{14}, $G$-функцию Мейера \cite{15}. Получены 
достаточное условие сходимости базисного аналога обобщенной 
$H$-функции, интегральные связи и некоторые формулы. 
 
\section{Базисный аналог обобщенной $H$-функции} 
 
{\bf Определение} (ср. \cite{7}). Базисным аналогом обобщенной 
$H$-функции назовем интеграл Мел\-ли\-на--Барнса 
\begin{equation} 
I_q(z\mid\theta(s))\equiv I_q(z\mid r,p_i,\ov m_i,\ov a_i,\ov\alpha_i)= 
\frac{1}{2\pi \omega}\int_{L_s}\theta(s)z^s ds, 
\end{equation} 
где 
$$
\frac{1}{\theta(s)}=\sin(\pi s)\sum^r_{i=1} 
Q^{p_i}_{\ov m_i,\ov a_i,\ov\alpha_i}(s),\quad r\in N,\quad
% 
Q^{p_i}_{\ov m_i,\ov a_i,\ov\alpha_i}(s)= 
\prod^{p_i}_{k=1}\Gamma^{m_{i k}}_q(A_{i k}+\alpha_{i k}s), 
$$
$q$-гамма-функция 
$\Gamma_q(s)=\frac{(q;q)_\infty}{(q^s;q)_\infty}(1-q)^{1-s}$ 
определена в \cite{1}, \cite{2}, 
$(a,q)_\infty=\prod\limits^\infty_{n=0} 
(1-a q^n)$, $|q|<1$, 
$A_{i k}=\frac{1}{2}-(a_{i k}-\frac{1}{2})\sign \alpha_{i k}$, $p_i$ 
--- натуральное число, $i=\ov{1,r}$, $k=\ov{1,p_i}$; 
$\ov m_i=(m_{i1},m_{i2},\dotsc,m_{i p_i})$, $m_{i k}$ --- целые числа; 
$\ov a_i=(a_{i1},a_{i2},\dotsc,a_{i p_i})$, $a_{i k}$ комплексные; 
$\ov\alpha_i=(\alpha_{i1},\alpha_{i2},,\dotsc,\alpha_{i p_i})$, 
$\alpha_{i k}$ --- действительные числа; 
$z^s$ определено в \cite{15}--\cite{17}, $\omega=(-1)^{1/2}$. 
Контур интегрирования $L_s$ идет от $-i\infty$ до $i\infty$, так что 
полюсы функций $\Gamma^{m_{i k}}_q(A_{i k}+\alpha_{i k}s)$ с $m_{i 
k}<0$, $\alpha_{i k}<0$ лежат справа от контура, а полюсы функций 
$\Gamma^{m_{ i k}}_q(A_{i k}+\alpha_{i k}s)$ с $m_{i k}<0$, $\alpha_{i 
k}>0$, $i=\ov{1,r}$, $k=\ov{1,p_i}$, лежат слева и находятся по крайней 
мере на некотором расстоянии $\varepsilon>0$ от контура. 
 
\begin{thm}[{\series{m}\shape{n}\selectfont ср. \cite{7}}] Базисный 
аналог обобщенной $H$-функции $(1)$ сходится, если $|\arg z|<\pi$. 
Если, кроме того, $m_{i k}=-1$ при $i=\ov{1,r}$, $k=\ov{1,m+n}$, $a_{i 
k}=a_k$, $\alpha_{i k}=\alpha_k>0$, $k=\ov{1,n};$ 
$a_{i k}=b_{k-n}$, $\alpha_{i k}=-\beta_{k-n}<0$, $k=\ov{n+1,n+m};$ 
$m_{i k}=1$ при $i=\ov{1,r}$, $k=\ov{m+n+1,A_i+B_i+2};$ 
$a_{i k}=a_{i(k-m)}$, $\alpha_{i k}=-\alpha_{i(k-m)}<0$, 
$k=\ov{m+n+1,m+A_i}$, $a_{i(A_i+m+1)}=1$, $\alpha_{i(A_i+m+1)}=-1;$ 
$a_{i k}=b_{i(k-A_i-1)}$, $\alpha_{i k}=\beta_{i(k-A_i-1)}>0$, 
$k=\ov{A_i+m+2,A_i+B_i+1};$ 
$a_{i(A_i+B_i+2)}=1$, $\alpha_{i(A_i+B_i+2)}=1$, $p_i=A_i+B_i+2$, 
то справедливо равенство 
$$ 
\lim_{q\to 1_-}I_q(z\mid r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)= 
I^{m,n}_{A_i,B_i} 
\left[z\left|\begin{matrix}(a_j,\alpha_i)_{1,n},\dotsc, 
(a_{i j},\alpha_{i j})_{n+1,A}\\ 
(b_j,\beta_j)_{1,m},\dotsc,(b_{i j},\beta_{i j})_{m+1,B_i} 
\end{matrix}
\right.\right], 
$$ 
в котором функция, стоящая справа, введена в $\cite{11}$. 
\end{thm} 
 
\begin{thm} 
Справедливы следующие утверждения$:$ 
 
{\em а)} 
$\displaystyle{
\int^{+\infty}_0 x^{s-1}I_q(x z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)dx=-\frac{z^{-s}}{\sin(\pi s)} 
\bigg(\,\sum^r_{i=1}Q^{p_i}_{\ov m_i,\ov a_i,-\ov\alpha_i}(s) 
\bigg)^{-1}}$, 

\noindent{}где $|\arg z|<\pi$, $0<\re s<1;$ 
 
{\em б)} 
$\displaystyle{
\int^{+\infty}_0 x^{-1/2}I_q(x\mid 
r_1, p_{1i},\ov m_{1i},\ov a_{1i},\ov\alpha_{1i}) 
I_q(z x\mid r_2, p_{2j},\ov m_{2j},\ov a_{2j},\ov\alpha_{2j})dx=}$ 
 
\hfill $=z^{-1/2}I_q(z^{1/2}\mid r_1+r_2, p_{1i}+p_{2j}, 
\ov m_{1i},\ov a_{1i},-\tfrac{1}{2}\ov\alpha_{1i}; 
\ov m_{2j},\tfrac{1}{2}\ov{2a_{2j}-\alpha_{2j}}, 
\tfrac{1}{2}\ov{\alpha_{2j}})$, 

\noindent{}где $|\arg z|<\pi$. 
\end{thm} 
 
Из (1), формул (1.10.5), (1.10.9), (1.10.11) в (\cite{18}, с.\,38) 
вытекает 
 
\begin{thm} 
Имеют место равенства
 
{\em а)} 
$z^{-k}I_q( z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)=(-1)^k I_q(z\mid 
r, p_i,\ov m_i,\ov{a_i+k\alpha_i},\ov\alpha_i)$, 
$k$ --- целое число$;$ 
\smallskip
 
{\em б)} 
$
q^{A_{k_0}}I_q(q^{\alpha_{k_0}}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)=I_q(z\mid 
r, p_i,\ov m_i,\ov{ a_i+k\alpha_i},\ov\alpha_i) +$ 

\begin{center} 
$+(q-1)I_q(z\mid r, p_i,\ov m_i,\ov a_i,\ov\alpha_i; 1,1, 
a_{k_0}+1,\alpha_{k_0}; 1,-1,a_{k_0},\alpha_{k_0})$, 
 
$m_{i k_0}=m_{k_0}<0,\quad a_{i k_0}=a_{k_0},\quad 
\alpha_{i k_0}=\alpha_{k_0},\quad 
i=\ov{1,r},\quad k_0=\ov{1,p_i}$; 
\end{center}
 
{\em в)} 
$q^{A_{k_0}-1}I_q(q^{\alpha_{k_0}}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)=I_q(z\mid 
r, p_i,\ov m_i,\ov{a_i+k\alpha_i},\ov\alpha_i)+$ 

\begin{center} 
$+(q-1)I_q(z\mid r, p_i,\ov m_i,\ov a_i,\ov\alpha_i; 
1,1, a_{k_0},\alpha_{k_0};1, -1,a_{k_0}-1,\alpha_{k_0})$, 
 
$m_{i k_0}=m_{k_0}>0,\quad a_{i k_0}=a_{k_0},\quad 
\alpha_{i k_0}=\alpha_{k_0},\quad 
i=\ov{1,r},\quad k_0=\ov{1,p_i}$.
\end{center} 
\end{thm} 
 
\begin{thm} 
Пусть $I_q(a_{i j}\pm1)$ является базисным аналогом обобщенной 
$H$-функции, получающейся из $(1)$, если в ее правой части множитель 
$\Gamma^{m_{i j}}_q(A_{i j}+\alpha_{i j}s)$ заменяется на 
$\Gamma^{\sign(m_{i j})}_q(A_{i j}\pm1+\alpha_{i j}s) 
\Gamma^{m_{i j}-\sign(m_{i j})}_q(A_{i j}+\alpha_{i j}s)$. 
Кроме этого, допустим, что $m_{i j_0}=m_{j_0}$, $m_{i k_0}=m_{k_0}$, 
$a_{i j_0}=a_{j_0}$, $a_{i k_0}=a_{k_0}$, $\alpha_{i j_0}=\alpha_{i 
k_0}=\alpha_{k_0}$, $i=\ov{1,r}$, $j_0=\ov{1,p_i}$, $k_0=\ov{1,p_i}$. 
Тогда справедливы равенства 
 
{\em а)} 
$(1-q)[I_q(\alpha_{j_0}+1)-I_q(\alpha_{k_0}+1)]= 
(q^{\pm\alpha_{k_0}}-q^{\pm\alpha_{j_0}}) 
I_q(q^{\mp\alpha_{k_0}}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)$, 
\\ $m_{j_0}\lessgtr0$, $m_{k_0}\lessgtr0$, $\alpha_{k_0}\lessgtr0;$ 
\smallskip
 
{\em б)} 
$(1-q)[I_q(\alpha_{j_0}-1)-I_q(\alpha_{k_0}-1)]= 
(q^{\pm(\alpha_{k_0}-1)}-q^{\pm(\alpha_{j_0}-1)}) 
I_q(q^{\mp\alpha_{k_0}}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)$, \\
$m_{j_0}\gtrless0$, $m_{k_0}\gtrless0$, $\alpha_{k_0}\lessgtr0;$
 
{\em в)} 
$(1-q)[I_q(\alpha_{j_0}+1)-I_q(\alpha_{k_0}-1)]= 
\left\{\begin{matrix}q^{1-\alpha_{k_0}}-q^{-\alpha_{j_0}} \\ 
q^{\alpha_{j_0}-1}-q^{\alpha_{k_0}}\end{matrix}\right\} 
I_q(q^{\mp\alpha_{k_0}}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i).$
\end{thm} 
 
Доказательство основано на формуле (1.10.5) в (\cite{18}, с.\,38) и 
(1). 
 
\begin{thm}[{\series{m}\shape{n}\selectfont ср. \cite{7}}] Имеют место 
следующие равенства$:$ 
\begin{gather*} 
(1-q)^{-\sigma}\int^1_0 x^{\sigma-1}E_q(q x)I_q(z x^{-\delta}\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)d_q x=\\ 
% 
= I_q\big((1-q)^{-\delta}z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i;1,1,\sigma,-\delta\big),\quad
\re\sigma>0,\quad \re\delta>0,\quad |\arg z|<\pi; \\
%
\Gamma^{-1}_q(\beta)\int^1_0 x^{\sigma-1}(1-q)_{\delta-\sigma-1} 
I_q(z x^{-\gamma}\mid r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)d_q x=\\ 
% 
=I_q(z\mid r, p_i,\ov m_i,\ov a_i,\ov\alpha_i; 
1,1,\alpha,-\gamma;1,-1,\alpha+\beta,-\gamma),\\ 
% 
\re\sigma>0,\quad \re\delta>0,\quad 
\re\gamma>0,\quad\re(\delta-\sigma)>0,\quad 
|\arg z|<\pi; \\[3mm]
%
\frac{(1-q^{\sigma-1})}{2\pi\omega}\int_L 
x^{-\sigma}e_q(q x)I_q(z x^\delta\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i)dx= \\ 
= I_q\big((1-q)^\delta z\mid 
r, p_i,\ov m_i,\ov a_i,\ov\alpha_i; 
1,-1,\sigma,-\delta\big), \quad
\re\sigma>0,\quad |\arg z|<\pi. 
\end{gather*} 
\end{thm}
