\begin{enumerate}
  \item $\varphi\left(a_{1}, \ldots, \quad a_{n} \omega\right)=\varphi\left(a_{1}\right) \ldots \varphi\left(a_{n}\right) \omega$ для любой $n$-арной операции $\omega \in \Omega_{F}$ и любой последовательности әлементов $a_{1}, \ldots, a_{n}$ системы $A$;
