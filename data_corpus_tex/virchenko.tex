\entitle{On one generalization of Bessel function}



\enauthor{N.~A.~Virchenko, M.~O.~Chetvertak}{V\,i\,r\,c\,h\,e\,n\,k\,o~N.~A., C\,h\,e\,t\,v\,e\,r\,t\,a\,k~M.~O.}

\enabstract{In this paper the generalized Bessel function $J_{\mu ,\omega } ( x )$ is introduced. The function $J_{\mu ,\omega } ( x )$ is given as one solution of the following differential equation: $$ x^2{y}''+x{y}'+\left( {x-\mu ^2} \right)\left( {x+\omega ^2} \right)y=0, \quad \mu , \omega \notin \mathbb Z. $$ The representation of the $J_{\mu ,\omega } ( x )$ by the power series is given. The theorem on integral representations of the function $J_{\mu ,\omega } ( x )$ is established. The main properties of the function $J_{\mu ,\omega } ( x )$ are studied. The integral transforms of Bessel type with the function $J_{\mu ,\omega } ( x )$ is constructed. Formula of inversion of this transform is received.}

\textbf{Введение.} 
Среди многообразия специальных функций особо выделяются функции Бесселя в силу своих многочисленных замечательных приложений в~теории дифференциальных и интегральных уравнений, в физике, механике и~др. (см., например библиографию в~[\citen{virch:1}--\citen{virch:10}]).
Функции Бесселя впервые возникли при рассмотрении задач о~распостранении тепла в твердом круглом цилиндре, при исследовании колебаний  растянутой круговой мембраны (работы Л.~Эйлера), встречаются в работе Ж.~Лагранжа по  эллиптическим движениям. Систематическое изучение функций Бесселя начато Ф.~Бесселем (1824~г.) в его научных работах.

Теперь функции Бесселя используются при решении широкого класса краевых  задач математической физики, механики сплошных сред, в теории интегральных  преобразований, так как они часто возникают при решении задач как прикладной, так и теоретической математики, в теории  специальных функций и других отраслях прикладного естествознания [\citen{virch:10}--\citen{virch:18}].


\smallskip
\Section{Обобщенная функция Бесселя}
Введем обобщенную функцию Бесселя $J_{\mu ,\omega } ( x )$ как  одно из решений следующего дифференциального уравнения: 
\begin{equation}
\label{vir:eq1}
x^2 y''+x y'+ (x-\mu ^2 )(x+\omega ^2)y=0,
\end{equation}
где $\mu$, $\omega \notin \mathbb Z$. 

Заметим, что при $\mu ^2=\omega ^2=\upsilon $ уравнение \eqref{vir:eq1} является известным  уравнением Бесселя [\citen{virch:1}]:
\[
x^2{y}''+x{y}'+( x^2-\upsilon ^2 )y=0.
\]

Решение дифференциального уравнения \eqref{vir:eq1} будем искать в виде 
\begin{equation}
\label{vir:eq2}
y=\sum\limits_{k=0}^\infty C_k z^{k+\rho },\quad  C_0 \ne 0, 
\end{equation}
где $\rho $ "--- корень характеристического уравнения, $C_k $ "--- некоторые аналитические 
функции при всех $x$.

Подставив \eqref{vir:eq2} в \eqref{vir:eq1} и приравняв коэффициенты при одинаковых степенях~$x$, 
получим следующую рекуррентную систему для определения $C_k $:
\begin{gather}
\label{vir:eq3}
( \rho +k )^2 C_k +C_{k-2} + (\mu ^2-\omega ^2 )C_{k-1} =0, \quad k=2, 3, \dots,
\\
\label{vir:eq4}
C_{2m} =-\frac{C_{2m-2} +( \omega ^2-\mu ^2) C_{2m-1} } {( \rho +2m )^2-\mu ^2\omega ^2}, \quad m=1, 2, \dots.
\end{gather}
После преобразования \eqref{vir:eq2} с учетом $C_k $, найденных  из \eqref{vir:eq3}, \eqref{vir:eq4}, получим
\begin{equation}
\label{vir:eq5}
y=J_{\mu ,\omega } ( x )=\Bigl( \frac{x}{2} \Bigr)^{\mu ^2}
\frac{1}{\Gamma \bigl( \frac{\mu ^2+\mu \omega +2}{2} \bigr)} \cdot{ }_1 F_2 
\Bigl( 1; \frac{\mu ^2-\mu \omega +2}{2}, \frac{\mu ^2+\mu \omega +2}{2}; -\frac{x^2}{4} \Bigr),
\end{equation}
где $\Gamma (x)$ "--- гамма-функция [\citen{virch:2}], 
${ }_1 F_2 ( a; c, d; x)$ "--- гипергеометрическая функция [\citen{virch:19}], определяемая следующим образом:
\begin{equation}
\label{vir:eq6}
{ }_1 F_2 (a; c, d; x )=\sum\limits_{n=0}^\infty 
\frac{( a )_n }{( c )_n ( d )_n }\frac{x^n}{n!},
\end{equation}
где $( a )_n $ "--- символ Похгаммера, определенный для $x\in \mathbb C$ и $n\in  \mathbb N_0 = \mathbb N\cup \{ 0 \}$:
\begin{equation}
\label{vir:eq7}
( a)_0 =1, \quad (a)_n =a( a+1)\dots ( a+n-1),\quad n\in \mathbb N.
\end{equation}

Легко убедиться, что при $\mu ^2=\omega ^2=\upsilon $ функция, определяемая формулой \eqref{vir:eq5}, совпадает с функцией  Бесселя $J_\upsilon ( x)$ [\citen{virch:1}]:
\[
J_\upsilon ( x )=\Bigl(\frac{x}{2}\Bigr)^\upsilon \frac{1}{\Gamma ( \upsilon +1)}\cdot { }_0 F_1 \Bigl(\upsilon 
+1; -\frac{x^2}{4} \Bigr).
\]

\Section{Интегральные представления функции $\boldsymbol{J_{\mu ,\omega } ( x )}$}

\begin{newthm}{Теорема (об интегральных представлениях функции $J_{\mu ,\omega } ( x)$)}
При условиях существования функции $J_{\mu ,\omega } ( x )$ справедливы следующие интегральные представления{\/\rm:}
\begin{equation}
\label{vir:real:eq8}
J_{\mu ,\omega } ( x )=
\frac{\mu ^2+\mu \omega }{2}
\int_0^1 \int_0^1 
\exp\Bigl({-t\tau \frac{x^2}4}\Bigr) 
(1-t)   ^{\frac{\mu ^2+\mu \omega }{2}-1} 
(1-\tau)^{\frac{\mu ^2-\mu \omega }{2}}\tau ^{-1} \, dt \,d\tau ,
\end{equation}
\begin{multline}
\label{vir:eq8}
J_{\mu ,\omega } ( x )=2 (\mu ^2+\mu \omega ) \int_0^\pi \int _0^\pi 
\exp\Bigl(-\sin ^2\phi \sin \theta \frac{x^2}{4}\Bigr) ( \cos \phi )^{\mu ^2+\mu \omega -1}\times \\
\times ( \cos \theta )^{\mu ^2+\mu \omega +1}\sin \phi \sin ^{-1}\theta \, d\phi \,d\theta,
\end{multline}
\begin{multline}
\label{vir:eq8}
J_{\mu ,\omega } ( x )=4\int_0^1 \int _0^1 
\exp\Bigl(-( 1-k^2)(1-l^2)\frac{x^2}{4}\Bigr) \times \\ 
\times
k^{\mu ^2+\mu  \omega } l^{\mu ^2-\mu \omega +1} ( 1-l^2)^{-1}\, dk\, dl. 
\end{multline}
\end{newthm}

\smallskip

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о\/} формул \eqref{vir:real:eq8}--\eqref{vir:eq8} осуществляется при помощи соответствующих  подстановок.

Учитывая для гипергеометрической функции ${ }_1 F_2 $ формулы
\begin{gather}
\label{vir:eq9}
{ }_1 F_2 ( 1;k{+}m{+}1, r; x)\, {=}\, ( k{+}m)\Gamma ( r )x^{\frac{1-r}{2}}\int_0^1 \!\!
I_{r-1} ( 2\sqrt {xt} ) t^{\frac{1-r}{2}}( 1-t)^{k+m-1}dt,
\\
\label{vir:eq10}
{ }_1 F_2 (1; k+m+1, r; x)=\frac{\Gamma (k+m+1)}{k!\Gamma ( m)}\int_0^1 { }_1 F_2 ( 1; m, r; xt)t^{m-1}(1-t)^k dt,
\end{gather}
где $I_{r-1} (x)$ "--- модифицированная функция Бесселя при $x\in \mathbb C$, 
можно получить еще ряд представлений для $J_{\mu ,\omega } ( x )$.

Заметим, что формулы \eqref{vir:eq9}, \eqref{vir:eq10} доказываются непосредственной проверкой.


Используя представление функции $J_{\mu ,\omega } ( x )$ в виде ряда
\begin{multline}
\label{vir:real:eq13}
J_{\mu ,\omega } ( x )=\Gamma \Bigl( \frac{\mu ^2-\mu \omega }{2}+1 \Bigr)\times 
\\
\times \sum\limits_{n=0}^\infty 
\frac{(-1)^n}{n!} \frac{1}{\Gamma \bigl( \frac{\mu ^2-\mu \omega }{2}+n+1 \bigr)\Gamma \bigl( \frac{\mu ^2+\mu \omega }{2}+n+1  \bigr)}
\Bigl( \frac{x}{2} \Bigr) ^{2n+\mu ^2},
\end{multline}
получаем формулу для дифференцирования:
\begin{equation}
\label{vir:eq11}
\frac{d}{dx}J_{\mu ,\omega } ( x )=A\sum\limits_{n=0}^\infty 
\frac{(-1)^n}{n!}\frac{( 2n+\mu ^2)}{2\Gamma 
(\upsilon _1 +n+1)\Gamma (\upsilon _2 +n+1)}\Bigl( \frac{x}{2}\Bigr)^{2n+\mu ^2-1},
\end{equation}
где
\[
\upsilon _1 =\frac{\mu ^2-\mu \omega }{2},\quad 
\upsilon _2 =\frac{\mu ^2+\mu \omega }{2},\quad 
\mu ^2=\upsilon _1 +\upsilon _2 ,\quad 
A=\Gamma \Bigl( \frac{\mu ^2-\mu \omega }{2}+1 \Bigr).
\]


\smallskip
\Section{Обобщенное интегральное преобразование} 
Введем обобщенное интегральное преобразование типа Бесселя:
\begin{equation}
\label{vir:eq12}
\bigl( I_{\mu ,\omega } f \bigr)( x )=
\int_0^\infty  J_{\mu ,\omega } \Bigl( a; c, d; -\frac{x^2}{4}t \Bigr) 
f( t )\,dt.
\end{equation}
Для простоты рассмотрим \eqref{vir:eq12} в~следующей форме:
\begin{equation}
\label{vir:eq13}
\bigl( I_{\mu ,\omega } f \bigr) ( x )=
\int_0^\infty 
J_{\mu ,\omega}  (a;c,d;-xt ) f( t )dt, \quad a, c, d \in \mathbb C,\quad \mathop{\rm Re} a>0.
\end{equation}
Легко заметить, что \eqref{vir:eq13} "--- интегральное $G$-преобразование [\citen{virch:3}]:
\begin{equation}
\label{vir:eq14}
\bigl( {Gf} \bigr) ( x )=\int_0^\infty G_{p,q}^{m,n} 
\left[ \begin{array}{*{2}c}
 \left. \begin{array}{*{2}c}
 ( a_i )_{1,p}   \\
 (\beta _j )_{1,q}  \\
\end{array}  \right| & xt \\
\end{array}  \right] f ( t )dt,
\end{equation}
где $G_{p,q}^{m,n} $ "--- функция Мейера [\citen{virch:20}]:
\begin{multline}
\label{vir:eq15}
G_{p,q}^{m,n} \left[ {{\begin{array}{*{20}c}
 {\left. {{\begin{array}{*{20}c}
 {\left( {a_i } \right)_{1,p} } \hfill \\
 {\left( {\beta _j } \right)_{1,q} } \hfill \\
\end{array} }} \right|} \hfill & z \hfill \\
\end{array} }} \right]=
\\
=
\frac{1}{2\pi i}\int_L 
{\frac{\prod _{j=1}^m {\Gamma \left( {\beta _j +s} \right)} 
\prod _{i=1}^n {\Gamma \left( {1-a_i -s} \right)} 
}{\prod _{i=n+1}^p {\Gamma \left( {a_i +s} \right)} 
\prod _{j=m+1}^q {\Gamma \left( {1-\beta _j -s} \right)} }} z^{-s}ds.
\end{multline}
С помощью интегрального преобразования Меллина получаем формулу обращения 
для (\ref{vir:eq13}). Справедлива следующая теорема.

\smallskip

\begin{newthm}{Теорема} 
Пусть $0<1-\upsilon <\mathop{\rm Re} a,$ $\alpha _0 =\max [ 1-\mathop{\rm Re} c, 1-\mathop{\rm Re} d ].$ 
Если 
$\upsilon >\alpha _0 ,$ $\mathop{\rm Re} \lambda >\upsilon -1,$ 
$( 1-\upsilon )+ \mathop{\rm Re} (a-c-d)=0,$ $f\in L_{\upsilon ,2} ,$ то имеет место следующая формула обращения{\/\rm:} 
\begin{equation}
\label{vir:eq16}
f( x )=x^{-\lambda }\frac{d}{dx}x^{\lambda +1}
\int_0^\infty G_{2,4}^{2,1} 
\left[ \begin{array}{*{20}c}
 {\left. \begin{array}{*{20}c}
 {-\lambda , 1-a} \hfill \\
 {c-1, d-1, 0, -\lambda -1} \hfill \\
\end{array}  \right|} \hfill & {xt} \hfill \\
\end{array}  \right] \bigl( { }_1F_2 f \bigr) ( t )dt.
\end{equation}

\end{newthm}
