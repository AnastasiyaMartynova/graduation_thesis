\section{Введение}

В~[1--3] начат цикл статей по методам и~инструментальным программным средствам 
аналитического моделирования процессов в~СтС со сложными 
нелинейностями, описываемыми такими специальными функциями, как Бесселевы целого 
и~дробного порядка, а~также связанными с~ними функциями.

Продолжим~[1--3] и~рассмотрим типовые эллиптические нелинейности Якоби (ЭНЯ); 
МСЛ ЭНЯ (см.\ разд.~2 и~приложения~П1 и~П2); 
МНА и~МСЛ 
для сложных СтС с~ЭНЯ (см.\ разд.~3).
В качестве тестового примера рассмотрим динамику эллиптического осциллятора 
Якоби (ЭОЯ) в~стохастической среде (см.\  разд.~4). В~заключении сформулированы основные 
выводы и~возможные обобщения.

\section{Эллиптические нелинейности Якоби и~их статистическая линеаризация}

Как известно~[4, 5], эллиптическая функция Якоби $\rE (u)\hm = \pq u$ определяется 
через эллиптический интеграл первого рода (см.\ приложение~П1):
\begin{equation*}
F\left(\vrp, k^2\right) = \int\limits_0^\vrp \fr{dt}{\sqrt{1- k^2 \sin^2 t}}\,,
%\label{e2.1-s1}
\end{equation*}
где $k$~--- модуль интеграла; $\vrp$~--- амплитуда. Пусть
\begin{equation*}
u= \int\limits_0^\vrp \sqrt{1-k^2 \sin^2 t}\,dt\,,\quad  
\vrp = \mathrm{am} u\,.
%\label{e2.2-s1}
\end{equation*}
Тогда~12~эллиптических функций Якоби определяются следующими формулами:
    \begin{equation*}
    \sn u =\sin\vrp\,;\enskip \cn u = \cos \vrp\,;\enskip 
    \dn  u = \Delta (\vrp) =  \sqrt{1-k^2 \sin^2 t}\,\,;
   % \label{e2.3-s1}
    \end{equation*}
    $$
    \cd u = \fr{\cn u}{\dn u}\,; \enskip 
    \dc u= \fr{\dn u}{\cn u}\,; \enskip 
    \ns u =\fr {1}{\sn u}\,;\enskip \sd u = \fr{\sn u}{\dn u}\,; \enskip 
    \nc u = \fr{1}{\cn u}\,;
    $$
   \begin{equation*}
     \ds u = \fr{\dn u }{\sn u}\,;\enskip 
     \nd u = \fr{1}{\dn u}\,; \enskip 
     \ssc u = \fr{\sn u}{\cn u}\,; \enskip
     \cs u = \fr{\cn u }{\sn u}\,.
     %\label{e2.4-s1}
     \end{equation*}
В общем случае, если p, q, r являются любыми тремя буквами  из четырех букв~s, c, d, 
$u$, то $\pq u/\qr u$ $(\mathrm{pp}\, u\hm\equiv 1)$, $\pq u \hm=1/\qp u$.

Обозначим через~$\bK$  и~$\bK'$ полные эллиптические интегралы, когда $\vrp \hm= \pi/2$:
    \begin{equation*}
    \bK (k) = F\left(\fr{\pi}{2}\,, k\right) = \bK'\left(k'\right)\,; \enskip 
    k'=\sqrt{1-k^2}\,;\enskip 
    \bK'(k) = F\left(\fr{\pi}{2}\,,k'\right) = \bK\left(k'\right)
   % \label{e2.5-s1}
    \end{equation*}
и дополнительные параметры Якоби~$q$  и~$q'$:
    \begin{equation*}
    q= q(k^2) = \exp \left( - \fr{\pi \bK'}{\bK}\right)\,;\enskip 
    q'= q'(k^2) = \exp \left( - \fr{\pi \bK}{\bK'}\right).
%    \label{e2.6-s1}
    \end{equation*}
Тогда разложение в~ряд по параметру Якоби и~аргументу~$v$ $(v\hm= \pi u /(2 \bK))$
 имеет следующий вид:
   \begin{equation}
   \left.
   \begin{array}{rl}
\sn \left(u\mid k^2\right) &= \fr{2\pi}{k\bK} \sss_{n=0}^\infty \fr{q^{n+1/2}}{1-q^{2n+1}}
     \sin \lk (2n+1)v\rk\,;\\[9pt]
         \cn \left(u\mid k^2\right) &= \fr{2\pi}{k\bK} \sss_{n=0}^\infty \fr{q^{n+1/2}}
    {1+q^{2n+1}} \cos \lk (2n+1)v\rk\,;\\[9pt]
    \dn \left(u\mid k^2\right) &= \fr{\pi}{2\bK} +\fr{2\pi}{\bK}
    \sss_{n=0}^\infty \fr{q^{n}}{1+q^{2n}} \cos  (2nv)\,;
    \end{array}
    \right\}
    \label{e2.7-s1}
\end{equation}

\vspace*{-9pt}


    \begin{align*}
    \cd \left(u\mid k^2\right) &= \fr{2\pi}{k\bK} \sss_{n=0}^\infty 
    \fr{(-1)^nq^{n+1/2}}{1-q^{2n+1}} \cos \lk (2n+1)v\rk\,;
\\
    \sd \left(u\mid k^2\right) &= \fr{2\pi}{\sqrt{k^2(1-k^2)}\bK} 
    \sss_{n=0}^\infty \fr{(-1)^n q^{n+1/2}}{1+q^{2n+1}} \sin \lk (2n+1)v\rk\,;
\\
    \nd \left(u\mid k^2\right) &= \fr{\pi}{2\sqrt{ 1-k^2}\bK} +
    \fr{2\pi}{\sqrt{1-k^2}\bK} \sss_{n=0}^\infty \fr{(-1)^nq^{n}}{1+q^{2n}} 
    \cos (2nv)\,;
\\
    \dc \left(u\mid k^2\right) &= \fr{\pi}{k\bK}\mathrm{sec}\,v +\fr{2\pi}{ \bK} 
    \sss_{n=0}^\infty \fr{(-1)^nq^{2n+1}}{1-q^{2n+1}} \cos \lk (2n+1)v\rk\,;
\\
    \nc \left(u\mid k^2\right) &= \fr{\pi}{\sqrt{1-k^2}\bK}\mathrm{sec}\, v -
    \fr{2\pi}{\sqrt{1-k^2}\, \bK} \sss_{n=0}^\infty \fr{(-1)^nq^{2n+1}}{1+q^{2n+1}} 
    \cos \lk (2n+1)v\rk\,;
\\
   \ssc \left(u\mid k^2\right) &= \fr{\pi}{2\sqrt{1-k^2}\bK}\,\tg v + 
   \fr{2\pi}{\sqrt{1-k^2}\, \bK} \sss_{n=0}^\infty 
   \fr{(-1)^nq^{2n}}{1+q^{2n}} \sin (2nv)\,;
   \\
   \ns \left(u\mid k^2\right) &= \fr{\pi}{2\bK}\,\mathrm{cosec}\, v - 
   \fr{2\pi}{\bK} \sss_{n=0}^\infty \fr{q^{2n+1}}{1-q^{2n+1}} \sin \lk (2n+1)v\rk\,;
\\
   \ds \left(u\mid k^2\right) &= \fr{\pi}{2\bK}\,\mathrm{cosec}\, v - \fr{2\pi}{\bK} 
   \sss_{n=0}^\infty \fr{q^{2n+1}}{1+q^{2n+1}} \sin \lk (2n+1)v\rk\,;
\\
       \cs \left(u\mid k^2\right) &= \fr{\pi}{2\bK}\,\ctg v - 
       \fr{2\pi}{\bK} \sss_{n=0}^\infty \fr{q^{2n1}}{1+q^{2n}} \sin  (2nv)\,.
%       \label{e2.8-s1}
       \end{align*}

Теоремы сложения, удвоения для половинных аргументов, 
а~также формулы для производных и~интегралов даны в~приложении~П2.

Рассмотрим скалярную ЭНЯ следующего вида:
     \begin{equation}
     Z= \rE (Y,t)\,.\label{e2.9-s1}
     \end{equation}
Применим статистическую линеаризацию ЭНЯ по Казакову~[6--8] 
при действительном несимметричном $(m_y\hm\ne 0)$ гауссовском (нормальном) входном 
сигнале~$Y_t$:
    \begin{equation*}
    Y_t = Y(t) = m_y + Y_t^0\,, \enskip Y_t^0 = Y_t - m_y\,,
    %\label{e2.10-s1}
    \end{equation*}
где $m_y = \mm Y_t$~--- математическое ожидание; $D_y \hm= \mm (Y_t^0)^2$~--- 
дисперсия; $Y_t^0$~--- центрированная составляющая. 
В~соответствии с~МСЛ зависимость~(\ref{e2.9-s1}) аппроксимируется следующим выражением:
    \begin{equation*}
    Z_t = \rE_0 \left(m_y, D_y, t\right) + k_1^\re \left(m_y, D_y, t\right) Y_t^0.
   % \label{e2.11-s1}
    \end{equation*}
Здесь $\rE_0 \hm= \rE_0 (m_y, D_y, t) $ и~$k_1^{\re} \hm= k_1^\re  (m_y, D_y, t)$~--- 
коэффициенты МСЛ для~(\ref{e2.9-s1}), определяемые по формулам:
   \begin{equation}
   \left.
   \begin{array}{rl}
    \rE_0 &= \fr{1}{\sqrt{2\pi D_y}} \iin \rE (\eta, t) \exp 
    \lk - \fr{(\eta-m_y)^2}{2 D_y}\rk d\eta\,;
   \\[6pt]
        k_1^\re &= \fr{1}{\sqrt{2\pi D_y}} \iin \left(\eta- m_y\right)\rE (\eta, t) 
        \exp \lk - \fr{(\eta-m_y)^2}{2 D_y}\rk d\eta\,.
        \end{array}
        \right\}
        \label{e2.12-s1}
        \end{equation}
Для нечетных ЭНЯ первая формула~(\ref{e2.12-s1}) принимает вид:
\begin{equation*}
\rE_0 = k_0^\re m_y\,,\quad
k_0^\re = \fr{1}{m_y\sqrt{2\pi D_y}} \iin \rE (\eta, t) \exp 
\lk - \fr{(\eta-m_y)^2}{2 D_y}\rk d\eta\,.
%\label{e2.13-s1}
\end{equation*}

В случае векторных и~матричных ЭНЯ полученные формулы имеют место для соответствующих 
компонент.

В приложении~П3 приведены выражения для коэффициентов статистической линеаризации ЭНЯ, 
полученные на основе~(\ref{e2.7-s1}) при $m_y \hm=0$ и~$m_y\hm \ne 0$.


\section{Алгоритмы аналитического моделирования нормальных процессов в~стохастических
системах с~эллиптическими нелинейностями Якоби}

Уравнения конечномерных непрерывных нелинейных систем со стохастическими возмущениями 
путем расширения вектора состояния СтС могут быть записаны в~виде следующего 
векторного стохастического дифференциального уравнения Ито~[6--8]:
\begin{equation}
dY_t = a^\re\left(Y_t, t\right) dt + b^\re \left(Y_t, t\right) dW_0+ 
\iii_{R_0} c^\re \left(Y_t, t, v\right) P^0 (dt, dv)\,,\enskip Y\left(t_0\right) = Y_0\,.
\label{e3.1-s1}
\end{equation}
Здесь $Y_t$~--- $(p\times 1)$-мер\-ный вектор состояния, $Y_t \hm\in \Delta_y$ 
($\Delta_y$~--- многообразие состояний);  $a^\re\hm=a^\re(Y_t, t)$ 
и~$b^\re\hm=b^\re(Y_t, t)$~--- известные  $(p\times 1)$-мер\-ная 
и~$(p\times m)$-мер\-ная функции~$Y_t$ и~$t$;  $W_0\hm= W_0(t)$~--- 
$(r\times 1)$-мер\-ный винеровский стохастический процесс (СтП) интенсивности  
$\nu_0 \hm= \nu_0(t)$; $c^\re\hm=c^\re(Y_t, t, v)$~--- 
$(p\times 1)$-мер\-ная функция~$Y_t$, $t$ и~вспомогательного $(q\times 1)$-мер\-но\-го 
па\-ра\-мет\-ра~$v$; $\iii\nolimits_{\Delta} dP^0 (t, A)$~--- 
центрированная пуассоновская мера, определяемая следующим образом:
    $$
    \iii_{\Delta} dP^0 (t, A) = \iii_{\Delta} dP (t,A) =
    \iii_{\Delta} \nu_P (t,A) \,dt\,.
    $$
При этом принято: $\iii\nolimits_{\Delta}$~--- число скачков пуассоновского СтП
в~интервале времени  $\Delta \hm= (t_1, t_2]$; $\nu_P (t, A)$~---
интенсивность пуассоновского СтП  $P(t,A)$; $A$~--- некоторое
борелевское множество пространства~$R_0^q$ с~выколотым началом.
Начальное значение~$Y_0$ представляет собой случайную величину, не зависящую 
от приращений~$W_0(t)$ и~$P(t,A)$ на
интервалах времени, следующих за~$t_0$, $t_0 \hm\le t_1\hm\le t_2$, для
любого множества~$A$. Элементы век\-тор\-но-мат\-рич\-ных функций $a(Y_t,t)$, 
$b(Y_t,t)$ и~$c(Y_t,t,v)$ являются сложные бесселевы нелинейности дробного порядка.

В~случае аддитивных гауссовских (нормальных) и~обобщенных пуассоновских 
возмущений уравнение~(\ref{e3.1-s1}) принимает  вид~[6--8]:
\begin{equation}
\dot Y = a^\re\left(Y_t,t\right)+ b_0 (t) V\,, \enskip 
V = \dot W\,,\enskip 
Y\left(t_0\right) = Y_0\,.
\label{e3.2-s1}
\end{equation}
Здесь $W$~--- СтП с~независимыми приращениями, представляющий собой смесь 
нормального и~обобщенного пуассоновского СтП.

Если существуют конечные вероятностные моменты второго порядка для моментов 
времени~$t_1$ и~$t_2$, то уравнения МНА примут следующий вид~[6--8]:
\begin{itemize}
\item  для характеристических функций:
    \begin{equation}
    \left.
    \begin{array}{rl}
    g_1^N (\lambda;t) &=\exp \lk i\lambda^{\mathrm{T}} m_t - \fr{1}{2}\, \lambda^{\mathrm{T}} 
    K_t \lambda\rk\,;\\[12pt]
    g_{t_1, t_2}^N \left(\lambda_1, \lambda_2;t_1, t_2 \right) &=
    \exp \lk i\bar \lambda^{\mathrm{T}} \bar m_2 - \fr{1}{2}\,\bar \lambda^{\mathrm{T}}
     \bar K_2 \lambda\rk\,,
     \end{array}
     \right\}
     \label{e3.3-s1}
     \end{equation}
где
    $$
    \bar \lambda =\lk \lambda_1^{\mathrm{T}}\lambda_2^{\mathrm{T}}\rk^{\mathrm{T}}\,; 
    \enskip\bar m_2 = \lk m_{t_1}^{\mathrm{T}} m_{t_2}^{\mathrm{T}}\rk^{\mathrm{T}}\,;
    \enskip \bar K_2= \begin{bmatrix}
    K(t_1, t_1)& K(t_1, t_2)\\[3pt]
    K(t_2, t_1)& K(t_2, t_2)\end{bmatrix}\,;
    $$

\item для математических ожиданий~$m_t$, ковариационной матрицы~$K_t$ 
и~матрицы ковариационных функций $K(t_1, t_2)$:
   \begin{equation}
   \left.
   \begin{array}{c}
    \dot m_t = a_1^\re \left(m_t, K_t, t\right)\,,\qquad m_0 = m\left(t_0\right)\,;
\\[6pt]
    \dot K_t = a_2^\re \left(m_t, K_t, t\right)\,,\qquad K_0 = K\left(t_0\right)\,;
\\[6pt]
    \fr{\prt K(t_1, t_2)}{\prt t_2 }= K\left(t_1, t_2\right) a_{21} 
    \left(m_{t_2}, K_{t_2}, t_2\right)^{\mathrm{T}}\,,\qquad 
    K\left(t_1, t_1\right) = K_{t_1}\,.
    \end{array}
    \right\}
    \label{e3.4-s1}
    \end{equation}
    \end{itemize}
Здесь приняты следующие обозначения:
    \begin{equation}
    \left.
    \begin{array}{c}
    m_t = {\sf M}_{\Delta_y}^N \lk Y_t\rk \,;\enskip 
    Y_t^0 = Y_t - m_t\,;\\[6pt]
    K_t = {\sf M}_{\Delta_y}^N \lk Y_t^0 Y_t^{0\mathrm{T}}\rk \,,
    \enskip K\left(t_1, t_2\right) = {\sf M}_{\Delta_y}^N \lk Y_{t_1}^{0} Y_{t_2}^{0\mathrm{T}}\rk \,;
\\[6pt]
    a_1^\re = a_1^\re \left(m_t, K_t, t\right) = 
    \mm_{\Delta_y}^N \lk a^\re\left(Y_t, t\right)\rk\,;
\\[6pt]
    a_2^\re = a_2^\re \left(m_t, K_t, t\right) = 
    a_{21}^\re \left(m_t, K_t, t\right)+ a_{21}^\re \left(m_t, K_t, t\right)^{\mathrm{T}} 
    +a_{22}^\re\left(m_t, K_t, t\right)\,;
\\[6pt]
    a_{21}^\re = a_{21}^\re\left(m_t, K_t, t\right)=  
    \mm_{\Delta_y}^N \lk a^\re\left(Y_t, t\right) Y_{t}^{0\mathrm{T}}\rk \,;
\\[6pt]
    a_{22}^\re = a_{22}^\re\left(m_t, K_t, t\right)= 
    \mm_{\Delta_y}^N\lk  \bar\sigma^\re \left(Y_t, t\right)\rk\,;
  \\[6pt]
    \sigma^\re \left(Y_t, t\right)=b^\re\left(Y_t, t\right) 
    \nu_0(t) b^\re\left(Y_t, t\right)^{\mathrm{T}}\,;
    \\[6pt]
    \bar\sigma \left(Y_t, t\right) = \sigma^\re \left(Y_t, t\right)+
   \displaystyle \iii_{R_0^q} c^\re \left(Y_t, t, v\right) c^\re\left(Y_t, t,v\right)^{\mathrm{T}} 
    \nu_P (t, dv)\,,
    \end{array}
    \right\}
    \label{e3.5-s1}
    \end{equation}
где ${\sf M}_{\Delta_y}^N$~--- символ вычисления математического ожидания для 
нормальных распределений~(\ref{e3.3-s1}) на гладком многообразии~$\Delta_y$.

Для стационарных СтС нормальные стационарные СтП~--- если они
существуют, то  $m_t \hm= m^*$, $ K_t\hm = K^*$, $K(t_1, t_2) \hm=
k(\tau)$ $(\tau \hm= t_1\hm-t_2)$,~--- определяются уравнениями~[6--8]:
    \begin{equation}
\left.
\begin{array}{c}
    a_1^\re \left(m^*, K^*\right) =0\,;\quad a_2^\re \left(m^*, K^*\right)=0\,;
   \\
\dot k_\tau (\tau) = a_{21}^\re \left(m^*, K^*\right) {K^*}^{-1} k(\tau)\,;\quad
k(0) =K^* \enskip (\forall\,\tau >0)\,;\\[6pt]
k(\tau) = k(-\tau)^{\mathrm{T}} \enskip (\forall\,\tau <0)\,.
\end{array}
\right\}
\label{e3.6-s1}
\end{equation}
При этом необходимо, чтобы матрица  $a_{21}^\re ( m^*, K^*)\hm= a^{\re *}_{21}$ 
была асимптотически устойчивой.

В случае СтС~(\ref{e3.2-s1}) уравнения МНА переходят в~известные уравнения МСЛ
[6--8]:

\noindent
\begin{equation}
\left.
\begin{array}{c}
     \dot m_t = a_1^\re \left(m_t, K_t, t\right)\,,\enskip 
     m_0 = m\left(t_0\right)\,;\\[6pt]
     \dot K_t = k_1^{\re a} \left(m_t, K_t, t\right) K_t + 
     K_t k_1^{\re a} \left(m_t, K_t, t\right)^{\mathrm{T}} +
     \sigma_0(t)\,,\enskip K_0 = K\left(t_0\right)\,;
\\[6pt]
     \fr{\prt K(t_1, t_2)}{\prt t_2} = K\left(t_1, t_2\right) K_{t_2} k_1^{\re a} 
     \left(m_{t_2}, K_{t_2}, t_2\right)^{\mathrm{T}}\,,\enskip 
     K\left(t_1, t_2\right) = K_{t_1}\,,
     \end{array}
     \right\}
     \label{e3.7-s1}
     \end{equation}
где
    \begin{gather*}
    a^\re\left(Y_t,t\right) = a_0^\re \left(m_t, K_t\right) + k_1^{\re a} 
    \left(m_t, K_t\right) Y_t^0\,;\\
    k_1^{\re a}  \left(m_t, K_t, t\right) =\lk \left(\fr{\prt}{\prt m_t} \right)
    a_0^\re \left(m_t, K_t, t\right)^{\mathrm{T}}\rk^{\mathrm{T}}\,;
\\
    b^\re\left(Y_t,t\right) = b_0^\re (t)\,;\quad 
    \sigma^\re\left(Y_t, t\right)= b_0^\re(t) \nu(t) 
    b_0^\re(t)^{\mathrm{T}} = \sigma_0^\re(t)\,.
    \end{gather*}

При условии асимптотической устойчивости матрицы $k_1^a (m^*, K^*)$ 
для стационарных СтС~(\ref{e3.2-s1}) в~основе МСЛ лежат уравнения~(\ref{e3.6-s1}), 
записанные в~виде:
    \begin{equation}
    \left.
    \begin{array}{c}
    a_0^\re \left(m^*, K^*\right) =0\,; \enskip 
    k_1^{\re a} \left( m^*,  K^*\right)  K^* +  K^* k_1^{\re a} \left( 
    m^*,  K^*\right)^{\mathrm{T}} +\bar \sigma_0 =0\,;
\\[6pt]
\dot k_\tau (\tau) = k_1^{\re a} \left( m^*, K^*\right)k(\tau)\,;\quad 
k(0) =K^* \enskip (\forall\,\tau >0)\,;\\[6pt] 
k(\tau) = k (-\tau)^{\mathrm{T}} \enskip (\forall\,\tau <0)\,.
\end{array}
\right\}
\label{e3.8-s1}
\end{equation}

Таким образом, приходим к следующим алгоритмам МНА и~МСЛ.
\begin{enumerate}[1.]
\item Если существуют интегралы~(\ref{e3.5-s1}), то 
уравнения~(\ref{e3.3-s1}) и~(\ref{e3.4-s1}) 
лежат в~основе нестационарных алгоритмов 
методов аналитического моделирования (МАМ) для негауссовских СтС~(\ref{e3.1-s1}), 
а~уравнения~(\ref{e3.7-s1})~---  для негауссовских  СтС~(\ref{e3.2-s1}).

\item Если СтС~(\ref{e3.1-s1}) и~(\ref{e3.2-s1}) 
стационарны и~существует стационарный нормальный процесс и~матрица~$a_{21}^*$ 
асимптотически устойчива, то уравнения~(\ref{e3.6-s1}) и~(\ref{e3.8-s1}) 
лежат в~основе стационарных алгоритмов МАМ.
\end{enumerate}


Наконец, отметим, что для гауссовских СтС алгоритмы упрощаются, если 
принять  $c^\re(Y_t, t, v)\hm\equiv 0$ в~(\ref{e3.1-s1}) 
и~$V\hm=V_0$, $\nu^V \hm= v^{V_0}$ в~(\ref{e3.2-s1}).

Для алгоритмизации МНА необходимо уметь вычислять следующие интегралы:
\begin{align}
I_0^{\re a} &= I_0^{\re a} \left(m_t, K_t, t\right) = a_1^\re \left(m_t, K_t, t\right)= 
\mm_{\Delta_y}^N \lk a^\re\left(Y_t, t\right)\rk\,;
\label{e3.9-s1}
\\
I_1^{\re a} &= I_1^{\re a} \left(m_t, K_t, t\right)= a_{21}^\re\left(m_t, K_t, t\right)= 
\mm_{\Delta_y}^N \lk a^\re\left(Y_t , t\right) Y_t^{0\mathrm{T}}\rk\,;\notag
%\label{e3.10-s1}
\\
I_0^{\re \bar \sigma} &= I_0^{\re \bar \sigma} \left(m_t, K_t, t\right) = 
a_{22}^\re\left(m_t, K_t, t\right) = \mm^N \lk \bar \sigma^\re \left(Y_t, t\right)\rk\,,\notag
%\label{e3.11-s1}
\end{align}
а для МСЛ достаточно вычислить первый интеграл в~(\ref{e3.9-s1}), 
причем интеграл~$I_1^{\re a}$ вычисляется по формуле~[6--8]:
    \begin{equation*}
    k_1^{\re a} = k_1^a \left(m_t, K_t, t\right)=
    \lk \left( \fr{\prt}{\prt m_t}\right) I_0^{\re a}
     \left(m_t, K_t, t\right)^{\mathrm{T}}\rk^{\mathrm{T}}\,.
     %\label{e3.12-s1}
     \end{equation*}

Уравнения МНА (МСЛ) содержат интегралы $I_0^{\re a}$, $I_1^{\re a}$ и~$I_0^{\re\sigma}$ 
в~виде соответствующих коэффициентов. Поэтому процедура вычисления интегралов должна
 быть согласована с~методом численного решения обыкновенных дифференциальных 
 уравнений для~$m_t$, $K_t$ и~$K(t_1, t_2)$. Эти коэффициенты допускают 
 дифференцирование по~$m_t$ и~$K_t$, так как под интегралом стоит сглаживающая 
 нормальная плот\-ность.


В~\cite{9-s1} изложены алгоритмы дискретного аналитического и~статистического
моделирования типовых распределений (в~том числе нормальных) в~нелинейных
СтС на многообразиях. Алгоритмы дискретного аналитического и~статистического
моделирования для СтС со сложной эллиптической нелинейностью, а~так\-же смешанные алгоритмы различной
степени точности относительно шага интегрирования также представлены в~\cite{9-s1}.

\section{Эллиптический осциллятор Якоби в~стохастической среде}

Рассмотрим нелинейную двумерную СтС, описываемую следующими уравнениями:
\begin{equation}
\left.
\begin{array}{rlrl}
\dot Y_1 &= Y_2\,; &\quad
\dot Y_2 &=-\w_0^2 \rE \left(Y_1\right) + L_0 - 2 \eps \w_0 Y_2 + h^2 V\,; \\[6pt]
    Y_1\left(t_0\right) &= Y_{10}\,;&\quad
     Y_2 \left(t_0\right) &= Y_{20}\,.
     \end{array}
     \right\}
     \label{e4.1-s1}
     \end{equation}
Здесь $Y_1$ и~$Y_2$~--- координата и~скорость; $\rE (Y_1)$~--- 
эллиптическая функция Якоби, определяющая ЭОЯ; 
$L_0$~--- постоянная или медленно меняющаяся на интервале времени  
$T\hm= 2\pi \w_0^{-1}$ величина; $-2\eps \w_0 Y_2$~--- 
линейная диссипативная составляющая; $h^2 V$~--- 
стохастическая составляющая, представляющая собой нормальный белый шум 
интенсивности~$h^2\nu$; $\eps$, $h$ и~$\w_0^2$~--- постоянные параметры;  
$Y_{10}$ и~$Y_{20}$~--- нормальные независимые начальные координата и~скорость.

При $L_0=0$, $\eps\hm =0$, $ h\hm=0$ и~$\rE (Y_1)\hm =\sn Y_1$  уравнения 
фазового портрета ЭОЯ и~интеграл сохранения энергии, если учесть выражение для 
интеграла от~$\sn  Y$ (см.\ приложение~П2), имеют следующий вид:
    \begin{align}
    \fr{dY_2}{dY_1} &=- \fr{\w_0^2 \sn Y_1 }{Y_2}\,;\notag
%    \label{e4.2-s1}
\\
    {\cal H} ={\cal H}\left(Y_1, Y_2\right) &= \fr{Y_2^2}{2}-\w_0^2 \Pi^\re (Y_1) = 
    {const}\,;
    \label{e4.3-s1}
\\
\Pi^\re \left(Y_1\right) &= \fr{1}{k} \ln \left(\dn Y_1 - k \cn Y_1\right)\,.
\label{e4.4-s1}
\end{align}

При $\eps>0$ и~$L_0\hm=0$ по теореме Гиббса~[6--8]
уравнения~(\ref{e4.1-s1}) допускают стационарный стохастический режим с~одномерной
 плот\-ностью:
\begin{equation}
    f_1^* \left(y_1, y_2\right) = c \exp \lk -\fr{4\eps\w_0^2}{\nu}\, {\cal H}
    \left (y_1,y_2\right)\rk\,.
    \label{e4.5-s1}
    \end{equation}

Из~(\ref{e4.3-s1}) и~(\ref{e4.4-s1}) следует, что точное одномерное распределение ЭОЯ 
по ско\-рости~$Y_2$ является гауссовским, а~по координате~$Y_1$ негауссовским 
следующего вида:
\begin{equation}
f_1^* \left(y_1\right) = c_1 \exp \lk - \fr{4\eps\w_0^2}{\nu} \Pi^\re 
\left(y_1\right)\rk\,,\label{e4.6-s1}
\end{equation}
где функция $\Pi^\re (y_1)$ определена~(\ref{e4.4-s1}).

Аналогично, пользуясь формулами приложения~П2 для $\rE(Y_1) \hm= \cn Y_1$ 
и~$\dn Y_1$,  получаем формулы~(\ref{e4.5-s1}) и~(\ref{e4.6-s1}), если учесть соотношения:
\begin{equation*}
\Pi^\cn \left(Y_1\right) = k^{-1} \mathrm{arccos}\, \left(\dn Y_1\right)\,;\quad 
\Pi^\dn \left(Y_1\right) =  \mathrm{arcsin}\, \left(\sn Y_1\right)\,.
%\label{e4.7-s1}
\end{equation*}

Применяя уравнения разд.~3 для $D_1 \hm= K_{11}$, $D_2 \hm= K_{22}$, $K_{12}$ 
и~различных ЭНЯ, получим искомые уравнения аналитического моделирования ЭОЯ:
\begin{gather*}
    \dot m_1 =0\,, \qquad m_1(t_0) = m_{10}\,;\\
\dot m_2 =-\w_0^2 \rE_0 \left(m_1, D_1\right) + L_0 - 2 \eps \w_0 m_2\,,\quad
m_2 \left(t_0\right)=m_{20}\,;
%\label{e4.8-s1}
\\
\dot D_1 =2K_{12}\,, \qquad D_1\left(t_0\right) = D_{10}\,;\\
\dot D_2 =-2\lk \w_0^2 \rE_0 \left(m_1, D_1\right) K_{12}+2\eps \w_0 D_2\rk + h^2\nu\,,
\qquad D_2 \left(t_0\right)=D_{20}\,;\\
\dot K_{12} = D_2 - \w_0^2 k_1^\re \left(m_1, D_1\right) D_1 - 2 \eps \w_0 K_{12}\,,
\qquad K_{12}\left(t_0\right)=K_{120}\,.
%\label{e4.9-s1}
\end{gather*}
Коэффициенты $\rE_0 \hm= \rE_0 (m_1, D_1)$ и~$k_1^\re\hm=k_1^\re  (m_1, D_1)$ 
для типовых ЭНЯ приведены в~приложении~П3.

В стационарном режиме при $L_0\hm=0$, $ \nu\hm=\nu^*$ для $ \rE(Y_1)\hm =\sn Y_1$ имеем:
\begin{equation*}
\rE_0  \left(0, D_1\right)=0\,;\enskip 
m_1^* =0\,; \enskip 
m_2^*=0\,;\enskip 
K_{12}^* =0\,;\enskip 
D_2^* = \fr{h^2\nu^*}{4 \eps \w_0}\,,
%\label{e4.10-s1}
\end{equation*}
    а~$D_1^*$ определяется из уравнения:
    \begin{equation*}
    k_1^\sn \left(0, D_1^*\right) D_1^* =\gamma\quad
    \left(\gamma= \fr{h^2 \nu^*}{4\eps \w_0^3}\right)\,.
    %\label{e4.11-s1}
    \end{equation*}
    Здесь введены следующие обозначения:
\begin{equation*}
k_1^\sn \left(0,D_1\right)= \kappa^\sn \left(D_1\right) 
\sss_{n=0}^\infty A_n^\sn (q) \exp\lk - \alpha_n^\sn \left(D_1\right)\rk\,,
\end{equation*}
где
\begin{gather*}
\kappa^\sn \left(D_1\right) = \fr{\pi D_1}{k\bK^2}\,; \qquad 
A_n^\sn (q) =\fr{(2n+1)q^{n+1/2}}{1-q^{2n+1}}\,;\\
 \alpha_n^\sn = \left(\fr{\pi (2n+1)}{\bK}\right)^2 \fr{D_1}{8}\,;\qquad
  q=\exp\lk -\fr{\pi \bK'(k)}{\bK(k)}\rk\,.
  \end{gather*}
%  \label{e4.12-s1}


Аналогично рассматриваются ЭОЯ для других эллиптических функций Якоби (см.\ приложение~П3).

При малых $k$ и~$k'$ имеют место простые приближенные выражения для 
коэффициентов статистической линеаризации, если воспользоваться асимптотическими 
формулами приложения~П2~(\ref{p2.6})--(\ref{p2.8}).

\section{Заключение}

Разработано методическое и~алгоритмическое обеспечение аналитического моделирования 
нормальных (гауссовских) процессов в~дифференциальных 
СтС с~нелинейностями, описываемыми эл\-лип\-ти\-че\-ски\-ми функциями Якоби. 
В~качестве тестового примера рассмотрена динамика осциллятора Якоби в~стохастической 
среде.
Алгоритмы положены в~основу разрабатываемого инструментального программного 
обеспечения и~комплекса тестовых примеров.


В приложениях~П1 и~П2 приведены необходимые сведения из теории эл\-лип\-ти\-че\-ских 
интегралов и~эллиптических функций Якоби, а~в~приложении~П3~--- 
коэффициенты статистической линеаризации при $m\hm\ne 0$ и~$m\hm=0$.

Результаты допускают обобщения на системы с~другими типами 
эллиптических нелинейностей, описываемых функциями Вейерштрасса, тэта-функ\-ци\-ями и~др.

\section*{\raggedleft Приложения}


\subsection*{П1.\ Сведения из теории эллиптических интегралов первого рода}

Эллиптический интеграл первого рода $F(\vrp,k)$ и~полный эллиптический интеграл 
первого рода~$\bK(k)$ определяются по формулам:
\begin{equation*}
F(\vrp,k) = \iii_0^\vrp \fr{dt}{\sqrt{1-k^2 \sin^2 t}}\,;
%\label{p1.1}
\end{equation*}
    \begin{equation*}
    \bK(k)= F\left(\fr{\pi}{2}, k\right) = \bK'\left(k'\right)\,,\quad 
    \fr{\pi}{2} \le \bK (k) \le \infty \enskip (0\le k\le 1)\,;
    %\label{p1.2}
    \end{equation*}
\begin{equation*}
\bK'(k) = F\left( \fr{\pi}{2}, k'\right) = \bK(k')\,,
%\label{p1.3}
\end{equation*}
где $k$ $(k^2<1)$~--- модуль; $\vrp$~--- амплитуда; $k'\hm=\sqrt{1-k^2}$~--- 
дополнительный модуль; $k^2 \hm+ {k'}^2 \hm=1$.

Преобразование интервала:
\begin{equation*}
F(-\vrp, k) =-F(\vrp,k)\,,\quad 
F(n\pi \pm \vrp,k) = 2n \bK(k)\pm F(\vrp,k)\,.
%\label{p1.5}
\end{equation*}

Степенные разложения:
\begin{equation*}
\bK \left(k^2\right) = \fr{\pi}{2} \lf 1 + 
\sss_{j=1}^\infty \fr{(2j)!}{(2^j\, j!)^2}\, k^{2j}\rf\,,\enskip 
\lv k \rv<1\,;
%\label{p1.6}
\end{equation*}

\vspace*{-12pt}

\noindent
\begin{multline*}
\bK\left(k^{'2}\right) = \fr{1}{2} \ln \left(\fr{16}{{k'}^2}\right)\lf 1+ 
\sss_{j=1}^\infty \lk \fr{(2j)!}{(2^j\, j!)^2} \left({k'}^2\right)^j \rk\rf+{}\\
{}+ \sss_{j=1}^\infty\lf \lk \fr{(2j)!}{(2^j\, j!)^2}\rk^2 
\sss_{l=1}^{2j} \lk (-1)^l \fr{2}{l}\rk({k'}^2)^l\rf\,;
%\label{p1.7}
\end{multline*}

\noindent
\begin{equation*}
\bK (s) =\fr{\pi}{2}\left(1+s\right) \sss_{j=1}^\infty 
\fr{\lk(2j-1)!!\rk^2}{\lk(2j-2)!!\rk^2}\, s^{2j-\eta}\,,\quad 
s=\fr{1-k'}{1_k'}\,.
%\label{p1.8}
\end{equation*}

Основной и~дополнительный параметры Якоби~$q$ и~$q'$:
  \begin{equation*}
  q=q\left(k^2\right)=\exp \left(-\fr{\pi \bK'}{\bK}\right)\,; \enskip 
  q'=q'\left(k^2\right)=\exp \left(-\fr{\pi \bK}{\bK'}\right)\,;\enskip 
  \ln \fr{1}{q} \cdot \ln \fr{1}{q'} =\pi^2\,.
  %\label{p1.9}
  \end{equation*}

Разложение в~$q$-ряд Якоби:
\begin{equation*}
\bK (k) = \fr{\pi}{2} +2\pi \sss_{j=1}^\infty\fr{q^j}{1+q^{2j}}\,.
%\label{p1.10}
\end{equation*}

\subsection*{П2.\ Сведения из теории эллиптических функций Якоби}

Соотношения между квадратами функций Якоби:
    \begin{gather*}
    -\dn^2 u + k^{'2}=-k^2 \cn^2 u = k^2 \sn^2 u - k^2\,;\\
    -k^{'2}\nd^2 u +k^{'2}=-k^2k^{'2}\sd^2u = k^2 \cd^2u - k^2\,,
    \\
    k^{'2}\ssc^2 u +k^{'2}=k^{'2}\nc^2 u = \dc^2 u - k^2\,; \qquad
    \cs^2u +k^{'2}=\ds^2 u =\ns^2 u - k^2\,.
    %\label{p2.1}
    \end{gather*}

Теоремы сложения для функций Якоби:
    \begin{equation}
    \left.
    \begin{array}{rl}
    \sn(n+v)&=\fr{\sn\, u\,\cn \,v\,\dn \,v+\sn\, v\,\cn\, v\,\dn\, u}
    {1-k^2\sn^2u\, \sn^2 v}\,;\\[6pt]
\\
    \cn(u+v)&= \fr{\cn\, u \,\cn\, v - \sn\, u\, \dn\, u\, \sn\, v\, \dn\, v}
    {1-k^2\sn^2u\,\sn^2v}\,;
  \\[6pt]
\dn(u+v)&= \fr{\dn\,n\,\dn\,v - k^2 \sn\,u\,\cn\,u \,\sn\,v\,\cn\,v}
{1-k^2\sn^2u\,\sn^2v}\,.
\end{array}
\right\}
\label{p2.2}
\end{equation}

Формулы удвоения аргумента получаются из~(\ref{p2.2}) при $u\hm=v$.

\pagebreak

Формулы для половинных аргументов функций Якоби:
\begin{equation*}
\sn^2 \left(\fr{u}{2}\right) = \fr{1-\cn\, u}{1+\dn\, u}\,;\enskip 
\cn^2 \left(\fr{u}{2}\right) = \fr{dn\, u +\cn\,  u}{1+\dn\, u}\,;\enskip
    \dn^2\left(\fr{u}{2}\right) =\fr{k^{'2} + \dn\, u +k^2\cn\, u}{1+\dn\, u}\,.
  %  \label{p2.3}
    \end{equation*}

Производные от функций Якоби:
   \begin{alignat*}{2}
    \left(\sn\, u\right)' &= \cn\,u\,\dn \, u\,;&\quad
    \left(\cn\, u\right)' &=- \sn\,u\,\dn \, u\,;
\\
        \left(\dn\, u\right)' &=-k^2 \sn\,u\,\cn \, u\,;&\quad
        \left(\cd\, u\right)' &=- k^{'2}\sd\,u\,\nd \, u\,;
\\
    \left(\sd\, u\right)' &= \cd\,u\,\nd \, u\,;&\quad
    \left(\nd\, u\right)' &=k^2 \sd\,u\,\cd \, u\,;
\\
        \left(\dc\, u\right)' &= k^{'2}\ssc\,u\,\nc \, u\,;&\quad
        \left(\nc\, u\right)' &= \ssc\,u\,\dc \, u\,;
\\
    \left(\ssc\, u\right)' &= \dc\,u\,\nc \, u\,;&\quad
    (\nc\, u)' &=- \ds\,u\,\cs \, u\,;
\\
        (\ds\, u)'& = -\cs\,u\,\ns \, u\,;&\quad
        (\cs\, u)' &=- \ns\,u\,\ds \, u\,.
%        \label{p2.4}
        \end{alignat*}

Интегралы от функций Якоби:
   \begin{alignat*}{2}
    \int \sn\, u\, du &= k^{-1} \ln (\dn\, u - k\,\cn\, u)\,;&\enskip 
    \int \cn\, u\, du &= k^{-1} \mathrm{arccos}\,(\dn\, u )\,;
\\
    \int \dn\, u\, du &= \mathrm{arcsin}\, (\sn\, u )\,;&\enskip 
    \int \cd\, u\, du &= k^{-1} \ln(\nd\, u +k\,\sd\,u))\,;
\\
     \int \sd\, u\, du &= \left(kk'\right)^{-1} \mathrm{arsin} 
     \left( - k\right)^{-1}\,\cd\, u)\,;&\enskip 
     \int \nd\, u\, du &= \left(k'\right)^{-1} \mathrm{arccos}\,(\cd\, u )\,;
\\
     \int \dc\, u\, du& = \ln (\nc\, u +\ssc\, u)\,;&\enskip 
     \int \nc\, u\, du &= \left(k'\right)^{-1} \ln(\dc\, u +k'\ssc\,u)\,;
\\
      \int \ssc\, u\, du &= \left(k'\right)^{-1} \ln 
      \left(\dc\, u + k'\,\nc\, u\right)\,;&\enskip 
      \int \nc\, u\, du &=\ln(\ds\, u -\cs\,u)\,;
\\
      \int \ds\, u\, du &=  \ln (\ns\, u - \cs\, u)\,;&\enskip
       \int \cs\, u\, du &=\ln(\ns\, u -\ds\,u)\,.
       %\label{p2.5}
       \end{alignat*}

Асимптотические представления $\sn\, u$, $\cn\,u$ и~$\dn\,u$ при малых~$k$ и~$k'$:
\begin{equation}
\left.
\begin{array}{rl}
    \sn\,u&\approx \sin u -\fr{1}{4}\, k^2 (u-\sin u \cos u)\cos u\,;\\[6pt]
    \sn\,u&\approx \mathrm{th}\, u +\fr{1}{4}\, k' 
    (\mathrm{sh}\, u \,\hbox{ch}\, u-u)\mathrm{sech}^2 u;
    \end{array}
    \right\}
\label{p2.6}
\end{equation}

\noindent
\begin{equation}
\left.
\begin{array}{rl}
   \cn\,u&\approx \cos u -\fr{1}{4}\, k^2 (u-\sin u \cos u)\sin u\,;\\[6pt]
   \cn\,u&\approx \mathrm{sech}\, u -\fr{1}{4}\, k^{'2} (\mathrm{sh}\, u \,\mathrm{ch}\, u-u)\hbox{th}\, u\ \hbox{sech}\, u\,;
   \end{array}
    \right\}
    \label{p2.7}
    \end{equation}
    
    \noindent
\begin{equation}
\left.
\begin{array}{rl}
\dn\,u&\approx 1- \fr{1}{2}\, k^2 \sin^2 u \,;\\[6pt]
    \dn\,u&\approx \mathrm{sech}\, u +\fr{1}{4}\,k^{'2} 
    (\mathrm{sh}\, u \,\mathrm{ch}\, u+u)\mathrm{th}\, u\ \mathrm{sech}\, u\,;
    \end{array}
        \right\}
        \label{p2.8}
        \end{equation}
        
        \noindent
\begin{equation}
\left.
\begin{array}{rl}
    \mathrm{arcsin}\,x -\mathrm{arcsin}\,y &=\mathrm{arcsin}\,
    \left(x\sqrt{1-y^2}-y\sqrt{1-x^2}\right)\,;\\
    \mathrm{arccos}\,x- \mathrm{arccos}\,y&= \mathrm{arccos}\,\left(
    xy+\sqrt{(1-x^2)(1-y^2)}\right)\,.
    \end{array}
    \right\}
    \label{p2.9}
    \end{equation}

\subsection*{П3.\ Коэффициенты статистической линеаризации эллиптических\newline
\hphantom{П3.\ }нелинейностей Якоби}

    \vspace*{3pt}

При $m=0$, пользуясь~(\ref{p2.7}) и~табличными интегралами
     \begin{align*}
      \iii_0^\infty e^{-p^2x^2} \cos ax \,dx &=\fr{1}{2p}
      \sqrt{\pi}\exp\left(-\fr{a^2}{4p^2}\right)\,;\\
\iii_0^\infty xe^{-p^2x^2} \sin ax\, dx &=\fr{a}{2p^3} \sss_{l=1}^\infty 
       \fr{1}{(2l-1)!!}\left(-\fr{a^2}{2p^2}\right)^{l-1}\,;
%\label{p3.1}
\\
\iin e^{-p^2x^2} \sin \left[a(x+x_0)\right]\, dx& =
\fr{\sqrt{\pi}}{p}\exp\left(-\fr{a^2}{4p^2}\right)\sin a x_0\,;\\
\iin xe^{-p^2x^2} \cos \left[a(x+x_0)\right] dx &=
\fr{\sqrt{\pi}}{p} \exp\left(-\fr{a^2}{4p^2}\right)\cos a x_0\,;
%\label{p3.2}
\\
    \iii_0^\infty xe^{-p^2x^2} \sin ax\, dx &=\fr{a\sqrt{\pi}}{4p^3}
    \exp\left(-\fr{a^2}{4p^2}\right)\,;\\
    \iii_0^\infty xe^{-p^2x^2} \cos ax\, dx &=\fr{1}{2p^2}-\fr{a}{4p^3}
    \sss_{l=0}^\infty \fr{(-1)^l l!}{(2l+1)!} \left(\fr{a}{p}\right)^{2l+2}\,,
%\label{p3.3}
\end{align*}
соответственно для $\rE (Y) \hm= \sn\, Y$, $\cn\, Y$ и~$\dn \, Y$ получаем:
   \begin{equation}
    \rE_0^\sn =0\,;\quad 
    k_1^\sn (0,D) = \kappa^\sn \sss_{n=0}^\infty \mathrm{A}_n^\sn 
    \exp\left(-\alpha_n^\sn\right)\,,
    \label{p3.4}
    \end{equation}
где
    \begin{equation*}
    \kappa^\sn =\fr{\pi D}{k\bK}\,,\quad \mathrm{A}_n^\sn =(2n+1) \fr{q^{n+1/2}}
    {1-q^{2n+1}}\,;
    \end{equation*}
    \begin{equation}
     \rE_0^\cn(0,D) = \kappa^\cn \sss_{n=0}^\infty \mathrm{A}_n^\cn \exp
     \left(-\alpha_n^\cn\right)\,,\quad k_1^\cn (0,D)=0\,,
     \label{p3.5}
     \end{equation}
     где
\begin{equation*}
\kappa^\cn =\fr{2\pi}{k\bK}\,,\quad \mathrm{A}_n^\cn = \fr{q^{n+1/2}}{1+q^{2n+1}}\,;
\end{equation*}
     \begin{equation}
     \rE_0^\dn (0,D) = \fr{\pi}{2\bK}+\kappa^\dn \sss_{n=1}^\infty \mathrm{A}_n^\dn 
     \exp\left(-\alpha_n^\dn\right)\,,\quad k_1^\dn =0\,,
     \label{p3.6}
     \end{equation}
     где
\begin{equation*}
\kappa^\dn =\fr{\pi}{\bK}\,, \enskip
\mathrm{A}_n^\dn = \fr{q^{n}}{1+q^{2n}}\,,\enskip 
\alpha_n^\dn =\fr{\pi^2 D n^2}{2 \bK^2}\,.
\end{equation*}
При $m\ne 0$, пользуясь~(\ref{e2.7-s1}) и~известными табличными интегралами
\begin{multline}
 S^{\alpha,\beta,\gamma}_{n,\w,x_0} =\iin x^n \exp\lk - 
 \left(\alpha x^2 +\beta x +\gamma\right)\rk \sin \left(\w x+x_0\right)\, dx ={}\\
{}=-\left(\fr{-1}{2\alpha}\right)^n \sqrt{\fr{\pi}{\alpha}}\, \exp\left(
\fr{\beta^2 -\w^2}{4\alpha} -\gamma\right)
\sss_{j=0}^{\mathrm{E}(n/2)} \fr{n!}{(n-2l)!l!}\times{}\\
{}\times \alpha^l \sss_{j=0}^{n-2l} 
\left(\fr{n-2l}{j}\right)\beta^{n-2l-j} \w^j \sin \left(
\fr{\w\beta}{2\alpha} - x_0 + \fr{\pi}{2}\, j\right)\,,
\label{p3.7}
\end{multline}

\vspace*{-12pt}

\noindent
\begin{multline}
C^{\alpha,\beta,\gamma}_{n,\w,x_0} =\iin x^n \exp\lk - 
\left(\alpha x^2 +\beta x +\gamma\right)\rk \cos \left(\w x+x_0\right) \,dx = {}\\
{}=\fr{(-1)^n}{2\alpha}\sqrt{\fr{\pi}{\alpha}} \exp\left(\fr{\beta^2 -\w^2}{4\alpha} -
\gamma\right)
\sss_{j=0}^{\mathrm{E}\,(n/2)} \fr{n!}{(n-2l)!l!} \times{}\\
{}\times \alpha^l 
\sss_{j=0}^{\mathrm{E}\,(n/2)} \left(\fr{n-2l}{j}\right)\beta^{n-2l-j} 
\w^j \cos \left(\fr{\w\beta}{2\alpha} - x_0 + \fr{\pi}{2}\,j\right)\,,
\label{p3.8}
\end{multline}
где $\begin{pmatrix}n\\ m\end{pmatrix} =
(n(n-1)\cdots (n-m))/m!$, придем к следующим результатам:
   \begin{equation}
    \hspace*{-2mm}\rE_0^\sn (m,D) =\kappa_0 (D) \sss_{n=0}^\infty \fr{q^{n+1/2}}{1-q^{2n+1}}\,
     S_{0,\W_{2n+1},0}^{\alpha,\beta,\gamma} (m,D)\,,\enskip 
     k_1^\sn (m,D) = \fr{\partial\rE_0^\sn}{\partial m} \left(m,D\right),\!\!
       \label{p3.9}
    \end{equation}
    где
     \begin{equation*}
     \kappa_0(D)=\kappa_0^\sn(D)=\sqrt{\fr{2\pi}{D}}\, \fr{\pi}{k\bK}\,;
        \end{equation*}
        
        \noindent
    \begin{equation}
    \hspace*{-2mm}\rE_0^\cn (m,D) =\kappa_0 (D) \sss_{n=0}^\infty \fr{q^{n+1/2}}{1+q^{2n+1}}\,
     C_{0,\W_{2n+1},0}^{\alpha,\beta,\gamma} (m,D)\,,\enskip
      k_1^\cn (m,D) = \fr{\partial\rE_0^\cn}{\partial m} \left(m,D\right),\!\!\!
      \label{p3.10}
      \end{equation}
      где
      \begin{equation*}
      \kappa_0(D)=\kappa_0^\cn(D)=\sqrt{\fr{2\pi}{D}}\, \fr{1}{k\bK}\,;
            \end{equation*}
            
            \vspace*{-12pt}
            
            \noindent
\begin{multline}
    \rE_0^\dn (m,D) =\fr{\pi}{2\bK} +\kappa_0^\dn (D) \sss \fr{q^{n}}{1+q^{2n}}\,
    C_{0,\W_{2n},0}^{\alpha,\beta,\gamma} (m,D)\,,\\
     k_1^\cn (m,D) = \fr{\partial\rE_0^\cn}{\partial m} \left(m,D\right)\,,
      \label{p3.11}
    \end{multline}
    где
     \begin{equation*}
     \kappa_0^\dn(D)=\sqrt{\fr{\pi}{D}}\,\fr{1}{\bK}\,.
         \end{equation*}
Здесь функции~$S$ и~$C$ с~соответствующими индексами определены~(\ref{p3.7}) и~(\ref{p3.8}), 
а
\begin{equation*}
\W_{2p+1} =\fr{2\pi (2p+1)}{2\bK}\,;\enskip 
\W_{2p}= \fr{2\pi p}{\bK}\,;\enskip 
\alpha= \fr{1}{2D}\,;\enskip 
\beta= - \fr{m}{D}\,;\enskip \gamma = \fr{m^2}{2D}\,. 
%\label{p3.12}
\end{equation*}
