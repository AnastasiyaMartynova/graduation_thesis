\maketitle


В работе построено замкнутое решение осесимметричной динамической задачи для спирально"=многослойного цилиндра, который формируется путём навивки 
предварительно натянутой металлической ленты на упругоподатливую оправку. 
В~отличие от [1] в~настоящей работе учитываются силы вязкого сопротивления.

Рассматриваются случай плоской деформации, а~также плоского напряжённого состояния при наличии осевой симметрии. 

Полый цилиндр подвержен действию произвольной радиально"=симметричной нагрузки, приложенной как на внешней, так и~внутренней поверхностях.

Используя уравнения движения осесимметричной пространственной задачи теории упругости в~перемещениях, краевую задачу осесимметричных колебаний 
спирально"=многослойного неоднородного цилиндра можно сформулировать в~следующем виде: 
\begin{multline}
\label{epishkin:eq1}
\frac{\partial^2 U^{\text{н}}(r, t)}{\partial r^2}+\frac{n+1}{r}\frac{\partial U^{\text{н}}(r,t)}{\partial r}-\frac{1-n\beta }{r^2}U^{\text{н}}(r,t)-\frac{1}{C_1^2 
}\frac{\partial ^2U^{\text{н}}(r,t)}{\partial t^2}= \\ =(\alpha +n\beta 
)\frac{T(r,t)}{E\delta r}-\frac{\beta }{E\delta }\frac{\partial 
T(r,t)}{\partial r}
\end{multline} 
при начальных и граничных условиях соответственно:
\begin{equation}
\label{epishkin:eq2}
U^{\text{н}}(r, 0)=f^{\text{н}}(r), \quad  \frac{\partial U^{\text{н}}(r, 0)}{\partial t}=g^{\text{н}}(r), \quad \text{при} \quad t=0, \; a\le r\le b;
\end{equation}
\begin{equation}
\label{epishkin:eq3}
\begin{array}{cl}
\displaystyle 
D\frac{\partial U^{\text{н}}(a,t)}{\partial r}+
 \frac{F}{a}U^{\text{н}}(a,t)=A^{\text{н}}(t), & \text{ при $r=a$;} \vspace{1mm} \\ 
\displaystyle 
D\frac{\partial U^{\text{н}}(b,t)}{\partial r}+
 \frac{F}{b}U^{\text{н}}(b,t)=B^{\text{н}}(t), & \text{ при $r=b$.}\\ 
 \end{array}
\end{equation}
Здесь 
$a$, $b$ ($\bar a = {a}/{b}$, $\bar b =1$) "--- внутренний и~наружный радиусы цилиндра; 
$U^{\text{н}}(r,t) $ "--- радиальное перемещение цилиндра;
$f^{\text{н}}(r)$, $g^{\text{н}}(r)$ "--- функции начальных перемещений и~начальных скоростей перемещений;
$C_1 =\sqrt {{D}/{\rho ^{\text{н}}}} $ "--- скорость волн расширения;
$\rho ^{\text{н}}=\rho \bar  r ^n$  "--- плотность материала навитой части цилиндра;
$\rho$ "--- плотность материала ленты; $\bar r = {r}/{b}$;
$n$ "--- показатель неоднородности, зависящий от упругой податливости оправки и закона натяжения ленты;
$\delta $ "--- толщина ленты;
$T(r,t) $ "--- закон натяжения ленты;
\begin{equation}
\label{epishkin:eq4}
 A^{\text{н}}(t)=\frac{A(t)}{\bar  a ^{\text{н}}}-\frac{FT(a,t)}{E\delta }, \quad 
 B^{\text{н}}(t)=\frac{B(t)}{b^{\text{н}}}-\frac{FT(b,t)}{E\delta };
\end{equation}
$A(t)$, $B(t)$ "--- динамическое радиальное давление на внутренней и~внешней поверхностях  цилиндра;
$\beta ={F}/{D}$, $\alpha =1-\beta $ "--- безразмерные коэффициенты, причём $F$ и $D $ определяются следующими выражениями:

\begin{itemize}
\item в случае плоской деформации
\begin{equation*}
%\label{epishkin:eq5}
%\begin{array}{l}
 D=\lambda +2\mu =\frac{E(1-\nu )}{(1-\nu )(1-2\nu )}, \quad
 F=\lambda =\frac{E\nu }{(1+\nu )(1-2\nu )}, \\ 
 %\end{array}
\end{equation*}

\item в случае плоского напряжённого состояния
\begin{equation*}
%\label{epishkin:eq6}
%\begin{array}{l}
 D=\frac{4\mu (\lambda +\mu )}{\lambda +2\mu }=\frac{E}{1-\nu ^2}, \quad
 F=\frac{2\mu \lambda }{\lambda +2\mu }=\frac{E\nu }{1-\nu ^2}, 
 %\end{array}
\end{equation*}
где
$\lambda$, $\mu$ "--- постоянные Ламе, а $E$, $\nu$ "--- модуль упругости и коэффициент Пуассона материала ленты.
\end{itemize}

В соответствии с [2] начально-краевая задача (\ref{epishkin:eq1})--(\ref{epishkin:eq3}) %для функции $W^{\text{н}}(r,t)$ 
приводится к~стандартной форме. Для этой цели для $U^{\text{н}}(r,t)$ вводим такое представление:
\begin{equation}
\label{epishkin:eq7}
U^{\text{н}}(r,t)=g_1^{\text{н}} (r)  A^{\text{н}}(t)+g_2^{\text{н}} (r)  B^{\text{н}}(t)+W^{\text{н}}(r,t),
\end{equation}
где $W^{\text{н}}(r,t)$, $g_1^{\text{н}} (r)$, $g_2^{\text{н}} (r)$ "--- дважды дифференцируемые 
функции определяемые ниже.

Подставляя выражение (\ref{epishkin:eq7}) в граничные условия (\ref{epishkin:eq3}) и требуя выполнения 
равенств
\begin{equation}
\label{epishkin:eq8}
\begin{array}{c}
g_1^{\text{н}} (b)=g_2^{\text{н}} (a)=g_1^{\text{н}\prime}  (a)=g_1^{\text{н}\prime}  (b)=g_2^{\text{н}\prime}  
(a)=g_2^{\text{н}\prime}  (b)=0; \\
g_1^{\text{н}} (a)={a}/{F}, \quad
g_2^{\text{н}} (b)={b}/{F},
\end{array}
\end{equation}
приходим к однородным граничным условиям для функции $W^{\text{н}}(r,t)$:
\begin{equation}
\label{epishkin:eq9}
\begin{array}{cl}
\displaystyle
 D\frac{\partial W^{\text{н}}(a,t)}{\partial r}+\frac{F}{a}W^{\text{н}}(a,t)=0, &\text{ при $r=a$;} \vspace{1mm} \\ 
\displaystyle
 D\frac{\partial W^{\text{н}}(b,t)}{\partial r}+\frac{F}{b}W^{\text{н}}(b,t)=0, &\text{ при $r=b$.} \\ 
 \end{array}
\end{equation}
С учетом (\ref{epishkin:eq7}) дифференциальное уравнение (\ref{epishkin:eq1}) и начальные условия (\ref{epishkin:eq2}) 
записываются следующим образом:
\begin{equation}
\label{epishkin:eq10}
\frac{\partial ^2W^{\text{н}}(r,t)}{\partial r^2}+\frac{n+1}{r}\frac{\partial 
W^{\text{н}}(r,t)}{\partial r}-\frac{1-n\beta }{r^2}W^{\text{н}}(r,t)-\frac{1}{C_1 
^2}\frac{\partial ^2W^{\text{н}}(r,t)}{\partial t^2}=p^{\text{н}}(r,t);
\end{equation}
\begin{equation}
\label{epishkin:eq11}
W^{\text{н}}(r,0)=G_1^{\text{н}} (r,0), \quad \frac{\partial W^{\text{н}}(r,0)}{\partial t}=G_2^{\text{н}} (r,0), \quad \text{при} \quad t=0.
\end{equation}
Здесь
\begin{multline}
\label{epishkin:eq12}
p^{\text{н}}(r,t)=\frac{1}{C_1^2 }  \left( {g_1 ^{\text{н}}  \ddot {A}^{\text{н}}(t)+g_2^{\text{н}}  \ddot {B}^{\text{н}}} \right)+
\\+
\Bigl( \frac{1-n\beta }{r^2}  g_1^{\text{н}} (r)-\frac{n+1}{r}g_1^{\text{н}\prime}  (r)-g_1^{\text{н}\prime\prime}(r) \Bigr)  A^{\text{н}}(t)+ \\ 
  +\Bigl( \frac{1-n\beta }{r^2}  g_2^{\text{н}} (r)-\frac{n+1}{r}g_2^{\text{н}\prime} 
 (r)-g_2^{\text{н}\prime\prime}(r) \Bigr)  B^{\text{н}}(t)+
\\+
(\alpha -n\beta 
)\frac{T(r,t)}{E\delta r}\frac{\beta }{E\delta r}-\frac{\beta }{E\delta }T'(r,t) ,
\end{multline}
\begin{equation}
\label{epishkin:eq13}
\begin{array}{c}
\displaystyle
 G_1^{\text{н}} (r,0)=f^{\text{н}}(r)-g_1^{\text{н}} (r)  A^{\text{н}}(0)-g_2^{\text{н}} (r)  B^{\text{н}}(0), \\ 
\displaystyle
 G_2^{\text{н}} (r,0)=g^{\text{н}} (r)-g_1^{\text{н}} (r)  \dot A^{\text{н}}(0)-g_2^{\text{н}} (r)  \dot B^{\text{н}}(0). \\ 
 \end{array}
\end{equation}
В соотношениях (\ref{epishkin:eq8})--(\ref{epishkin:eq13}) и~далее штрих означает дифференцирование по $r$, а~точка "--- по $t$.

Функции $g_1^{\text{н}} (r)$, $g_2^{\text{н}} (r)$ можно представить в~виде полиномов третьей степени. Тогда, используя условия (\ref{epishkin:eq8}), выражения для $g_1^{\text{н}} (r)$, $g_2^{\text{н}} (r)$ можно  получить в виде
\begin{equation}
\label{epishkin:eq14}
\begin{array}{l}
 g_1^{\text{н}} (r)=\frac{a}{F(a-b)^3}  \left[ {b^2(3  a-b)-6  b  
a  r+3  (b+a)  r^2-2  r^3} \right], \\ 
 g_2^{\text{н}} (r)=\frac{b}{F(b-a)^3}  \left[ {a^2(3  b-a)-6  a  
b  r+3  (a+b)  r^2-2  r^3} \right]. \\ 
 \end{array}
\end{equation}

Таким образом, исходная начально-краевая (\ref{epishkin:eq1})--(\ref{epishkin:eq3}) задача преобразуется к~(\ref{epishkin:eq9})--(\ref{epishkin:eq11}) для функции $W^{\text{н}}(r,t)$ с~соответствующими однородными граничными условиями. 
Её решение осуществляется структурным алгоритмом метода конечных интегральных преобразований.

Вводим на сегменте $[a, b]$ конечное интегральное преобразование с~весовой функцией $r^{n+1}$ и~неизвестным пока ядром 
$K(\xi _i r)$, т.\,е. умножаем все члены уравнения (\ref{epishkin:eq10}) и~начальных условий (\ref{epishkin:eq11}) на $r^{n+1}K(\xi _i r)$ и~интегрируем в~пределах от $a$ до~$b$:
\begin{gather}
\label{epishkin:eq15}
\bar  W ^{\text{н}}(\xi _i t)=\int\limits_a^b {r^{n+1}} W^{\text{н}}(r, t)  K(\xi _i r)dr,
\\
\label{epishkin:eq16}
\begin{array}{l}
\displaystyle 
 \int\limits_a^b {r^{n+1}}   W^{\text{н}}(r,0)  K(\xi _i r)dr=\int\limits_a^b r^{n+1}   G_1^{\text{н}}(r)  K(\xi _i r)dr, \vspace{2mm}\\ 
\displaystyle 
 \frac{d}{dt}\int\limits_a^b r^{n+1}   W^{\text{н}}(r,0)  K(\xi _i r)dr=\int\limits_a^b r^{n+1}   G_2^{\text{н}}(r)  K(\xi _i r)dr,
 \end{array}
\end{gather}
где $\xi _i $ "--- параметры, образующие счётное множество.

Выполняя теперь интегрирование по частям и используя условия
\begin{equation}
\label{epishkin:eq17}
r^{n+1}  \left[ K(\xi _i r)\frac{\partial W^{\text{н}}(r,t)}{\partial r}-K'(\xi _i r)  W^{\text{н}}(r, t)\right]=0 \quad (r=a, b);
\end{equation}
\begin{multline}
\label{epishkin:eq18}
\int\limits_a^b W^{\text{н}}(r,t)  
\Bigl[ r^{n+1}{K}''(\xi _i r)+(n+1)r^n  K'(\xi _i r)-
(1-n\beta )r^{n-1}K(\xi _i r) \Bigr]dr= \\ 
= -\xi _i^2 \int\limits_a^b r^{n+1}  W^{\text{н}}(r,t)  K(\xi _i r)dr 
\end{multline}
(условие  равенства нулю билинейной формы на концах интервала $[a, b]$ и~операционное свойство), получаем счётную систему задач Коши для 
трансформанты $\bar  {W^{\text{н}}} (\xi _i t)$
\begin{equation}
\label{epishkin:eq19}
\frac{d^2\bar  W ^{\text{н}}(\xi _i ,t )}{dt^2}+\omega _i^2   \bar  W ^{\text{н}}(\xi _i ,t)=
-C_1^2   \bar  p ^{\text{н}}( \xi _i ,t)
\end{equation}
и однородную краевую задачу для ядра преобразования $K(\xi _i r)$. 
Действительно, из (\ref{epishkin:eq18}) имеем
\begin{equation}
\label{epishkin:eq20}
K''(\xi _i r)+\frac{n+1}{r}  K'(\xi _i r)+\left(\xi _i^2 -\frac{1-n\beta }{r^2}  \right)K(\xi _i r)=0,
\end{equation}
а равенства (\ref{epishkin:eq17}) в сочетании с однородными относительно функции $W^{\text{н}}(r,t)$ условиями (\ref{epishkin:eq3}) дают
\begin{equation}
\label{epishkin:eq21}
 D  K'(\xi _i a)+\frac{F}{a}K(\xi _i a)=0,  \quad 
 D  K'(\xi _i b)+\frac{F}{b}K(\xi _i b)=0.
\end{equation}

В соответствии с методом квазинормальных координат [3] вводим в уравнение (\ref{epishkin:eq19}) силы упруговязкого сопротивления (внутреннего трения). 
Такой приём основан на экспериментально подтвержденном факте, согласно которому силы вязкого сопротивления практически не оказывают влияния на формы колебаний конструкции $K(\xi _i r)$, и их следует вводить в~математическую модель после отделения пространственной переменной $r$. Обозначив через $\gamma_{i}^{\ast }$ коэффициенты потерь для каждой моды колебаний $i$, силу внутреннего трения, следуя скорректированной частотно"=независимой гипотезе Фойхта, можно представить в виде [3] 
\begin{equation}
\label{epishkin:eq22}
R(\xi _i ,t)=\gamma _i^\ast   \omega _i   \frac{d\bar  W^{\text{н}} (\xi _i ,t)}{dt}.
\end{equation}
С учётом (\ref{epishkin:eq22}) уравнение (\ref{epishkin:eq19}) принимает вид
\begin{equation}
\label{epishkin:eq23}
\frac{d^2\bar  W^{\text{н}}(\xi _i ,t)} {dt^2}+\gamma _i^\ast  \omega _i   \frac{d\bar  W ^{\text{н}}(\xi _i ,t)}{dt}+
\omega _i^2   \bar  W ^{\text{н}}(\xi _i ,t)=-C_1^2   \bar  p ^{\text{н}}(\xi _i ,t),
\end{equation}
где  $\omega _i =C_1   \xi _i $ ($i \in \mathbb N$) "--- круговые частоты радиально"=симметричных колебаний;
$\bar  p ^{\text{н}}\left( {\xi _i ,t} \right)$, $\bar  G _1^{\text{н}} (\xi _i )$, $\bar  G _2^{\text{н}} (\xi _i )$ "--- трансформанты (13) функций (\ref{epishkin:eq12}), (\ref{epishkin:eq13}).


Начальные условия (\ref{epishkin:eq16}) записываются следующим образом:
\begin{equation}
\label{epishkin:eq24}
\bar  W ^{\text{н}}(\xi _i, 0)=\bar  G _1^{\text{н}} (\xi _i ), \quad 
\frac{d\bar  W ^{\text{н}}(\xi _i, 0)}{dt}=\bar  G _2^{\text{н}} (\xi_i ).
\end{equation}
Решением уравнения (\ref{epishkin:eq23}) при начальных условиях (\ref{epishkin:eq24}) является выражение
\begin{multline}
\label{epishkin:eq25}
 \bar  W ^{\text{н}}(\xi _i , t)=
 \exp  \left(-\frac{\gamma _i^\ast }{2}  \omega _i   t \right)  
 \Biggl[ \bar  G _1^{\text{н}} (\xi_i)  \cos  (\omega _i   \varepsilon _i   t) \frac{\bar  G _2^{\text{н}} (\xi _i )}{\omega _i }  
 \sin (\omega _i   \varepsilon _i   t)- \\ 
-\frac{C_1^2 }{\omega _i   \varepsilon _i }
\int\limits_0^t 
\bar  p^{\text{н}} ( \xi _i ,\tau )  
\exp  \left(\frac{\gamma _i^\ast }{2}  \omega _i   \tau  \right)  
\sin \bigl(\omega _i   \varepsilon _i   (t-\tau )\bigr)d\tau
\Biggr], 
\end{multline}
где $ \varepsilon _i =\sqrt {1-\frac{\left( {\gamma _i^\ast } 
\right)^2}{4}} .$

Разыскивая нетривиальные решения дифференциального уравнения (\ref{epishkin:eq20}) при 
граничных условиях (\ref{epishkin:eq21}), получаем общий интеграл в виде [4]
\begin{multline}
\label{epishkin:eq26}
K(\xi_i r)=(\xi _i r)^{-\frac{n}{2}}  \Bigl\{
J_p (\xi_i r) \bigl[ \xi _i Y_{p-1} (\xi _i a)-h_1 Y_p (\xi _i a) \bigr]-\\ -Y_p (\xi _i r)\bigl[ \xi _i J_{p-1} (\xi _i a)-h_1 J_p (\xi 
_i a)  \bigr] \Bigr\},
\end{multline}
где $J_p (\xi_i r)$, $Y_p (\xi_i r)$ "--- цилиндрические функции I и II родов порядка $$p=\sqrt {\frac{n^2}{4}-n\beta +1};$$
$\xi_i$ "--- положительные корни следующего трансцендентного 
уравнения:
\begin{multline}
\label{epishkin:eq28}
 \bigl( \xi _i  J_{p-1} (\xi _i b)-k_1 J_p (\xi _i b) \bigr)
