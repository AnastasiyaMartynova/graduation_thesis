\Section{Постановка задачи}
Рассмотрим задачу на собственные значения
\begin{gather}
    \left.\begin{array}{rl}
        \Delta\varphi+\lambda\varphi=0;\\
        \varphi\bigr|_\Gamma=0;\\
        \varphi\in C^2(\D)\cap C(\bar \D), &
        \D\subset\mathbb{R}^n.
    \end{array}
    \right\}
\label{kuzn:task1}    
\end{gather}

Множество собственных значений $\set{\lambda_k}$ этой задачи не имеет конечных предельных точек, $\lambda_k>0$, $\lambda_1$ "--- простое ($\lambda_1<\lambda_2\leqslant\lambda_3\leqslant\ldots$). Собственная функция $\varphi_1(x)$ неотрицательна, $\varphi_k$ можно выбрать вещественными и ортонормированными в $\mathcal{L}_2(\D)$.

Пусть $\Green(x, y)$ "--- функция Грина для задачи \eqref{kuzn:task1}, тогда \eqref{kuzn:task1} эквивалентно задаче на собственные значения для интегрального оператора $\IGreen$ с ядром $\Green(x, y)$:
\[
u(x)=\lambda\int_{\D}\Green(x, y)u(y)dy, \quad 
        u\in C(\bar\D).
\]

Интегральный оператор $\IGreen$ будем назвать \emph{оператором Грина}, норму в $\mathcal{L}_2(\D)$ будем обозначать $\norm{\cdot}$, в~$C(\bar \D)$ "--- $\norm{\cdot}_C$.

Для определения $\lambda_1$ можно использовать \emph{метод Келлога} \cite{kuzn:Vladimirov}. Пусть \linebreak
$\varphi^{(0)}(x){\geqslant}0$; $\norm{\varphi^{(0)}}{=}1$; $\varphi^{(p)}{=}\IGreen^p\varphi^{(0)}$;
$\varphi_{(p)}{=}{\varphi^{(p)}}/{\norm{\varphi^{(p)}}}$; $\lambda_{(p)}{=}{\norm{\varphi^{(p-1)}}}/{\norm{\varphi^{(p)}}}$. \linebreak
Тогда последовательность $\seq{\lambda_{(p)}}$ сходится, монотонно убывая, к $\lambda_1$, а последовательность $\seq{\varphi_{(p)}}$ сходится к $\varphi_1$ в $\mathcal{L}_2(\D)$ и в $C(\bar \D)$, причём справедливы оценки
\[
    \begin{aligned}
        0\leqslant \lambda_{(p)}-\lambda_1 &\leqslant \frac{\lambda_1}{2}\left(\frac{\lambda_1}{\lambda_2}\right)^{2p-2}\frac{1-c_1^2}{c_1^2},&p&=2, 3, \ldots;\\
        \norm{\varphi_{(p)}-\varphi_1}&\leqslant\left(\frac{\lambda_1}{\lambda_2}\right)^p\frac{\sqrt{1-c_1^2}}{c_1},&p&=2, 3, \ldots;\\
        \norm{\varphi_{(p)}-\varphi_1}_C&\leqslant L\lambda_2\left(\frac{\lambda_1}{\lambda_2}\right)^p\frac{\sqrt{1-c_1^2}}{c_1},&p&=2, 3, \ldots,
    \end{aligned}
\]
где $c_1=\scp{\varphi^{(0)}}{\varphi_1}$, $\displaystyle L^2=\lmax_{x\in\bar D}\int_D\left(\Green(x, y)\right)^2dy$.

Построению стохастических алгоритмов для определения $\lambda$ посвящено много работ. Наиболее общие результаты представлены в статье \cite{kuzn:MikMak}. В данной работе анализируются проблемы реализации лишь одного стохастического алгоритма. В определённом смысле она является продолжением работы \cite{kuzn:KuzSip}.

\Section{Вероятностный смысл метода Келлога для оператора Грина}
С оператором Лапласа связан винеровский процесс, а именно, $\frac{1}{2}\Delta$ является характеристическим оператором процесса. Пусть $\tau=\tau(x)$ "--- момент первого выхода винеровского процесса $\xi(t)$ из множества $\bar \D$, если $\xi(0)=x$, тогда справедлива

\begin{lemma}{}
Пусть $\varphi^{(0)}(x)=\varphi_0=\const>0${\rm ,} тогда $\varphi^{(p)}(x)=\frac{\varphi_0}{2^pp!}\expect_x\tau^p${\rm .}
\end{lemma}

\begin{proof}
    Пусть $\lambda<0$ и $\abs{\lambda}<\lambda_1$, тогда для решения краевой задачи
    \begin{gather}
        \Delta v+\lambda v=-\varphi_0, \quad v\bigr|_\Gamma=0
        \label{kuzn:ltask}
    \end{gather}
    справедливо вероятностное представление
    \[
        v(x)=\frac{1}{2}\expect_x\int_0^\tau e^{\lambda t/2}\varphi_0dt=\frac{\varphi_0}{2}\expect_x\lsum_{p=0}^\infty\left(\frac{\lambda}{2}\right)^p\frac{\tau^{p+1}}{(p+1)!}.
    \]
В последнем ряде можно поменять местами знаки математического ожидания и суммы, т.~к. $\expect_x e^{\abs{\lambda}\tau/2}<+\infty$.
    Тогда
    \[
        v(x)=\frac{\varphi_0}{2}\lsum_{p=0}^\infty\left(\frac{\lambda}{2}\right)^p\frac{\expect_x\tau^{p+1}}{(p+1)!}.
    \]

    С другой стороны, задача \eqref{kuzn:ltask} эквивалентна интегральному уравнению $v(x)=\lambda\IGreen v+\IGreen \varphi_0$, которое имеет решение, представимое при малых $\lambda$ рядом
    \begin{equation*}
        v(x)=\lsum_{p=0}^{\infty}\lambda^p\IGreen^{p+1}\varphi_0.
    \end{equation*}
Сравнивая коэффициенты при $\lambda^p$, имеем $\varphi^{(p)}=\IGreen^p\varphi_0=\varphi_0\frac{\expect_x\tau^p}{2^p p!}$.
\end{proof}

\begin{newthm}{Следствие}
    \[
        \lambda_1=\llim_{p\to\infty}2p\left[\frac{\displaystyle\int_{\D}\left(\expect_x\tau^{p-1}\right)^2dx}{\displaystyle\int_{\D}\left(\expect_x\tau^p\right)^2dx}\right]^{1/2};\quad \varphi_1(x)=\llim_{p\to\infty}\frac{\expect_x\tau^p}
{\left[\displaystyle\int_{\D}\left(\expect_x\tau^p\right)^2dx\right]^{1/2}}.
    \]
\end{newthm}

\begin{lemma}{}
    Пусть $\lambda_{(p)}(x)=2p\frac{\expect_x\tau^{p-1}}{\expect_x\tau^p}${\rm ,} тогда $\llim_{p\to\infty}\lambda_{(p)}(x)=\lambda_1${\rm .}
\end{lemma}

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
    \[      \lambda_{(p)}(x)=\frac{\varphi^{(p-1)}(x)}{\varphi^{(p)}(x)}=\frac{\varphi_{(p-1)}(x)}{\varphi_{(p)}(x)}\frac{\norm{\varphi^{(p-1)}}}{\norm{\varphi^{(p)}}}=\frac{\varphi_{(p-1)}(x)}{\varphi_{(p)}(x)}\lambda_{(p)};
    \]
    $\llim_{p\to\infty}\varphi_{(p)}(x)=\varphi_1(x)>0$, $\llim_{p\to\infty}\lambda_{(p)}=\lambda_1$, поэтому $\llim_{p\to\infty}\lambda_{(p)}(x)=\frac{\varphi_1(x)}{\varphi_1(x)}\lambda_1=\lambda_1.~\square $

\Section{Построение статистических оценок итераций оператора Грина}
    Один из вариантов метода Монте"--~Карло для вычисления $\lambda_1$ может быть основан на определении функции $$v_n(x)=(\IGreen^n\One)(x)=\frac{1}{2^nn!}\expect_x\tau^n,$$ т.~е. на оценке моментов случайной величины $\tau$. Займёмся получением несмещённых оценок величин $v_n$.
    
    \smallskip

{\bf 3.1. Преобразование сдвига.}
    Для построения несмещённых оценок для $v_n(x)$ используем уравнение
    \begin{gather}
        \Delta u +\lambda u =0, \quad u\bigr|_\Gamma=1.
            \label{kuzn:eq-v}
    \end{gather}
При $\abs{\lambda}<\lambda_1$ решение этого уравнения имеет вид
    \[
        u(x, \lambda)=\expect_x e^{\lambda\tau/2}=\lsum_{n=0}^\infty\lambda^n v_n(x), \quad v_0(x)=1.
    \]

    Пусть $c>0$, $\D\subset\mathbb{R}^3$. Свяжем с задачей \eqref{kuzn:eq-v} интегральное уравнение. Для шара $K_R(x)$ радиуса $R$ с центром в точке $x$ ($K_R(x)\subset \D$) имеем \cite{kuzn:ENS}:
    \begin{equation}\label{kuzn:eq-u}
        u(x)=q\int_{S_R}u(y)d\omega+(1-q)\int_{K_R}p(x, y)\Bigl(1+\frac{\lambda}{c^2}\Bigr)u(y)dy,
    \end{equation}
    где $q=\frac{Rc}{\sh Rc}$, $\omega$ "--- равномерное распределение на сфере, $p(x, y)=\frac{c^2}{4\pi\abs{x-y}}\hm\times\frac{\sh((R-\abs{x-y})c)}{(\sh cR-cR)}$ "--- переходная плотность.

    Соотношение \eqref{kuzn:eq-u} позволяет строить несмещённые оценки для $u(x, \lambda)$ на траекториях $\seq{x_i}_{i=0}^\infty$ случайного процесса, для которого $x_0=x$, $x_{i+1}$ распределено с вероятностью $q_i$ равномерно на сфере $S_{R(x_i)}$ и с вероятностью $(1-q_i)$ в шаре с плотностью $p(x_i, y)$. Нетрудно доказать, что с вероятностью~$1$ существует $x_\infty=\llim_{i\to\infty} x_i$ и $x_\infty\in \Gamma$.

    С процессом $\seq{x_i}_{i=0}^\infty$ естественно связать мартингал несмещённых оценок для $u(x, \lambda)$:
    $\xi_n=\left(1+\lambda/c^2\right)^{N_n}u(x_n)$, $n=1, 2, \ldots$, где $N_n$ --- число переходов в шар на траектории $x_0, x_1, \ldots, x_n$. Известно \cite{kuzn:ENS}, что п.~н. существует $N\hm=\llim_{n\to\infty}N_n<+\infty$, причём конечны моменты $\expect_x N^s$, $s=1, 2, \ldots$.

\begin{lemma}{}
        Пусть $\abs{\lambda}<\rho<\lambda_1/3$ и $\rho<c^2${\rm ,} тогда мартингал $\xi_n$ квадратично интегрируемый{\rm .}
    \end{lemma}

\smallskip
{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
        \[
            \abs{\xi_n^2}\leqslant\Bigl(1+\frac{\rho}{c^2}\Bigr)^{2N_n}\abs{u(x_n)}^2\leqslant
            \Bigl(1+\frac{2\rho}{c^2}+\frac{\rho^2}{c^4}\Bigr)^{N_n}\abs{u(x_n)}^2\leqslant
            \Bigl(1+\frac{3\rho}{c^2}\Bigr)^{N_n}\abs{u(x_n)}^2;
        \]
$
            \abs{u^2(x)}=\abs{\left(\expect_xe^{\lambda\tau/2}\right)^2}\leqslant
            \left(\expect_xe^{\tau\rho/2}\right)^2\leqslant \expect_xe^{\rho\tau}\leqslant
            \expect_x e^{3\rho\tau/2}=u(x, 3\rho).
$
        Тогда $\abs{\xi_n^2(x, \lambda)}\leqslant \left(1+3\rho/c^2\right)^{N_n}u(x_n, 3\rho)$, значит, \mbox{$\abs{\expect_x\xi_n^2(x, \lambda)}\leqslant u(x, 3\rho)<\infty.\,\square$}


\begin{newthm}{Следствие}
        Случайная величина $\xi=\left(1+\lambda/c^2\right)^N$ является несмещённой оценкой для $u(x, \lambda)$.
\end{newthm}

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
        Действительно, квадратично интегрируемый мартингал равномерно интегрируем, поэтому
\begin{multline*}
u(x,\lambda)=\expect_x\xi_n(x, \lambda)=\llim_{n\to\infty}\expect_x\xi_n(x, \lambda)=\expect_x\llim_{n\to\infty}\xi_n(x, \lambda)=
\\
=\expect_x\xi\cdot\llim_{n\to\infty}u(x_n, \lambda)=\expect_x\xi.~\square
\end{multline*}


    Представим $\xi$ в другой форме: $$\xi=\Bigl(1+\frac{\lambda}{c^2}\Bigr)^N=\lsum_{n=0}^N C_N^n\frac{\lambda^n}{c^{2n}}.$$ Обозначим $N^{[n]}=N(N-1)\cdots(N-n+1),$ тогда
    \begin{equation}\label{kuzn:eq-xi}
        \xi=\lsum_{n=0}^\infty\lambda^n\frac{N^{[n]}}{n!c^{2n}},
    \end{equation}
    причём ряд \eqref{kuzn:eq-xi} мажорируется рядом $\lsum_{n=0}^\infty\rho^n\frac{N^{[n]}}{n!c^{2n}},$ поэтому
    \[
        u(x, \lambda)=\expect_x\xi=\lsum_{n=0}^\infty\lambda^n\expect_x\Bigl(\frac{N^{[n]}}{n!c^{2n}}\Bigr),
    \]
    следовательно, справедлива
    \begin{theorem}{}
        Случайная величина $\frac{N^{[n]}}{n!c^{2n}}$ является несмещённой оценкой $v_n(x)$ с конечной дисперсией.
    \end{theorem}

\smallskip
{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
        Осталось проверить конечность дисперсии. Оценим дисперсию величины ${\zeta_n=\frac{N^{[n]}}{n!c^{2n}}}$. Можно считать, что $N\geqslant n,$ иначе ${\zeta_n=0}$.
\begin{multline*}
            \zeta_n^2=\left[\frac{2N\cdot 2(N-1) \cdots 2(N-n+1)}{2n\cdot2(n-1) \cdots 2}\right]^2\cdot\frac{1}{c^{4n}}=
\\            
         =   \frac{(2N)^{[2n]}}{(2n)!c^{4n}}\cdot\frac{\frac{2N}{2N-1}\cdot\frac{2N-2}{2N-3} \cdots \frac{2N-2n+2}{2N-2n+1}}{\frac{2n}{2n-1}\cdot\frac{2n-2}{2n-3} \cdots \frac{2}{1}}.
\end{multline*}

        Функция $\frac{2k}{2k-1}$ убывает, поэтому $\frac{2(N-k)}{2(N-k)-1}<\frac{2(n-k)}{2(n-k)-1}$, следовательно,
        \[
            \zeta_n^2\leqslant\frac{(2N)^{[2n]}}{(2n)!c^{2\cdot2n}}=\eta_n.
        \]

        Докажем, что $\expect_x\eta_n<+\infty$. Действительно,
        \[
        \abs{\expect_x\xi^2}\leqslant\abs{\expect_x\Bigl(1+\frac{\lambda}{c^2}\Bigr)^{2N}}\leqslant u(x, 3\rho) \quad\text{и}\quad \expect_x\xi^2=\lsum_{k=0}^\infty\lambda^k\expect_x\Bigl(\frac{(2N)^{[k]}}{k!c^{2k}}\Bigr),
        \]
        отсюда
        \[
            \expect_x\frac{(2N)^{[k]}}{k!c^{2k}}<+\infty.\quad \square
        \]

\smallskip

{\bf 3.2. Реализация оценки сдвига.}
    Оценка $\zeta_n$ не является реализуемой в силу невозможности моделирования траектории бесконечной длины. Для получения реализуемой оценки остановим мартингал $\xi_i$ в момент первого попадания процесса в $\varepsilon$-окрестность границы. Обозначим $N_\varepsilon$ "--- число переходов в шар до попадания в $\varepsilon$-окрестность границы, $x_\varepsilon$ "--- последнюю точку процесса, тогда $\xi_\varepsilon=\left(1+\lambda/c^2\right)^{N_\varepsilon}u(x_\varepsilon)$ является несмещённой оценкой $u(x, \lambda)$:
    \[
\xi_\varepsilon(\lambda)=\lsum_{k=0}^\infty\lambda^k\frac{N_\varepsilon^{[k]}}{k!c^{2k}}\lsum_{m=0}^\infty\lambda^mv_m(x_\varepsilon)=\lsum_{n=0}^\infty\lambda^n\lsum_{k=0}^n\frac{N_\varepsilon^{[k]}}{k!c^{2k}}v_{n-k}(x_\varepsilon).
    \]
    Отсюда
    \begin{equation}\label{kuzn:eq-v-n}
        v_{n}(x)=\expect_x\lsum_{k=0}^n\frac{N_\varepsilon^{[k]}}{k!c^{2k}}v_{n-k}(x_\varepsilon).
    \end{equation}

    Смещённая оценка $v_n(x)$ получится, если в \eqref{kuzn:eq-v-n} отбросить слагаемые, для которых $k<n$.
    Она имеет вид
    \begin{equation}\label{kuzn:eq-xi-n-e}
        \xi_{n, \varepsilon}=\frac{N_\varepsilon^{[n]}}{n!c^{2n}}.
    \end{equation}

{\bf 3.3. Оценка смещения и трудоёмкость алгоритма.}
    Оценим величину смещения $v_n(x)-\expect_x\xi_{n,\varepsilon}$.
    Пусть $g(\varepsilon)$ "--- модуль непрерывности функции $v_1(x)=(\IGreen\One)(x)$, т.~е. ${\abs{\IGreen\One(x)}<g(\varepsilon)}$, при ${\dist(x, \Gamma)<\varepsilon}$, значит, $v_k(x)=\IGreen^{k-1}v_1(x)\hm\leqslant\norm{\IGreen^{k-1}}_Cg(\varepsilon)$.
    Тогда смещение $v_n$ не превзойдёт
    \[
        g(\varepsilon)\lsum_{k=0}^{n-1}\norm{\IGreen^{n-k-1}}_C\cdot\norm{\IGreen^k}_C\leqslant g(\varepsilon)n\norm{\IGreen}^{n-1}_C.
    \]

    Как правило, $g(\varepsilon)=g\cdot\varepsilon$, и для достижения смещения $\delta$ необходимо взять $\varepsilon={\delta} /({ng\|{\IGreen}\|^{n-1}_C})$.

    Для выпуклой области можно показать, что среднее число шагов процесса до попадания в $\varepsilon$-окрестность границы имеет порядок $\abs{\ln\varepsilon}$.

{\bf 4.1. Оценки итераций с помощью рядов.}
    Рассмотрим задачу \eqref{kuzn:task1} в кубе $\D=[0,\,1]^3$. Собственные числа такой задачи известны:
    \[
        \lambda_{k,l,m}=\pi^2(k^2+l^2+m^2), \quad k, l, m = 1, 2, 3, \ldots.
    \]
    В частности, $\lambda_1=3\pi^2, \lambda_2=6\pi^2$, причём $\lambda_2$ имеет кратность $3$.

    Собственные функции имеют вид $\varphi_{k,l,m}(x, y, z)=X_k(x)Y_l(y)Z_m(z)$, где $X_k(x)=\sqrt{2}\sin\pi kx$, $Y_l$ и $Z_m$ строятся аналогично.

    Используя разложение $\One$ по собственным функциям, можно получить тождество
\begin{multline}\label{kuzn:eq-scp}
\scp{\One}{\IGreen^p\One}=\lsum_{k, l, m=0}^\infty\frac{8^3}{\pi^6}\frac{1}{(2k+1)^2(2l+1)^2(2m+1)^2}\times
\\
\times\frac{1}{\left\lbrace\pi^2\left[(2k+1)^2+(2l+1)^2+(2m+1)^2\right]\right\rbrace^p}.
\end{multline}
Для определения суммы ряда \eqref{kuzn:eq-scp} можно вычислять его частичные суммы $S_n=\lsum_{k,l,m=0}^n\ldots$.

    Воспользовавшись неравенством $a+b+c\geqslant 3\sqrt[3]{abc}\quad(a\geqslant0, b\geqslant 0, c\geqslant0)$, получим
    \[
        \frac{1}{[(2k+1)^2+(2l+1)^2+(2m+1)^2]^p}\leqslant\frac{1}{[3\sqrt[3]{(2k+1)^2(2l+1)^2(2m+1)^2]^p}},
    \]
    что влечёт оценку остатка $R_N=\scp{\One}{\IGreen^p\One}-S_N$. А именно, $R_N\leqslant\tilde R_N$, где $\tilde R_N$ построено для ряда
    \[
        \lsum_{k,l,m=0}^\infty\frac{8^3}{3^p\pi^{6+2p}}\frac{1}{(2k+1)^{2+2p/3}(2l+1)^{2+2p/3}(2m+1)^{2+2p/3}},
    \]
    который распадается в произведение трёх рядов.

    Пусть $A_p=\lsum_{k=0}^\infty\frac{1}{(2k+1)^{2+2p/3}}$, $B_p=\frac{8^3}{3^p\pi^{6+2p}}$, тогда
    \[
        \tilde R_N=3B_p\Bigl[A^2_p\lsum_{k=N+1}^\infty\frac{1}{(2k+1)^{2+2p/3}}\Bigr]=3B_pA^2_pr_N.
    \]

    Нетрудно получить оценку для $r_N$ и $A_p$:
    \begin{align*}
        r_N&\leqslant\frac{1}{2}\frac{1}{(1+2p/3)(2N+1)^{1+2p/3}};\quad
        A_p\leqslant1+r_0=1+\frac{1}{2(1+2p/3)};\\
        R_N&\leqslant\tilde R_N\leqslant3B_p\left(1+\frac{3}{6+4p}\right)^2\frac{3}{6+4p}\frac{1}{(2N+1)^{1+2p/3}}.
    \end{align*}

По заданной точности $E$ вычислялись $N$ и итерации оператора. Результаты сведены в табл.~\ref{kuzn:tab-green}.

{\bf 4.2. Реализация оценки (7).}
    Наибольшую трудность при реализации оценки \eqref{kuzn:eq-xi-n-e} представляет моделирование плотности вероятности перехода $\tilde p(x, y)=q\delta_R(y)+(1-q)p(x, y)$, где $\delta_R$ --- равномерное распределение на сфере с центром в точке $x$, а 
$
        p(x, y)=\frac{c^2}{4\pi r}\frac{\sh(cR-cr)}{\sh cR-cR}.
 $

    Моделирование $\tilde p(x, y)$ осуществлялось методом отбора.

    Пусть $a=cR$ и $p(r)$ --- плотность на $[0,a]$, $\eta$ --- случайная величина с плотностью $p(r)$.

    \begin{lemma}{}
        Пусть $\frac{r\sh (a-r)}{\sh a}<p(r)${\rm ,} тогда метод отбора моделирует смесь
        \begin{equation}\label{kuzn:eq-mix}
            \frac{a}{\sh a}\delta(a-r)+\frac{r\sh(a-r)}{\sh a}.
        \end{equation}
    \end{lemma}

{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
        Определим случайную величину $\zeta$ условиями:
        \[
            \zeta=\begin{cases}
                \eta, &\text{если $\alpha p(\eta)<\frac{\eta\sh(a-\eta)}{\sh a}$;}\\
                a,&\text{в противном случае.}\\
            \end{cases}
        \]
Далее
\begin{multline*}
P(\zeta=a)=P\Bigl(\alpha p(\eta)\geqslant\frac{\eta\sh(a-\eta)}{\sh a}\Bigr)=
                \int_0^ap(r)\Bigl(1-\frac{r\sh(a-r)}{\sh a\cdot p(r)}\Bigr)dr=
\\
=1-\int_0^a\frac{r\sh(a-r)}{\sh a}dr=\frac{a}{\sh a}.
\end{multline*}

        Пусть $x<a$, тогда
\begin{multline*}
F_\zeta(x)=P(\zeta<x)=P\Bigl(\eta<x, \,\alpha p(\eta)<\frac{\eta\sh(a-\eta)}{\sh a}\Bigr)=
\\
=\int_0^x p(r)\frac{r\sh(a-r)}{\sh a\cdot p(r)}dr
                =\int_0^x \frac{r\sh(a-r)}{\sh a}dr.\quad \square
\end{multline*}

\smallskip

    В качестве $p(r)$ можно выбрать $p(r)=r\ch(a-r)/(\ch a-1)$, тогда, очевидно,
    \[
        \frac{r\sh(a-r)}{\sh a}<\frac{r\ch(a-r)}{\ch a -1}.
    \]

    \begin{lemma}{}
        Пусть $\xi_1$  и $\xi_2$ независимые случайные величины{\rm ,} имеющие экспоненциальное распределение с плотностью $e^{-x}${\rm ,} $x>0${\rm ,} $z_1=\xi_1-a\left[{\xi_1}/{a}\right]${\rm ,} $z_2=\xi_2-a[{\xi_2}/{a}]${\rm ,} тогда случайная величина
        \[
            \eta=\begin{cases}
                z_1+z_2,&\text{если $z_1+z_2<a$;}\\
                2a-(z_1+z_2),&\text{если $z_1+z_2\geqslant a;$}
            \end{cases}
        \]
        имеет плотность $p(r)${\rm .}
\end{lemma}

\smallskip
{\it Д\,о\,к\,а\,з\,а\,т\,е\,л\,ь\,с\,т\,в\,о.}
Ясно, что $0\leqslant z_1<a$. Далее
\[P(z_1<x)=\lsum_{k=0}^\infty P(ka\leqslant \xi_1<x+ka)=\lsum_{k=0}^\infty e^{-ka}-e^{-ka-x}=\frac{1-e^{-x}}{1-e^{-a}};\]
\begin{multline*}
                P(z<x)=P(z_1+z_2<x)+P(2a-(z_1+z_2)<x,\,z_1+z_2\geqslant a)=
                \\
                =\frac{1}{\left(1-e^{-a}\right)^2}\left(1-(x+1)e^{-x}+(x-1)e^{x-2a}+e^{-2a}\right);
\end{multline*}
\begin{multline*}
            p_z(x)=\frac{1}{\left(1-e^{-a}\right)^2}\left((x+1)e^{-x}-e^{-x}+e^{x-2a}+(x-1)e^{x-2a}\right)=
            \\
        =\frac{1}{\left(1-e^{-a}\right)^2}\left(xe^{-x}+xe^{x-2a}\right)=\frac{2e^{-a}x\ch(a-x)}{1-2e^{-a}+e^{-2a}}=\frac{x\ch(x-a)}{\ch a -1}.\quad \square
    \end{multline*}


    Моделирование переходной плотности $\tilde p(x, y)$ теперь не вызывает затруднений.

    \begin{lemma}{}
        Пусть случайная величина $\eta$ имеет плотность \eqref{kuzn:eq-mix} $(a=cR),$ $\omega$ "--- равномерно распределено на единичной сфере{\rm ,} тогда случайный вектор $y=x+{\eta}\omega/c$ имеет плотность распределения $\tilde p(x, y)${\rm .}
    \end{lemma}

        \item использование достаточного количества блоков (thread block) в сетке (grid) с последующим суммированием и осреднением результатов, полученных различными потоками сетки, не на видеокарте, а на ПК с двойной точностью.
    \end{itemize}
